// server/services/visionService.ts

import vision from "@google-cloud/vision";

const client = new vision.ImageAnnotatorClient();

/**
 * Analyze image using Google Vision
 * Returns:
 * - detected faces
 * - OCR text
 */
export async function analyzeImage(imageUrl: string) {
  const [result] = await client.annotateImage({
    image: { source: { imageUri: imageUrl } },
    features: [
      { type: "FACE_DETECTION" },
      { type: "TEXT_DETECTION" }
    ]
  });

  const faces = result.faceAnnotations || [];
  const text = result.fullTextAnnotation?.text || "";

  return {
    faces,
    text,
  };
}

// server/services/providerVerification.ts

import { analyzeImage } from "./visionService";

/**
 * Score face quality + confidence
 */
function scoreFace(faces: any[]) {
  if (faces.length !== 1) return 0;

  const face = faces[0];

  let score = 50;

  if (face.detectionConfidence > 0.8) score += 20;
  if (face.underExposedLikelihood === "VERY_UNLIKELY") score += 10;
  if (face.blurredLikelihood === "VERY_UNLIKELY") score += 10;

  return Math.min(score, 100);
}

/**
 * Provider verification logic
 * Runs AFTER submit (async)
 */
export async function verifyProvider(documentUrl: string, selfieUrl: string) {
  const doc = await analyzeImage(documentUrl);
  const selfie = await analyzeImage(selfieUrl);

  const docScore = scoreFace(doc.faces);
  const selfieScore = scoreFace(selfie.faces);

  const finalScore = Math.round((docScore + selfieScore) / 2);

  return {
    score: finalScore,
    ocrText: doc.text,
    facesDetected:
      doc.faces.length === 1 && selfie.faces.length === 1
  };
}

// server/routes/provider.ts

import express from "express";
import { verifyProvider } from "../services/providerVerification";
import db from "../db";

const router = express.Router();

/**
 * Provider submits application
 * IMPORTANT:
 * - saves immediately
 * - returns immediately
 * - verification runs async
 */
router.post("/apply", async (req, res) => {
  const { userId, documentUrl, selfieUrl } = req.body;

  const application = await db.provider_applications.create({
    user_id: userId,
    document_url: documentUrl,
    selfie_url: selfieUrl,
    status: "pending"
  });

  // RETURN FAST (NO WAITING)
  res.json({
    status: "pending",
    applicationId: application.id
  });

  // BACKGROUND PROCESS
  setImmediate(async () => {
    try {
      const result = await verifyProvider(documentUrl, selfieUrl);

      let status = "manual_review";

      if (result.score >= 80 && result.facesDetected) {
        status = "approved";
      } else if (result.score < 50) {
        status = "rejected";
      }

      await db.provider_applications.update(application.id, {
        status,
        face_score: result.score,
        ocr_data: result.ocrText,
        reviewed_at: new Date(),
        decision_reason: `score=${result.score}`
      });

    } catch (err) {
      console.error("Verification failed", err);
    }
  });
});

export default router;

// server/services/discount.ts

/**
 * Returns discount percentage
 * RULE:
 * - special approved → 10%
 * - club member → 5%
 * - else → 0%
 */
export function getUserDiscount(user: any) {

  if (
    user.discount_status === "approved" &&
    user.discount_program === "special_10"
  ) {
    return 0.10;
  }

  if (user.is_club_member) {
    return 0.05;
  }

  return 0;
}
