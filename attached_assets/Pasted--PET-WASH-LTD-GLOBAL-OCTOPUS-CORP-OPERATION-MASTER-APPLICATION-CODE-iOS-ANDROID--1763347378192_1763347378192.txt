 * amount DECIMAL(10, 2),
 * date_incurred DATE DEFAULT CURRENT_DATE,
 * status VARCHAR(20) DEFAULT 'Pending' -- 'Pending', 'Approved'

import React, { useState } from 'react';
import { 
  StyleSheet, 
  Text, 
  View, 
  TextInput, 
  TouchableOpacity, 
  ScrollView, 
  SafeAreaView, 
  StatusBar, 
  Alert,
  Modal,
  Platform
} from 'react-native';

// --- MOCK DATA & STATE ---
// In production, fetch this from your API
const INITIAL_EXPENSES = [
  { id: 1, type: 'Hardware', desc: 'HP Laptop Model 250 G9', amount: 1250.00, status: 'Approved', user: 'Owner' },
  { id: 2, type: 'Office', desc: '55" TV for Office Display', amount: 699.00, status: 'Pending', user: 'Manager' },
  { id: 3, type: 'Materials', desc: 'Polyester Uniform Roll', amount: 320.50, status: 'Pending', user: 'Staff' },
];

export default function App() {
  // Navigation State
  const [currentScreen, setCurrentScreen] = useState('DASHBOARD'); // 'DASHBOARD' or 'FORM'
  
  // Data State
  const [expenses, setExpenses] = useState(INITIAL_EXPENSES);
  
  // Form State
  const [category, setCategory] = useState('Client Lunch');
  const [desc, setDesc] = useState('');
  const [amount, setAmount] = useState('');
  const [showCategoryModal, setShowCategoryModal] = useState(false);

  // --- ACTIONS ---

  const handleSubmit = () => {
    if (!amount || !desc) {
      Alert.alert("Missing Info", "Please fill out Description and Amount.");
      return;
    }

    const newExpense = {
      id: Math.random(), // Temporary ID
      type: category,
      desc: desc,
      amount: parseFloat(amount),
      status: 'Submitted',
      user: 'Current User'
    };

    // Add to top of list
    setExpenses([newExpense, ...expenses]);
    
    // Reset and go back
    setDesc('');
    setAmount('');
    setCategory('Client Lunch');
    setCurrentScreen('DASHBOARD');
    Alert.alert("Success", "Expense uploaded to Global HQ.");
  };

  // --- RENDER HELPERS ---

  const CategoryButton = ({ label }) => (
    <TouchableOpacity 
      style={[styles.catOption, category === label && styles.catOptionSelected]} 
      onPress={() => { setCategory(label); setShowCategoryModal(false); }}
    >
      <Text style={[styles.catOptionText, category === label && styles.catOptionTextSelected]}>{label}</Text>
    </TouchableOpacity>
  );

  // --- SCREEN: DASHBOARD ---
  const renderDashboard = () => (
    <View style={styles.screenContainer}>
      <View style={styles.header}>
        <Text style={styles.headerTitle}>Pet Wash Ltd.</Text>
        <Text style={styles.headerSubtitle}>Global Operations • Octopus Corp</Text>
      </View>

      <View style={styles.summaryCard}>
        <Text style={styles.summaryLabel}>TOTAL PENDING APPROVAL</Text>
        <Text style={styles.summaryAmount}>
          $ {expenses.filter(e => e.status !== 'Approved').reduce((sum, item) => sum + item.amount, 0).toFixed(2)}
        </Text>
      </View>

      <View style={styles.listContainer}>
        <Text style={styles.sectionHeader}>Recent Expenses</Text>
        <ScrollView showsVerticalScrollIndicator={false}>
          {expenses.map((item) => (
            <View key={item.id} style={styles.expenseCard}>
              <View style={styles.cardTop}>
                <Text style={styles.cardCategory}>{item.type}</Text>
                <Text style={styles.cardAmount}>${item.amount.toFixed(2)}</Text>
              </View>
              <Text style={styles.cardDesc}>{item.desc}</Text>
              <View style={styles.cardBottom}>
                <Text style={styles.cardUser}>{item.user}</Text>
                <Text style={[
                  styles.statusBadge, 
                  item.status === 'Approved' ? styles.statusGreen : styles.statusOrange
                ]}>{item.status}</Text>
              </View>
            </View>
          ))}
          <View style={{height: 100}} /> 
        </ScrollView>
      </View>

      <TouchableOpacity 
        style={styles.fab} 
        onPress={() => setCurrentScreen('FORM')}
      >
        <Text style={styles.fabText}>+ NEW EXPENSE</Text>
      </TouchableOpacity>
    </View>
  );

  // --- SCREEN: EXPENSE FORM ---
  const renderForm = () => (
    <View style={styles.screenContainer}>
      <View style={styles.header}>
        <TouchableOpacity onPress={() => setCurrentScreen('DASHBOARD')} style={styles.backBtn}>
          <Text style={styles.backBtnText}>← Cancel</Text>
        </TouchableOpacity>
        <Text style={styles.headerTitle}>New Claim</Text>
      </View>

      <ScrollView style={styles.formContainer}>
        
        {/* Category Selector */}
        <Text style={styles.label}>Expense Category</Text>
        <TouchableOpacity style={styles.inputSelector} onPress={() => setShowCategoryModal(true)}>
          <Text style={styles.inputText}>{category}</Text>
          <Text style={styles.arrow}>▼</Text>
        </TouchableOpacity>

        {/* Description Input */}
        <Text style={styles.label}>Description / Item</Text>
        <TextInput 
          style={styles.input} 
          placeholder="e.g. Polyester uniforms, Office TV..." 
          placeholderTextColor="#999"
          value={desc}
          onChangeText={setDesc}
        />

        {/* Amount Input */}
        <Text style={styles.label}>Amount ($)</Text>
        <TextInput 
          style={styles.input} 
          placeholder="0.00" 
          placeholderTextColor="#999"
          keyboardType="numeric"
          value={amount}
          onChangeText={setAmount}
        />

        {/* Purpose Input (Optional) */}
        <Text style={styles.label}>Client / Purpose (Optional)</Text>
        <TextInput 
          style={styles.input} 
          placeholder="e.g. Lunch with Paws Ltd." 
          placeholderTextColor="#999"
        />

        {/* Submit Button */}
        <TouchableOpacity style={styles.submitBtn} onPress={handleSubmit}>
          <Text style={styles.submitBtnText}>SUBMIT TO HQ</Text>
        </TouchableOpacity>

      </ScrollView>

      {/* Category Modal */}
      <Modal visible={showCategoryModal} transparent={true} animationType="slide">
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalHeader}>Select Category</Text>
            <ScrollView>
              <CategoryButton label="Client Lunch / Meals" />
              <CategoryButton label="Car Rental / Travel" />
              <CategoryButton label="Office Equipment (TV, Furniture)" />
              <CategoryButton label="Hardware (Laptops, Phones)" />
              <CategoryButton label="Materials (Polyester, Fabrics)" />
              <CategoryButton label="Utilities / Bills" />
            </ScrollView>
            <TouchableOpacity style={styles.closeBtn} onPress={() => setShowCategoryModal(false)}>
              <Text style={styles.closeBtnText}>Close</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </View>
  );

  return (
    <SafeAreaView style={styles.safeArea}>
      <StatusBar barStyle="light-content" />
      {currentScreen === 'DASHBOARD' ? renderDashboard() : renderForm()}
    </SafeAreaView>
  );
}
