-- Enable PostGIS
CREATE EXTENSION IF NOT EXISTS postgis;

CREATE TABLE providers (
  id UUID PRIMARY KEY,
  name TEXT NOT NULL,
  is_online BOOLEAN NOT NULL DEFAULT FALSE,
  is_busy BOOLEAN NOT NULL DEFAULT FALSE,
  service_types TEXT[] NOT NULL DEFAULT '{}',
  last_seen_at TIMESTAMPTZ,
  location GEOGRAPHY(POINT, 4326),
  rating NUMERIC(3,2) DEFAULT 5.00,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX providers_location_idx
ON providers USING GIST (location);

CREATE TABLE jobs (
  id UUID PRIMARY KEY,
  customer_id UUID NOT NULL,
  service_type TEXT NOT NULL,
  status TEXT NOT NULL CHECK (status IN ('pending','offering','assigned','in_progress','completed','canceled')),
  pickup_address TEXT NOT NULL,
  pickup_place_id TEXT,
  pickup_location GEOGRAPHY(POINT, 4326) NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE job_offers (
  id UUID PRIMARY KEY,
  job_id UUID REFERENCES jobs(id) ON DELETE CASCADE,
  provider_id UUID REFERENCES providers(id),
  status TEXT CHECK (status IN ('sent','accepted','rejected','expired','canceled')),
  sent_at TIMESTAMPTZ DEFAULT NOW(),
  expires_at TIMESTAMPTZ,
  responded_at TIMESTAMPTZ
);

CREATE UNIQUE INDEX job_offer_unique ON job_offers(job_id, provider_id);

async function geocodePlaceId(placeId: string) {
  const url = `https://maps.googleapis.com/maps/api/place/details/json?place_id=${placeId}&key=${process.env.GOOGLE_API_KEY}`
  const r = await fetch(url)
  const data = await r.json()
  return data.result.geometry.location
}

app.post("/jobs", async (req, res) => {
  const { customerId, serviceType, pickupAddress, pickupPlaceId } = req.body

  const geo = await geocodePlaceId(pickupPlaceId)
  const jobId = crypto.randomUUID()

  await pg.query(
    `INSERT INTO jobs (id, customer_id, service_type, status, pickup_address, pickup_place_id, pickup_location)
     VALUES ($1,$2,$3,'offering',$4,$5,ST_GeogFromText($6))`,
    [jobId, customerId, serviceType, pickupAddress, pickupPlaceId, `POINT(${geo.lng} ${geo.lat})`]
  )

  dispatchOffers(jobId, serviceType, geo.lat, geo.lng)

  res.json({ jobId })
})

async function dispatchOffers(jobId: string, serviceType: string, lat: number, lng: number) {

  const point = `POINT(${lng} ${lat})`
  const radius = 6000
  const batchSize = 5
  const ttl = 20

  const result = await pg.query(
    `SELECT id FROM providers
     WHERE is_online = TRUE
       AND is_busy = FALSE
       AND service_types @> ARRAY[$1]::text[]
       AND ST_DWithin(location, ST_GeogFromText($2), $3)
     ORDER BY ST_Distance(location, ST_GeogFromText($2)) ASC, rating DESC
     LIMIT $4`,
    [serviceType, point, radius, batchSize]
  )

  for (const row of result.rows) {
    const offerId = crypto.randomUUID()

    await pg.query(
      `INSERT INTO job_offers (id, job_id, provider_id, status, expires_at)
       VALUES ($1,$2,$3,'sent',NOW()+INTERVAL '${ttl} seconds')`,
      [offerId, jobId, row.id]
    )

    io.to(`provider:${row.id}`).emit("job_offer", {
      jobId,
      offerId,
      expiresIn: ttl
    })
  }
}

==========================================
8. ACCEPT OFFER (ATOMIC FIRST WINS)
==========================================
***************************************************************************************************/

app.post("/jobs/:jobId/accept", async (req, res) => {
  const { offerId, providerId } = req.body
  const { jobId } = req.params

  const client = await pg.connect()

  try {
    await client.query("BEGIN")

    const offer = await client.query(
      `UPDATE job_offers
       SET status='accepted', responded_at=NOW()
       WHERE id=$1 AND job_id=$2 AND provider_id=$3
       AND status='sent' AND expires_at > NOW()
       RETURNING id`,
      [offerId, jobId, providerId]
    )

    if (!offer.rowCount) {
      await client.query("ROLLBACK")
      return res.json({ ok:false })
    }

    const job = await client.query(
      `UPDATE jobs
       SET status='assigned'
       WHERE id=$1 AND status='offering'
       RETURNING id`,
      [jobId]
    )

    if (!job.rowCount) {
      await client.query("ROLLBACK")
      return res.json({ ok:false })
    }

    await client.query(`UPDATE providers SET is_busy=TRUE WHERE id=$1`, [providerId])
    await client.query(`UPDATE job_offers SET status='canceled' WHERE job_id=$1 AND status='sent'`, [jobId])

    await client.query("COMMIT")
    res.json({ ok:true })

  } catch(e) {
    await client.query("ROLLBACK")
    res.status(500).json({ error:true })
  } finally {
    client.release()
  }
})

async function startLocationLoop(providerId: string) {
  const { status } = await Location.requestForegroundPermissionsAsync()
  if (status !== "granted") return

  await Location.watchPositionAsync(
    { accuracy: Location.Accuracy.Balanced, timeInterval: 7000 },
    async (pos) => {
      await fetch(`https://your-api.com/providers/${providerId}/location`, {
        method: "POST",
        headers: { "Content-Type":"application/json" },
        body: JSON.stringify({
          lat: pos.coords.latitude,
          lng: pos.coords.longitude
        })
      })
    }
  )
}

async function acceptOffer(jobId: string, offerId: string, providerId: string) {
  await fetch(`https://your-api.com/jobs/${jobId}/accept`, {
    method:"POST",
    headers:{ "Content-Type":"application/json" },
    body: JSON.stringify({ offerId, providerId })
  })
}
