# FILE: .github/workflows/firebase-hosting-deploy.yml
#
# Pet Wash marketplace - CI deploy to Firebase Hosting
# Uses modern Google service account JSON (Base64) instead of firebase login:ci

name: Deploy PetWash to Firebase Hosting

on:
  push:
    branches:
      - main
  workflow_dispatch:

env:
  # Update if your Firebase or GCP project id is different
  PROJECT_ID: nifty-quanta-475212-v3

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      # 1. Checkout repository
      - name: Checkout repository
        uses: actions/checkout@v4

      # 2. Setup Node for build
      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      # 3. Install dependencies
      - name: Install dependencies
        run: npm ci

      # 4. Build project (adjust if your build script is different)
      - name: Build app
        run: npm run build

      # 5. Decode the Base64 service account JSON from GitHub secret
      #    and export GOOGLE_APPLICATION_CREDENTIALS pointing to it.
      - name: Prepare Google service account credentials
        run: |
          echo "${{ secrets.GOOGLE_APPLICATION_CREDENTIALS_JSON }}" | base64 -d > $HOME/gcp-key.json
          echo "GOOGLE_APPLICATION_CREDENTIALS=$HOME/gcp-key.json" >> $GITHUB_ENV

      # 6. Authenticate with Google Cloud using the service account
      - name: Google Cloud auth (service account JSON)
        uses: google-github-actions/auth@v2
        with:
          credentials_json: '${{ secrets.GOOGLE_APPLICATION_CREDENTIALS_JSON }}'
          project_id: ${{ env.PROJECT_ID }}

      # 7. Install Firebase CLI
      - name: Install Firebase CLI
        run: npm install -g firebase-tools

      # 8. Confirm Firebase project configuration
      - name: Show Firebase project info
        run: firebase projects:list || true

      # 9. Deploy to Firebase Hosting
      - name: Deploy to Firebase Hosting
        run: |
          firebase deploy \
            --only hosting \
            --project $PROJECT_ID \
            --non-interactive
