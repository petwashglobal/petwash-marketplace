/* =========================================================
   1. SHARED TYPES AND CONFIG
   ========================================================= */

export type BookingPlatform =
  | "K9000_WASH"
  | "WALK_MY_PET"
  | "PETTREK_TRANSPORT"
  | "SITTER_SUITE"
  | "GROOMING"
  | "PET_TRAINER_ACADEMY";

export type BookingMode =
  | "SINGLE_SLOT"
  | "MULTI_DAY"
  | "ARRIVAL_WINDOW";

export interface AvailabilitySlot {
  id: string;
  platform: BookingPlatform;
  providerId: string;
  // ISO strings in UTC
  start: string;
  end: string;
  // example: "AVAILABLE", "HELD", "BOOKED"
  status: "AVAILABLE" | "HELD" | "BOOKED";
  // optional metadata (station, area, dog size, etc)
  meta?: Record<string, any>;
}

export interface BookingLock {
  lockId: string;
  bookingId?: string;
  platform: BookingPlatform;
  slotId: string;
  userId: string;
  expiresAt: string; // ISO
}

export interface BookingCalendarProps {
  platform: BookingPlatform;
  userId: string;
  // how many days ahead to show
  daysAhead?: number;
  // fetch availability from backend for given range
  fetchAvailability: (params: {
    platform: BookingPlatform;
    from: string;
    to: string;
  }) => Promise<AvailabilitySlot[]>;
  // called when booking finally confirmed and paid
  onBookingConfirmed?: (data: { bookingId: string }) => void;
}

// Auto behavior config per platform - from your spec
const PLATFORM_BOOKING_MODE: Record<BookingPlatform, BookingMode> = {
  K9000_WASH: "SINGLE_SLOT",
  WALK_MY_PET: "SINGLE_SLOT",
  PETTREK_TRANSPORT: "ARRIVAL_WINDOW",
  SITTER_SUITE: "MULTI_DAY",
  GROOMING: "SINGLE_SLOT",
  PET_TRAINER_ACADEMY: "SINGLE_SLOT",
};

// Booking lock duration in seconds (5 minutes)
const BOOKING_LOCK_SECONDS = 5 * 60;

/* Simple date helpers without external libs */

function startOfDay(d: Date): Date {
  return new Date(d.getFullYear(), d.getMonth(), d.getDate(), 0, 0, 0, 0);
}

function addDays(d: Date, days: number): Date {
  const copy = new Date(d);
  copy.setDate(copy.getDate() + days);
  return copy;
}

function formatDateLabel(d: Date): string {
  return d.toLocaleDateString("en-IL", {
    weekday: "short",
    day: "2-digit",
    month: "short",
  });
}

function isSameDay(a: Date, b: Date): boolean {
  return (
    a.getFullYear() === b.getFullYear() &&
    a.getMonth() === b.getMonth() &&
    a.getDate() === b.getDate()
  );
}

function formatTimeRange(startIso: string, endIso: string): string {
  const s = new Date(startIso);
  const e = new Date(endIso);
  const opts: Intl.DateTimeFormatOptions = { hour: "2-digit", minute: "2-digit" };
  return `${s.toLocaleTimeString("en-IL", opts)} - ${e.toLocaleTimeString(
    "en-IL",
    opts
  )}`;
}

/* =========================================================
   2. FRONTEND - BOOKING CALENDAR COMPONENT (REACT 18)
   Pure white background, iOS style, payment lock flow
   ========================================================= */

import React, { useEffect, useMemo, useState } from "react";

type LockState =
  | { status: "IDLE" }
  | {
      status: "LOCKED";
      lockId: string;
      slotId: string;
      expiresAt: Date;
      secondsLeft: number;
    }
  | { status: "CONFIRMING"; lockId: string; slotId: string }
  | { status: "ERROR"; message: string };

export const BookingCalendar: React.FC<BookingCalendarProps> = ({
  platform,
  userId,
  daysAhead = 14,
  fetchAvailability,
  onBookingConfirmed,
}) => {
  const [currentDay, setCurrentDay] = useState<Date>(startOfDay(new Date()));
  const [availability, setAvailability] = useState<AvailabilitySlot[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [lockState, setLockState] = useState<LockState>({ status: "IDLE" });
  const [error, setError] = useState<string | null>(null);
  const [selectedRange, setSelectedRange] = useState<{
    start?: Date;
    end?: Date;
  }>({});

  const bookingMode = useMemo(
    () => PLATFORM_BOOKING_MODE[platform],
    [platform]
  );

  // Build days array for the visible period
  const days = useMemo(() => {
    const result: Date[] = [];
    for (let i = 0; i < daysAhead; i++) {
      result.push(addDays(currentDay, i));
    }
    return result;
  }, [currentDay, daysAhead]);

  // Group slots by day for display
  const slotsByDay = useMemo(() => {
    const map = new Map<string, AvailabilitySlot[]>();
    for (const slot of availability) {
      const dayKey = startOfDay(new Date(slot.start)).toISOString();
      if (!map.has(dayKey)) map.set(dayKey, []);
      map.get(dayKey)!.push(slot);
    }
    // sort each day's slots by start time
    for (const [key, list] of map.entries()) {
      list.sort(
        (a, b) =>
          new Date(a.start).getTime() - new Date(b.start).getTime()
      );
    }
    return map;
  }, [availability]);

  // Load availability when day or platform changes
  useEffect(() => {
    let cancelled = false;
    const load = async () => {
      setIsLoading(true);
      setError(null);
      try {
        const from = startOfDay(currentDay).toISOString();
        const to = addDays(currentDay, daysAhead).toISOString();
        const data = await fetchAvailability({ platform, from, to });
        if (!cancelled) setAvailability(data);
      } catch (e: any) {
        if (!cancelled) setError(e?.message || "Failed to load availability");
      } finally {
        if (!cancelled) setIsLoading(false);
      }
    };
    load();
    return () => {
      cancelled = true;
    };
  }, [currentDay, daysAhead, fetchAvailability, platform]);

  // Countdown for lock
  useEffect(() => {
    if (lockState.status !== "LOCKED") return;
    const interval = setInterval(() => {
      setLockState(prev => {
        if (prev.status !== "LOCKED") return prev;
        const secondsLeft =
          Math.max(
            0,
            Math.floor(
              (prev.expiresAt.getTime() - Date.now()) / 1000
            )
          );
        if (secondsLeft <= 0) {
          return { status: "ERROR", message: "Reservation expired" };
        }
        return { ...prev, secondsLeft };
      });
    }, 1000);
    return () => clearInterval(interval);
  }, [lockState.status]);

  /* 2.1 Handle day selection (for multi day behavior) */

  function handleDayClick(day: Date) {
    if (bookingMode !== "MULTI_DAY") {
      setCurrentDay(day);
      return;
    }
    setSelectedRange(prev => {
      // no start yet -> set start
      if (!prev.start || (prev.start && prev.end)) {
        return { start: day, end: undefined };
      }
      // start exists and end empty -> set end (ensure correct order)
      if (prev.start && !prev.end) {
        if (day.getTime() < prev.start.getTime()) {
          return { start: day, end: prev.start };
        }
        return { start: prev.start, end: day };
      }
      return { start: day, end: undefined };
    });
  }

  function dayInRange(day: Date): boolean {
    if (bookingMode !== "MULTI_DAY") return false;
    if (!selectedRange.start || !selectedRange.end) return false;
    return (
      day.getTime() >= selectedRange.start.getTime() &&
      day.getTime() <= selectedRange.end.getTime()
    );
  }

  /* 2.2 Lock slot via backend */

  async function lockSlot(slot: AvailabilitySlot) {
    if (lockState.status === "LOCKED") {
      // optional: release previous lock before new one
    }
    try {
      setLockState({ status: "IDLE" });
      setError(null);
      const res = await fetch("/api/bookings/lock", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          platform,
          userId,
          slotId: slot.id,
          lockSeconds: BOOKING_LOCK_SECONDS,
        }),
      });
      if (!res.ok) {
        const text = await res.text();
        throw new Error(text || "Unable to reserve time slot");
      }
      const data: BookingLock = await res.json();
      const exp = new Date(data.expiresAt);
      const secondsLeft = Math.max(
        0,
        Math.floor((exp.getTime() - Date.now()) / 1000)
      );
      setLockState({
        status: "LOCKED",
        lockId: data.lockId,
        slotId: data.slotId,
        expiresAt: exp,
        secondsLeft,
      });
    } catch (e: any) {
      setLockState({ status: "ERROR", message: e?.message || "Lock error" });
    }
  }

  async function confirmBooking() {
    if (lockState.status !== "LOCKED") return;
    try {
      setLockState({
        status: "CONFIRMING",
        lockId: lockState.lockId,
        slotId: lockState.slotId,
      });
      const res = await fetch("/api/bookings/confirm-and-pay", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          lockId: lockState.lockId,
          userId,
          platform,
          // here you can pass payment method, Nayax token, etc
        }),
      });
      if (!res.ok) {
        const text = await res.text();
        throw new Error(text || "Booking confirmation failed");
      }
      const data: { bookingId: string } = await res.json();
      setLockState({ status: "IDLE" });
      if (onBookingConfirmed) onBookingConfirmed(data);
    } catch (e: any) {
      setLockState({ status: "ERROR", message: e?.message || "Confirm error" });
    }
  }

  /* 2.3 Render */

  const isLockActive = lockState.status === "LOCKED";

  return (
    <div className="pw-calendar-root">
      {/* iOS style card container */}
      <div className="pw-calendar-card">
        <header className="pw-calendar-header">
          <div>
            <div className="pw-calendar-title">
              {platformLabel(platform)} - Booking
            </div>
            <div className="pw-calendar-subtitle">
              {bookingModeLabel(bookingMode)}
            </div>
          </div>

          <div className="pw-calendar-nav">
            <button
              className="pw-nav-button"
              onClick={() =>
                setCurrentDay(addDays(currentDay, -7))
              }
            >
              {"<"}
            </button>
            <span className="pw-nav-label">
              {currentDay.toLocaleDateString("en-IL", {
                month: "long",
                year: "numeric",
              })}
            </span>
            <button
              className="pw-nav-button"
              onClick={() => setCurrentDay(addDays(currentDay, 7))}
            >
              {">"}
            </button>
          </div>
        </header>

        {/* Day strip */}
        <div className="pw-day-strip">
          {days.map(day => {
            const isToday = isSameDay(day, new Date());
            const isSelected =
              bookingMode === "MULTI_DAY"
                ? dayInRange(day) ||
                  (selectedRange.start &&
                    isSameDay(day, selectedRange.start)) ||
                  (selectedRange.end &&
                    isSameDay(day, selectedRange.end))
                : isSameDay(day, currentDay);
            return (
              <button
                key={day.toISOString()}
                className={[
                  "pw-day-pill",
                  isSelected ? "pw-day-pill-selected" : "",
                  isToday ? "pw-day-pill-today" : "",
                ]
                  .filter(Boolean)
                  .join(" ")}
                onClick={() => handleDayClick(day)}
              >
                <span className="pw-day-label">
                  {formatDateLabel(day)}
                </span>
              </button>
            );
          })}
        </div>

        {/* Availability list */}
        <div className="pw-slot-list">
          {isLoading && (
            <div className="pw-info-text">Loading availability...</div>
          )}
          {!isLoading && availability.length === 0 && (
            <div className="pw-info-text">
              No available times in the selected period.
            </div>
          )}

          {!isLoading &&
            days.map(day => {
              const key = startOfDay(day).toISOString();
              const slots = slotsByDay.get(key) || [];
              if (slots.length === 0) return null;
              return (
                <div className="pw-day-section" key={key}>
                  <div className="pw-day-section-title">
                    {formatDateLabel(day)}
                  </div>
                  <div className="pw-slot-chips">
                    {slots.map(slot => {
                      const lockedForUser =
                        isLockActive &&
                        lockState.slotId === slot.id;
                      const disabled =
                        slot.status !== "AVAILABLE" && !lockedForUser;
                      return (
                        <button
                          key={slot.id}
                          className={[
                            "pw-slot-chip",
                            lockedForUser ? "pw-slot-chip-locked" : "",
                          ]
                            .filter(Boolean)
                            .join(" ")}
                          disabled={disabled}
                          onClick={() => lockSlot(slot)}
                        >
                          <span>{formatTimeRange(slot.start, slot.end)}</span>
                          {lockedForUser && (
                            <span className="pw-slot-chip-timer">
                              {lockState.secondsLeft}s
                            </span>
                          )}
                        </button>
                      );
                    })}
                  </div>
                </div>
              );
            })}
        </div>

        {/* Error and lock bar */}
        {error && (
          <div className="pw-error-banner">
            {error}
          </div>
        )}

        {lockState.status === "LOCKED" && (
          <div className="pw-lock-bar">
            <div>
              Time slot reserved for you for{" "}
              <strong>{lockState.secondsLeft}</strong> seconds.
              Complete payment to confirm.
            </div>
            <button
              className="pw-primary-button"
              onClick={confirmBooking}
            >
              Confirm and Pay
            </button>
          </div>
        )}

        {lockState.status === "CONFIRMING" && (
          <div className="pw-lock-bar">
            Confirming and processing payment...
          </div>
        )}
      </div>
    </div>
  );
};

/* Helper labels for UI */

function platformLabel(p: BookingPlatform): string {
  switch (p) {
    case "K9000_WASH":
      return "K9000 Station";
    case "WALK_MY_PET":
      return "Walk My Pet";
    case "PETTREK_TRANSPORT":
      return "PetTrek Transport";
    case "SITTER_SUITE":
      return "Sitter Suite";
    case "GROOMING":
      return "Grooming";
    case "PET_TRAINER_ACADEMY":
      return "Pet Trainer Academy";
    default:
      return "Pet Wash";
  }
}

function bookingModeLabel(m: BookingMode): string {
  switch (m) {
    case "SINGLE_SLOT":
      return "Choose one time";
    case "MULTI_DAY":
      return "Choose start and end date";
    case "ARRIVAL_WINDOW":
      return "Choose preferred time window";
    default:
      return "Booking";
  }
}

/* =========================================================
   3. BACKEND - EXPRESS ROUTES FOR BOOKING LOCK & CONFIRM
   Compatible with Node 18+, TypeScript
   Integrate this into your existing Express app

// Backend imports - adjust paths as needed
import express, { Request, Response } from "express";
import crypto from "crypto";

// Example in-memory storage - in production use PostgreSQL + Drizzle
const locksInMemory = new Map<string, BookingLock>();

// Example Express router
export const bookingRouter = express.Router();

// Helper to generate IDs
function createId(prefix: string): string {
  return `${prefix}_${crypto.randomBytes(8).toString("hex")}`;
}

// POST /api/bookings/lock
bookingRouter.post(
  "/lock",
  express.json(),
  async (req: Request, res: Response) => {
    try {
      const { platform, userId, slotId, lockSeconds } = req.body as {
        platform: BookingPlatform;
        userId: string;
        slotId: string;
        lockSeconds?: number;
      };

      if (!platform || !userId || !slotId) {
        return res
          .status(400)
          .send("platform, userId and slotId are required");
      }

      const lockId = createId("lock");
      const expires = new Date(
        Date.now() + (lockSeconds || BOOKING_LOCK_SECONDS) * 1000
      );

      const lock: BookingLock = {
        lockId,
        slotId,
        platform,
        userId,
        expiresAt: expires.toISOString(),
      };

      // In production: persist in database
      locksInMemory.set(lockId, lock);

      return res.status(200).json(lock);
    } catch (err) {
      console.error("Lock error", err);
      return res.status(500).send("Internal server error");
    }
  }
);

// POST /api/bookings/confirm-and-pay
bookingRouter.post(
  "/confirm-and-pay",
  express.json(),
  async (req: Request, res: Response) => {
    try {
      const { lockId, userId, platform } = req.body as {
        lockId: string;
        userId: string;
        platform: BookingPlatform;
      };

      if (!lockId || !userId || !platform) {
        return res
          .status(400)
          .send("lockId, userId and platform are required");
      }

      const lock = locksInMemory.get(lockId);
      if (!lock) return res.status(410).send("Reservation not found");

      const now = Date.now();
      const exp = new Date(lock.expiresAt).getTime();
      if (now > exp) {
        locksInMemory.delete(lockId);
        return res.status(410).send("Reservation expired");
      }

      if (lock.userId !== userId) {
        return res
          .status(403)
          .send("Reservation belongs to another user");
      }

      // Here you integrate Nayax or Stripe Connect:
      // 1. Create payment intent
      // 2. Confirm payment
      // 3. Store booking in database
      // For now we simulate a booking record

      const bookingId = createId("booking");

      // Remove lock after booking
      locksInMemory.delete(lockId);

      // In production: write booking to database and attach payment info

      return res.status(200).json({ bookingId });
    } catch (err) {
      console.error("Confirm error", err);
      return res.status(500).send("Internal server error");
    }
  }
);

// POST /api/bookings/release
bookingRouter.post(
  "/release",
  express.json(),
  async (req: Request, res: Response) => {
    try {
      const { lockId, userId } = req.body as {
        lockId: string;
        userId: string;
      };
      if (!lockId || !userId) {
        return res
          .status(400)
          .send("lockId and userId are required");
      }
      const lock = locksInMemory.get(lockId);
      if (!lock) return res.status(200).json({ released: false });

      if (lock.userId !== userId) {
        return res.status(403).send("Reservation not owned by user");
      }

      locksInMemory.delete(lockId);
      return res.status(200).json({ released: true });
    } catch (err) {
      console.error("Release error", err);
      return res.status(500).send("Internal server error");
    }
  }
);
