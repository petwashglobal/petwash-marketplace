type Role =
  | "PUBLIC"
  | "USER"
  | "PROVIDER"
  | "EMPLOYEE"
  | "ADMIN"
  | "SUPER_ADMIN"

type ResourceType = "HUMAN" | "MACHINE"

type PricingType = "HOURLY" | "SESSION" | "MINUTES" | "DISTANCE"

type BookingStatus =
  | "DRAFT"
  | "QUOTED"
  | "CONFIRMED"
  | "IN_PROGRESS"
  | "COMPLETED"
  | "REFUNDED"

type TransactionType =
  | "PAID"
  | "COMPLIMENTARY"
  | "PROMO"
  | "ADMIN"

type Currency = "ILS" | "USD" | "EUR"

/* ================================================================================
 SECTION 4 – UTILITIES
================================================================================ */

function uuid(): string {
  return crypto.randomUUID()
}

function now(): Date {
  return new Date()
}

function calculateVAT(amount: number, rate = 0.17): number {
  return Math.round(amount * rate)
}

/* ================================================================================
 SECTION 5 – BOOKING ENGINE (CORE LOGIC)
================================================================================ */

class BookingEngine {
  async createDraft(input: {
    serviceId: string
    resourceId: string
    userId: string
    startTime: Date
    endTime: Date
  }): Promise<Booking> {
    return {
      id: uuid(),
      platform: "PETWASH",
      serviceId: input.serviceId,
      resourceId: input.resourceId,
      userId: input.userId,
      startTime: input.startTime,
      endTime: input.endTime,
      status: "DRAFT",
      priceSnapshot: {
        gross: 0,
        vat: 0,
        net: 0,
        currency: "ILS",
        breakdown: {}
      },
      createdAt: now()
    }
  }

  async quote(booking: Booking, price: number): Promise<Booking> {
    const vat = calculateVAT(price)
    booking.priceSnapshot = {
      gross: price,
      vat,
      net: price - vat,
      currency: "ILS",
      breakdown: {
        base: price
      }
    }
    booking.status = "QUOTED"
    return booking
  }

  async confirm(
    booking: Booking,
    transaction: Transaction
  ): Promise<{ booking: Booking; transaction: Transaction }> {
    booking.status = "CONFIRMED"
    return { booking, transaction }
  }

  async start(booking: Booking): Promise<Booking> {
    booking.status = "IN_PROGRESS"
    return booking
  }

  async complete(booking: Booking): Promise<Booking> {
    booking.status = "COMPLETED"
    return booking
  }
}

/* ================================================================================
 SECTION 6 – TRANSACTION STAMP (FINANCIAL TRUTH)
================================================================================ */

function createTransaction(
  bookingId: string,
  amount: number,
  provider: string,
  providerRef?: string
): Transaction {
  const vat = calculateVAT(amount)
  return {
    id: uuid(),
    bookingId,
    type: amount === 0 ? "COMPLIMENTARY" : "PAID",
    gross: amount,
    vat,
    net: amount - vat,
    currency: "ILS",
    provider,
    providerRef,
    stampedAt: now()
  }
}

/* ================================================================================
 SECTION 7 – EVENT LOGGING (ADMIN + HR)
================================================================================ */

function logEvent(
  actorId: string,
  action: string,
  bookingId?: string,
  meta?: Record<string, any>
): EventLog {
  return {
    id: uuid(),
    actorId,
    bookingId,
    action,
    meta,
    createdAt: now()
  }
}

/* ================================================================================
 SECTION 8 – ADMIN OPERATIONS
================================================================================ */

async function adminGrantFreeWash(params: {
  adminId: string
  machineId: string
  bay: string
  startTime: Date
  minutes: number
}) {
  const engine = new BookingEngine()

  const booking = await engine.createDraft({
    serviceId: "PET_WASH_PUBLIC",
    resourceId: `${params.machineId}_${params.bay}`,
    userId: "SYSTEM",
    startTime: params.startTime,
    endTime: new Date(params.startTime.getTime() + params.minutes * 60000)
  })

  await engine.quote(booking, 0)

  const transaction = createTransaction(booking.id, 0, "ADMIN")

  await engine.confirm(booking, transaction)

  logEvent(
    params.adminId,
    "ADMIN_GRANTED_FREE_WASH",
    booking.id,
    { bay: params.bay, minutes: params.minutes }
  )

  return { booking, transaction }
}

/* ================================================================================
 SECTION 9 – FRONTEND FLOW (SIMPLIFIED)
================================================================================ */

async function frontendBookingFlow(input: {
  serviceId: string
  resourceId: string
  userId: string
  startTime: Date
  endTime: Date
  price: number
}) {
  const engine = new BookingEngine()

  let booking = await engine.createDraft(input)

  booking = await engine.quote(booking, input.price)

  const transaction = createTransaction(
    booking.id,
    input.price,
    "NAYAX",
    "provider-reference"
  )

  await engine.confirm(booking, transaction)

  return {
    bookingId: booking.id,
    status: booking.status,
    transactionId: transaction.id
  }
}
