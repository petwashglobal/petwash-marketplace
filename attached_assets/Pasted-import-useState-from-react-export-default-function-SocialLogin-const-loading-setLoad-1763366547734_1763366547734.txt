import { useState } from 'react';

export default function SocialLogin() {
  const [loading, setLoading] = useState(false);

  // --- CONFIGURATION (Replace with your actual Client IDs) ---
  const FB_APP_ID = 'YOUR_FB_APP_ID'; // from developers.facebook.com
  const TIKTOK_CLIENT_KEY = 'YOUR_TIKTOK_CLIENT_KEY'; // from developers.tiktok.com
  const REDIRECT_URI = 'https://your-pet-wash-app.com/api/auth/callback';

  // --- 1. FACEBOOK / META DIRECT LOGIN ---
  const handleFacebookLogin = () => {
    setLoading(true);
    // 2025 Standard: Use the official SDK or direct OAuth URL
    const fbUrl = `https://www.facebook.com/v18.0/dialog/oauth?client_id=${FB_APP_ID}&redirect_uri=${REDIRECT_URI}/facebook&state={st=state123abc,ds=123456789}&scope=public_profile,email`;
    window.location.href = fbUrl;
  };

  // --- 2. TIKTOK DIRECT LOGIN (v2 API) ---
  const handleTikTokLogin = () => {
    setLoading(true);
    // TikTok requires a random CSRF state for security
    const csrfState = Math.random().toString(36).substring(2);
    
    // Scopes: user.info.basic is standard for 2025 identity
    const tiktokUrl = `https://www.tiktok.com/v2/auth/authorize/?client_key=${TIKTOK_CLIENT_KEY}&scope=user.info.basic&response_type=code&redirect_uri=${REDIRECT_URI}/tiktok&state=${csrfState}`;
    
    window.location.href = tiktokUrl;
  };

  return (
    <div className="flex flex-col gap-4 w-full max-w-sm mx-auto p-6 bg-white rounded-xl shadow-sm">
      <h3 className="text-center text-gray-700 font-semibold mb-2">
        Join the Pet Wash Club
      </h3>

      {/* FACEBOOK / INSTAGRAM (META) BUTTON */}
      <button
        onClick={handleFacebookLogin}
        disabled={loading}
        className="flex items-center justify-center gap-3 w-full bg-[#1877F2] text-white py-3 px-4 rounded-lg font-medium hover:bg-[#166fe5] transition-all"
      >
        <svg className="w-6 h-6" fill="currentColor" viewBox="0 0 24 24">
          <path d="M24 12.073c0-6.627-5.373-12-12-12s-12 5.373-12 12c0 5.99 4.388 10.954 10.125 11.854v-8.385H7.078v-3.47h3.047V9.43c0-3.007 1.792-4.669 4.533-4.669 1.312 0 2.686.235 2.686.235v2.953H15.83c-1.491 0-1.956.925-1.956 1.874v2.25h3.328l-.532 3.47h-2.796v8.385C19.612 23.027 24 18.062 24 12.073z"/>
        </svg>
        Continue with Facebook / Instagram
      </button>

      {/* TIKTOK BUTTON */}
      <button
        onClick={handleTikTokLogin}
        disabled={loading}
        className="flex items-center justify-center gap-3 w-full bg-black text-white py-3 px-4 rounded-lg font-medium hover:bg-gray-900 transition-all"
      >
        <svg className="w-6 h-6" fill="currentColor" viewBox="0 0 24 24">
          <path d="M12.525.02c1.31-.02 2.61-.01 3.91-.02.08 1.53.63 3.09 1.75 4.17 1.12 1.11 2.7 1.62 4.24 1.79v4.03c-1.44-.05-2.89-.35-4.2-.97-.57-.26-1.1-.65-1.62-1.12-1.64 2.7-1.58 6.08.25 8.67 1.85 2.63 5.46 3.18 7.85 1.23.49-.4.93-.86 1.3-1.37.58-1.06.86-2.31.76-3.53l-4.18-.01c.01 1.45-.67 2.96-1.95 3.66-1.34.72-3.09.4-4.06-.79-.96-1.17-1.06-2.88-.24-4.17.8-1.27 2.38-1.92 3.83-1.59l.01-4.16c-3.98-.08-7.5 2.66-8.28 6.59-.84 4.26 1.71 8.65 5.74 9.86 4.78 1.43 10.14-1.05 11.69-5.61.81-2.37.67-5.01-.35-7.29-.81-1.79-2.13-3.32-3.84-4.27z"/>
        </svg>
        Continue with TikTok
      </button>
    </div>
  );
}
