set -euo pipefail

echo "== 0) Go to repo =="
# Change this path if your repo is elsewhere
cd ~
if [ ! -d "petwash-marketplace" ]; then
  echo "Repo folder not found in ~. Listing directories:"
  ls -la
  echo "If your repo has a different folder name, cd into it and re-run from step 1."
  exit 1
fi
cd petwash-marketplace

echo
echo "== 1) Git sanity =="
git remote -v
git status
git branch --show-current

echo
echo "== 2) Pull latest and show last commits =="
git pull --rebase
git log -5 --oneline

echo
echo "== 3) Confirm the health changes exist in working tree =="
# These are just quick greps, adjust paths if your server entry differs
grep -R --line-number "api/health/strict" . || true
grep -R --line-number "serverReady" . || true
grep -R --line-number "status\":\"OK\"" . || true

echo
echo "== 4) If changes are present, commit and push =="
# If nothing to commit, this will stop and show you why
git add -A
git commit -m "Fix health endpoints and non-blocking startup" || {
  echo "No commit created. Either nothing changed or commit failed."
  echo "Run: git status"
  exit 1
}
git push

echo
echo "== 5) Show recent Cloud Run revisions and image digests (needs gcloud auth set up) =="
# Set these if your names differ
SERVICE="petwash-api"
REGION="me-west1"
PROJECT="$(gcloud config get-value project)"

echo "Project: $PROJECT"
gcloud run revisions list --service "$SERVICE" --region "$REGION" --format="table(metadata.name, status.conditions[0].status, metadata.creationTimestamp)"

echo
echo "== 6) Get the latest revision image URL (must change after deploy) =="
LATEST_REV="$(gcloud run revisions list --service "$SERVICE" --region "$REGION" --sort-by="~metadata.creationTimestamp" --limit=1 --format="value(metadata.name)")"
echo "Latest revision: $LATEST_REV"
gcloud run revisions describe "$LATEST_REV" --region "$REGION" --format="value(spec.containers[0].image)"

echo
echo "== 7) Test health endpoints on the service URL =="
SERVICE_URL="$(gcloud run services describe "$SERVICE" --region "$REGION" --format="value(status.url)")"
echo "Service URL: $SERVICE_URL"

echo
echo "GET /health"
curl -sS -i "$SERVICE_URL/health" | sed -n '1,20p'
echo
echo "GET /api/health"
curl -sS -i "$SERVICE_URL/api/health" | sed -n '1,40p'
echo
echo "GET /api/health/strict"
curl -sS -i "$SERVICE_URL/api/health/strict" | sed -n '1,40p'
