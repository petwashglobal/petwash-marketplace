/*

- export only what each destination truly needs

const PII_FIELD_POLICY = {
  FULL_NAME: "PII",
  EMAIL: "PII",
  PHONE: "PII",
  MESSAGE: "PII",
  BOOKING_ID: "SAFE",
  MACHINE_ID: "SAFE",
  PROVIDER_ID: "SAFE",
  DOCUMENT_ID: "SAFE",
  AMOUNT: "SAFE_FINANCIAL",
  VAT: "SAFE_FINANCIAL",
  STATUS: "SAFE",
};

export function redactEmail(email?: string | null): string | null {
  if (!email) return null;
  const [name, domain] = email.split("@");
  if (!domain) return "***";
  return `${name.slice(0, 2)}***@${domain}`;
}

export function redactPhone(phone?: string | null): string | null {
  if (!phone) return null;
  const last4 = phone.slice(-4);
  return `***${last4}`;
}

export function minimizeExportPayload(
  channel: "SHEETS" | "DRIVE" | "FORMS",
  payload: Record<string, any>
): Record<string, any> {
  // Rules:
  // SHEETS: never export full email/phone unless specifically whitelisted for operational need
  // DRIVE: full PDF allowed only for legal docs
  // FORMS: intake only, not financial export

  const out: Record<string, any> = { ...payload };

  if (channel === "SHEETS") {
    if ("email" in out) out.email = redactEmail(out.email);
    if ("phone" in out) out.phone = redactPhone(out.phone);
    if ("fullName" in out) out.fullName = out.fullName ? `${String(out.fullName).slice(0, 1)}***` : null;
    if ("message" in out) out.message = "[REDACTED_FOR_SHEETS]";
  }

  return out;
}
