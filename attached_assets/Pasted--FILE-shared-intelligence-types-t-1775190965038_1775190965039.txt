import { db } from "../db";
import { sql } from "drizzle-orm";

export class CustomerIntelligenceService {
  async recomputeForUser(userId: string) {
    const stats = await db.execute(sql`
      SELECT
        COUNT(*)::int AS booking_history_count,
        COUNT(*) FILTER (WHERE status = 'cancelled')::int AS cancelled_count,
        COUNT(*) FILTER (WHERE status = 'completed')::int AS completed_count,
        COUNT(*) FILTER (WHERE status = 'no_show')::int AS no_show_count,
        COUNT(DISTINCT provider_id)::int AS repeat_usage_count,
        MAX(updated_at) AS last_activity_at
      FROM booking_requests
      WHERE owner_id = ${userId}
    `);

    const row = stats.rows[0] as any;

    const bookingHistoryCount = Number(row?.booking_history_count || 0);
    const cancelledCount = Number(row?.cancelled_count || 0);
    const completedCount = Number(row?.completed_count || 0);
    const noShowCount = Number(row?.no_show_count || 0);
    const repeatUsageCount = Number(row?.repeat_usage_count || 0);

    const cancellationRate =
      bookingHistoryCount > 0 ? cancelledCount / bookingHistoryCount : 0;

    let recentActivityDaysAgo: number | null = null;
    if (row?.last_activity_at) {
      const last = new Date(row.last_activity_at).getTime();
      const now = Date.now();
      recentActivityDaysAgo = Math.floor((now - last) / (1000 * 60 * 60 * 24));
    }

    let trustScore = 50;
    trustScore += Math.min(completedCount * 1.5, 25);
    trustScore -= Math.min(cancellationRate * 50, 25);
    trustScore -= Math.min(noShowCount * 10, 20);
    trustScore = Math.max(0, Math.min(100, Number(trustScore.toFixed(2))));

    let behaviorScore = 40;
    behaviorScore += Math.min(bookingHistoryCount * 2, 30);
    behaviorScore += Math.min(repeatUsageCount * 3, 20);
    if (recentActivityDaysAgo !== null && recentActivityDaysAgo <= 7) behaviorScore += 10;
    if (recentActivityDaysAgo !== null && recentActivityDaysAgo > 30) behaviorScore -= 10;
    behaviorScore = Math.max(0, Math.min(100, Number(behaviorScore.toFixed(2))));

    const riskLevel = Number((100 - trustScore).toFixed(2));

    await db.execute(sql`
      INSERT INTO user_intelligence_profiles (
        user_id,
        user_type,
        trust_score,
        behavior_score,
        risk_level,
        booking_history_count,
        cancellation_rate,
        no_show_count,
        repeat_usage_count,
        recent_activity_days_ago,
        preferences,
        last_computed_at,
        created_at,
        updated_at
      )
      VALUES (
        ${userId},
        'customer',
        ${trustScore},
        ${behaviorScore},
        ${riskLevel},
        ${bookingHistoryCount},
        ${cancellationRate},
        ${noShowCount},
        ${repeatUsageCount},
        ${recentActivityDaysAgo},
        ${JSON.stringify({})}::jsonb,
        NOW(),
        NOW(),
        NOW()
      )
      ON CONFLICT (user_id)
      DO UPDATE SET
        trust_score = EXCLUDED.trust_score,
        behavior_score = EXCLUDED.behavior_score,
        risk_level = EXCLUDED.risk_level,
        booking_history_count = EXCLUDED.booking_history_count,
        cancellation_rate = EXCLUDED.cancellation_rate,
        no_show_count = EXCLUDED.no_show_count,
        repeat_usage_count = EXCLUDED.repeat_usage_count,
        recent_activity_days_ago = EXCLUDED.recent_activity_days_ago,
        preferences = EXCLUDED.preferences,
        last_computed_at = NOW(),
        updated_at = NOW()
    `);

    const result = await db.execute(sql`
      SELECT
        user_id AS "userId",
        user_type AS "userType",
        trust_score AS "trustScore",
        behavior_score AS "behaviorScore",
        risk_level AS "riskLevel",
        booking_history_count AS "bookingHistoryCount",
        cancellation_rate AS "cancellationRate",
        no_show_count AS "noShowCount",
        repeat_usage_count AS "repeatUsageCount",
        recent_activity_days_ago AS "recentActivityDaysAgo",
        preferences,
        last_computed_at AS "lastComputedAt"
      FROM user_intelligence_profiles
      WHERE user_id = ${userId}
      LIMIT 1
    `);

    return result.rows[0];
  }

  async getForUser(userId: string) {
    const result = await db.execute(sql`
      SELECT
        user_id AS "userId",
        user_type AS "userType",
        trust_score AS "trustScore",
        behavior_score AS "behaviorScore",
        risk_level AS "riskLevel",
        booking_history_count AS "bookingHistoryCount",
        cancellation_rate AS "cancellationRate",
        no_show_count AS "noShowCount",
        repeat_usage_count AS "repeatUsageCount",
        recent_activity_days_ago AS "recentActivityDaysAgo",
        preferences,
        last_computed_at AS "lastComputedAt"
      FROM user_intelligence_profiles
      WHERE user_id = ${userId}
      LIMIT 1
    `);

    return result.rows[0] || null;
  }
}

import { db } from "../db";
import { sql } from "drizzle-orm";
import type { JourneyState } from "../../shared/intelligence-types";

const JOURNEY_ORDER: JourneyState[] = [
  "visitor",
  "browsing",
  "authenticated",
  "ready_to_book",
  "booked",
];

export class JourneyStateService {
  async getState(userId: string): Promise<JourneyState> {
    const result = await db.execute(sql`
      SELECT COALESCE(journey_state, 'visitor') AS state
      FROM users
      WHERE id = ${userId}
      LIMIT 1
    `);

    return (result.rows[0]?.state as JourneyState) || "visitor";
  }

  async setState(userId: string, targetState: JourneyState) {
    const currentState = await this.getState(userId);
    const currentIdx = JOURNEY_ORDER.indexOf(currentState);
    const targetIdx = JOURNEY_ORDER.indexOf(targetState);

    if (targetIdx <= currentIdx) {
      return currentState;
    }

    await db.execute(sql`
      UPDATE users
      SET journey_state = ${targetState}
      WHERE id = ${userId}
    `);

    return targetState;
  }
}

export const journeyStateService = new JourneyStateService();

  try {
    const userId = req.user.uid;
    let profile = await customerIntelligenceService.getForUser(userId);

      profile = await customerIntelligenceService.recomputeForUser(userId);
    }
    res.json({ profile });

import { Router } from "express";
import { requireAuth } from "../middleware/requireAuth";
import { db } from "../db";
import { sql } from "drizzle-orm";
import { eventBus } from "../services/eventBus";

export const providerArrivingRouter = Router();

providerArrivingRouter.post(
  "/api/booking-requests/:requestId/arriving",
  requireAuth,
  async (req: any, res) => {
    try {
      const userId = req.user.uid;
      const { requestId } = req.params;
      const { eta } = req.body || {};

      const bookingResult = await db.execute(sql`
        SELECT id, owner_id, provider_id, service_type, status
        FROM booking_requests
        WHERE id = ${requestId}
        LIMIT 1
      `);

      const booking = bookingResult.rows[0] as any;

      if (!booking) {
        return res.status(404).json({ error: "Booking request not found" });
      }

      if (booking.provider_id !== userId) {
        return res.status(403).json({ error: "Only the assigned provider can signal arrival" });
      }

      if (!["accepted", "confirmed", "in_progress"].includes(booking.status)) {
        return res.status(400).json({ error: "Booking is not in a valid state for arrival update" });
      }

      await eventBus.publish({
        eventType: "provider.arriving",
        timestamp: new Date().toISOString(),
        platform: "marketplace",
        userId,
        data: {
          requestId,
          ownerId: booking.owner_id,
          providerId: booking.provider_id,
          serviceType: booking.service_type,
          eta: eta || null,
        },
      });

      return res.json({
        success: true,
        message: "Arrival signal sent to customer",
      });
    } catch (error) {
      console.error("[provider-arriving] failed", error);
      return res.status(500).json({ error: "Failed to signal arrival" });
    }
  }
);

import cron from "node-cron";
import { db } from "../db";
import { sql } from "drizzle-orm";
import { sendEmail } from "./sendEmail";

async function alreadyQueued(idempotencyKey: string) {
  const result = await db.execute(sql`
    SELECT id
    FROM notification_logs
    WHERE idempotency_key = ${idempotencyKey}
    LIMIT 1
  `);
  return !!result.rows[0];
}

async function markQueued(idempotencyKey: string, userId: string) {
  await db.execute(sql`
    INSERT INTO notification_logs (
      user_id,
      type,
      idempotency_key,
      created_at
    )
    VALUES (
      ${userId},
      'recovery_winback_30d',
      ${idempotencyKey},
      NOW()
    )
  `);
}

export async function runInactiveUserReengagement() {
  const users = await db.execute(sql`
    SELECT id, email, first_name
    FROM users
    WHERE activation_status = 'active'
      AND last_login_at < NOW() - INTERVAL '30 days'
      AND last_login_at > NOW() - INTERVAL '90 days'
  `);

  for (const user of users.rows as any[]) {
    const idempotencyKey = `recovery_winback_30d_${user.id}`;
    const exists = await alreadyQueued(idempotencyKey);
    if (exists) continue;

    await sendEmail({
      to: user.email,
      subject: "🐾 חסרים לנו!",
      html: `
        <p>שלום ${user.first_name || ""},</p>
        <p>לא ראינו אותך כבר 30 יום - נשמח לראות אותך שוב.</p>
        <p>חזור עכשיו וקבל עדיפות בהזמנה הבאה שלך.</p>
      `,
    });

    await markQueued(idempotencyKey, user.id);
  }
}

export function registerRecoveryAutomationCron() {
  cron.schedule("0 10 * * *", async () => {
    try {
      await runInactiveUserReengagement();
    } catch (error) {
      console.error("[RecoveryAutomation] winback cron failed", error);
    }
  });

  console.info(
    "[RecoveryAutomation] Cron jobs registered (signup: 60min, booking: 30min, winback: daily 10:00)"
  );
}

export async function apiGet<T>(url: string): Promise<T> {
  const res = await fetch(url, {
    credentials: "include",
  });
  if (!res.ok) {
    throw new Error(`GET ${url} failed`);
  }
  return res.json();
}

export async function apiPost<T>(url: string, body?: any): Promise<T> {
  const res = await fetch(url, {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
    },
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) {
    const text = await res.text();
    throw new Error(`POST ${url} failed: ${text}`);
  }
  return res.json();
}

import { useEffect, useState } from "react";

export function useUserIntelligence() {
  const [profile, setProfile] = useState<UserIntelligenceProfile | null>(null);
  const [loading, setLoading] = useState(true);

  async function load() {
    setLoading(true);
    try {
      const data = await apiGet<{ profile: UserIntelligenceProfile }>("/api/user/intelligence");
      setProfile(data.profile);
    } finally {
      setLoading(false);
    }
  }

  async function recompute() {
    const data = await apiPost<{ profile: UserIntelligenceProfile }>(
      "/api/user/intelligence/recompute"
    );
    setProfile(data.profile);
  }

  useEffect(() => {
    load().catch(console.error);
  }, []);

  return {
    profile,
    loading,
    recompute,
  };
}

import { useEffect, useState } from "react";
import { apiGet, apiPost } from "../lib/api";
import type { JourneyState } from "../../../shared/intelligence-types";

export function useJourneyState() {
  const [state, setState] = useState<JourneyState>("visitor");
  const [loading, setLoading] = useState(true);

  async function load() {
    setLoading(true);
    try {
      const data = await apiGet<{ state: JourneyState }>("/api/user/journey-state");
      setState(data.state);
    } finally {
      setLoading(false);
    }
  }

  async function advance(nextState: JourneyState) {
    const data = await apiPost<{ state: JourneyState }>("/api/user/journey-state", {
      state: nextState,
    });
    setState(data.state);
    return data.state;
  }

  useEffect(() => {
    load().catch(console.error);
  }, []);

  return {
    state,
    loading,
    advance,
  };
}

// ============================================================

import { useEffect } from "react";
import type { RealtimeMarketplaceEvent } from "../../../shared/intelligence-types";

interface Options {
  onMatchingStarted?: (event: RealtimeMarketplaceEvent) => void;
  onProviderAccepted?: (event: RealtimeMarketplaceEvent) => void;
  onProviderArriving?: (event: RealtimeMarketplaceEvent) => void;
}

export function useMarketplaceEvents(options: Options) {
  useEffect(() => {
    const ws = new WebSocket(
      `${window.location.protocol === "https:" ? "wss" : "ws"}://${window.location.host}/ws/match`
    );

    ws.onmessage = (message) => {
      try {
        const event: RealtimeMarketplaceEvent = JSON.parse(message.data);

        if (event.eventType === "matching.started" && options.onMatchingStarted) {
          options.onMatchingStarted(event);
        }

        if (event.eventType === "provider.accepted" && options.onProviderAccepted) {
          options.onProviderAccepted(event);
        }

        if (event.eventType === "provider.arriving" && options.onProviderArriving) {
          options.onProviderArriving(event);
        }
      } catch (error) {
        console.error("[useMarketplaceEvents] parse failed", error);
      }
    };

    ws.onerror = (error) => {
      console.error("[useMarketplaceEvents] ws error", error);
    };

    return () => {
      ws.close();
    };
  }, []);
}

export function ProviderArrivingButton({ requestId }: { requestId: string }) {
  const [eta, setEta] = useState("10 minutes");
  const [loading, setLoading] = useState(false);
  const [done, setDone] = useState(false);

  async function handleArriving() {
    setLoading(true);
    try {
      await apiPost(`/api/booking-requests/${requestId}/arriving`, { eta });

      setLoading(false);
    }
  }

  return (
    <div className="rounded-xl border p-3 bg-white">
      <div className="text-sm font-medium mb-2">Arrival update</div>
      <input
        value={eta}
        onChange={(e) => setEta(e.target.value)}
        className="mb-2 w-full rounded-lg border px-3 py-2"
        placeholder="ETA"
      />
      <button
        onClick={handleArriving}
        disabled={loading || done}
        className="rounded-lg bg-green-600 px-4 py-2 text-white"
      >
        {done ? "Arrival Sent" : loading ? "Sending..." : "I'm Arriving"}
      </button>
    </div>
  );
}

import type { JourneyState } from "../../../shared/intelligence-types";

export function resolveSmartRoute(state: JourneyState, requestedPath: string) {
  if (requestedPath.startsWith("/book") && state === "visitor") {
    return "/signin";
  }

  if (requestedPath.startsWith("/book") && state === "authenticated") {
    return "/complete-profile";
  }

  return requestedPath;

// ============================================================

// ============================================================

import { journeyStateService } from "./services/journeyStateService";
import { eventBus } from "./services/eventBus";

export async function onBookingCreated(ownerId: string) {
  await journeyStateService.setState(ownerId, "ready_to_book");
}

export async function onProviderAccepted(params: {
  requestId: string;
  ownerId: string;
  providerId: string;
  serviceType: string;
}) {
  await journeyStateService.setState(params.ownerId, "booked");

  await eventBus.publish({
    eventType: "provider.accepted",
    timestamp: new Date().toISOString(),
    platform: "marketplace",
    userId: params.providerId,
    data: {
      requestId: params.requestId,
      ownerId: params.ownerId,
      providerId: params.providerId,
      serviceType: params.serviceType,
      newStatus: "accepted",
    },
  });
}

export async function onMatchingStarted(params: {
  callerId?: string;
  serviceType: string;
  source?: string;
  lat?: number;
  lng?: number;
  totalCandidates: number;
}) {
  await eventBus.publish({
    eventType: "matching.started",
    timestamp: new Date().toISOString(),
    platform: "marketplace",
    userId: params.callerId,
    data: {
      serviceType: params.serviceType,
      location: {
        source: params.source || "unknown",
        lat: params.lat || null,
        lng: params.lng || null,
      },
      totalCandidates: params.totalCandidates,
    },
  });
}
