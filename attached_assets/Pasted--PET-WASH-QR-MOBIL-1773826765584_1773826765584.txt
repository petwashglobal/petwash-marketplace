function nowIso(): string {
  return new Date().toISOString();
}

function safeEqual(a: string, b: string): boolean {
  const aBuf = Buffer.from(a, "utf8");
  const bBuf = Buffer.from(b, "utf8");
  if (aBuf.length !== bBuf.length) return false;
  return crypto.timingSafeEqual(aBuf, bBuf);
}

function signQrPayload(machineId: string, locationId: string, ts: number, nonce: string): string {
  const raw = `${machineId}:${locationId}:${ts}:${nonce}`;
  return crypto.createHmac("sha256", QR_SECRET).update(raw).digest("hex");
}

function verifyQrSignature(payload: QrPayload): boolean {
  const expected = signQrPayload(payload.machineId, payload.locationId, payload.ts, payload.nonce);
  return safeEqual(expected, payload.sig);
}

function createActivationToken(sessionId: string, userId: string, machineId: string): string {
  const raw = `${sessionId}:${userId}:${machineId}:${Date.now()}:${crypto.randomUUID()}`;
  return crypto.createHmac("sha256", APP_SESSION_SECRET).update(raw).digest("hex");
}

function getIp(req: Request): string {
  return (
    (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() ||
    req.socket.remoteAddress ||
    "unknown"
  );
}

function checkRateLimit(ip: string): boolean {
  const now = Date.now();
  const current = ipRateLimitMap.get(ip);

  if (!current || now > current.resetAt) {
    ipRateLimitMap.set(ip, { count: 1, resetAt: now + 60_000 });
    return true;
  }

  if (current.count >= 10) {
    return false;
  }

  current.count += 1;
  ipRateLimitMap.set(ip, current);
  return true;
}

qrActivationRouter.post("/api/qr/activate", async (req: Request, res: Response) => {
  try {
    const ip = getIp(req);

    if (!checkRateLimit(ip)) {
      return res.status(429).json({
        success: false,
        errorCode: "RATE_LIMITED",
        message: "Too many attempts. Please try again shortly.",
      } satisfies ActivationResult);
    }

    const user = getAuthenticatedUser(req);
    if (!user) {
      return res.status(401).json({
        success: false,
        errorCode: "UNAUTHORIZED",
        message: "User authentication required.",
      } satisfies ActivationResult);
    }

    const idempotencyKey = (req.headers["idempotency-key"] as string | undefined)?.trim();
    if (idempotencyKey && idempotencyCache.has(idempotencyKey)) {
      return res.json(idempotencyCache.get(idempotencyKey));
    }

    const payload = req.body;
    if (!assertValidQrPayload(payload)) {
      return res.status(400).json({
        success: false,
        errorCode: "INVALID_QR_PAYLOAD",
        message: "QR payload is invalid.",
      } satisfies ActivationResult);
    }

    const currentTs = Math.floor(Date.now() / 1000);
    if (currentTs - payload.ts > QR_MAX_AGE_SECONDS) {
      return res.status(400).json({
        success: false,
        errorCode: "QR_EXPIRED",
        message: "QR code expired. Please scan again.",
      } satisfies ActivationResult);
    }

    if (!verifyQrSignature(payload)) {
      return res.status(403).json({
        success: false,
        errorCode: "INVALID_SIGNATURE",
        message: "QR signature verification failed.",
      } satisfies ActivationResult);
    }

    if (usedQrNonces.has(payload.nonce)) {
      return res.status(409).json({
        success: false,
        errorCode: "QR_ALREADY_USED",
        message: "This QR code was already used.",
      } satisfies ActivationResult);
    }

    const machine = machines.get(payload.machineId);
    if (!machine || machine.locationId !== payload.locationId) {
      return res.status(404).json({
        success: false,
        errorCode: "MACHINE_NOT_FOUND",
        message: "Machine not found.",
      } satisfies ActivationResult);
    }

    if (!machine.isActive || !machine.isOnline) {
      return res.status(409).json({
        success: false,
        errorCode: "MACHINE_OFFLINE",
        message: "Machine is currently unavailable.",
      } satisfies ActivationResult);
    }

    if (machine.isBusy) {
      return res.status(409).json({
        success: false,
        errorCode: "MACHINE_BUSY",
        message: "Machine is currently in use.",
      } satisfies ActivationResult);
    }

    const priceCents = calculateSessionPriceCents(machine, user);

    const sessionId = crypto.randomUUID();
    const activationToken = createActivationToken(sessionId, user.id, machine.id);

    const session: ActivationSession = {
      id: sessionId,
      userId: user.id,
      machineId: machine.id,
      locationId: machine.locationId,
      status: "created",
      activationToken,
      priceCents,
      currency: DEFAULT_CURRENCY,
      qrNonce: payload.nonce,
      createdAt: nowIso(),
    };

    activationSessions.set(sessionId, session);

    const nayaxAuthorization = await createNayaxAuthorization({
      machine,
      user,
      session,
    });

    if (!nayaxAuthorization.success || !nayaxAuthorization.nayaxSessionId) {
      session.status = "failed";
      session.failureReason = nayaxAuthorization.message || "Nayax authorization failed";
      activationSessions.set(sessionId, session);

      return res.status(502).json({
        success: false,
        errorCode: "PAYMENT_AUTH_FAILED",
        message: "Unable to authorize payment session.",
      } satisfies ActivationResult);
    }

    session.status = "authorized";
    session.nayaxSessionId = nayaxAuthorization.nayaxSessionId;
    activationSessions.set(sessionId, session);

    usedQrNonces.add(payload.nonce);

    const result: ActivationResult = {
      success: true,
      sessionId,
      activationToken,
      nayaxSessionId: nayaxAuthorization.nayaxSessionId,
      machineId: machine.id,
      startWindowSeconds: ACTIVATION_TOKEN_TTL_SECONDS,
      message: "Session authorized. Ready to start.",
    };

    if (idempotencyKey) {
      idempotencyCache.set(idempotencyKey, result);
    }

    return res.json(result);
  } catch (error: any) {
    console.error("[QR Activation] Failed:", error?.message || error);
    return res.status(500).json({
      success: false,
      errorCode: "SERVER_ERROR",
      message: "Activation failed.",
    } satisfies ActivationResult);
  }
});

qrActivationRouter.post("/api/qr/start", async (req: Request, res: Response) => {
  try {
    const user = getAuthenticatedUser(req);
    if (!user) {
      return res.status(401).json({
        success: false,
        errorCode: "UNAUTHORIZED",
        message: "User authentication required.",
      });
    }

    const { sessionId, activationToken } = req.body as {
      sessionId?: string;
      activationToken?: string;
    };

    if (!sessionId || !activationToken) {
      return res.status(400).json({
        success: false,
        errorCode: "INVALID_START_REQUEST",
        message: "Missing session data.",
      });
    }

    const session = activationSessions.get(sessionId);
    if (!session) {
      return res.status(404).json({
        success: false,
        errorCode: "SESSION_NOT_FOUND",
        message: "Session not found.",
      });
    }

    if (session.userId !== user.id) {
      return res.status(403).json({
        success: false,
        errorCode: "SESSION_USER_MISMATCH",
        message: "This session does not belong to the current user.",
      });
    }

    if (session.activationToken !== activationToken) {
      return res.status(403).json({
        success: false,
        errorCode: "INVALID_ACTIVATION_TOKEN",
        message: "Activation token invalid.",
      });
    }

    if (session.status !== "authorized") {
      return res.status(409).json({
        success: false,
        errorCode: "SESSION_NOT_AUTHORIZED",
        message: "Session is not ready to start.",
      });
    }

    const createdAtMs = new Date(session.createdAt).getTime();
    if (Date.now() - createdAtMs > ACTIVATION_TOKEN_TTL_SECONDS * 1000) {
      session.status = "failed";
      session.failureReason = "Activation token expired";
      activationSessions.set(session.id, session);

      return res.status(410).json({
        success: false,
        errorCode: "ACTIVATION_TOKEN_EXPIRED",
        message: "Activation window expired. Please scan again.",
      });
    }

    const machine = machines.get(session.machineId);
    if (!machine) {
      return res.status(404).json({
        success: false,
        errorCode: "MACHINE_NOT_FOUND",
        message: "Machine not found.",
      });
    }

    if (machine.isBusy) {
      return res.status(409).json({
        success: false,
        errorCode: "MACHINE_BUSY",
        message: "Machine is already in use.",
      });
    }

    const startResult = await startMachineProgram({ machine, session });

    if (!startResult.success) {
      session.status = "failed";
      session.failureReason = startResult.message || "Machine start failed";
      activationSessions.set(session.id, session);

      return res.status(502).json({
        success: false,
        errorCode: "MACHINE_START_FAILED",
        message: "Could not start machine.",
      });
    }

    machine.isBusy = true;
    machines.set(machine.id, machine);

    session.status = "running";
    session.startedAt = nowIso();
    activationSessions.set(session.id, session);

    return res.json({
      success: true,
      message: "Machine started successfully.",
      sessionId: session.id,
      machineId: machine.id,
      startedAt: session.startedAt,
      estimatedSeconds: machine.defaultProgramSeconds,
    });
  } catch (error: any) {
    console.error("[QR Start] Failed:", error?.message || error);
    return res.status(500).json({
      success: false,
      errorCode: "SERVER_ERROR",
      message: "Unable to start machine.",
    });
  }
});

import { useCallback, useMemo, useState } from "react";

type HookState = {
  activationState: ActivationState;
  sessionId: string | null;
  activationToken: string | null;
  machineId: string | null;
  nayaxSessionId: string | null;
  error: string | null;
};

export function useQrMachineActivation() {
  const [state, setState] = useState<HookState>({
    activationState: "IDLE",
    sessionId: null,
    activationToken: null,
    machineId: null,
    nayaxSessionId: null,
    error: null,
  });

  const reset = useCallback(() => {
    setState({
      activationState: "IDLE",
      sessionId: null,
      activationToken: null,
      machineId: null,
      nayaxSessionId: null,
      error: null,
    });
  }, []);

  const activate = useCallback(async (qrPayload: QrPayload) => {
    try {
      setState((prev) => ({ ...prev, activationState: "VERIFYING", error: null }));

      const result = await activateQrSession(qrPayload);

      if (!result.success || !result.sessionId || !result.activationToken) {
        throw new Error(result.message || "Activation failed");
      }

      setState((prev) => ({
        ...prev,
        activationState: "READY",
        sessionId: result.sessionId || null,
        activationToken: result.activationToken || null,
        machineId: result.machineId || null,
        nayaxSessionId: result.nayaxSessionId || null,
        error: null,
      }));
    } catch (error: any) {
      setState((prev) => ({
        ...prev,
        activationState: "FAILED",
        error: error?.message || "Activation failed",
      }));
    }
  }, []);

  const start = useCallback(async () => {
    try {
      if (!state.sessionId || !state.activationToken) {
        throw new Error("Missing session or activation token");
      }

      setState((prev) => ({ ...prev, activationState: "ACTIVATING", error: null }));

      await startQrSession({
        sessionId: state.sessionId,
        activationToken: state.activationToken,
      });

      setState((prev) => ({
        ...prev,
        activationState: "RUNNING",
        error: null,
      }));
    } catch (error: any) {
      setState((prev) => ({
        ...prev,
        activationState: "FAILED",
        error: error?.message || "Could not start session",
      }));
    }
  }, [state.sessionId, state.activationToken]);

  const complete = useCallback(async () => {
    try {
      if (!state.sessionId) {
        throw new Error("Missing session");
      }

      await completeQrSession({ sessionId: state.sessionId });

      setState((prev) => ({
        ...prev,
        activationState: "COMPLETED",
        error: null,
      }));
    } catch (error: any) {
      setState((prev) => ({
        ...prev,
        activationState: "FAILED",
        error: error?.message || "Could not complete session",
      }));
    }
  }, [state.sessionId]);

  const canStart = useMemo(() => state.activationState === "READY", [state.activationState]);
  const isRunning = useMemo(() => state.activationState === "RUNNING", [state.activationState]);

  return {
    ...state,
    activate,
    start,
    complete,
    reset,
    canStart,
    isRunning,
  };
}

  const parsed = JSON.parse(raw);

    typeof parsed.machineId !== "string" ||
    typeof parsed.locationId !== "string" ||
    typeof parsed.ts !== "number" ||
    typeof parsed.nonce !== "string" ||
    typeof parsed.sig !== "string"

    throw new Error("QR code format invalid");
  }

  return parsed as QrPayload;
}

export function generateMachineQrPayload(machineId: string, locationId: string): QrPayload {
  const ts = Math.floor(Date.now() / 1000);
  const nonce = crypto.randomUUID();
  const sig = signQrPayload(machineId, locationId, ts, nonce);

  return {
    machineId,
    locationId,
    ts,
    nonce,
    sig,
  };
}
