//// server.js
//// npm i express helmet cors rate-limiter-flexible pg zod nodemailer
//// node server.js

const express = require("express");
const helmet = require("helmet");
const cors = require("cors");
const { Pool } = require("pg");
const { z } = require("zod");
const { RateLimiterMemory } = require("rate-limiter-flexible");

const app = express();

// ---------- ENV ----------
const PORT = Number(process.env.PORT || 5000);
const DATABASE_URL = process.env.DATABASE_URL;
if (!DATABASE_URL) {
  console.error("Missing DATABASE_URL env var");
  process.exit(1);
}
const ALLOWED_ORIGINS = (process.env.ALLOWED_ORIGINS || "").split(",").map(s => s.trim()).filter(Boolean);

// ---------- DB ----------
const pool = new Pool({ connectionString: DATABASE_URL });

// ---------- SECURITY ----------
app.use(helmet({
  contentSecurityPolicy: false, // enable CSP after you finalize script/style hosting
}));
app.use(express.json({ limit: "1mb" }));

app.use(cors({
  origin: function(origin, cb) {
    if (!origin) return cb(null, true); // allow server-to-server, curl
    if (ALLOWED_ORIGINS.length === 0) return cb(null, true); // if not set, allow all (set in prod)
    if (ALLOWED_ORIGINS.includes(origin)) return cb(null, true);
    return cb(new Error("CORS blocked"), false);
  },
  credentials: true
}));

// ---------- RATE LIMITING (basic) ----------
const feedbackLimiter = new RateLimiterMemory({
  points: 10, // 10 requests
  duration: 60, // per 60 seconds
});

async function rateLimit(limiter, req, res, next) {
  try {
    await limiter.consume(req.ip || "unknown");
    next();
  } catch {
    res.status(429).json({ error: "RATE_LIMITED" });
  }
}

// ---------- DB MIGRATION (run once on boot) ----------
async function ensureTables() {
  const sql = `
  CREATE TABLE IF NOT EXISTS accessibility_feedback (
    id SERIAL PRIMARY KEY,
    email TEXT NULL,
    message TEXT NOT NULL,
    page_url TEXT NULL,
    user_agent TEXT NULL,
    ip_address TEXT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
  );

  CREATE TABLE IF NOT EXISTS accessibility_audit_log (
    id SERIAL PRIMARY KEY,
    action TEXT NOT NULL,
    component TEXT NULL,
    details JSONB NULL,
    ip_address TEXT NULL,
    user_agent TEXT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
  );

  CREATE INDEX IF NOT EXISTS idx_accessibility_feedback_created_at ON accessibility_feedback(created_at DESC);
  CREATE INDEX IF NOT EXISTS idx_accessibility_audit_created_at ON accessibility_audit_log(created_at DESC);
  `;
  await pool.query(sql);
}

function getClientIp(req) {
  const xf = req.headers["x-forwarded-for"];
  if (typeof xf === "string" && xf.length) return xf.split(",")[0].trim();
  return req.ip || null;
}

// ---------- SCHEMAS ----------
const FeedbackSchema = z.object({
  email: z.string().email().optional().or(z.literal("")),
  message: z.string().min(5).max(4000),
  pageUrl: z.string().max(2048).optional().or(z.literal("")),
});

const AuditSchema = z.object({
  action: z.string().min(2).max(128),
  component: z.string().max(256).optional(),
  details: z.any().optional(),
});

// ---------- ROUTES ----------
app.get("/api/accessibility-statement", async (req, res) => {
  // Update these fields to your real company details
  res.json({
    complianceLevel: "WCAG 2.1 AA",
    standardIsrael: "Israeli Standard 5568 (based on WCAG)",
    lastAuditDate: "2026-02-19",
    knownLimitations: [
      "Some third-party embedded content may have partial accessibility support.",
      "We are continuously improving form flows for keyboard and screen readers."
    ],
    contact: {
      title: "Accessibility Coordinator",
      email: "accessibility@example.com",
      phone: "+972000000000",
      responseTime: "Up to 3 business days"
    }
  });
});

app.post("/api/accessibility-feedback", (req, res, next) => rateLimit(feedbackLimiter, req, res, next), async (req, res) => {
  const parsed = FeedbackSchema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ error: "INVALID_INPUT", details: parsed.error.flatten() });
  }

  const { email, message, pageUrl } = parsed.data;

  await pool.query(
    `INSERT INTO accessibility_feedback(email, message, page_url, user_agent, ip_address)
     VALUES ($1, $2, $3, $4, $5)`,
    [
      (email || "").trim() || null,
      message.trim(),
      (pageUrl || "").trim() || null,
      String(req.headers["user-agent"] || "").slice(0, 512) || null,
      getClientIp(req)
    ]
  );

  res.json({ success: true });
});

app.post("/api/accessibility-audit", async (req, res) => {
  // Optional: protect this route (admin token) if you want.
  const parsed = AuditSchema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ error: "INVALID_INPUT", details: parsed.error.flatten() });
  }

  const { action, component, details } = parsed.data;

  await pool.query(
    `INSERT INTO accessibility_audit_log(action, component, details, user_agent, ip_address)
     VALUES ($1, $2, $3, $4, $5)`,
    [
      action,
      component || null,
      details ? JSON.stringify(details) : null,
      String(req.headers["user-agent"] || "").slice(0, 512) || null,
      getClientIp(req)
    ]
  );

  res.json({ success: true });
});

// Health check
app.get("/api/health", async (req, res) => {
  res.json({ ok: true, time: new Date().toISOString() });
});

// ---------- START ----------
ensureTables()
  .then(() => {
    app.listen(PORT, () => console.log(`Accessibility server running on :${PORT}`));
  })
  .catch((e) => {
    console.error("DB init failed:", e);
    process.exit(1);
  });

    if (focusable) focusable.focus();
  }

  function trapFocus(modalEl) {
    const focusable = $$(
      'button,[href],input,select,textarea,[tabindex]:not([tabindex="-1"])',
      modalEl
    ).filter(el => !el.hasAttribute("disabled") && el.getAttribute("aria-hidden") !== "true");

    if (focusable.length === 0) return () => {};

    const first = focusable[0];
    const last = focusable[focusable.length - 1];

    function onKeyDown(e) {
      if (e.key !== "Tab") return;
      if (e.shiftKey) {
        if (document.activeElement === first) {
          e.preventDefault();
          last.focus();
        }
      } else {
        if (document.activeElement === last) {
          e.preventDefault();
          first.focus();
        }
      }
    }

    modalEl.addEventListener("keydown", onKeyDown);
    return () => modalEl.removeEventListener("keydown", onKeyDown);
  }

    // Modal close
    const closeBtn = $(".acc-close", modal);
    const closeModalBtn = $('[data-acc-action="closeModal"]', modal);
    const modalCard = $(".acc-modal-card", modal);
    let untrap = null;
    let lastFocus = null;

    function closeModal() {
      modal.setAttribute("aria-hidden", "true");
      if (untrap) untrap();
      untrap = null;
      if (lastFocus) lastFocus.focus();
    }

    closeBtn.addEventListener("click", closeModal);
    if (closeModalBtn) closeModalBtn.addEventListener("click", closeModal);

    modal.addEventListener("click", (e) => {
      if (e.target === modal) closeModal();
    });

    document.addEventListener("keydown", (e) => {
      if (modal.getAttribute("aria-hidden") === "false" && e.key === "Escape") {
        e.preventDefault();
        closeModal();
      }
    });

    async function openModal(modalEl, mode) {
      lastFocus = document.activeElement;
      modalEl.setAttribute("aria-hidden", "false");

      // Toggle content
      const feedbackBox = $("#acc-feedback-box", modalEl);
      const statementBox = $("#acc-statement-box", modalEl);
      if (mode === "feedback") {
        feedbackBox.style.display = "block";
        await renderStatement(statementBox, false);
        await audit("ACCESSIBILITY_MODAL_OPEN", "feedback", {});
      } else {
        feedbackBox.style.display = "none";
        await renderStatement(statementBox, true);
        await audit("ACCESSIBILITY_MODAL_OPEN", "statement", {});
      }

      untrap = trapFocus(modalCard);
      focusFirstFocusable(modalCard);
    }

    // Feedback submit
    const form = $("#acc-feedback-form", modal);
    const alert = $("#acc-feedback-alert", modal);

    form.addEventListener("submit", async (e) => {
      e.preventDefault();
      alert.textContent = "";

      const email = String($("#acc-email", modal).value || "").trim();
      const message = String($("#acc-msg", modal).value || "").trim();

      if (message.length < 5) {
        alert.setAttribute("role", "alert");
        alert.textContent = "נא לכתוב לפחות 5 תווים.";
        return;
      }

      try {
        const resp = await fetch(CFG.apiFeedback, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            email: email || "",
            message,
            pageUrl: location.href
          })
        });
        const data = await resp.json().catch(() => ({}));
        if (!resp.ok) {
          alert.setAttribute("role", "alert");
          alert.textContent = data && data.error ? `שגיאה: ${data.error}` : "שגיאה בשליחה.";
          return;
        }
        alert.setAttribute("role", "status");
        alert.textContent = "נשלח בהצלחה. תודה.";
        $("#acc-msg", modal).value = "";
        await audit("ACCESSIBILITY_FEEDBACK_SENT", "feedback", { hasEmail: !!email });
      } catch {
        alert.setAttribute("role", "alert");
        alert.textContent = "שגיאת רשת. נסו שוב.";
      }
    });

    // Statement load
    async function renderStatement(box, showContact) {
      box.innerHTML = `<h3>הצהרת נגישות</h3><small>טוען מידע...</small>`;
      try {
        const resp = await fetch(CFG.apiStatement, { method: "GET" });
        const st = await resp.json();

        const contact = st.contact || {};
        const lines = [
          `רמת תאימות: ${st.complianceLevel || "WCAG 2.1 AA"}`,
          `תקן: ${st.standardIsrael || "5568"}`,
          `תאריך בדיקה אחרונה: ${st.lastAuditDate || "N/A"}`,
        ];

        const limitations = Array.isArray(st.knownLimitations) ? st.knownLimitations : [];
        const limHtml = limitations.length
          ? `<ul>${limitations.map(x => `<li>${escapeHtml(String(x))}</li>`).join("")}</ul>`
          : `<p style="margin:0;">אין מגבלות ידועות.</p>`;

        const contactHtml = showContact
          ? `
            <div style="margin-top:10px;">
              <h3 style="margin:0 0 6px; font-size:13px;">איש קשר</h3>
              <p style="margin:0;">${escapeHtml(contact.title || "רכז נגישות")}</p>
              <p style="margin:0;">אימייל: ${escapeHtml(contact.email || "accessibility@example.com")}</p>
              <p style="margin:0;">טלפון: ${escapeHtml(contact.phone || "+972...")}</p>
              <p style="margin:0;">זמן תגובה: ${escapeHtml(contact.responseTime || "N/A")}</p>
            </div>
          `
          : "";

        box.innerHTML = `
          <h3>הצהרת נגישות</h3>
          <p style="margin:0 0 10px;">${lines.map(l => escapeHtml(l)).join("<br/>")}</p>
          <div class="acc-box" style="padding:10px; margin:0;">
            <h3 style="margin:0 0 6px; font-size:13px;">מגבלות ידועות</h3>
            ${limHtml}
          </div>
          ${contactHtml}
        `;
      } catch {
        box.innerHTML = `<h3>הצהרת נגישות</h3><p style="margin:0;">לא ניתן לטעון מידע כרגע.</p>`;
      }
    }

    function escapeHtml(s) {
      return s.replace(/[&<>"']/g, (c) => {
        const m = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#039;" };
        return m[c] || c;
      });
    }
  }

  // ---------- INIT ----------
  function init() {
    injectStyles();

    const state = loadState();

    // If OS already prefers reduced motion, enable unless user explicitly disabled before
    if (prefersReducedMotion() && !localStorage.getItem(KEY)) {
      state.reduceMotion = true;
      saveState(state);
    }

    applyState(state);
    buildUI(state);
    audit("ACCESSIBILITY_INIT", "widget", { version: "2026" });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init);
  } else {
    init();
  }
})();
