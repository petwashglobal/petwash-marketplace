interface SignedRedeemTokenPayload {
  passId: string;
  userId: string;
  tokenVersion: number;
  machineId?: string | null;
  bayId?: "LEFT" | "RIGHT" | null;
  nonce: string;
  issuedAt: number;
  expiresAt: number;
}

function generateSignedRedeemToken(input: {
  passId: string;
  userId: string;
  tokenVersion: number;
  machineId?: string | null;
  bayId?: "LEFT" | "RIGHT" | null;
  ttlSeconds?: number;
}): string {
  const now = Math.floor(Date.now() / 1000);
  const ttl = input.ttlSeconds ?? 45;

  const payload: SignedRedeemTokenPayload = {
    passId: input.passId,
    userId: input.userId,
    tokenVersion: input.tokenVersion,
    machineId: input.machineId ?? null,
    bayId: input.bayId ?? null,
    nonce: crypto.randomUUID(),
    issuedAt: now,
    expiresAt: now + ttl,
  };

  // HMAC-signed compact token
  const data = Buffer.from(JSON.stringify(payload)).toString("base64url");
  const sig = crypto
    .createHmac("sha256", PASS_SIGNING_SECRET)
    .update(data)
    .digest("base64url");

  return `${data}.${sig}`;
}

function verifySignedRedeemToken(token: string): SignedRedeemTokenPayload {
  const [data, sig] = token.split(".");
  if (!data || !sig) throw new Error("INVALID_TOKEN_FORMAT");

  const expected = crypto
    .createHmac("sha256", PASS_SIGNING_SECRET)
    .update(data)
    .digest("base64url");

  if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) {
    throw new Error("INVALID_SIGNATURE");
  }

  const payload = JSON.parse(
    Buffer.from(data, "base64url").toString("utf8")
  ) as SignedRedeemTokenPayload;

  const now = Math.floor(Date.now() / 1000);
  if (payload.expiresAt < now) throw new Error("TOKEN_EXPIRED");

  return payload;
}

function buildGoogleWalletClassId(suffix: string) {
  return `${GOOGLE_WALLET_ISSUER_ID}.${suffix}`;
}

function buildGoogleWalletObjectId(passId: string) {
  // Google object IDs are issuerId.objectSuffix
  const suffix = passId.replace(/[^A-Za-z0-9._-]/g, "_");
  return `${GOOGLE_WALLET_ISSUER_ID}.${suffix}`;
}

function buildGoogleWalletPayload(record: PetWashPassRecord, qrToken: string) {
  const classId = buildGoogleWalletClassId("petwash_premium_pass_v1");
  const objectId = buildGoogleWalletObjectId(record.passId);

  const genericClass: GoogleWalletClass = {
    id: classId,
    issuerName: "PetWash",
    reviewStatus: "UNDER_REVIEW",
  };

  const genericObject: GoogleWalletObject = {
    id: objectId,
    classId,
    state: "ACTIVE",
    cardTitle: {
      defaultValue: { language: "en-US", value: "PetWash" },
    },
    subheader: {
      defaultValue: { language: "en-US", value: `${record.tier} Tier` },
    },
    header: {
      defaultValue: {
        language: "en-US",
        value: record.primaryPetName
          ? `${record.ownerName} • ${record.primaryPetName}`
          : record.ownerName,
      },
    },
    barcode: {
      type: "QR_CODE",
      value: qrToken,
      alternateText: record.passId,
    },
    // Optional rotating barcode support if you implement device-side / platform-compatible flow.
    rotatingBarcode: {
      initialRotatingBarcodeValues: [qrToken],
      periodMillis: "60000",
    },
    textModulesData: [
      {
        id: "available_credit",
        header: "Available Credit",
        body: `₪${record.availableCreditIls.toFixed(2)}`,
      },
      {
        id: "member_id",
        header: "Member ID",
        body: record.passId,
      },
      {
        id: "status",
        header: "Status",
        body: record.status,
      },
      ...(record.validUntil
        ? [
            {
              id: "valid_until",
              header: "Valid Until",
              body: record.validUntil,
            },
          ]
        : []),
    ],
    linksModuleData: {
      uris: [
        {
          uri: `${BASE_URL}/dashboard/pass/${encodeURIComponent(record.passId)}`,
          description: "Open PetWash Pass",
        },
      ],
    },
    validTimeInterval: record.validUntil
      ? {
          end: { date: record.validUntil },
        }
      : undefined,
    expiryNotification: {
      enableNotification: true,
    },
  };

  return { genericClass, genericObject };
}

app.post("/wallet/redeem", async (req, res) => {
  try {
    const { scannedToken, machineId, bayId, amountIls } = req.body as {
      scannedToken: string;
      machineId: string;
      bayId?: "LEFT" | "RIGHT";
      amountIls: number;
    };

    const payload = verifySignedRedeemToken(scannedToken);

    if (payload.machineId && payload.machineId !== machineId) {
      return res.status(403).json({ error: "TOKEN_MACHINE_MISMATCH" });
    }

    if (payload.bayId && payload.bayId !== (bayId ?? null)) {
      return res.status(403).json({ error: "TOKEN_BAY_MISMATCH" });
    }

    await assertNonceUnusedAndConsume(payload.nonce);

    const pass = await loadPassForRedeem(payload.passId);
    if (!pass) return res.status(404).json({ error: "PASS_NOT_FOUND" });
    if (pass.status !== "ACTIVE") return res.status(403).json({ error: "PASS_NOT_ACTIVE" });

    if (pass.validUntil && new Date(pass.validUntil).getTime() < Date.now()) {
      return res.status(403).json({ error: "PASS_EXPIRED" });
    }

    if (pass.availableCreditIls < amountIls) {
      return res.status(402).json({ error: "INSUFFICIENT_BALANCE" });
    }

    await debitPassAtomically(pass.passId, amountIls, {
      source: "K9000",
      machineId,
      bayId: bayId ?? null,
      nonce: payload.nonce,
    });

    await startK9000Machine({ machineId, bayId });

    // After debit, your backend should trigger:
    // - Apple pass update notification
    // - Google Wallet object update

    return res.json({
      ok: true,
      passId: pass.passId,
      machineId,
      bayId: bayId ?? null,
      debitedIls: amountIls,
    });
  } catch (err: any) {
    return res.status(400).json({ error: err.message || "REDEEM_FAILED" });
  }
});
