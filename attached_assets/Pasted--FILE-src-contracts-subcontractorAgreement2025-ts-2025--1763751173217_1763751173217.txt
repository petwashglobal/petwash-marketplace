/**
 * 2. מודל נתונים - איך זה נשמר במסד הנתונים
 *    אפשר להתאים ל־Prisma, Drizzle, TypeORM וכו.
 *    כאן זו דוגמה כללית עם TypeScript.
 */

export type DigitalSignatureMethod =
  | "typed_name"          // הקלדת שם מלא בתיבה מאושרת
  | "drawn_signature"     // חתימה עם טאצ'
  | "otp_code"            // קוד חד פעמי
  | "external_provider";  // DocuSign / Adobe Sign וכו

export interface SubcontractorSignature {
  id: string;                        // UUID
  subcontractorId: string;           // מזהה קבלן משנה במערכת שלך
  fullName: string;
  email: string;
  phone?: string;
  agreementVersion: string;         // למשל "2025.01"
  signedAt: string;                 // ISO date string
  ipAddress?: string;
  userAgent?: string;
  deviceInfo?: string;
  signatureMethod: DigitalSignatureMethod;
  signaturePayload: string;         // base64 של החתימה הגרפית או hash של הנתונים
  agreementSnapshotJson: string;    // שמירה של ההסכם כפי שהיה בעת החתימה (JSON מלא)
  agreedToPrivacy: boolean;
  agreedToTerms: boolean;
  // שדה נוסף לחותמת מערכת
  auditTrailId?: string;            // מזהה לרשומת לוג מורחבת במערכת
}

/**
 * 3. פונקציית שרות שיוצרת רשומת חתימה חדשה מהפרונט
 *    - כאן זה ללא DB אמיתי, אבל המפתח יכול לחבר ל־Postgres
 */

import crypto from "crypto";

export function createSubcontractorSignature(input: {
  subcontractorId: string;
  fullName: string;
  email: string;
  phone?: string;
  ipAddress?: string;
  userAgent?: string;
  deviceInfo?: string;
  signatureMethod: DigitalSignatureMethod;
  rawSignatureData: string; // לדוגמה: JSON מהפרונט עם נתוני Canvas או קוד OTP מאושר
}): SubcontractorSignature {
  const now = new Date().toISOString();

  // hash סימון של החתימה - לא חייבים לשמור גראפיקה גולמית אם לא רוצים
  const signatureHash = crypto
    .createHash("sha256")
    .update(input.rawSignatureData + now + input.subcontractorId)
    .digest("hex");

  return {
    id: crypto.randomUUID(),
    subcontractorId: input.subcontractorId,
    fullName: input.fullName,
    email: input.email,
    phone: input.phone,
    agreementVersion: SUBCONTRACTOR_AGREEMENT_2025.version,
    signedAt: now,
    ipAddress: input.ipAddress,
    userAgent: input.userAgent,
    deviceInfo: input.deviceInfo,
    signatureMethod: input.signatureMethod,
    signaturePayload: signatureHash,
    agreementSnapshotJson: JSON.stringify(SUBCONTRACTOR_AGREEMENT_2025),
    agreedToPrivacy: true,
    agreedToTerms: true,
    auditTrailId: `audit_${Date.now()}_${Math.floor(Math.random() * 9999)}`
  };
}

/**
 * 4. דוגמת API ב־Express:
 *    - POST /api/subcontractors/agreements/2025/sign
 *    - מקבל את פרטי הקבלן ונתוני החתימה ושומר למסד
 *
 *    שים לב:
 *    - צריך להחליף את fakeDbSave ב־ORM אמיתי.
 *    - רצוי להוסיף אימות JWT כניסה וכדומה.
 */

import type { Request, Response } from "express";

// סימולציה של שמירה במסד
const fakeSignaturesDb: SubcontractorSignature[] = [];

function fakeDbSave(sig: SubcontractorSignature) {
  fakeSignaturesDb.push(sig);
  return sig;
}

export async function signSubcontractorAgreement2025Handler(
  req: Request,
  res: Response
) {
  try {
    // 4.1 ולידציה בסיסית - המפתח יכול להחליף ל־Zod או Joi
    const {
      subcontractorId,
      fullName,
      email,
      phone,
      signatureMethod,
      rawSignatureData
    } = req.body || {};

    if (!subcontractorId || !fullName || !email || !signatureMethod || !rawSignatureData) {
      return res.status(400).json({
        ok: false,
        error: "Missing required fields"
      });
    }

    // 4.2 יצירת רשומת חתימה
    const sig = createSubcontractorSignature({
      subcontractorId,
      fullName,
      email,
      phone,
      signatureMethod,
      rawSignatureData,
      ipAddress: req.ip,
      userAgent: req.headers["user-agent"],
      deviceInfo: req.headers["x-device-info"] as string | undefined
    });

    // 4.3 שמירה "למסד נתונים"
    const saved = fakeDbSave(sig);

    return res.status(201).json({
      ok: true,
      data: {
        signatureId: saved.id,
        agreementVersion: saved.agreementVersion,
        signedAt: saved.signedAt
      }
    });
  } catch (err) {
    console.error("signSubcontractorAgreement2025Handler error", err);
    return res.status(500).json({
      ok: false,
      error: "Internal server error"
    });
  }
}

/**
 * 5. פונקציה להחזרת פרטי החתימה עבור מסך "מסמכים משפטיים"
 *    GET /api/subcontractors/agreements/2025/:signatureId
 */

export async function getSubcontractorSignatureHandler(
  req: Request,
  res: Response
) {
  const { signatureId } = req.params;
  const record = fakeSignaturesDb.find((s) => s.id === signatureId);

  if (!record) {
    return res.status(404).json({ ok: false, error: "Signature not found" });
  }

  return res.json({
    ok: true,
    data: {
      signature: record,
      // אפשר להחזיר גם HTML ייצוגי אם תרצו
      agreement: JSON.parse(record.agreementSnapshotJson)
    }
  });
}
