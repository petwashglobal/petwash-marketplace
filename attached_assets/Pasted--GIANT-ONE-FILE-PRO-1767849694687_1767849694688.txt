function getPlatformById(id: string): PlatformConfig | undefined {
  return PlatformRegistry.find((p) => p.id === id && p.isActive);
}

function getPlatformByHost(host?: string): PlatformConfig | undefined {
  if (!host) return undefined;
  const clean = host.split(":")[0].toLowerCase();
  return PlatformRegistry.find((p) => p.isActive && p.domains.map((d) => d.toLowerCase()).includes(clean));
}

function now(): number {
  return Date.now();
}

function uid(prefix = ""): string {
  const rnd = crypto.randomBytes(16).toString("hex");
  return prefix ? `${prefix}_${rnd}` : rnd;
}

function safeJsonError(res: Response, code: string, message: string, requestId: string, status = 400) {
  const payload: ApiErrorShape = { error: { code, message, requestId } };
  return res.status(status).json(payload);
}

      id: p.id,

      description: p.description,
      features: p.features,
    }));
  }
}

type AuthContext = {
  requestId: string;
  platform: PlatformConfig;
  user?: User;
};

declare global {
  namespace Express {
    interface Request {
      ctx?: AuthContext;
    }
  }
}

function signJwt(payload: { sub: string; platformId: string; role: Role }): string {
  return jwt.sign(payload, env.JWT_SECRET, { expiresIn: env.JWT_EXPIRES_IN });
}

function verifyJwt(token: string): { sub: string; platformId: string; role: Role } | null {
  try {
    return jwt.verify(token, env.JWT_SECRET) as any;
  } catch {
    return null;
  }
}

function requireAuth(req: Request, res: Response, next: NextFunction) {
  const requestId = req.ctx?.requestId || uid("req");
  const platform = req.ctx?.platform;
  if (!platform) return safeJsonError(res, "NO_PLATFORM", "Platform not resolved", requestId, 400);

  const authHeader = req.headers.authorization || "";
  const bearer = authHeader.startsWith("Bearer ") ? authHeader.slice(7) : "";
  const token = bearer || (req.cookies?.session as string | undefined) || "";

  if (!token) return safeJsonError(res, "UNAUTHENTICATED", "Missing token", requestId, 401);

  const decoded = verifyJwt(token);
  if (!decoded) return safeJsonError(res, "UNAUTHENTICATED", "Invalid token", requestId, 401);

  if (decoded.platformId !== platform.id) {
    return safeJsonError(res, "PLATFORM_MISMATCH", "Token is for a different platform", requestId, 403);
  }

  const user = db.users.get(decoded.sub);
  if (!user || user.platformId !== platform.id) {
    return safeJsonError(res, "UNAUTHENTICATED", "User not found", requestId, 401);
  }
  if (user.flags.isBanned) return safeJsonError(res, "FORBIDDEN", "Account disabled", requestId, 403);

  req.ctx = { ...(req.ctx as any), user };
  next();
}

function requireRole(roles: Role[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    const requestId = req.ctx?.requestId || uid("req");
    const user = req.ctx?.user;
    if (!user) return safeJsonError(res, "UNAUTHENTICATED", "Login required", requestId, 401);
    if (!roles.includes(user.role)) return safeJsonError(res, "FORBIDDEN", "Insufficient role", requestId, 403);
    next();
  };
}

function audit(req: Request, action: string, target?: { type: string; id: string }, meta?: Record<string, any>) {
  const platformId = req.ctx?.platform?.id || "unknown";
  const ev: AuditEvent = {
    id: uid("aud"),
    platformId,
    actorUserId: req.ctx?.user?.id,
    action,
    target,
    ip: req.ip,
    ua: req.headers["user-agent"],
    meta,
    createdAt: now(),
  };
  db.audit.set(ev.id, ev);
}

/* =========================
   PLATFORM RESOLUTION MIDDLEWARE
========================= */

function resolvePlatform(req: Request, res: Response, next: NextFunction) {
  const requestId = uid("req");

  const headerKey = env.PLATFORM_HEADER.toLowerCase();
  const headerVal = (req.headers[headerKey] as string | undefined) || "";
  const byHeader = headerVal ? getPlatformById(headerVal) : undefined;

  const host = req.headers.host;
  const byHost = getPlatformByHost(host);

  const platform = byHeader || byHost;

  if (!platform) {
    req.ctx = { requestId, platform: { id: "unknown", title: "Unknown", description: "", domains: [], features: [], isActive: false } as any };
    return safeJsonError(res, "NO_PLATFORM", `Platform not found. Provide header ${env.PLATFORM_HEADER} or use a mapped domain.`, requestId, 400);
  }

  req.ctx = { requestId, platform };
  next();
}

/* =========================
   BOOKING PRICING ENGINE (Simple)
========================= */

function computeBookingQuote(listing: Listing, startAt: number, endAt: number) {
  const durationMs = Math.max(0, endAt - startAt);
  const durationHours = Math.max(1, Math.ceil(durationMs / (60 * 60 * 1000)));

  let subtotal = listing.price.amountCents;

  if (listing.price.unit === "hour") {
    subtotal = listing.price.amountCents * durationHours;
  }

  const feePct = 0.08; // platform fee example
  const fees = Math.round(subtotal * feePct);
  const total = subtotal + fees;

  return {
    currency: listing.price.currency,
    subtotalCents: subtotal,
    feesCents: fees,
    totalCents: total,
    breakdown: [
      { label: "Subtotal", amountCents: subtotal },
      { label: "Platform fees", amountCents: fees },
    ],
  };
}

app.post("/api/auth/signup", async (req, res) => {
  const requestId = req.ctx!.requestId;
  const platform = req.ctx!.platform;

  const parsed = AuthSignupSchema.safeParse(req.body);
  if (!parsed.success) return safeJsonError(res, "VALIDATION", parsed.error.message, requestId, 400);

  const { email, password, displayName } = parsed.data;

  if (db.findUserByEmail(platform.id, email)) {
    return safeJsonError(res, "EMAIL_EXISTS", "Email already registered", requestId, 409);
  }

  const passwordHash = await bcrypt.hash(password, env.BCRYPT_ROUNDS);

  const user: User = {
    id: uid("usr"),
    platformId: platform.id,
    email: email.toLowerCase(),
    passwordHash,
    role: "user",
    displayName,
    createdAt: now(),
    updatedAt: now(),
    flags: { isBanned: false, isEmailVerified: false },
  };

  db.users.set(user.id, user);

  const token = signJwt({ sub: user.id, platformId: platform.id, role: user.role });

  audit(req, "auth.signup", { type: "user", id: user.id });

  res
    .cookie("session", token, {
      httpOnly: true,
      secure: env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: 7 * 24 * 60 * 60 * 1000,
    })
    .json({
      token,
      user: { id: user.id, email: user.email, displayName: user.displayName, role: user.role, platformId: user.platformId },
      requestId,
    });
});

app.post("/api/auth/login", async (req, res) => {
  const requestId = req.ctx!.requestId;
  const platform = req.ctx!.platform;

  const parsed = AuthLoginSchema.safeParse(req.body);
  if (!parsed.success) return safeJsonError(res, "VALIDATION", parsed.error.message, requestId, 400);

  const { email, password } = parsed.data;
  const user = db.findUserByEmail(platform.id, email);

  if (!user) return safeJsonError(res, "INVALID_CREDENTIALS", "Invalid email or password", requestId, 401);
  if (user.flags.isBanned) return safeJsonError(res, "FORBIDDEN", "Account disabled", requestId, 403);

  const ok = await bcrypt.compare(password, user.passwordHash);
  if (!ok) return safeJsonError(res, "INVALID_CREDENTIALS", "Invalid email or password", requestId, 401);

  const token = signJwt({ sub: user.id, platformId: platform.id, role: user.role });

  audit(req, "auth.login", { type: "user", id: user.id });

  res
    .cookie("session", token, {
      httpOnly: true,
      secure: env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: 7 * 24 * 60 * 60 * 1000,
    })
    .json({
      token,
      user: { id: user.id, email: user.email, displayName: user.displayName, role: user.role, platformId: user.platformId },
      requestId,
    });
});

/* =========================
   SEARCH
   - Default search is platform-scoped
   - Admin/Super admin can search across platforms using ?platformId=all
========================= */

app.get("/api/search", (req, res) => {
  const requestId = req.ctx!.requestId;
  const platform = req.ctx!.platform;

  const parsed = SearchSchema.safeParse(req.query);
  if (!parsed.success) return safeJsonError(res, "VALIDATION", parsed.error.message, requestId, 400);

  const { q, type, tag, city, limit, platformId } = parsed.data;

  let effectivePlatformIds: string[] = [platform.id];

  // Optional cross-platform search control:
  // - If platformId is provided and user is admin/super_admin, allow it.
  const authHeader = req.headers.authorization || "";
  const bearer = authHeader.startsWith("Bearer ") ? authHeader.slice(7) : "";
  const token = bearer || (req.cookies?.session as string | undefined) || "";
  const decoded = token ? verifyJwt(token) : null;
  const requester = decoded ? db.users.get(decoded.sub) : undefined;

  if (platformId) {
    if (!requester || !["admin", "super_admin"].includes(requester.role)) {
      return safeJsonError(res, "FORBIDDEN", "Cross-platform search requires admin role", requestId, 403);
    }
    if (platformId === "all") {
      effectivePlatformIds = PlatformRegistry.filter((p) => p.isActive).map((p) => p.id);
    } else {
      const p = getPlatformById(platformId);
      if (!p) return safeJsonError(res, "NOT_FOUND", "Platform not found", requestId, 404);
      effectivePlatformIds = [p.id];
    }
  }

  const needle = (q || "").trim().toLowerCase();

  const all = Array.from(db.listings.values()).filter((l) => effectivePlatformIds.includes(l.platformId));

  const filtered = all.filter((l) => {
    if (type && l.type !== type) return false;
    if (tag && !l.tags.map((t) => t.toLowerCase()).includes(tag.toLowerCase())) return false;
    if (city && (l.location.city || "").toLowerCase() !== city.toLowerCase()) return false;

    if (needle) {
      const hay = `${l.title} ${l.description} ${l.tags.join(" ")} ${(l.location.city || "")}`.toLowerCase();
      if (!hay.includes(needle)) return false;
    }
    return l.availability?.isActive !== false;
  });

  // naive rank: shorter title match first
  filtered.sort((a, b) => a.title.length - b.title.length);

  const results = filtered.slice(0, limit);

  res.json({
    platformScope: effectivePlatformIds,
    results,
    count: results.length,
    requestId,
  });
});

   - Stored as records with status workflow
========================= */

app.post(
  "/api/docs/upload",
  requireAuth,
  upload.array("files", 10),
  (req: Request, res: Response) => {
    const requestId = req.ctx!.requestId;
    const platform = req.ctx!.platform;
    const user = req.ctx!.user!;

    if (!platform.features.includes("docs_verification")) {
      return safeJsonError(res, "FEATURE_DISABLED", "Docs verification not enabled", requestId, 403);
    }

    const metaParsed = DocUploadMetaSchema.safeParse(req.body);
    if (!metaParsed.success) return safeJsonError(res, "VALIDATION", metaParsed.error.message, requestId, 400);

    const files = (req.files as Express.Multer.File[]) || [];
    if (!files.length) return safeJsonError(res, "NO_FILES", "No files uploaded", requestId, 400);

    const created: DocumentRecord[] = [];

    for (const f of files) {
      const rec: DocumentRecord = {
        id: uid("doc"),
        platformId: platform.id,
        userId: user.id,
        type: metaParsed.data.type,
        originalFileName: f.originalname,
        mimeType: f.mimetype,
        sizeBytes: f.size,
        storageKey: f.filename, // local filename; replace with cloud key if needed
        status: "pending_review",
        createdAt: now(),
        updatedAt: now(),
      };
      db.documents.set(rec.id, rec);
      created.push(rec);
    }

    audit(req, "docs.upload", undefined, { count: created.length, type: metaParsed.data.type });

    res.status(201).json({ documents: created, requestId });
  }
);

  user.updatedAt = now();

  audit(req, "admin.user.set_role", { type: "user", id: user.id }, { role });

  res.json({ ok: true, user: { id: user.id, role: user.role }, requestId });
});

app.post("/api/bookings/quote", requireAuth, (req, res) => {
  const requestId = req.ctx!.requestId;
  const platform = req.ctx!.platform;
  const user = req.ctx!.user!;

  const parsed = BookingQuoteSchema.safeParse(req.body);
  if (!parsed.success) return safeJsonError(res, "VALIDATION", parsed.error.message, requestId, 400);

  const booking = db.bookings.get(parsed.data.bookingId);
  if (!booking || booking.platformId !== platform.id) return safeJsonError(res, "NOT_FOUND", "Booking not found", requestId, 404);
  if (booking.buyerUserId !== user.id) return safeJsonError(res, "FORBIDDEN", "Not your booking", requestId, 403);

  const listing = db.listings.get(booking.listingId);
  if (!listing) return safeJsonError(res, "NOT_FOUND", "Listing not found", requestId, 404);

  const quote = computeBookingQuote(listing, booking.schedule.startAt, booking.schedule.endAt);

  booking.quote = quote;
  booking.status = "quoted";
  booking.updatedAt = now();
  db.bookings.set(booking.id, booking);

  audit(req, "booking.quote", { type: "booking", id: booking.id }, { quote });

  res.json({ booking, requestId });
});

app.post("/api/bookings/confirm", requireAuth, async (req, res) => {
  const requestId = req.ctx!.requestId;
  const platform = req.ctx!.platform;
  const user = req.ctx!.user!;

  if (!platform.features.includes("payments")) {
    return safeJsonError(res, "FEATURE_DISABLED", "Payments not enabled", requestId, 403);
  }

  const parsed = BookingConfirmSchema.safeParse(req.body);
  if (!parsed.success) return safeJsonError(res, "VALIDATION", parsed.error.message, requestId, 400);

  const booking = db.bookings.get(parsed.data.bookingId);
  if (!booking || booking.platformId !== platform.id) return safeJsonError(res, "NOT_FOUND", "Booking not found", requestId, 404);
  if (booking.buyerUserId !== user.id) return safeJsonError(res, "FORBIDDEN", "Not your booking", requestId, 403);

  if (!booking.quote) return safeJsonError(res, "NO_QUOTE", "Quote required before confirm", requestId, 400);

  const intent = await payments.createIntent({
    amountCents: booking.quote.totalCents,
    currency: booking.quote.currency,
    metadata: { bookingId: booking.id, platformId: platform.id, buyerUserId: user.id },
  });

  booking.payment = { provider: "placeholder", intentId: intent.id, status: intent.status };
  booking.status = "confirmed";
  booking.updatedAt = now();
  db.bookings.set(booking.id, booking);

  audit(req, "booking.confirm", { type: "booking", id: booking.id }, { intentId: intent.id });

  res.json({
    booking,
    payment: { intentId: intent.id, status: intent.status, amountCents: intent.amountCents, currency: intent.currency },
    requestId,
  });
});
