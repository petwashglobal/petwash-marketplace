    <script>
        const overlay = document.getElementById('luxury-popup');
        const closeBtn = document.getElementById('close-popup');

        // Elegant Entrance
        window.addEventListener('load', () => {
            setTimeout(() => {
                overlay.classList.add('active');
            }, 1500); // 1.5s delay for a premium feel
        });

        closeBtn.onclick = () => {
            overlay.classList.remove('active');
        };

        // Backend Integration Logic
        document.getElementById('loyalty-form').onsubmit = async (e) => {
            e.preventDefault();
            const email = document.getElementById('email').value;

            // Update UI to show "Processing"
            const btn = e.target.querySelector('button');
            btn.innerText = "Verifying...";
            btn.style.opacity = "0.7";

            try {
                // Change URL to your actual backend endpoint
                const response = await fetch('/api/subscribe', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ email: email, tier: 'Luxury Member' })
                });

                if (response.ok) {
                    btn.innerText = "Welcome";
                    setTimeout(() => overlay.classList.remove('active'), 1000);
                }
            } catch (err) {
                // Fallback for demo
                console.log("Connect this to your Node.js server.js file.");
                btn.innerText = "Success";
                setTimeout(() => overlay.classList.remove('active'), 1000);
            }
        };
