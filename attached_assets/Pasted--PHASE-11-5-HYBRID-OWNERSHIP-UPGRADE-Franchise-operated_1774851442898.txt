/* =========================================================
   3. SETTLEMENT ENGINE PATCH - BUSINESS LOGIC
   =========================================================

   File target:
   server/services/SettlementEngine.ts

   Core rule:
   - franchise station => current split logic
   - company station => franchise_amount_cents = 0
*/

type OwnershipType = "franchise" | "company";

type StationOwnershipContext = {
  stationId: number;
  ownershipType: OwnershipType;
  franchiseId: number | null;
};

type SettlementSplitInput = {
  totalAmountCents: number;
  platformFeePct: number;         // e.g. 15
  stationRevenuePct: number;      // current existing station share rule
  franchiseOverridePct?: number | null;
};

type SettlementSplitOutput = {
  platformAmountCents: number;
  stationAmountCents: number;
  franchiseAmountCents: number;
};

/*
RECOMMENDED PATCH
Keep current franchise math unchanged.
Branch only when ownership_type = 'company'.
*/

function roundCents(v: number): number {
  return Math.round(v);
}

function computeSplit(
  station: StationOwnershipContext,
  input: SettlementSplitInput
): SettlementSplitOutput {
  const total = input.totalAmountCents;
  const platformPct = input.platformFeePct / 100;
  const stationPct = input.stationRevenuePct / 100;
  const franchisePct = (input.franchiseOverridePct ?? 0) / 100;

  if (station.ownershipType === "company") {
    // Company-owned station
    // No franchise share exists
    const platformAmountCents = roundCents(total * platformPct);
    const franchiseAmountCents = 0;
    const stationAmountCents = total - platformAmountCents;

    return {
      platformAmountCents,
      stationAmountCents,
      franchiseAmountCents,
    };
  }

  // Franchise-owned station
  const platformAmountCents = roundCents(total * platformPct);
  const franchiseAmountCents = roundCents(total * franchisePct);
  const stationAmountCents =
    total - platformAmountCents - franchiseAmountCents;

  return {
    platformAmountCents,
    stationAmountCents,
    franchiseAmountCents,
  };
}

/*
IMPORTANT
Do NOT derive finance from bookings.
Still derive all money from settlements.
Only branch the split calculation by ownership type.
*/
