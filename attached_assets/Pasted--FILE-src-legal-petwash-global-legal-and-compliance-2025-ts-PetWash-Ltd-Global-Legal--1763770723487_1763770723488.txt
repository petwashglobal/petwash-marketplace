import crypto from "crypto";

function generateId(prefix: string): string {
  return `${prefix}_${crypto.randomBytes(16).toString("hex")}`;
}

export function createSubcontractorSignature(input: {
  subcontractorId: string;
  fullName: string;
  email: string;
  phone?: string;
  ipAddress?: string;
  userAgent?: string;
  deviceInfo?: string;
  signatureMethod: DigitalSignatureMethod;
  rawSignatureData: string;
}): SubcontractorSignature {
  const now = new Date().toISOString();
  const signatureHash = crypto
    .createHash("sha256")
    .update(input.rawSignatureData + now + input.subcontractorId)
    .digest("hex");

  return {
    id: generateId("sig"),
    subcontractorId: input.subcontractorId,
    fullName: input.fullName,
    email: input.email,
    phone: input.phone,
    agreementVersion: SUBCONTRACTOR_AGREEMENT_2025.version,
    signedAt: now,
    ipAddress: input.ipAddress,
    userAgent: input.userAgent,
    deviceInfo: input.deviceInfo,
    signatureMethod: input.signatureMethod,
    signaturePayload: signatureHash,
    agreementSnapshotJson: JSON.stringify(SUBCONTRACTOR_AGREEMENT_2025),
    agreedToPrivacy: true,
    agreedToTerms: true,
    auditTrailId: `audit_${Date.now()}_${Math.floor(Math.random() * 9999)}`,
  };
}

export function createConsentRecord(input: {
  userId?: string;
  channel: ConsentRecord["channel"];
  type: ConsentType;
  granted: boolean;
  ipAddress?: string;
  userAgent?: string;
  data?: Record<string, any>;
}): ConsentRecord {
  const now = new Date().toISOString();
  return {
    id: generateId("consent"),
    userId: input.userId,
    channel: input.channel,
    type: input.type,
    granted: input.granted,
    createdAt: now,
    updatedAt: now,
    ipAddress: input.ipAddress,
    userAgent: input.userAgent,
    data: input.data ?? {},
  };
}

export function createPrivacyRequest(input: {
  userId?: string;
  requestType: PrivacyRequestType;
  email: string;
  details?: string;
}): PrivacyRequestRecord {
  const now = new Date().toISOString();
  return {
    id: generateId("pr"),
    userId: input.userId,
    requestType: input.requestType,
    status: "received",
    createdAt: now,
    updatedAt: now,
    email: input.email,
    details: input.details,
  };
}

import type { Request, Response, Router } from "express";

/**
 * LEGAL DOCUMENTS ROUTE
 * GET /api/legal/:docKey
 */
export function registerLegalRoutes(router: Router) {
  router.get("/api/legal/:docKey", (req: Request, res: Response) => {
    const { docKey } = req.params;
    const doc = (legalDocumentsRegistry as any)[docKey];

    if (!doc) {
      return res.status(404).json({ ok: false, error: "Document not found" });
    }

    return res.json({
      ok: true,
      data: doc,
    });
  });
}

/**
 * SUBCONTRACTOR AGREEMENT SIGN
 * POST /api/subcontractors/agreements/2025/sign
 */
export async function signSubcontractorAgreement2025Handler(
  req: Request,
  res: Response
) {
  try:
    const {
      subcontractorId,
      fullName,
      email,
      phone,
      signatureMethod,
      rawSignatureData,
    } = req.body || {};

    if (
      !subcontractorId ||
      !fullName ||
      !email ||
      !signatureMethod ||
      !rawSignatureData
    ) {
      return res.status(400).json({
        ok: false,
        error: "Missing required fields",
      });
    }

    const sig = createSubcontractorSignature({
      subcontractorId,
      fullName,
      email,
      phone,
      signatureMethod,
      rawSignatureData,
      ipAddress: req.ip,
      userAgent: req.headers["user-agent"],
      deviceInfo: req.headers["x-device-info"] as string | undefined,
    });

    fakeSubcontractorSignaturesDb.push(sig);

    return res.status(201).json({
      ok: true,
      data: {
        signatureId: sig.id,
        agreementVersion: sig.agreementVersion,
        signedAt: sig.signedAt,
      },
    });
  } catch (err) {
    console.error("signSubcontractorAgreement2025Handler error", err);
    return res.status(500).json({
      ok: false,
      error: "Internal server error",
    });
  }
}

/**
 * GET ONE SUBCONTRACTOR SIGNATURE
 * GET /api/subcontractors/agreements/2025/:signatureId
 */
export async function getSubcontractorSignatureHandler(
  req: Request,
  res: Response
) {
  const { signatureId } = req.params;
  const record = fakeSubcontractorSignaturesDb.find((s) => s.id === signatureId);

  if (!record) {
    return res.status(404).json({ ok: false, error: "Signature not found" });
  }

  return res.json({
    ok: true,
    data: {
      signature: record,
      agreement: JSON.parse(record.agreementSnapshotJson),
    },
  });
}

/**
 * CONSENT API
 * POST /api/consents
 * Body example:
 * {
 *   "userId": "user_123",
 *   "channel": "web",
 *   "type": "cookies",
 *   "granted": true,
 *   "data": {
 *     "necessary": true,
 *     "analytics": false,
 *     "marketing": false,
 *     "functional": true
 *   }
 * }
 */
export async function postConsentHandler(req: Request, res: Response) {
  try {
    const { userId, channel, type, granted, data } = req.body || {};

    if (!channel || !type || typeof granted !== "boolean") {
      return res.status(400).json({
        ok: false,
        error: "Missing required fields",
      });
    }

    const record = createConsentRecord({
      userId,
      channel,
      type,
      granted,
      ipAddress: req.ip,
      userAgent: req.headers["user-agent"],
      data,
    });

    fakeConsentsDb.push(record);

    return res.status(201).json({
      ok: true,
      data: record,
    });
  } catch (err) {
    console.error("postConsentHandler error", err);
    return res.status(500).json({
      ok: false,
      error: "Internal server error",
    });
  }
}

/**
 * PRIVACY REQUESTS API
 * POST /api/privacy/requests

 *   "requestType": "delete",
 *   "email": "user@example.com",
 *   "details": "Please delete all my data."
 * }
 */
export async function postPrivacyRequestHandler(req: Request, res: Response) {
  try {
    const { userId, requestType, email, details } = req.body || {};

    if (!requestType || !email) {
      return res.status(400).json({
        ok: false,
        error: "Missing required fields",
      });
    }

    const record = createPrivacyRequest({
      userId,
      requestType,
      email,
      details,
    });

    fakePrivacyRequestsDb.push(record);

    return res.status(201).json({
      ok: true,
      data: {
        requestId: record.id,
        status: record.status,
        createdAt: record.createdAt,
      },
    });
  } catch (err) {
    console.error("postPrivacyRequestHandler error", err);
    return res.status(500).json({
      ok: false,
      error: "Internal server error",
    });
  }
}

/**
 * GET ONE PRIVACY REQUEST
 * GET /api/privacy/requests/:id
 */
export async function getPrivacyRequestHandler(req: Request, res: Response) {
  const { id } = req.params;
  const record = fakePrivacyRequestsDb.find((r) => r.id === id);

  if (!record) {
    return res.status(404).json({ ok: false, error: "Request not found" });
  }

  return res.json({
    ok: true,
    data: record,
  });
}
