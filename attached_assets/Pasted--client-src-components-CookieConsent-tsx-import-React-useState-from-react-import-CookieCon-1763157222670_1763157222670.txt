// This is the main component
const GDPRCookieConsent = () => {
  const [showModal, setShowModal] = useState(false);

  // Function to run when user accepts
  const handleAccept = () => {
    // This is where you would initialize analytics, tracking, etc.
    Cookies.set('cookieConsent', 'true', { expires: 365 });
    console.log('Cookie consent given. Initializing analytics...');
  };

  // Function to run when user rejects
  const handleDecline = () => {
    // User rejected. Do not load any non-essential scripts.
    Cookies.set('cookieConsent', 'false', { expires: 365 });
    console.log('Cookie consent declined.');
  };

  // Check if consent has already been given
  const isConsentGiven = getCookieConsentValue('cookieConsent') === 'true';

  if (isConsentGiven) {
    // User has already consented, don't show the banner
    // You might want to re-run your analytics init here on page load
    return null;
  }

  return (
    <>
      {showModal && <CookieManagerModal onClose={() => setShowModal(false)} />}
      
      <CookieConsent
        location="bottom"
        style={{ background: '#2B373B', zIndex: 9999 }}
        buttonText="Accept All"
        onAccept={handleAccept}
        
        // This adds the "Reject" button
        enableDeclineButton
        declineButtonText="Reject All"
        onDecline={handleDecline}
        
        // This adds the "Manage" button
        customButtonProps={{ 'aria-label': 'Manage Preferences' }}
        buttonId="manage-button"
        buttonWrapperClasses="cookie-manage-wrapper"
        onButtonClick={() => {
          // This overrides the "Accept" click to open the modal instead
          // We will rename the default "Accept" button to "Manage"
        }}
      >
        This website uses cookies to ensure you get the best experience. 
        <span style={{ fontSize: '12px', marginLeft: '10px' }}>
          <a href="/legal/PrivacyPolicy" style={{ color: 'white' }}>Privacy Policy</a>
        </span>
      </CookieConsent>

      {/* This is a bit of a hack to "rename" the main button.
        We find the button the library created and change its text and action.
      */}
      <style>{`
        #manage-button {
          background: #4A90E2 !important; /* Blue color for 'Manage' */
          margin-right: 15px;
        }
        .cookie-manage-wrapper button {
          background-color: #4CAF50 !important; /* Green for 'Accept All' */
        }
      `}</style>
      <script>
        {`
          // Find the library's button and change it to "Manage Preferences"
          const manageBtn = document.getElementById('manage-button');
          if (manageBtn) {
            manageBtn.innerText = 'Manage Preferences';
            manageBtn.onclick = (e) => {
              e.preventDefault();
              document.dispatchEvent(new Event('openCookieModal'));
            };
          }
          // Listen for our custom event
          document.addEventListener('openCookieModal', () => {
            document.querySelector('.cookie-modal-overlay').style.display = 'block';
          });
        `}
      </script>
    </>
  );
};

export default GDPRCookieConsent;
