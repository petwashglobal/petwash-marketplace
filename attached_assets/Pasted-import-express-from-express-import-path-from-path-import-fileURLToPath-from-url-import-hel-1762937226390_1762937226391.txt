import express from "express"
import path from "path"
import { fileURLToPath } from "url"
import helmet from "helmet"
import cors from "cors"
import compression from "compression"
import morgan from "morgan"
import rateLimit from "express-rate-limit"
import cookieParser from "cookie-parser"
import csrf from "csurf"
import crypto from "crypto"
import fetch from "node-fetch"
import { OAuth2Client } from "google-auth-library"

const __filename = fileURLToPath(import.meta.url)
const __dirname = path.dirname(__filename)

const PORT = Number(process.env.PORT || 3000)
const NODE_ENV = process.env.NODE_ENV || "production"
const ORIGIN_1 = process.env.APP_ORIGIN_1 || "https://petwash.co.il"
const ORIGIN_2 = process.env.APP_ORIGIN_2 || "https://www.petwash.co.il"
const ORIGIN_3 = process.env.APP_ORIGIN_3 || ""
const CORS_ORIGINS = [ORIGIN_1, ORIGIN_2, ORIGIN_3].filter(Boolean)
const COOKIE_SECRET = process.env.COOKIE_SECRET || crypto.randomBytes(32).toString("hex")
const SESSION_COOKIE = process.env.SESSION_COOKIE || "__pw_sid"
const GOOGLE_CLIENT_ID = process.env.GOOGLE_CLIENT_ID || ""
const GOOGLE_CLIENT_SECRET = process.env.GOOGLE_CLIENT_SECRET || ""
const GOOGLE_REDIRECT_URI = process.env.GOOGLE_REDIRECT_URI || `${ORIGIN_1}/auth/google/callback`
const STATIC_DIR = process.env.STATIC_DIR || path.resolve(__dirname, "../dist")
const TRUST_PROXY = process.env.TRUST_PROXY === "true" || true
const API_RATE_WINDOW_MS = Number(process.env.API_RATE_WINDOW_MS || 60_000)
const API_RATE_MAX = Number(process.env.API_RATE_MAX || 120)
const HEALTH_KEY = process.env.HEALTH_KEY || ""
const CSP_SELF = ["'self'"]
const CSP_SRC = (extra: string[]) => [...CSP_SELF, ...extra].join(" ")

const app = express()
if (TRUST_PROXY) app.set("trust proxy", 1)

app.disable("x-powered-by")

app.use(helmet({
  contentSecurityPolicy: {
    useDefaults: true,
    directives: {
      "default-src": ["'self'"],
      "script-src": ["'self'", "'unsafe-inline'", "'unsafe-eval'", "https://accounts.google.com", "https://www.googletagmanager.com", "https://www.google.com", "https://maps.googleapis.com"],
      "style-src": ["'self'", "'unsafe-inline'", "https://fonts.googleapis.com"],
      "img-src": ["'self'", "data:", "https://*.googleusercontent.com", "https://www.google.com", "https://maps.gstatic.com", "https://maps.googleapis.com"],
      "font-src": ["'self'", "https://fonts.gstatic.com", "data:"],
      "connect-src": ["'self'", ORIGIN_1, ORIGIN_2, "https://www.google-analytics.com", "https://maps.googleapis.com", "https://oauth2.googleapis.com", "https://www.googleapis.com"],
      "frame-src": ["'self'", "https://accounts.google.com", "https://www.google.com"],
      "frame-ancestors": ["'self'"],
      "upgrade-insecure-requests": []
    }
  },
  hsts: { maxAge: 31536000, includeSubDomains: true, preload: true },
  referrerPolicy: { policy: "no-referrer" },
  crossOriginResourcePolicy: { policy: "same-site" },
  crossOriginOpenerPolicy: { policy: "same-origin" },
  crossOriginEmbedderPolicy: false
}))

app.use(cors({
  origin: (origin, cb) => {
    if (!origin) return cb(null, true)
    if (CORS_ORIGINS.includes(origin)) return cb(null, true)
    return cb(new Error("CORS"), false)
  },
  credentials: true
}))

app.use(compression())
app.use(morgan(NODE_ENV === "production" ? "combined" : "dev"))
app.use(express.json({ limit: "2mb" }))
app.use(express.urlencoded({ extended: true }))
app.use(cookieParser(COOKIE_SECRET))

const limiter = rateLimit({ windowMs: API_RATE_WINDOW_MS, max: API_RATE_MAX, standardHeaders: true, legacyHeaders: false })
app.use("/api", limiter)

const csrfProtection = csrf({ cookie: { key: "__pw_csrf", httpOnly: true, sameSite: "lax", secure: true, signed: false } })

const oauth = new OAuth2Client({
  clientId: GOOGLE_CLIENT_ID,
  clientSecret: GOOGLE_CLIENT_SECRET,
  redirectUri: GOOGLE_REDIRECT_URI
})

const signCookie = (payload: object) => {
  const data = Buffer.from(JSON.stringify(payload)).toString("base64url")
  const sig = crypto.createHmac("sha256", COOKIE_SECRET).update(data).digest("base64url")
  return `${data}.${sig}`
}

const verifyCookie = (value: string | undefined) => {
  if (!value) return null
  const [data, sig] = value.split(".")
  if (!data || !sig) return null
  const expect = crypto.createHmac("sha256", COOKIE_SECRET).update(data).digest("base64url")
  if (crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expect))) {
    try { return JSON.parse(Buffer.from(data, "base64url").toString()) } catch { return null }
  }
  return null
}

app.get("/_health", (req, res) => {
  if (HEALTH_KEY && req.headers["x-health-key"] !== HEALTH_KEY) return res.status(401).json({ ok: false })
  res.set("Cache-Control", "no-store").json({ ok: true, ts: Date.now(), env: NODE_ENV })
})

app.get("/robots.txt", (req, res) => {
  res.type("text/plain").send("User-agent: *\nAllow: /\nSitemap: https://www.petwash.co.il/sitemap.xml")
})

app.get("/.well-known/security.txt", (req, res) => {
  res.type("text/plain").send("Contact: mailto:security@petwash.co.il\nExpires: 2030-01-01T00:00:00.000Z\nPreferred-Languages: en\nPolicy: https://www.petwash.co.il/security")
})

app.get("/api/csrf", csrfProtection, (req, res) => {
  res.json({ token: req.csrfToken() })
})

app.get("/api/session", (req, res) => {
  const user = verifyCookie(req.cookies[SESSION_COOKIE])
  res.json({ authenticated: !!user, user: user ? { id: user.id, email: user.email, name: user.name, picture: user.picture } : null })
})

app.post("/api/logout", (req, res) => {
  res.clearCookie(SESSION_COOKIE, { httpOnly: true, sameSite: "lax", secure: true, path: "/" })
  res.status(204).end()
})

app.get("/auth/google", (req, res) => {
  const state = crypto.randomBytes(16).toString("hex")
  const url = oauth.generateAuthUrl({
    access_type: "offline",
    prompt: "consent",
    scope: ["openid", "email", "profile"],
    state
  })
  res.redirect(url)
})

app.get("/auth/google/callback", async (req, res) => {
  try {
    const code = String(req.query.code || "")
    if (!code) return res.status(400).send("Missing code")
    const { tokens } = await oauth.getToken({ code, redirect_uri: GOOGLE_REDIRECT_URI })
    oauth.setCredentials(tokens)
    const idinfo = await oauth.verifyIdToken({ idToken: String(tokens.id_token || ""), audience: GOOGLE_CLIENT_ID })
    const payload = idinfo.getPayload() || {}
    const session = signCookie({
      id: payload.sub,
      email: payload.email,
      name: payload.name,
      picture: payload.picture,
      iat: Date.now()
    })
    res.cookie(SESSION_COOKIE, session, { httpOnly: true, sameSite: "lax", secure: true, path: "/", maxAge: 1000 * 60 * 60 * 24 * 7 })
    const target = req.query.state && typeof req.query.state === "string" && req.query.state.startsWith("/") ? req.query.state : "/"
    res.redirect(String(target))
  } catch (e) {
    res.status(500).send("Auth error")
  }
})

app.get("/api/geocode", limiter, async (req, res) => {
  try {
    const address = String(req.query.q || "")
    if (!address) return res.status(400).json({ error: "q required" })
    const apiKey = process.env.GOOGLE_MAPS_KEY || ""
    if (!apiKey) return res.status(500).json({ error: "maps key missing" })
    const r = await fetch(`https://maps.googleapis.com/maps/api/geocode/json?address=${encodeURIComponent(address)}&key=${apiKey}`)
    const json = await r.json()
    res.json(json)
  } catch {
    res.status(500).json({ error: "failed" })
  }
})

app.use("/assets", express.static(path.join(STATIC_DIR, "assets"), { immutable: true, maxAge: "365d" }))
app.use(express.static(STATIC_DIR, { maxAge: "1h", setHeaders: (res, p) => { if (p.endsWith("index.html")) res.setHeader("Cache-Control", "no-store") } }))

app.use((req, res, next) => {
  if (req.method !== "GET") return next()
  if (req.path.startsWith("/api")) return next()
  if (req.path.startsWith("/auth")) return next()
  if (req.accepts("html")) return res.sendFile(path.join(STATIC_DIR, "index.html"))
  return next()
})

app.use((req, res) => {
  res.status(404).json({ error: "not found" })
})

app.use((err: any, req: express.Request, res: express.Response, _next: express.NextFunction) => {
  const id = crypto.randomBytes(6).toString("hex")
  console.error(`[ERR:${id}]`, err && err.stack ? err.stack : err)
  res.status(500).json({ error: "server", id })
})

app.listen(PORT, () => {
  console.log(`petwash server on ${PORT}`)
})
