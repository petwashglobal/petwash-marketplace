import * as functions from "firebase-functions";
import * as admin from "firebase-admin";

admin.initializeApp();
const db = admin.firestore();
const fcm = admin.messaging();

/* -------------------------
   Tunables
   ------------------------- */
const WAVE_PLAN = [
  { wave: 1, radiusKm: 3, ttlSec: 20, topN: 5, nextAfterSec: 30 },
  { wave: 2, radiusKm: 6, ttlSec: 20, topN: 8, nextAfterSec: 30 },
  { wave: 3, radiusKm: 10, ttlSec: 20, topN: 12, nextAfterSec: 30 },
] as const;

const SCORE_WEIGHTS = {
  distance: 0.50,
  rating: 0.25,
  acceptance: 0.15,
  activity: 0.10,
  premiumBoost: 0.10, // applied as multiplier, not as additive weight
};

/* -------------------------
   Helpers
   ------------------------- */
function tsNow() {
  return admin.firestore.Timestamp.now();
}

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

function haversineKm(a: { lat: number; lng: number }, b: { lat: number; lng: number }) {
  const R = 6371;
  const dLat = (Math.PI / 180) * (b.lat - a.lat);
  const dLng = (Math.PI / 180) * (b.lng - a.lng);
  const lat1 = (Math.PI / 180) * a.lat;
  const lat2 = (Math.PI / 180) * b.lat;
  const x =
    Math.sin(dLat / 2) ** 2 +
    Math.sin(dLng / 2) ** 2 * Math.cos(lat1) * Math.cos(lat2);
  return 2 * R * Math.asin(Math.sqrt(x));
}

function computeScore(params: {
  distanceKm: number;
  ratingAvg: number;
  acceptanceRate: number;
  completedJobs30d: number;
  premium: boolean;
}) {
  // Normalize to 0..1
  const distanceScore = 1 - clamp(params.distanceKm / 10, 0, 1); // best within 0km, worst at 10km+
  const ratingScore = clamp(params.ratingAvg / 5, 0, 1);
  const acceptanceScore = clamp(params.acceptanceRate, 0, 1);
  const activityScore = clamp(params.completedJobs30d / 30, 0, 1); // 30 jobs/month as "high"

  let s =
    distanceScore * SCORE_WEIGHTS.distance +
    ratingScore * SCORE_WEIGHTS.rating +
    acceptanceScore * SCORE_WEIGHTS.acceptance +
    activityScore * SCORE_WEIGHTS.activity;

  // Premium boost as multiplier (feels fairer)
  if (params.premium) s = s * (1 + SCORE_WEIGHTS.premiumBoost);

  return Number(s.toFixed(6));
}

async function writeAudit(type: string, jobId: string, providerId?: string, meta?: any) {
  const id = `${type}_${jobId}_${Date.now()}_${Math.floor(Math.random() * 9999)}`;
  await db.collection("audit").doc(id).set({
    type,
    jobId,
    providerId: providerId || null,
    meta: meta || {},
    at: tsNow(),
  });
}

/* -------------------------
   1) Trigger: Job created -> start wave 1 dispatch
   - Create this call wherever you create a job after payment hold ok
   ------------------------- */
export const dispatch_start = functions.https.onCall(async (data, context) => {
  const uid = context.auth?.uid;
  if (!uid) throw new functions.https.HttpsError("unauthenticated", "Login required");

  const jobId = String(data?.jobId || "");
  if (!jobId) throw new functions.https.HttpsError("invalid-argument", "jobId required");

  const jobRef = db.collection("jobs").doc(jobId);
  const jobSnap = await jobRef.get();
  if (!jobSnap.exists) throw new functions.https.HttpsError("not-found", "job not found");

  const job = jobSnap.data() as any;
  if (job.createdByUid !== uid) {
    throw new functions.https.HttpsError("permission-denied", "Not your job");
  }
  if (!job.payment?.holdOk) {
    throw new functions.https.HttpsError("failed-precondition", "Payment hold not confirmed");
  }
  if (["assigned", "cancelled", "completed"].includes(job.status)) {
    return { ok: true, note: "No dispatch, job already finalized" };
  }

  await writeAudit("JOB_CREATED", jobId, undefined, { serviceType: job.serviceType });

  // Wave 1 now
  await runDispatchWave(jobId, 1);

  return { ok: true };
});

/* -------------------------
   2) Core: Run a dispatch wave
   - Batched to top N by score
   - Creates offers with TTL
   - Sends FCM to those providers
   - Schedules next wave by time check (Firebase-first)
   ------------------------- */
async function runDispatchWave(jobId: string, waveNumber: number) {
  const plan = WAVE_PLAN.find((x) => x.wave === waveNumber);
  if (!plan) return;

  const jobRef = db.collection("jobs").doc(jobId);
  const jobSnap = await jobRef.get();
  if (!jobSnap.exists) return;
  const job = jobSnap.data() as any;

  if (job.status === "assigned" || job.status === "cancelled" || job.status === "completed") return;

  const pickup = job.pickup;
  if (!pickup?.lat || !pickup?.lng) return;

  // Update job dispatch state
  const expiresAt = admin.firestore.Timestamp.fromMillis(Date.now() + plan.ttlSec * 1000);

  await jobRef.set(
    {
      status: "offered",
      updatedAt: tsNow(),
      dispatch: {
        wave: waveNumber,
        radiusKm: plan.radiusKm,
        offeredCount: admin.firestore.FieldValue.increment(0),
        offerExpiresAt: expiresAt,
      },
    },
    { merge: true }
  );

  // Query candidates (Firebase limitation: no geo radius query).
  // Approach: prefilter by city or geohash prefix buckets, then exact distance filter.
  // For simplicity here: query online providers for serviceType. Add city filter in your real code.
  const providersSnap = await db
    .collection("providers")
    .where("status", "==", "online")
    .where("services", "array-contains", job.serviceType)
    .limit(300)
    .get();

  const candidates: {
    providerId: string;
    distanceKm: number;
    score: number;
    fcmTokens: string[];
  }[] = [];

  const now = tsNow();

  providersSnap.forEach((doc) => {
    const p = doc.data() as any;
    const providerId = doc.id;

    // Cooldown
    if (p.cooldownUntil && p.cooldownUntil.toMillis && p.cooldownUntil.toMillis() > now.toMillis()) return;

    // Presence hard checks
    if (!p.geo?.lat || !p.geo?.lng) return;
    if (!p.lastSeenAt) return;

    // Stale presence: must be seen within last 3 minutes
    const lastSeenMs = p.lastSeenAt.toMillis();
    if (Date.now() - lastSeenMs > 3 * 60 * 1000) return;

    const d = haversineKm(
      { lat: pickup.lat, lng: pickup.lng },
      { lat: p.geo.lat, lng: p.geo.lng }
    );
    if (d > plan.radiusKm) return;

    const premium = !!(p.premiumBadge && p.subscriptionActive);
    const score = computeScore({
      distanceKm: d,
      ratingAvg: Number(p.ratingAvg || 0),
      acceptanceRate: Number(p.acceptanceRate || 0),
      completedJobs30d: Number(p.completedJobs30d || 0),
      premium,
    });

    const tokens = Array.isArray(p.fcmTokens) ? p.fcmTokens.filter(Boolean) : [];
    candidates.push({ providerId, distanceKm: d, score, fcmTokens: tokens });
  });

  // Pick top N
  candidates.sort((a, b) => b.score - a.score);
  const chosen = candidates.slice(0, plan.topN);

  // Create offers (id is providerId, so
