-- Policy versions (immutable history)
CREATE TABLE IF NOT EXISTS policy_versions (
  id SERIAL PRIMARY KEY,
  policy_key VARCHAR(100) NOT NULL,
  version INT NOT NULL,
  config JSONB NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'draft', -- draft | active | archived
  created_at TIMESTAMP DEFAULT NOW(),
  activated_at TIMESTAMP,
  created_by TEXT
);

-- Rollout tracking
CREATE TABLE IF NOT EXISTS policy_rollouts (
  id SERIAL PRIMARY KEY,
  policy_key VARCHAR(100) NOT NULL,
  version INT NOT NULL,
  scope_type VARCHAR(20) NOT NULL, -- global | franchise | station | ownership
  scope_key TEXT, -- franchise_owner_id / station_id / ownershipType
  rollout_status VARCHAR(20) NOT NULL DEFAULT 'planned', -- planned | active | paused | rolled_back | completed
  start_date TIMESTAMP,
  end_date TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW(),
  created_by TEXT
);

-- Rollout evaluation (links to 12.22 outcomes)
CREATE TABLE IF NOT EXISTS policy_rollout_evaluations (
  id SERIAL PRIMARY KEY,
  rollout_id INT,
  measured_at TIMESTAMP DEFAULT NOW(),
  success_rate DECIMAL(5,2),
  avg_margin_delta DECIMAL(6,2),
  avg_friction_delta DECIMAL(6,2),
  sample_size INT,
  FOREIGN KEY (rollout_id) REFERENCES policy_rollouts(id)
);

import { computeOutcomeSummary } from "./outcome-measurement";

export function evaluateRollout(outcomes: any, rollout: any) {
  const relevant = outcomes.caseOutcomes.filter((c: any) => {
    if (!c.hasBaseline) return false;

    if (rollout.scope_type === "station") {
      return String(c.entityId) === rollout.scope_key;
    }
    if (rollout.scope_type === "franchise") {
      return c.entityType === "network";
    }
    return true;
  });

  const resolved = relevant.filter((c: any) => c.status === "resolved");

  const success = resolved.filter((c: any) => c.verdict === "improved");

  const successRate =
    resolved.length > 0 ? (success.length / resolved.length) * 100 : null;

  const avgMarginDelta =
    resolved.reduce((acc: number, c: any) => acc + (c.marginDelta || 0), 0) /
      (resolved.length || 1);

  const avgFrictionDelta =
    resolved.reduce((acc: number, c: any) => acc + (c.frictionDelta || 0), 0) /
      (resolved.length || 1);

  return {
    successRate,
    avgMarginDelta,
    avgFrictionDelta,
    sampleSize: resolved.length,
  };
}

import express from "express";
import { db } from "../db";
import { evaluateRollout } from "../lib/policy-rollout";
import { computeOutcomeSummary } from "../lib/outcome-measurement";

const router = express.Router();

/* CREATE NEW POLICY VERSION */
router.post("/version", async (req, res) => {
  const { policyKey, config } = req.body;

  const versionResult = await db.query(
    `SELECT COALESCE(MAX(version),0)+1 as v FROM policy_versions WHERE policy_key=$1`,
    [policyKey]
  );

  const version = versionResult.rows[0].v;

  const result = await db.query(
    `INSERT INTO policy_versions (policy_key, version, config, created_by)
     VALUES ($1,$2,$3,$4) RETURNING *`,
    [policyKey, version, config, "admin"]
  );

  res.json({ policy: result.rows[0] });
});

/* ACTIVATE POLICY */
router.post("/version/:id/activate", async (req, res) => {
  const { id } = req.params;

  await db.query(
    `UPDATE policy_versions SET status='archived' WHERE policy_key =
     (SELECT policy_key FROM policy_versions WHERE id=$1)`,
    [id]
  );

  const result = await db.query(
    `UPDATE policy_versions SET status='active', activated_at=NOW()
     WHERE id=$1 RETURNING *`,
    [id]
  );

  res.json({ policy: result.rows[0] });
});

/* CREATE ROLLOUT */
router.post("/rollout", async (req, res) => {
  const { policyKey, version, scopeType, scopeKey } = req.body;

  const result = await db.query(
    `INSERT INTO policy_rollouts (policy_key, version, scope_type, scope_key, start_date)
     VALUES ($1,$2,$3,$4,NOW()) RETURNING *`,
    [policyKey, version, scopeType, scopeKey]
  );

  res.json({ rollout: result.rows[0] });
});

/* ROLLBACK */
router.post("/rollout/:id/rollback", async (req, res) => {
  const { id } = req.params;

  await db.query(
    `UPDATE policy_rollouts SET rollout_status='rolled_back', end_date=NOW()
     WHERE id=$1`,
    [id]
  );

  res.json({ status: "rolled_back" });
});

/* EVALUATE ROLLOUT */
router.get("/rollout/:id/evaluate", async (req, res) => {
  const { id } = req.params;

  const rolloutRes = await db.query(
    `SELECT * FROM policy_rollouts WHERE id=$1`,
    [id]
  );

  const rollout = rolloutRes.rows[0];

  const outcomes = await computeOutcomeSummary();

  const evaluation = evaluateRollout(outcomes, rollout);

  res.json({ evaluation });
});

export default router;

import { useEffect, useState } from "react";

export default function PolicyRollout() {
  const [rollouts, setRollouts] = useState<any[]>([]);

  useEffect(() => {
    fetch("/api/expansion/policy-rollout")
      .then(r => r.json())
      .then(setRollouts);
  }, []);

  return (
    <div>
      <h1>Policy Rollout Control</h1>

      {rollouts.map(r => (
        <div key={r.id} style={{ border: "1px solid #ccc", margin: 10, padding: 10 }}>
          <div>Policy: {r.policy_key} v{r.version}</div>
          <div>Status: {r.rollout_status}</div>
          <div>Scope: {r.scope_type} - {r.scope_key}</div>
        </div>
      ))}
    </div>
  );
}
