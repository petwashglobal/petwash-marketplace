/**
 * PET WAG / PETWASH PLATFORM 2026 - Auth + Onboarding (Hardened)
 * - Role via Firebase Custom Claims (server authority)
 * - Token refresh after claim set
 * - Firestore: user can only edit safe fields
 * - Address: Places Autocomplete with fetchDetails
 */

import React, { useEffect, useMemo, useState } from "react";
import { View, Text, Button, ActivityIndicator, Platform } from "react-native";
import auth from "@react-native-firebase/auth";
import firestore from "@react-native-firebase/firestore";
import functions from "@react-native-firebase/functions";
import { GooglePlacesAutocomplete } from "react-native-google-places-autocomplete";
import * as AppleAuthentication from "expo-apple-authentication";
import { GoogleSignin } from "@react-native-google-signin/google-signin";

type Role = "public" | "pending_staff" | "staff" | "admin" | "hr" | "management";

const SAFE_PROFILE_FIELDS = ["displayName", "phone", "marketingOptIn", "locale"] as const;

export default function PetWagOnboarding() {
  const [user, setUser] = useState<ReturnType<typeof auth>["currentUser"] | null>(null);
  const [role, setRole] = useState<Role>("public");
  const [loading, setLoading] = useState(true);

  // Configure Google sign-in once
  useEffect(() => {
    GoogleSignin.configure({
      // webClientId is required for idToken on Android
      // Put it in env/config, do NOT hardcode in repo
      webClientId: process.env.GOOGLE_WEB_CLIENT_ID,
    });
  }, []);

  // Listen for auth changes + claims
  useEffect(() => {
    const unsub = auth().onAuthStateChanged(async (u) => {
      try {
        setUser(u);
        if (!u) {
          setRole("public");
          return;
        }

        // Force refresh to pick up newly set claims after signup
        const tokenRes = await u.getIdTokenResult(true);
        setRole((tokenRes.claims.role as Role) || "public");
      } finally {
        setLoading(false);
      }
    });
    return unsub;
  }, []);

  const AddressInput = useMemo(
    () => (
      <GooglePlacesAutocomplete
        placeholder="Home address"
        fetchDetails={true}
        onPress={(data, details) => {
          const formatted =
            details?.formatted_address ||
            (details as any)?.description ||
            data?.description ||
            "";

          // Recommendation: store only what you need (city/suburb) not full address
          // Example below stores full formatted address as "addressLabel"
          if (formatted) {
            firestore().collection("users").doc(auth().currentUser?.uid).set(
              {
                addressLabel: formatted,
                addressUpdatedAt: firestore.FieldValue.serverTimestamp(),
              },
              { merge: true }
            );
          }
        }}
        query={{
          key: process.env.GOOGLE_PLACES_API_KEY,
          language: "en",
          types: "address",
        }}
        styles={{
          textInput: { height: 50, borderWidth: 1, borderColor: "#ddd", borderRadius: 12, paddingHorizontal: 12 },
          listView: { borderRadius: 12 },
        }}
      />
    ),
    []
  );

  const handleGoogle = async () => {
    try {
      const { idToken } = await GoogleSignin.signIn();
      const cred = auth.GoogleAuthProvider.credential(idToken);
      await auth().signInWithCredential(cred);

      // Trigger a callable to ensure profile initialization (more reliable than onCreate alone)
      await functions().httpsCallable("ensureUserProfile")({});
      await auth().currentUser?.getIdToken(true);
    } catch (e: any) {
      console.log("Google auth error:", e?.message || e);
    }
  };

  const handleApple = async () => {
    try {
      if (Platform.OS !== "ios") return;

      const appleCred = await AppleAuthentication.signInAsync({
        requestedScopes: [
          AppleAuthentication.AppleAuthenticationScope.FULL_NAME,
          AppleAuthentication.AppleAuthenticationScope.EMAIL,
        ],
      });

      const { identityToken, nonce } = appleCred;
      if (!identityToken) throw new Error("No Apple identityToken");

      const provider = new auth.AppleAuthProvider();
      const credential = provider.credential(identityToken, nonce);
      await auth().signInWithCredential(credential);

      await functions().httpsCallable("ensureUserProfile")({});
      await auth().currentUser?.getIdToken(true);
    } catch (e: any) {
      console.log("Apple auth error:", e?.message || e);
    }
  };

  const updateSafeProfile = async (patch: Record<string, any>) => {
    const uid = auth().currentUser?.uid;
    if (!uid) return;

    // Only allow safe fields client-side
    const safePatch: Record<string, any> = {};
    for (const k of SAFE_PROFILE_FIELDS) {
      if (patch[k] !== undefined) safePatch[k] = patch[k];
    }

    await firestore().collection("users").doc(uid).set(safePatch, { merge: true });
  };

  if (loading) return <ActivityIndicator />;

  return (
    <View style={{ padding: 20, gap: 12 }}>
      {!user ? (
        <View style={{ gap: 10 }}>
          <Text style={{ fontSize: 18, fontWeight: "600" }}>Welcome. Join our loyalty program (free).</Text>
          <Button title="Sign in with Google" onPress={handleGoogle} />
          {Platform.OS === "ios" ? <Button title="Sign in with Apple" onPress={handleApple} /> : null}
        </View>
      ) : (
        <View style={{ gap: 12 }}>
          <Text style={{ fontSize: 16 }}>Signed in as: {user.email || user.uid}</Text>
          <Text style={{ fontSize: 16 }}>Role: {role}</Text>

          {role === "public" && (
            <View style={{ gap: 12 }}>
              <Text>Welcome gift: points will appear after profile init.</Text>
              {AddressInput}
              <Button
                title="Opt in to notifications"
                onPress={() => updateSafeProfile({ marketingOptIn: true })}
              />
            </View>
          )}

          {["admin", "hr", "management", "staff", "pending_staff"].includes(role) && (
            <View style={{ padding: 12, borderWidth: 1, borderColor: "#f2c", borderRadius: 12 }}>
              <Text style={{ fontWeight: "700" }}>Staff dashboard</Text>
              <Text>Access controlled by server claims (not UI).</Text>
            </View>
          )}

          <Button title="Sign out" onPress={() => auth().signOut()} />
        </View>
      )}
    </View>
  );
}
