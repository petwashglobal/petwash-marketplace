export interface ConsentEventRecord {
  id: string;                        // UUID
  userId?: string;                   // internal user id if logged in
  sessionId?: string;
  jurisdiction: LegalJurisdiction;
  channel: ConsentChannel;
  createdAt: string;                 // ISO date string
  ipAddress?: string;
  userAgent?: string;
  // For cookie choices:
  cookiesAccepted: CookieType[];
  cookiesRejected: CookieType[];
  // For form based consent:
  agreedToPrivacyPolicy?: boolean;
  agreedToTermsOfUse?: boolean;
  marketingOptInEmail?: boolean;
  marketingOptInSms?: boolean;
  // Technical proof:
  snapshotHash?: string;             // hash of current privacy text at time of consent
}

/**
 * Helper to create a new consent record, to be saved into DB by the backend.
 * The backend developer should connect this to Postgres or another DB.
 */
import crypto from "crypto";

export function createConsentEvent(
  partial: Omit<ConsentEventRecord, "id" | "createdAt" | "snapshotHash">
): ConsentEventRecord {
  const now = new Date().toISOString();

  const hashSource =
    JSON.stringify({
      ...partial,
      version: "2025.01",
      law: IsraelPrivacyCompliance2025.lawName,
      amendment: IsraelPrivacyCompliance2025.amendment,
    }) + now;

  const snapshotHash = crypto.createHash("sha256").update(hashSource).digest("hex");

  return {
    id: crypto.randomUUID(),
    createdAt: now,
    snapshotHash,
    ...partial,
  };
}

/* 5. Data subject rights: access, correction, deletion                       */

export type DataSubjectRequestType =
  | "ACCESS"    // request to see what data is stored
  | "CORRECT"   // request to correct inaccurate data
  | "DELETE"    // right to be forgotten where applicable
  | "OPPOSE"    // objection to certain processing
  | "PORTABILITY";

export type DataSubjectRequestStatus =
  | "RECEIVED"
  | "IN_REVIEW"
  | "ADDITIONAL_INFO_REQUIRED"
  | "APPROVED"
  | "REJECTED"
  | "COMPLETED";

export interface DataSubjectRequestRecord {
  id: string;                // UUID
  jurisdiction: LegalJurisdiction;
  type: DataSubjectRequestType;
  fullName: string;
  email: string;
  phone?: string;
  userId?: string;           // internal system id if available
  description?: string;      // free text from user
  createdAt: string;         // ISO
  updatedAt: string;         // ISO
  status: DataSubjectRequestStatus;
  handledBy?: string;        // internal staff id or email
  resolutionNote?: string;
}

/**
 * Helper functions for DSAR (data subject access requests).
 * Again, real DB writes must be implemented by the backend.
 */

export function createDataSubjectRequest(input: {
  jurisdiction: LegalJurisdiction;
  type: DataSubjectRequestType;
  fullName: string;
  email: string;
  phone?: string;
  userId?: string;
  description?: string;
}): DataSubjectRequestRecord {
  const now = new Date().toISOString();
  return {
    id: crypto.randomUUID(),
    jurisdiction: input.jurisdiction,
    type: input.type,
    fullName: input.fullName,
    email: input.email,
    phone: input.phone,
    userId: input.userId,
    description: input.description,
    createdAt: now,
    updatedAt: now,
    status: "RECEIVED",
  };
}

export function updateDataSubjectRequestStatus(
  record: DataSubjectRequestRecord,
  status: DataSubjectRequestStatus,
  handledBy?: string,
  resolutionNote?: string
): DataSubjectRequestRecord {
  return {
    ...record,
    status,
    handledBy: handledBy ?? record.handledBy,
    resolutionNote: resolutionNote ?? record.resolutionNote,
    updatedAt: new Date().toISOString(),
  };
}
