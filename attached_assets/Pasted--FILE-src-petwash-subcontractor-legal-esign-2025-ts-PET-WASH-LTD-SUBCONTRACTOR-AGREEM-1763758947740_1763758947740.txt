  email: z.string().email(),
  phone: z.string().min(6).max(50).optional(),

  clientDevice: clientDeviceInfoSchema.optional(),
  agreedToPrivacy: z.boolean(),
  agreedToTerms: z.boolean()
});

export type SignSubcontractorAgreementRequest =
  z.infer<typeof signSubcontractorAgreementRequestSchema>;

/* ===========================================
 * 6. SIGNATURE CREATION SERVICE
 * ===========================================
 */

export function createSubcontractorSignatureRecord(
  input: SignSubcontractorAgreementRequest & {
    ipAddress?: string;
    userAgent?: string;
    agreement?: SubcontractorAgreementTemplate;
  }
): SubcontractorSignatureRecord {
  const now = new Date().toISOString();
  const agreement = input.agreement ?? SUBCONTRACTOR_AGREEMENT_IL_2025;

  // Create a strong hash as legal fingerprint:
  // combine agreement version, full name, subcontractor id, rawSignatureData, timestamp.
  const hash = crypto
    .createHash("sha256")
    .update(
      [
        agreement.version,
        input.subcontractorId,
        input.fullName,
        input.email,
        input.rawSignatureData,
        now
      ].join("|")
    )
    .digest("hex");

  const truncated =
    input.rawSignatureData.length > 200
      ? input.rawSignatureData.slice(0, 200)
      : input.rawSignatureData;

  return {
    id: crypto.randomUUID(),
    subcontractorId: input.subcontractorId,
    agreementVersion: agreement.version,
    agreementLanguage: agreement.language,
    signedAt: now,
    ipAddress: input.ipAddress,
    userAgent: input.userAgent,
    clientDevice: input.clientDevice ?? {},
    signatureMethod: input.signatureMethod,
    signatureHash: hash,
    rawSignatureDataTruncated: truncated,
    agreementSnapshotJson: JSON.stringify(agreement),
    agreedToPrivacy: input.agreedToPrivacy,
    agreedToTerms: input.agreedToTerms,
    auditTrailId: `audit_${Date.now()}_${Math.floor(Math.random() * 999999)}`
  };
}

/* ===========================================

export interface SubcontractorSignatureRepository {
  saveSignature(
    record: SubcontractorSignatureRecord
  ): Promise<SubcontractorSignatureRecord>;

  getSignatureById(
    id: string
  ): Promise<SubcontractorSignatureRecord | null>;

  listSignaturesForSubcontractor(
    subcontractorId: string
  ): Promise<SubcontractorSignatureRecord[]>;
}

/**
 * Example in memory implementation for development only.
 * Production must use a real DB implementation.
 */
export class InMemorySubcontractorSignatureRepository
  implements SubcontractorSignatureRepository
{
  private records: SubcontractorSignatureRecord[] = [];

  async saveSignature(
    record: SubcontractorSignatureRecord
  ): Promise<SubcontractorSignatureRecord> {
    this.records.push(record);
    return record;
  }

  async getSignatureById(
    id: string
  ): Promise<SubcontractorSignatureRecord | null> {
    return this.records.find((r) => r.id === id) ?? null;
  }

  async listSignaturesForSubcontractor(
    subcontractorId: string
  ): Promise<SubcontractorSignatureRecord[]> {
    return this.records.filter((r) => r.subcontractorId === subcontractorId);
  }
}

/**
 * Extract IP and user agent in a consistent way.
 * Works behind proxies if trust proxy is configured in Express.
 */
function extractRequestContext(req: Request): {
  ipAddress?: string;
  userAgent?: string;
} {
  const ip =
    (req.headers["x-forwarded-for"] as string | undefined)?.split(",")[0]?.trim() ||
    req.ip ||
    undefined;
  const userAgent = (req.headers["user-agent"] as string | undefined) ?? undefined;
  return { ipAddress: ip, userAgent };
}

/**
 * Wrap async handlers to catch errors.
 */
export function asyncHandler<
  P = any,
  ResBody = any,
  ReqBody = any,
  ReqQuery = any
>(fn: (req: Request<P, ResBody, ReqBody, ReqQuery>, res: Response, next: NextFunction) => Promise<any>) {
  return (req: Request<P, ResBody, ReqBody, ReqQuery>, res: Response, next: NextFunction) =>
    fn(req, res, next).catch(next);
}

 * Endpoints:
 * - POST /api/subcontractors/agreements/2025/sign
 * - GET  /api/subcontractors/agreements/:signatureId
 * - GET  /api/subcontractors/agreements/by-subcontractor/:subcontractorId
 */

export function buildSubcontractorAgreementRouter(
  repo: SubcontractorSignatureRepository
) {
  const express = require("express");
  const router = express.Router();

  /**
   * POST - sign subcontractor agreement 2025
   * Body: SignSubcontractorAgreementRequest
   * Returns: minimal data for confirmation screen.
   */
  router.post(
    "/agreements/2025/sign",
    asyncHandler(async (req: Request, res: Response) => {
      const parseResult = signSubcontractorAgreementRequestSchema.safeParse(req.body);
      if (!parseResult.success) {
        return res.status(400).json({
          ok: false,
          error: "Validation error",
          details: parseResult.error.flatten()
        });
      }

      const input = parseResult.data;
      if (!input.agreedToPrivacy || !input.agreedToTerms) {
        return res.status(400).json({
          ok: false,
          error: "Must agree to privacy and terms before signing."
        });
      }

      const { ipAddress, userAgent } = extractRequestContext(req);

      const record = createSubcontractorSignatureRecord({
        ...input,
        ipAddress,
        userAgent,
        agreement: SUBCONTRACTOR_AGREEMENT_IL_2025
      });

      const saved = await repo.saveSignature(record);

      return res.status(201).json({
        ok: true,
        data: {
          signatureId: saved.id,
          agreementVersion: saved.agreementVersion,
          signedAt: saved.signedAt,
          auditTrailId: saved.auditTrailId
        }
      });
    })
  );

  /**
   * GET - single signature by id
   * For control panel or legal export.
   */
  router.get(
    "/agreements/:signatureId",
    asyncHandler(async (req: Request, res: Response) => {
      const { signatureId } = req.params;
      if (!signatureId) {
        return res.status(400).json({ ok: false, error: "Missing signatureId" });
      }

      const record = await repo.getSignatureById(signatureId);
      if (!record) {
        return res.status(404).json({ ok: false, error: "Signature not found" });
      }

      return res.json({
        ok: true,
        data: {
          signature: record,
          agreement: JSON.parse(record.agreementSnapshotJson)
        }
      });
    })
  );

  /**
   * GET - list signatures for a subcontractor
   * For "legal documents" page inside subcontractor portal.
   */
  router.get(
    "/agreements/by-subcontractor/:subcontractorId",
    asyncHandler(async (req: Request, res: Response) => {
      const { subcontractorId } = req.params;
      if (!subcontractorId) {
        return res.status(400).json({ ok: false, error: "Missing subcontractorId" });
      }

      const records = await repo.listSignaturesForSubcontractor(subcontractorId);
      return res.json({
        ok: true,
        data: records
      });
    })
  );

  return router;
}
