import { initTRPC, TRPCError } from "@trpc/server";
import { z } from "zod";
import { auth } from "@clerk/nextjs/server";

// --- CONTEXT (Biometric Check) ---
export const createTRPCContext = async (opts: { headers: Headers }) => {
  const session = auth();
  return {
    userId: session.userId,
    role: session.sessionClaims?.metadata?.role || "arm_operator",
    biometricVerified: session.sessionClaims?.amr?.includes("face"), // Checks if FaceID was used
    ...opts,
  };
};

const t = initTRPC.context<typeof createTRPCContext>().create();

// --- MIDDLEWARE ---
const isBiometricAdmin = t.middleware(({ ctx, next }) => {
  // STRICT: Must be Admin AND have used FaceID/Fingerprint to login
  if (ctx.role !== "admin" || !ctx.biometricVerified) {
    throw new TRPCError({ code: "FORBIDDEN", message: "Biometric Admin Auth Required" });
  }
  return next();
});

// --- DATA SCHEMAS (Zod) ---
const EmergencySchema = z.object({
  code: z.string().min(8), // e.g., "RED-OCTOPUS-2025"
  reason: z.string(),
  targetRegion: z.enum(["GLOBAL", "EU", "US", "APAC"]),
});

// --- ROUTER ---
export const appRouter = t.router({
  
  // ARM 1: FRANCHISE REPORTING
  franchise: t.router({
    reportMetrics: t.procedure
      .input(z.object({ 
        dailyWashCount: z.number(), 
        revenue: z.number(),
        luxuryUpsells: z.number() // Count of "Gold Shampoo" used
      }))
      .mutation(({ input }) => {
        // AI Logic: Detect if luxury upsells are trending down
        if (input.luxuryUpsells < 5) console.log("⚠️ ALERT: Low Luxury Sales");
        return { status: "RECEIVED" };
      }),
  }),

  // HEAD: EMERGENCY PROTOCOL (The "Red Button")
  admin: t.router({
    executeProtocol: t.procedure
      .use(isBiometricAdmin) // 🔒 SECURE
      .input(EmergencySchema)
      .mutation(({ input }) => {
        console.log(`🚨 EMERGENCY PROTOCOL: ${input.code} TRIGGERED BY BIO-ADMIN`);
        
        // 1. Lock all frontend dashboards
        // 2. Send push notification to all mobile apps
        // 3. Killswitch payment gateways
        
        return { 
          systemStatus: "LOCKDOWN", 
          timestamp: new Date().toISOString(),
          deployedBy: "FACE_ID_VERIFIED_USER"
        };
      }),
  }),
});

export type AppRouter = typeof appRouter;
