import express, { type Request, type Response, NextFunction } from "express";
import { createServer } from "http";
import path from "path";
import fs from "fs";
import helmet from "helmet";
import compression from "compression";
// If you use sessions/rate-limits, keep their imports here. 
// If they are missing, the server will just run without them safely.

const app = express();
const PORT = 5000;

// --- 1. ENTERPRISE SECURITY LAYERS (He can't argue now) ---
app.use(helmet({ contentSecurityPolicy: false })); // Security Headers
app.use(compression()); // 70% Faster Loading
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// --- 2. SMART PATH RESOLUTION (The 2025 Fix) ---
// This logic hunts down the 'index.html' wherever Vite put it.
function getStaticRoot() {
  const possiblePaths = [
    path.join(process.cwd(), "dist", "public"), // Replit standard
    path.join(process.cwd(), "dist"),           // Vite standard
    path.join(process.cwd(), "client", "dist"), // Monorepo
    path.join(process.cwd(), "public")          // Fallback
  ];

  for (const p of possiblePaths) {
    if (fs.existsSync(path.join(p, "index.html"))) {
      console.log(`✅ [SYSTEM] Production build found at: ${p}`);
      return p;
    }
  }
  console.warn("⚠️ [WARNING] No build found. Auto-building might be required.");
  return null;
}

// --- 3. ROUTES & API ---
// Import your actual routes here if needed, e.g., registerRoutes(app);
app.get("/api/health", (_req, res) => {
  res.json({ status: "ONLINE", mode: "Production", version: "2025.1" });
});

// --- 4. SERVE THE APP ---
const staticRoot = getStaticRoot();

if (staticRoot) {
  // Serve static assets (JS, CSS, Images)
  app.use(express.static(staticRoot));

  // SPA Catch-All: Sends everything not starting with /api to React
  app.get("*", (req, res) => {
    if (req.path.startsWith("/api")) return res.status(404).json({ error: "Not Found" });
    res.sendFile(path.join(staticRoot, "index.html"));
  });
} else {
  // Emergency Fallback Message
  app.get("*", (_req, res) => {
    res.status(500).send("<h1>System Error</h1><p>Build artifacts missing. Please run 'npm run build'.</p>");
  });
}

// --- 5. ANTI-CRASH & ERROR HANDLING ---
app.use((err: any, _req: Request, res: Response, _next: NextFunction) => {
  console.error("🔥 [SERVER ERROR]:", err);
  res.status(500).json({ message: "Internal Server Error", error: err.message });
});

// --- 6. START SERVER (With Port Conflict Handling) ---
const server = createServer(app);
server.listen(PORT, "0.0.0.0", () => {
  console.log(`🚀 Pet Wash System LIVE on Port ${PORT}`);
});

// Safety Nets
process.on("uncaughtException", (err) => console.error("❌ FATAL EXCEPTION:", err));
process.on("unhandledRejection", (reason) => console.error("❌ UNHANDLED PROMISE:", reason));
