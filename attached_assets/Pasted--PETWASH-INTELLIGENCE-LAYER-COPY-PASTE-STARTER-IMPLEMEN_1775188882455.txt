export function normalizeEmail(email?: string | null): string | null {
  if (!email) return null;
  return email.trim().toLowerCase();
}

export function normalizePhone(phone?: string | null): string | null {
  if (!phone) return null;
  return phone.replace(/\s+/g, "").trim();
}

export function cleanName(value?: string | null): string | null {
  if (!value) return null;
  return value.replace(/\s+/g, " ").trim();
}

export function toRadians(value: number): number {
  return (value * Math.PI) / 180;
}

export function calculateDistanceKm(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number
): number {
  const earthRadiusKm = 6371;
  const dLat = toRadians(lat2 - lat1);
  const dLng = toRadians(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRadians(lat1)) *
      Math.cos(toRadians(lat2)) *
      Math.sin(dLng / 2) ** 2;

  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return earthRadiusKm * c;
}

export function buildGoogleMapsUrl(lat: number, lng: number): string {
  return `https://www.google.com/maps?q=${lat},${lng}`;
}

export function buildWazeUrl(lat: number, lng: number): string {
  return `https://waze.com/ul?ll=${lat},${lng}&navigate=yes`;
}

interface DbClient {
  query<T = any>(sql: string, params?: any[]): Promise<{ rows: T[] }>;
}

export class AddressRepository {
  constructor(private db: DbClient) {}

  async listByUser(userId: string): Promise<SavedAddress[]> {
    const result = await this.db.query<SavedAddress>(
      `
      SELECT
        id,
        user_id as "userId",
        label,
        formatted_address as address,
        lat,
        lng,
        postcode,
        city,
        country,
        is_default as "isDefault",
        usage_count as "usageCount",
        created_at as "createdAt",
        updated_at as "updatedAt"
      FROM user_addresses
      WHERE user_id = $1
      ORDER BY is_default DESC, usage_count DESC, updated_at DESC
      `,
      [userId]
    );
    return result.rows;
  }

  async upsertAddress(userId: string, input: AddressInput): Promise<SavedAddress> {
    const nearby = await this.db.query<SavedAddress>(
      `
      SELECT
        id,
        user_id as "userId",
        label,
        formatted_address as address,
        lat,
        lng,
        postcode,
        city,
        country,
        is_default as "isDefault",
        usage_count as "usageCount",
        created_at as "createdAt",
        updated_at as "updatedAt"
      FROM user_addresses
      WHERE user_id = $1
      `,
      [userId]
    );

    const existing = nearby.rows.find((row) => {
      const distanceKm = calculateDistanceKm(row.lat, row.lng, input.lat, input.lng);
      return distanceKm < 0.05;
    });

    if (existing) {
      const updated = await this.db.query<SavedAddress>(
        `
        UPDATE user_addresses
        SET
          label = COALESCE($2, label),
          formatted_address = $3,
          lat = $4,
          lng = $5,
          postcode = $6,
          city = $7,
          country = $8,
          usage_count = usage_count + 1,
          updated_at = NOW()
        WHERE id = $1
        RETURNING
          id,
          user_id as "userId",
          label,
          formatted_address as address,
          lat,
          lng,
          postcode,
          city,
          country,
          is_default as "isDefault",
          usage_count as "usageCount",
          created_at as "createdAt",
          updated_at as "updatedAt"
        `,
        [
          existing.id,
          input.label ?? null,
          input.address,
          input.lat,
          input.lng,
          input.postcode ?? null,
          input.city ?? null,
          input.country ?? null,
        ]
      );
      return updated.rows[0];
    }

    const inserted = await this.db.query<SavedAddress>(
      `
      INSERT INTO user_addresses (
        user_id, label, formatted_address, lat, lng, postcode, city, country
      )
      VALUES ($1,$2,$3,$4,$5,$6,$7,$8)
      RETURNING
        id,
        user_id as "userId",
        label,
        formatted_address as address,
        lat,
        lng,
        postcode,
        city,
        country,
        is_default as "isDefault",
        usage_count as "usageCount",
        created_at as "createdAt",
        updated_at as "updatedAt"
      `,
      [
        userId,
        input.label ?? null,
        input.address,
        input.lat,
        input.lng,
        input.postcode ?? null,
        input.city ?? null,
        input.country ?? null,
      ]
    );
    return inserted.rows[0];
  }

  async setDefault(userId: string, addressId: string): Promise<void> {
    await this.db.query(`UPDATE user_addresses SET is_default = FALSE WHERE user_id = $1`, [userId]);
    await this.db.query(
      `UPDATE user_addresses SET is_default = TRUE, updated_at = NOW() WHERE id = $1 AND user_id = $2`,
      [addressId, userId]
    );
  }

  async remove(userId: string, addressId: string): Promise<void> {
    await this.db.query(`DELETE FROM user_addresses WHERE id = $1 AND user_id = $2`, [addressId, userId]);
  }
}

export function calculateCustomerTrustScore(input: {
  cancellationRate: number;
  completedBookings: number;
  noShowCount?: number;
}): number {
  let score = 50;
  score += Math.min(input.completedBookings * 1.5, 25);
  score -= Math.min(input.cancellationRate * 50, 25);
  score -= Math.min((input.noShowCount ?? 0) * 10, 20);
  return Math.max(0, Math.min(100, score));
}

export function calculateProviderTrustScore(provider: ProviderProfile): number {
  let score = 50;
  score += Math.min(provider.ratingAverage * 8, 35);
  score += provider.verified ? 10 : 0;
  score += Math.min(provider.completedBookings / 20, 10);
  score -= Math.min(provider.cancellationRate * 40, 25);
  return Math.max(0, Math.min(100, score));
}

export function calculateBehaviorScore(input: {
  bookingHistoryCount: number;
  repeatUsage?: number;
  recentActivityDaysAgo?: number;
}): number {
  let score = 40;
  score += Math.min(input.bookingHistoryCount * 2, 30);
  score += Math.min((input.repeatUsage ?? 0) * 3, 20);

  if ((input.recentActivityDaysAgo ?? 999) <= 7) score += 10;
  if ((input.recentActivityDaysAgo ?? 999) > 30) score -= 10;

  return Math.max(0, Math.min(100, score));
}

/* =========================================================
   6. SMART MATCHING ENGINE
========================================================= */

export interface MatchingContext {
  request: BookingRequestInput;
  customerProfile: UserIntelligenceProfile | null;
  candidateProviders: ProviderProfile[];
  previousBookingsWithProviderIds: string[];
  marketplaceSettings: AdminMarketplaceSettings;
}

export function scoreProvider(
  provider: ProviderProfile,
  context: MatchingContext
): ProviderMatchScore {
  const { request, previousBookingsWithProviderIds, marketplaceSettings } = context;

  const effectiveRadiusKm =
    provider.serviceRadiusKm || marketplaceSettings.matchingRadiusKmDefault;

  const providerLat = provider.currentLat;
  const providerLng = provider.currentLng;

  const farAwayDistance = 9999;
  const distanceKm =
    providerLat !== null && providerLng !== null
      ? calculateDistanceKm(request.address.lat, request.address.lng, providerLat, providerLng)
      : farAwayDistance;

  const distanceScore =
    distanceKm <= effectiveRadiusKm
      ? Math.max(0, 30 - distanceKm * 3)
      : 0;

  const availabilityScore = Math.max(0, Math.min(provider.availabilityScore, 20));
  const ratingScore = Math.max(0, Math.min(provider.ratingAverage * 4, 20));

  const repeatCustomerScore = previousBookingsWithProviderIds.includes(provider.userId) ? 12 : 0;

  const responseSpeedScore =
    provider.responseTimeMinutes <= 5
      ? 10
      : provider.responseTimeMinutes <= 15
      ? 7
      : provider.responseTimeMinutes <= 30
      ? 4
      : 1;

  const trustScore = Math.max(0, Math.min(calculateProviderTrustScore(provider) / 10, 10));

  const budgetScore =
    request.budget && request.budget > 0
      ? 5
      : 0;

  const boostScore =
    provider.isBoosted || marketplaceSettings.boostedProviderIds.includes(provider.userId)
      ? 5
      : 0;

  const totalScore =
    distanceScore +
    availabilityScore +
    ratingScore +
    repeatCustomerScore +
    responseSpeedScore +
    trustScore +
    budgetScore +
    boostScore;

  return {
    providerId: provider.userId,
    totalScore,
    distanceKm,
    scoreBreakdown: {
      distanceScore,
      availabilityScore,
      ratingScore,
      repeatCustomerScore,
      responseSpeedScore,
      trustScore,
      budgetScore,
      boostScore,
    },
  };
}

export function rankProviders(context: MatchingContext): RankedProviderResult[] {
  const activeProviders = context.candidateProviders.filter((provider) => {
    if (provider.isPaused) return false;
    if (context.marketplaceSettings.pausedProviderIds.includes(provider.userId)) return false;
    return true;
  });

  const ranked = activeProviders
    .map((provider) => ({
      provider,
      score: scoreProvider(provider, context),
    }))
    .sort((a, b) => b.score.totalScore - a.score.totalScore)
    .slice(0, 3)
    .map((entry, index): RankedProviderResult => ({
      provider: entry.provider,
      score: entry.score,
      bucket:
        index === 0 ? "best_match" :
        index === 1 ? "great_option" :
        "budget_option",
    }));

  return ranked;
}

export async function buildProviderDashboard(
  db: DbClient,
  providerId: string
): Promise<ProviderDashboardData> {
  const todayEarningsResult = await db.query<{ amount: number }>(
    `
    SELECT COALESCE(SUM(provider_payout_amount), 0) as amount
    FROM bookings
    WHERE provider_id = $1
      AND status = 'completed'
      AND DATE(updated_at) = CURRENT_DATE
    `,
    [providerId]
  );

  const upcomingResult = await db.query<{ count: string }>(
    `
    SELECT COUNT(*) as count
    FROM bookings
    WHERE provider_id = $1
      AND status IN ('accepted', 'provider_arriving', 'in_progress')
      AND service_date >= CURRENT_DATE
    `,
    [providerId]
  );

  const completedJobsResult = await db.query<{ count: string }>(
    `
    SELECT COUNT(*) as count
    FROM bookings
    WHERE provider_id = $1
      AND status = 'completed'
    `,
    [providerId]
  );

  const missedBookingsResult = await db.query<{ count: string }>(
    `
    SELECT COUNT(*) as count
    FROM bookings
    WHERE provider_id = $1
      AND status = 'cancelled'
      AND cancelled_by = 'provider'
      AND updated_at >= NOW() - INTERVAL '7 days'
    `,
    [providerId]
  );

  const alerts: string[] = [];
  const suggestions: string[] = [];

  const missedCount = Number(missedBookingsResult.rows[0]?.count ?? 0);
  if (missedCount > 0) {
    alerts.push(`You missed ${missedCount} bookings in the last 7 days`);
  }

  const upcomingBookings = Number(upcomingResult.rows[0]?.count ?? 0);
  if (upcomingBookings < 3) {
    suggestions.push("Increase availability to improve your booking volume");
  }

  if (missedCount >= 2) {
    suggestions.push("Improve response speed and keep calendar up to date");
  }

  return {
    todayEarnings: Number(todayEarningsResult.rows[0]?.amount ?? 0),
    upcomingBookings,
    completedJobs: Number(completedJobsResult.rows[0]?.count ?? 0),
    alerts,
    optimizationSuggestions: suggestions,
  };
}

export async function handleRecoveryTriggers(input: {
  userId: string;
  eventType: "signup_not_completed" | "booking_abandoned" | "inactive_30_days";
  email?: string | null;
}): Promise<void> {
  if (input.eventType === "signup_not_completed") {
    await sendEmail("signup_incomplete_reminder", {
      userId: input.userId,
      email: input.email ?? null,
    });
  }

  if (input.eventType === "booking_abandoned") {
    await sendEmail("booking_abandoned_followup", {
      userId: input.userId,
      email: input.email ?? null,
    });
  }

  if (input.eventType === "inactive_30_days") {
    await sendEmail("reengagement_campaign", {
      userId: input.userId,
      email: input.email ?? null,
    });
  }
}

        paused_provider_ids as "pausedProviderIds",
        created_at as "createdAt",
        updated_at as "updatedAt"
      FROM admin_marketplace_settings
      ORDER BY created_at DESC
      LIMIT 1
      `
    );

    const settings =
      settingsResult.rows[0] ??
      {
        id: "default",
        frictionEnabled: true,
        matchingRadiusKmDefault: 10,
        allowSoftCaptchaFallback: true,
        smsRequiredForProviders: true,
        smsRequiredForCustomers: false,
        boostedProviderIds: [],
        pausedProviderIds: [],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      };

    const ranked = rankProviders({
      request: {
        customerId: booking.customerId,
        providerType: booking.providerType,
        serviceDate: booking.serviceDate,
        serviceTime: booking.serviceTime,
        address: {
          address: booking.addressText,
          lat: booking.lat,
          lng: booking.lng,
          postcode: booking.postcode ?? null,
          city: booking.city ?? null,
          country: booking.country ?? null,
        },
      },
      customerProfile: null,
      candidateProviders: providersResult.rows.map((provider) => ({
        ...provider,
        isBoosted: settings.boostedProviderIds.includes(provider.userId),
        isPaused: settings.pausedProviderIds.includes(provider.userId),
      })),
      previousBookingsWithProviderIds: previousResult.rows.map((row) => row.providerId),
      marketplaceSettings: settings,
    });

    await this.db.query(
      `UPDATE bookings SET status = 'matching', updated_at = NOW() WHERE id = $1`,
      [bookingId]
    );

    await emitBookingEvent(this.eventBus, "matching.started", {
      ...booking,
      status: "matching",
    });

    return ranked;
  }

  async assignProvider(bookingId: string, providerId: string): Promise<BookingRecord> {
    const result = await this.db.query<BookingRecord>(
      `
      UPDATE bookings
      SET provider_id = $2,
          status = 'accepted',
          updated_at = NOW()
      WHERE id = $1
      RETURNING
        id,
        customer_id as "customerId",
        provider_id as "providerId",
        provider_type as "providerType",
        service_date as "serviceDate",
        service_time as "serviceTime",
        address_text as "addressText",
        lat,
        lng,
        postcode,
        city,
        country,
        status,
        created_at as "createdAt",
        updated_at as "updatedAt"
      `,
      [bookingId, providerId]
    );

    const booking = result.rows[0];
    if (!booking) throw new Error("Booking not found");

    await emitBookingEvent(this.eventBus, "provider.accepted", booking);
    return booking;
  }

  async markProviderArriving(bookingId: string): Promise<BookingRecord> {
    const result = await this.db.query<BookingRecord>(
      `
      UPDATE bookings
      SET status = 'provider_arriving',
          updated_at = NOW()
      WHERE id = $1
      RETURNING
        id,
        customer_id as "customerId",
        provider_id as "providerId",
        provider_type as "providerType",
        service_date as "serviceDate",
        service_time as "serviceTime",
        address_text as "addressText",
        lat,
        lng,
        postcode,
        city,
        country,
        status,
        created_at as "createdAt",
        updated_at as "updatedAt"
      `,
      [bookingId]
    );

    const booking = result.rows[0];
    if (!booking) throw new Error("Booking not found");

    await emitBookingEvent(this.eventBus, "provider.arriving", booking);
    return booking;
  }
}

import express, { Request, Response } from "express";

interface AuthenticatedRequest extends Request {
  user?: {
    id: string;
    email?: string | null;
    role?: string | null;
  };
}

export function createIntelligenceRouter(
  db: DbClient,
  eventBus: EventBus
): express.Router {
  const router = express.Router();
  const addressRepo = new AddressRepository(db);
  const bookingService = new BookingIntelligenceService(db, addressRepo, eventBus);

  // Saved addresses
  router.get("/api/addresses", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      const items = await addressRepo.listByUser(req.user.id);
      return res.json({ items });
    } catch (error) {
      console.error("GET /api/addresses failed", error);
      return res.status(500).json({ error: "Failed to load addresses" });
    }
  });

  router.post("/api/addresses", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });

      const input = req.body as AddressInput;
      if (input.lat == null || input.lng == null || !input.address) {
        return res.status(400).json({ error: "Address with lat/lng required" });
      }

      const item = await addressRepo.upsertAddress(req.user.id, input);
      return res.json({ item });
    } catch (error) {
      console.error("POST /api/addresses failed", error);
      return res.status(500).json({ error: "Failed to save address" });
    }
  });

  router.post("/api/addresses/:id/default", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      await addressRepo.setDefault(req.user.id, req.params.id);
      return res.json({ success: true });
    } catch (error) {
      console.error("POST /api/addresses/:id/default failed", error);
      return res.status(500).json({ error: "Failed to set default address" });
    }
  });

  router.delete("/api/addresses/:id", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      await addressRepo.remove(req.user.id, req.params.id);
      return res.json({ success: true });
    } catch (error) {
      console.error("DELETE /api/addresses/:id failed", error);
      return res.status(500).json({ error: "Failed to delete address" });
    }
  });

  // Create booking draft
  router.post("/api/bookings/draft", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });

      const input = req.body as Omit<BookingRequestInput, "customerId">;
      const booking = await bookingService.createDraftBooking({
        ...input,
        customerId: req.user.id,
      });

      return res.json({ booking });
    } catch (error) {
      console.error("POST /api/bookings/draft failed", error);
      return res.status(500).json({ error: "Failed to create booking draft" });
    }
  });

  // Find intelligent matches
  router.post("/api/bookings/:id/match", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });

      const ranked = await bookingService.findMatchesForBooking(req.params.id);

      return res.json({
        bestMatch: ranked.find((x) => x.bucket === "best_match") ?? null,
        greatOption: ranked.find((x) => x.bucket === "great_option") ?? null,
        budgetOption: ranked.find((x) => x.bucket === "budget_option") ?? null,
        ranked,
      });
    } catch (error) {
      console.error("POST /api/bookings/:id/match failed", error);
      return res.status(500).json({ error: "Failed to match providers" });
    }
  });

  // Provider accepts
  router.post("/api/bookings/:id/accept", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      const booking = await bookingService.assignProvider(req.params.id, req.user.id);
      return res.json({ booking });
    } catch (error) {
      console.error("POST /api/bookings/:id/accept failed", error);
      return res.status(500).json({ error: "Failed to accept booking" });
    }
  });

  // Provider arriving
  router.post("/api/bookings/:id/arriving", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      const booking = await bookingService.markProviderArriving(req.params.id);
      return res.json({ booking });
    } catch (error) {
      console.error("POST /api/bookings/:id/arriving failed", error);
      return res.status(500).json({ error: "Failed to update booking state" });
    }
  });

  // Provider dashboard
  router.get("/api/provider/dashboard", async (req: AuthenticatedRequest, res: Response) => {
    try {
      if (!req.user?.id) return res.status(401).json({ error: "Unauthorized" });
      const dashboard = await buildProviderDashboard(db, req.user.id);
      return res.json({ dashboard });
    } catch (error) {
      console.error("GET /api/provider/dashboard failed", error);
      return res.status(500).json({ error: "Failed to load provider dashboard" });
    }
  });

  return router;
}

type MatchResponse = {
  bestMatch: RankedProviderResult | null;
  greatOption: RankedProviderResult | null;
  budgetOption: RankedProviderResult | null;
  ranked: RankedProviderResult[];
};

export function SmartBookingFlow() {
  const [address, setAddress] = useState<AddressInput | null>(null);
  const [serviceType, setServiceType] = useState<ProviderType>("walker");
  const [date, setDate] = useState("");
  const [time, setTime] = useState("");
  const [matches, setMatches] = useState<MatchResponse | null>(null);
  const [loading, setLoading] = useState(false);

  async function handleFindMatch() {
    if (!address) {
      alert("Select address first");
      return;
    }

    setLoading(true);
    try {
      const createResponse = await fetch("/api/bookings/draft", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({
          providerType: serviceType,
          serviceDate: date,
          serviceTime: time,
          address,
        }),
      });

      const createData = await createResponse.json();
      const bookingId = createData.booking.id;

      const matchResponse = await fetch(`/api/bookings/${bookingId}/match`, {
        method: "POST",
        credentials: "include",
      });

      const matchData = await matchResponse.json();
      setMatches(matchData);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ display: "grid", gap: 16 }}>
      <h2>Find the perfect match</h2>

      <select
        value={serviceType}
        onChange={(e) => setServiceType(e.target.value as ProviderType)}
      >
        <option value="walker">Walker</option>
        <option value="sitter">Sitter</option>
        <option value="groomer">Groomer</option>
        <option value="trainer">Trainer</option>
        <option value="transporter">Transporter</option>
      </select>

      <input type="date" value={date} onChange={(e) => setDate(e.target.value)} />
      <input type="time" value={time} onChange={(e) => setTime(e.target.value)} />

      <AddressPicker value={address} onChange={setAddress} />

      <button type="button" onClick={handleFindMatch} disabled={loading}>
        {loading ? "Finding your match..." : "Find my best match"}
      </button>

      {matches && (
        <div style={{ display: "grid", gap: 12 }}>
          {matches.bestMatch && (
            <MatchCard title="Best Match ⭐" match={matches.bestMatch} />
          )}
          {matches.greatOption && (
            <MatchCard title="Great Option" match={matches.greatOption} />
          )}
          {matches.budgetOption && (
            <MatchCard title="Budget Option" match={matches.budgetOption} />
          )}
        </div>
      )}
    </div>
  );
}

function MatchCard({ title, match }: { title: string; match: RankedProviderResult }) {
  const mapsUrl = buildGoogleMapsUrl(
    match.provider.currentLat ?? 0,
    match.provider.currentLng ?? 0
  );
  const wazeUrl = buildWazeUrl(
    match.provider.currentLat ?? 0,
    match.provider.currentLng ?? 0
  );

  return (
    <div style={{ border: "1px solid #ddd", padding: 12, borderRadius: 12 }}>
      <h3>{title}</h3>
      <div><strong>{match.provider.headline}</strong></div>
      <div>Rating: {match.provider.ratingAverage.toFixed(1)}</div>
      <div>Response: {match.provider.responseTimeMinutes} min</div>
      <div>Distance: {match.score.distanceKm.toFixed(1)} km</div>
      <div>Verified: {match.provider.verified ? "Yes" : "No"}</div>
      <div>Total score: {match.score.totalScore.toFixed(1)}</div>

      <div style={{ display: "flex", gap: 8, marginTop: 8 }}>
        <a href={mapsUrl} target="_blank" rel="noreferrer">Google Maps</a>
        <a href={wazeUrl} target="_blank" rel="noreferrer">Waze</a>
      </div>
    </div>
  );
}

export function shouldSkipHeavyOnboarding(input: {
  socialLogin: boolean;
  hasDisplayName: boolean;
  existingRole?: string | null;
}): boolean {
  if (input.existingRole) return true;
  if (input.socialLogin && input.hasDisplayName) return true;
  return false;
}

export function resolvePostLoginRoute(input: {
  socialLogin: boolean;
  hasDisplayName: boolean;
  existingRole?: string | null;
  returnUrl?: string | null;
}): string {
  if (shouldSkipHeavyOnboarding(input)) {
    return input.returnUrl || "/home";
  }
  return "/choose-role";
}
