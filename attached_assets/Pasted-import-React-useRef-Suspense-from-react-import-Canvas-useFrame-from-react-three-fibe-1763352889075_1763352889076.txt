import React, { useRef, Suspense } from 'react';
import { Canvas, useFrame } from '@react-three/fiber/native';
import { useGLTF, Environment, Float, Sparkles, MeshTransmissionMaterial, ContactShadows } from '@react-three/drei/native';
import * as THREE from 'three';

// 👑 7-STAR MATERIALS LIBRARY
const MATERIALS = {
  // 1. "Dubai Gold" - Pure 24k Mirror Finish
  gold: <meshStandardMaterial 
    color="#FFD700" 
    metalness={1} 
    roughness={0.15} 
    envMapIntensity={3} 
  />,

  // 2. "VVS Diamond" - Real Physics Refraction
  diamond: <MeshTransmissionMaterial 
    backside 
    backsideThickness={2} 
    thickness={5} 
    chromaticAberration={1} 
    anisotropy={1} 
    distortion={0.5} 
    distortionScale={0.5} 
    temporalDistortion={0.2} 
    ior={2.4} // Index of Refraction for Diamond
    color="#ffffff" 
    background={new THREE.Color('#ffffff')}
  />
};

// 💎 THE ASSET LOADER
function HighPolyAsset({ type, variant }: { type: string, variant: 'gold' | 'diamond' }) {
  // TEAM: Replace with local .glb file (e.g., require('./assets/crown.glb'))
  // For now, this generic mesh simulates the shape
  const meshRef = useRef<THREE.Mesh>(null);

  useFrame((state) => {
    if (!meshRef.current) return;
    const t = state.clock.getElapsedTime();
    // "Breathing" Animation - Slow, heavy, expensive feel
    meshRef.current.rotation.y = Math.sin(t / 4) * 0.2; 
    meshRef.current.rotation.x = Math.cos(t / 4) * 0.1;
  });

  return (
    <group dispose={null}>
      <mesh ref={meshRef} scale={1.5} castShadow receiveShadow>
        {/* Replace SphereGeometry with your .glb nodes */}
        <sphereGeometry args={[1, 64, 64]} /> 
        {MATERIALS[variant]}
      </mesh>
    </group>
  );
}

// 🎬 THE STAGE
export default function LuxuryEmoji({ type = "crown", variant = "gold" }) {
  return (
    <Canvas 
      shadows 
      camera={{ position: [0, 0, 6], fov: 35 }} 
      style={{ height: 250, width: 250 }} // Square format
    >
      {/* A. CINEMATIC LIGHTING */}
      <ambientLight intensity={0.5} />
      <spotLight position={[10, 10, 10]} angle={0.15} penumbra={1} intensity={2} castShadow />
      
      {/* B. REAL-WORLD REFLECTIONS (The "Shiny" Secret) */}
      <Environment preset="city" /> 

      {/* C. THE GLITTER EFFECT (Particle System) */}
      <Sparkles 
        count={30} 
        scale={3} 
        size={4} 
        speed={0.4} 
        opacity={0.7} 
        color={variant === 'gold' ? "#FFD700" : "#FFFFFF"} 
      />

      {/* D. THE OBJECT */}
      <Suspense fallback={null}>
        <Float speed={1.5} rotationIntensity={0.5} floatIntensity={0.5}>
          <HighPolyAsset type={type} variant={variant} />
        </Float>
      </Suspense>

      {/* E. GROUND SHADOW */}
      <ContactShadows opacity={0.5} scale={10} blur={2} far={4} color="#000000" />
    </Canvas>
  );
}
