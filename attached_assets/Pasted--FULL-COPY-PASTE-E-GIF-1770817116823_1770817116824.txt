function assertNonEmpty(v: any, name: string) {
  if (!v || typeof v !== "string" || !v.trim()) {
    throw new functions.https.HttpsError("invalid-argument", `${name} is required`);
  }
}

function nowIso() {
  return new Date().toISOString();
}

function safeUpperName(name: string) {
  return name.trim().replace(/\s+/g, " ");
}

function getInitials(name: string, locale: "he" | "en") {
  const parts = safeUpperName(name).split(" ").filter(Boolean);
  if (parts.length === 0) return "";
  if (locale === "he") {
    // Hebrew initials: first letters, keep as is (no uppercasing)
    const first = parts[0].slice(0, 1);
    const last = parts.length > 1 ? parts[parts.length - 1].slice(0, 1) : "";
    return (first + (last ? " " + last : "")).trim();
  }
  const first = parts[0].slice(0, 1).toUpperCase();
  const last = parts.length > 1 ? parts[parts.length - 1].slice(0, 1).toUpperCase() : "";
  return (first + (last ? " " + last : "")).trim();
}

function pickFont(locale: "he" | "en") {
  // Front end can load better fonts; for SVG keep safe defaults.
  // If you host custom fonts, you can embed via @font-face in SVG.
  return locale === "he"
    ? `system-ui, -apple-system, "Segoe UI", Arial, sans-serif`
    : `system-ui, -apple-system, "Segoe UI", Inter, Arial, sans-serif`;
}

function isHebrew(locale: string) {
  return (locale || "").toLowerCase().startsWith("he");
}

function formatText(locale: "he" | "en", en: string, he: string) {
  return locale === "he" ? he : en;
}

function cardGradient(style: "black" | "gold" | "green" | "pink") {
  if (style === "gold") return { a: "#a87b2d", b: "#e7c873", c: "#b8892f" };
  if (style === "green") return { a: "#2f5d3a", b: "#4d8a5e", c: "#1f3f2a" };
  if (style === "pink") return { a: "#c34f7b", b: "#f29db7", c: "#8b2e54" };
  return { a: "#0e0e0e", b: "#1b1b1b", c: "#000000" };
}

function svgEscape(s: string) {
  return s
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll(`"`, "&quot;")
    .replaceAll(`'`, "&#39;");
}

   egifts/{cardId}:

     style: "black" | "gold" | "green" | "pink"
     washesTotal: number
     washesRemaining: number
     startsAtNis: number
     validUntil: string
     ldLimit: string

     redeemedCount: number
     status: "active" | "expired" | "void"
     lastRedeemedAt?: string
   egiftEvents/{eventId}:
     type: "ISSUED" | "REDEEMED" | "VOIDED"
     cardId
     createdAt
     meta
*/

/* -------------------------
   API: Create e-gift purchase
   ------------------------- */
app.post("/api/egift/create", async (req, res) => {
  try {
    const { name, locale, style, washes, startsAtNis, validUntil, ldLimit } = req.body || {};

    assertNonEmpty(name, "name");

    const loc: "he" | "en" = isHebrew(locale) ? "he" : "en";
    const s: "black" | "gold" | "green" | "pink" = ["black", "gold", "green", "pink"].includes(style) ? style : "black";

    const w = Number.isFinite(Number(washes)) ? Number(washes) : 1;
    const starts = Number.isFinite(Number(startsAtNis)) ? Number(startsAtNis) : 55;

    const vu = typeof validUntil === "string" && validUntil.trim() ? validUntil.trim() : "12/26";
    const ld = typeof ldLimit === "string" && ldLimit.trim() ? ldLimit.trim() : "12/26";

    const cleanName = safeUpperName(name);
    const initials = getInitials(cleanName, loc);

    // Unique cardId
    const cardId = `PWL${Date.now().toString().slice(-8)}${Math.floor(Math.random() * 90 + 10)}`;
    const redeemUrl = `${PROJECT_DOMAIN}/redeem/${encodeURIComponent(cardId)}`;

    const doc = {
      cardId,
      name: cleanName,
      initials,
      locale: loc,
      style: s,
      washesTotal: w,
      washesRemaining: w,
      startsAtNis: starts,
      validUntil: vu,
      ldLimit: ld,
      createdAt: nowIso(),
      redeemedCount: 0,
      status: "active" as const,
    };

    await db.collection("egifts").doc(cardId).set(doc, { merge: false });

    const eventId = `${cardId}_ISSUED_${Date.now()}`;
    await db.collection("egiftEvents").doc(eventId).set({
      type: "ISSUED",
      cardId,
      createdAt: nowIso(),
      meta: { style: s, locale: loc, washes: w },
    });

    res.json({
      ok: true,
      cardId,
      redeemUrl,
      previewFront: `${PROJECT_DOMAIN}/api/egift/${encodeURIComponent(cardId)}/front.svg`,
      previewBack: `${PROJECT_DOMAIN}/api/egift/${encodeURIComponent(cardId)}/back.svg`,
    });
  } catch (e: any) {
    const msg = e?.message || "Unknown error";
    res.status(400).json({ ok: false, error: msg });
  }
});

/* -------------------------
   API: Get Front SVG
   ------------------------- */
app.get("/api/egift/:cardId/front.svg", async (req, res) => {
  try {
    const cardId = String(req.params.cardId || "");
    const snap = await db.collection("egifts").doc(cardId).get();
    if (!snap.exists) return res.status(404).send("Not found");

    const d = snap.data() as any;

    const svg = buildFrontSvg({
      cardId: d.cardId,
      name: d.name,
      initials: d.initials || getInitials(d.name, d.locale),
      locale: d.locale,
      style: d.style,
      washes: d.washesTotal,
      startsAtNis: d.startsAtNis || 55,
      packageDiscountLabel: d.washesTotal >= 3 ? `${d.washesTotal} WASHES` : "SINGLE",
    });

    res.setHeader("Content-Type", "image/svg+xml; charset=utf-8");
    res.setHeader("Cache-Control", "public, max-age=300");
    res.send(svg);
  } catch {
    res.status(500).send("Server error");
  }
});

/* -------------------------
   API: Get Back SVG
   ------------------------- */
app.get("/api/egift/:cardId/back.svg", async (req, res) => {
  try {
    const cardId = String(req.params.cardId || "");
    const snap = await db.collection("egifts").doc(cardId).get();
    if (!snap.exists) return res.status(404).send("Not found");

    const d = snap.data() as any;
    const redeemUrl = `${PROJECT_DOMAIN}/redeem/${encodeURIComponent(cardId)}`;

    const svg = await buildBackSvg({
      cardId: d.cardId,
      name: d.name,
      initials: d.initials || getInitials(d.name, d.locale),
      locale: d.locale,
      style: d.style,
      washes: d.washesTotal,
      validUntil: d.validUntil || "12/26",
      ldLimit: d.ldLimit || "12/26",
      redeemUrl,
    });

    res.setHeader("Content-Type", "image/svg+xml; charset=utf-8");
    res.setHeader("Cache-Control", "public, max-age=300");
    res.send(svg);
  } catch {
    res.status(500).send("Server error");
  }
});
