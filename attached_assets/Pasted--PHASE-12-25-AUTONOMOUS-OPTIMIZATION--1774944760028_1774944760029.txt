/* =========================================================
PHASE 12.25 - AUTONOMOUS OPTIMIZATION (CONTROLLED)
PURPOSE:
System proposes improvements using real outcomes,
but NEVER executes changes directly.

FLOW:
measure (12.22)
→ learn (12.23)
→ propose (12.25)
→ execute via control (12.24)

NO SHORTCUTS
========================================================= */

/* =========================
DB: optimization_proposals
========================= */

CREATE TABLE optimization_proposals (
  id SERIAL PRIMARY KEY,
  policy_key TEXT NOT NULL,
  proposal_type TEXT NOT NULL,
  current_config JSONB NOT NULL,
  proposed_config JSONB NOT NULL,
  rationale JSONB NOT NULL,
  confidence TEXT NOT NULL,
  evidence_count INT NOT NULL,
  status TEXT DEFAULT 'proposed', -- proposed | accepted | rejected | promoted
  created_at TIMESTAMP DEFAULT NOW(),
  reviewed_at TIMESTAMP
);


/* =========================
CORE ENGINE
server/lib/optimizer.ts
========================= */

import { computePolicyFeedback } from "./learning-policy";

export async function generateProposals(activeConfigs: any[]) {
  const feedback = await computePolicyFeedback();

  // HARD RULE: no automation if not enough real data
  if (feedback.dataMaturity.resolvedWithBaseline < 3) {
    return [];
  }

  const proposals = [];

  // Example rule: approval threshold too strict
  if (
    feedback.frictionFeedback.approvalDelayIls > 0 &&
    (feedback.frictionFeedback.frictionResolutionRate ?? 0) > 50
  ) {
    const current = activeConfigs.find(p => p.policyKey === "approval_threshold");
    if (current) {
      const proposed = {
        ...current.config,
        thresholdIls: Math.round(current.config.thresholdIls * 1.15),
      };

      proposals.push({
        policy_key: "approval_threshold",
        proposal_type: "relax",
        current_config: current.config,
        proposed_config: proposed,
        rationale: {
          summary: "Approval delays create friction without improving outcomes",
        },
        confidence: "medium",
        evidence_count: feedback.dataMaturity.resolvedWithBaseline,
      });
    }
  }

  return proposals;
}


/* =========================
ROUTES
server/routes/optimizer.ts
========================= */

router.post("/generate", async (_req, res) => {
  const configs = await db.execute(sql`
    SELECT policy_key as "policyKey", config
    FROM policy_configs
    WHERE status='active'
  `);

  const proposals = await generateProposals(configs.rows);

  const inserted = [];

  for (const p of proposals) {
    const r = await db.execute(sql`
      INSERT INTO optimization_proposals
      (policy_key, proposal_type, current_config, proposed_config, rationale, confidence, evidence_count)
      VALUES (
        ${p.policy_key},
        ${p.proposal_type},
        ${JSON.stringify(p.current_config)}::jsonb,
        ${JSON.stringify(p.proposed_config)}::jsonb,
        ${JSON.stringify(p.rationale)}::jsonb,
        ${p.confidence},
        ${p.evidence_count}
      )
      RETURNING *
    `);
    inserted.push(r.rows[0]);
  }

  res.json({ proposals: inserted });
});


/* =========================
ACCEPT / REJECT
========================= */

router.post("/:id/accept", async (req, res) => {
  const r = await db.execute(sql`
    UPDATE optimization_proposals
    SET status='accepted', reviewed_at=NOW()
    WHERE id=${Number(req.params.id)}
    RETURNING *
  `);
  res.json({ proposal: r.rows[0] });
});

router.post("/:id/reject", async (req, res) => {
  const r = await db.execute(sql`
    UPDATE optimization_proposals
    SET status='rejected', reviewed_at=NOW()
    WHERE id=${Number(req.params.id)}
    RETURNING *
  `);
  res.json({ proposal: r.rows[0] });
});


/* =========================
PROMOTE → PHASE 12.24
CRITICAL CONTROL POINT
========================= */

router.post("/:id/promote", async (req, res) => {
  const p = await db.execute(sql`
    SELECT * FROM optimization_proposals WHERE id=${Number(req.params.id)}
  `);

  const proposal = p.rows[0];
  if (!proposal || proposal.status !== "accepted") {
    return res.status(400).json({ error: "Must accept first" });
  }

  const version = await db.execute(sql`
    SELECT COALESCE(MAX(version),0)+1 as v
    FROM policy_configs
    WHERE policy_key=${proposal.policy_key}
  `);

  const newConfig = await db.execute(sql`
    INSERT INTO policy_configs (policy_key, version, config, status)
    VALUES (
      ${proposal.policy_key},
      ${version.rows[0].v},
      ${JSON.stringify(proposal.proposed_config)}::jsonb,
      'draft'
    )
    RETURNING *
  `);

  await db.execute(sql`
    UPDATE optimization_proposals SET status='promoted'
    WHERE id=${proposal.id}
  `);

  res.json({ draft: newConfig.rows[0] });
});


/* =========================
BUSINESS LOGIC SUMMARY
=========================

This phase does NOT change the system behavior directly.

It does:
- observe outcomes (12.22)
- learn patterns (12.23)
- suggest improvements (12.25)

It does NOT:
- activate policies
- bypass approvals
- modify finance logic

Every change STILL goes through:
draft → activate → rollout → evaluate → rollback

=========================

WHY THIS IS CORRECT

Without 12.25:
→ system is static

With uncontrolled 12.25:
→ system becomes dangerous

With THIS design:
→ system improves safely over time

========================= */
