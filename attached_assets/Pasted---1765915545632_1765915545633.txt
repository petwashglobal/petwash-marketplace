import datetime

# --- SIMULATED OCTOPUS INTEGRATION SERVICE ---
# (Assumes octopus_service.py logic is here)
def submit_booking_to_octopus(booking_data):
    """
    SIMULATED CALL to the Octopus Control Panel API for booking submission.
    """
    if "error" in booking_data.get('client_name', '').lower():
         return False, {"error": "Invalid client data", "detail": "Octopus rejected client name due to formatting"}
    
    unique_id = "PWBK" + str(datetime.datetime.utcnow().timestamp()).replace('.', '')[-8:]
    return True, {
        "octopus_booking_id": unique_id,
        "status": "Confirmed",
    }

# --- FLASK APPLICATION ---
app = Flask(__name__)

@app.route('/api/bookings', methods=['POST'])
def create_booking():
    """API endpoint for mobile app to submit a booking."""
    data = request.get_json()
    
    if not all(k in data for k in ('client_name', 'client_phone', 'pet_type', 'service')):
        return jsonify({"error": "Missing required fields"}), 400

    data['booking_date'] = datetime.datetime.utcnow().isoformat()
    data['service_code'] = data['service'].upper().replace(' ', '_')

    success, result_data = submit_booking_to_octopus(data)

    if success:
        return jsonify({
            "message": "Booking confirmed by Octopus Control Panel.", 
            "booking_id": result_data['octopus_booking_id'],
            "status": result_data['status']
        }), 201
    else:
        return jsonify({
            "error": "Booking failed at the core system.",
            "details": result_data.get("error", "Unknown communication error"),
            "system_message": result_data
        }), 400
