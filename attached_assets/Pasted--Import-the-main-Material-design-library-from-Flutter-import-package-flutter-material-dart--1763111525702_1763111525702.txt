// Import the main "Material" design library from Flutter
import 'package:flutter/material.dart';

// The 'main' function is the starting point for all Flutter apps.
void main() {
  // 'runApp' tells Flutter to run the app defined by the 'PetWashApp' widget.
  runApp(const PetWashApp());
}

// 'PetWashApp' is the root widget of your application.
// It's a 'StatelessWidget' because its properties don't change over time.
class PetWashApp extends StatelessWidget {
  const PetWashApp({super.key});

  @override
  Widget build(BuildContext context) {
    // 'MaterialApp' is the main container for your app,
    // giving it the standard Google Material Design look and feel.
    return MaterialApp(
      // Disables the "debug" banner in the top-right corner
      debugShowCheckedModeBanner: false, 
      title: 'Pet Wash',
      theme: ThemeData(
        // This sets the main color theme for the app
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.blueAccent),
        useMaterial3: true, // Use the new Material 3 (M3) design
        brightness: Brightness.light,
      ),
      // 'HomePage' is the widget that will be displayed as the home screen.
      home: const HomePage(),
    );
  }
}
