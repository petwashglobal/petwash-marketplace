    <script>
        // --- 1. GATE LOGIC ---
        function dismissGate() {
            document.getElementById('entry-gate').classList.add('dismissed');
        }

        document.getElementById('loyalty-form').onsubmit = (e) => {
            e.preventDefault();
            // Simulate API hit
            alert("Welcome to the Elite Circle.");
            dismissGate();
        };

        // --- 2. ELITE BOOKING & COMMISSION ENGINE ---
        const PRICING = {
            dogNight: 65.00,
            catNight: 40.00,
            transportBase: 35.00,
            commissionRate: 0.15 // 15% Platform Fee
        };

        function runCalculation() {
            const nights = 7;
            const dogs = 2;
            const cats = 1;
            const hasTransport = true;

            const hostingTotal = (dogs * PRICING.dogNight * nights) + (cats * PRICING.catNight * nights);
            const transport = hasTransport ? PRICING.transportBase : 0;
            const gross = hostingTotal + transport;
            
            const platformFee = gross * PRICING.commissionRate;
            const contractorNet = gross - platformFee;

            const summaryHtml = `
                <div class="calc-row"><span>Dog Hosting (${dogs})</span> <span>$${(dogs * PRICING.dogNight * nights).toFixed(2)}</span></div>
                <div class="calc-row"><span>Cat Hosting (${cats})</span> <span>$${(cats * PRICING.catNight * nights).toFixed(2)}</span></div>
                <div class="calc-row"><span>Private Transport</span> <span>$${transport.toFixed(2)}</span></div>
                <div class="total-box">
                    <div class="calc-row"><strong>Total Client Price</strong> <strong>$${gross.toFixed(2)}</strong></div>
                    <div class="calc-row" style="color: #B38728; font-size: 0.7rem;"><span>Contractor Payout (85%)</span> <span>$${contractorNet.toFixed(2)}</span></div>
                </div>
            `;
            document.getElementById('price-summary').innerHTML = summaryHtml;
        }

        // --- 3. SAVE PROGRESS ---
        function saveBooking() {
            const data = {
                notes: document.getElementById('contractor-notes').value,
                timestamp: new Date().toISOString()
            };
            localStorage.setItem('atelier_saved_booking', JSON.stringify(data));
            alert("Progress saved securely.");
        }

        window.onload = runCalculation;
