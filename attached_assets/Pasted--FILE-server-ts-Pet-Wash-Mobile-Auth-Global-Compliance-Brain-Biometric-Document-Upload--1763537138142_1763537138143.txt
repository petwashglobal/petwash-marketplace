function generateAccessToken(user: User): string {
  const payload: JwtPayload = { sub: user.id, roles: user.roles };
  return jwt.sign(payload, JWT_ACCESS_SECRET, { expiresIn: ACCESS_TOKEN_TTL_SEC });
}

function generateRefreshToken(user: User, deviceId?: string): string {
  const payload: JwtPayload & { deviceId?: string } = {
    sub: user.id,
    roles: user.roles,
    deviceId,
  };
  return jwt.sign(payload, JWT_REFRESH_SECRET, {
    expiresIn: `${REFRESH_TOKEN_TTL_DAYS}d`,
  });
}

function addDays(date: Date, days: number): Date {
  const d = new Date(date);
  d.setDate(d.getDate() + days);
  return d;
}

async function saveRefreshToken(userId: string, token: string, deviceId?: string) {
  const tokenHash = await bcrypt.hash(token, 10);
  const expiresAt = addDays(new Date(), REFRESH_TOKEN_TTL_DAYS);
  await prisma.refreshToken.create({
    data: { userId, deviceId, tokenHash, expiresAt },
  });
}

async function revokeRefreshToken(token: string) {
  const candidates = await prisma.refreshToken.findMany({
    where: { revokedAt: null, expiresAt: { gt: new Date() } },
    take: 1000,
  });
  for (const row of candidates) {
    const match = await bcrypt.compare(token, row.tokenHash);
    if (match) {
      await prisma.refreshToken.update({
        where: { id: row.id },
        data: { revokedAt: new Date() },
      });
      return true;
    }
  }
  return false;
}

async function findUserById(id: string): Promise<User | null> {
  return prisma.user.findUnique({ where: { id } });
}

function requireRole(roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.authRoles) return res.status(401).json({ error: "unauthorized" });
    const has = req.authRoles.some((r) => roles.includes(r));
    if (!has) return res.status(403).json({ error: "forbidden" });
    next();
  };
}

async function authMiddleware(req: Request, res: Response, next: NextFunction) {
  const authHeader = req.headers.authorization;
  if (!authHeader) return res.status(401).json({ error: "missing Authorization header" });
  const [scheme, token] = authHeader.split(" ");
  if (scheme !== "Bearer" || !token) {
    return res.status(401).json({ error: "invalid Authorization header" });
  }
  try {
    const decoded = jwt.verify(token, JWT_ACCESS_SECRET) as JwtPayload;
    const user = await findUserById(decoded.sub);
    if (!user) return res.status(401).json({ error: "user not found" });
    req.user = user;
    req.authUserId = user.id;
    req.authRoles = decoded.roles;
    next();
  } catch {
    return res.status(401).json({ error: "invalid or expired token" });
  }
}

const loginLimiter = rateLimit({

  max: 20,
  standardHeaders: true,
  legacyHeaders: false,
});

// helper: store uploaded file and return relative path
async function storeUploadedIdentityFile(file: Express.Multer.File): Promise<string> {
  // In production you can replace this with upload to S3 or GCS and return URL
  const relPath = path.relative(process.cwd(), file.path);
  return relPath;
}

app.post("/api/auth/login", loginLimiter, async (req: Request, res: Response) => {
  const { email, password, deviceId, platform, osVersion, appVersion, pushToken } = req.body || {};
  if (!email || !password) return res.status(400).json({ error: "email and password required" });

  const user = await prisma.user.findUnique({ where: { email } });
  if (!user) return res.status(401).json({ error: "invalid credentials" });

  const ok = await bcrypt.compare(password, user.passwordHash);
  if (!ok) return res.status(401).json({ error: "invalid credentials" });

  let finalDeviceId: string | undefined = deviceId;
  if (deviceId && platform) {
    await prisma.device.upsert({
      where: { id: deviceId },
      update: {
        userId: user.id,
        platform,
        osVersion,
        appVersion,
        pushToken,
        lastSeenAt: new Date(),
        isBlocked: false,
      },
      create: {
        id: deviceId,
        userId: user.id,
        platform,
        osVersion,
        appVersion,
        pushToken,
      },
    });
  }

  const accessToken = generateAccessToken(user);
  const refreshToken = generateRefreshToken(user, finalDeviceId);
  await saveRefreshToken(user.id, refreshToken, finalDeviceId);

  return res.json({
    accessToken,
    refreshToken,
    expiresIn: ACCESS_TOKEN_TTL_SEC,
    user: {
      id: user.id,
      email: user.email,
      roles: user.roles,
      fullName: user.fullName,
    },
  });
});

app.post("/api/auth/refresh", async (req: Request, res: Response) => {
  const { refreshToken, deviceId } = req.body || {};
  if (!refreshToken) return res.status(400).json({ error: "refreshToken required" });

  try {
    const decoded = jwt.verify(refreshToken, JWT_REFRESH_SECRET) as JwtPayload & { deviceId?: string };
    const user = await findUserById(decoded.sub);
    if (!user) return res.status(401).json({ error: "user not found" });

    if (deviceId && decoded.deviceId && decoded.deviceId !== deviceId) {
      return res.status(401).json({ error: "device mismatch" });
    }

    const rows = await prisma.refreshToken.findMany({
      where: { userId: user.id, revokedAt: null, expiresAt: { gt: new Date() } },
      take: 500,
    });

    let foundRow = null;
    for (const row of rows) {
      const match = await bcrypt.compare(refreshToken, row.tokenHash);
      if (match) {
        foundRow = row;
        break;
      }
    }
    if (!foundRow) return res.status(401).json({ error: "refresh token not valid" });

    const newAccessToken = generateAccessToken(user);
    const newRefreshToken = generateRefreshToken(user, deviceId || decoded.deviceId);
    await saveRefreshToken(user.id, newRefreshToken, deviceId || decoded.deviceId);

    return res.json({
      accessToken: newAccessToken,
      refreshToken: newRefreshToken,
      user: {
        id: user.id,
        email: user.email,
        roles: user.roles,
        fullName: user.fullName,
      },
    });
  } catch (err) {
    console.error("refresh error", err);
    return res.status(401).json({ error: "invalid refresh token" });
  }
});

app.post("/api/auth/logout", async (req: Request, res: Response) => {
  const { refreshToken } = req.body || {};
  if (!refreshToken) return res.status(400).json({ error: "refreshToken required" });
  await revokeRefreshToken(refreshToken);
  return res.json({ ok: true });
});

// POST /api/contractors/:id/identity-docs/upload
// mobile sends multipart/form-data with fields:
// - documentType: "passport" | "drivers_license" | "id_card"
// - country: country code, for example "IL"
// - provider: optional
// - biometricHint: optional flag from client like "device_face_match_ok"
// - files: one or more images or pdfs
app.post(
  "/api/contractors/:id/identity-docs/upload",
  authMiddleware,
  uploadIdentityDocs.array("files", 10),
  async (req: Request, res: Response) => {
    const contractorId = req.params.id;
    const { documentType, country, provider, biometricHint } = req.body || {};
    const files = req.files as Express.Multer.File[] | undefined;

    if (!documentType) {
      return res.status(400).json({ error: "documentType required" });
    }
    if (!files || files.length === 0) {
      return res.status(400).json({ error: "at least one file required" });
    }

    const contractor = await prisma.contractor.findUnique({
      where: { id: contractorId },
      include: { user: true },
    });
    if (!contractor) return res.status(404).json({ error: "contractor not found" });

    // if user is a contractor, make sure he uploads only for his own profile
    if (req.authRoles && req.authRoles.includes("contractor")) {
      if (contractor.userId !== req.authUserId) {
        return res.status(403).json({ error: "cannot upload for another contractor" });
      }
    }

    const details: any = {
      country: country || contractor.country || null,
      source: "mobile_upload",
      biometricHint: biometricHint || null,
      biometricVerified: false, // will be set to true only after external provider verification
      uploadedFileCount: files.length,
    };

    const verification = await prisma.identityVerification.create({
      data: {
        contractorId,
        documentType,
        status: "pending",
        provider: provider || "mobile_app",
        details,
      },
    });

    const storedFiles: { id: string; filePath: string; mimeType: string }[] = [];
    for (const file of files) {
      const relPath = await storeUploadedIdentityFile(file);
      const row = await prisma.identityDocumentFile.create({
        data: {
          identityVerificationId: verification.id,
          filePath: relPath,
          mimeType: file.mimetype || "application/octet-stream",
        },
      });
      storedFiles.push({ id: row.id, filePath: row.filePath, mimeType: row.mimeType });
    }

    // optional: create audit log
    await prisma.complianceAuditLog.create({
      data: {
        contractorId,
        assignmentId: null,
        decisionId: null,
        action: "identity_upload",
        actorType: req.authRoles?.includes("admin") ? "admin" : "system",
        actorId: req.authUserId,
        notes: `Uploaded ${storedFiles.length} identity files, documentType=${documentType}`,
      },
    });

    return res.json({
      verification: {
        ...verification,
        details,
      },
      files: storedFiles,
    });
  }
);

// Admin route to mark biometric passport as verified after external provider check
// POST /api/compliance/identity/:verificationId/approve-biometric
app.post(
  "/api/compliance/identity/:verificationId/approve-biometric",
  authMiddleware,
  requireRole(["admin"]),
  async (req: Request, res: Response) => {
    const verificationId = req.params.verificationId;
    const { biometricVerified, country } = req.body || {};

    const verification = await prisma.identityVerification.findUnique({
      where: { id: verificationId },
    });
    if (!verification) return res.status(404).json({ error: "verification not found" });

    const details: any = verification.details || {};
    if (typeof biometricVerified === "boolean") {
      details.biometricVerified = biometricVerified;
    } else {
      details.biometricVerified = true;
    }
    if (country) {
      details.country = country;
    }

    const updated = await prisma.identityVerification.update({
      where: { id: verificationId },
      data: {
        status: "approved",
        details,
      },
    });

    await prisma.complianceAuditLog.create({
      data: {
        contractorId: updated.contractorId,
        assignmentId: null,
        decisionId: null,
        action: "identity_biometric_approval",
        actorType: "admin",
        actorId: req.authUserId,
        notes: `Biometric identity approved for verification ${verificationId}`,
      },
    });

    return res.json({ verification: updated });
  }
);

type ComplianceEvaluationResult = {
  decision: "approved" | "pending" | "rejected" | "blocked";
  score: number;
  reasons: string[];
  triggeredRules: string[];
  requirementsMissing: string[];
};

function isBiometricApprovedPassport(ver: any): boolean {
  if (!ver || ver.documentType !== "passport" || ver.status !== "approved") return false;

  const details: any = ver.details || {};
  const biometricVerified = !!details.biometricVerified || !!details.biometric_verification_passed;
  const countryRaw: string | undefined =
    details.country || details.countryCode || details.country_code || undefined;

  if (!biometricVerified || !countryRaw) return false;

  const countryNorm = countryRaw.toString().trim().toUpperCase();
  return APPROVED_PASSPORT_COUNTRIES.includes(countryNorm);
}

async function evaluateAssignmentEligibility(
  contractorId: string,
  assignmentId?: string
): Promise<ComplianceEvaluationResult> {
  const contractor = await prisma.contractor.findUnique({
    where: { id: contractorId },
    include: {
      identityVerifications: { include: { files: true } },
      livenessChecks: { orderBy: { createdAt: "desc" }, take: 1 },
      criminalBackgroundChecks: { orderBy: { createdAt: "desc" }, take: 1 },
      driverSafetyProfile: true,
      incidents: { orderBy: { occurredAt: "desc" }, take: 10 },
      ratings: true,
    },
  });

  if (!contractor) {
    return {
      decision: "rejected",
      score: 0,
      reasons: ["contractor not found"],
      triggeredRules: ["MISSING_CONTRACTOR"],
      requirementsMissing: ["contractor_profile"],
    };
  }

  let score = 100;
  const reasons: string[] = [];
  const rules: string[] = [];
  const missing: string[] = [];
  const now = new Date();

  // identity + passport logic
  const biometricPassport = contractor.identityVerifications.find((v) => isBiometricApprovedPassport(v));
  const anyApprovedId = contractor.identityVerifications.find(
    (v) => v.status === "approved" && (!v.expiresAt || v.expiresAt > now)
  );

  if (biometricPassport) {
    score += 5;
    rules.push("BIOMETRIC_PASSPORT_STRONG_ID");
  } else if (!anyApprovedId) {
    score -= 30;
    reasons.push("no valid identity verification");
    rules.push("IDENTITY_MISSING_OR_EXPIRED");
    missing.push("identity_verification");
  }

  const lastLiveness = contractor.livenessChecks[0];
  if (!lastLiveness) {
    score -= 15;
    reasons.push("no liveness check on file");
    rules.push("LIVENESS_MISSING");
    missing.push("liveness_check");
  } else if (lastLiveness.status !== "passed") {
    score -= 40;
    reasons.push("last liveness check failed");
    rules.push("LIVENESS_FAILED");
  }

  const lastCriminal = contractor.criminalBackgroundChecks[0];
  if (!lastCriminal) {
    score -= 20;
    reasons.push("no criminal background check");
    rules.push("CRIMINAL_CHECK_MISSING");
    missing.push("criminal_background_check");
  } else if (lastCriminal.status === "record_found") {
    score -= 60;
    reasons.push("criminal record found");
    rules.push("CRIMINAL_RECORD");
  }

  if (contractor.driverSafetyProfile) {
    const dsp = contractor.driverSafetyProfile;
    if (dsp.riskLevel === "medium") {
      score -= 10;
      reasons.push("medium driver risk level");
      rules.push("DRIVER_RISK_MEDIUM");
    } else if (dsp.riskLevel === "high") {
      score -= 40;
      reasons.push("high driver risk level");
      rules.push("DRIVER_RISK_HIGH");
    }
  } else {
    score -= 10;
    reasons.push("no driver safety profile");
    rules.push("DRIVER_PROFILE_MISSING");
    missing.push("driver_safety_profile");
  }

  const recentCriticalIncident = contractor.incidents.find(
    (i) =>
      i.severity === "critical" &&
      i.occurredAt > new Date(now.getTime() - 1000 * 60 * 60 * 24 * 180)
  );
  if (recentCriticalIncident) {
    score -= 50;
    reasons.push("recent critical incident on record");
    rules.push("RECENT_CRITICAL_INCIDENT");
  }

  if (contractor.ratings.length >= 3) {
    const avg = contractor.ratings.reduce((sum, r) => sum + r.score, 0) / contractor.ratings.length;
    if (avg < 3) {
      score -= 20;
      reasons.push("low average rating");
      rules.push("LOW_AVERAGE_RATING");
    } else if (avg > 4.5) {
      score += 5;
      rules.push("HIGH_AVERAGE_RATING");
    }
  }

  if (score < 0) score = 0;
  if (score > 100) score = 100;

  let decision: ComplianceEvaluationResult["decision"] = "pending";
  if (score >= 80 && missing.length === 0) {
    decision = "approved";
  } else if (score >= 50) {
    decision = "pending";
  } else if (score >= 20) {
    decision = "rejected";
  } else {
    decision = "blocked";
  }

  return {
    decision,
    score,
    reasons,
    triggeredRules: rules,
    requirementsMissing: missing,
  };
}

app.post(
  "/api/compliance/evaluate",
  authMiddleware,
  requireRole(["admin", "dispatcher"]),
  async (req: Request, res: Response) => {
    const { contractorId, assignmentId } = req.body || {};
    if (!contractorId) return res.status(400).json({ error: "contractorId required" });

    const result = await evaluateAssignmentEligibility(contractorId, assignmentId);

    const decisionRow = await prisma.complianceDecision.create({
      data: {
        contractorId,
        assignmentId,
        decision: result.decision,
        score: result.score,
        reasons: result.reasons,
        triggeredRules: result.triggeredRules,
        decidedBy: "system",
      },
    });

    await prisma.complianceAuditLog.create({
      data: {
        contractorId,
        assignmentId: assignmentId || null,
        decisionId: decisionRow.id,
        action: "evaluate",
        actorType: "system",
        actorId: null,
        notes: result.reasons.join("; "),
      },
    });

    await prisma.contractor.update({
      where: { id: contractorId },
      data: { status: result.decision === "approved" ? "approved" : result.decision },
    });

    return res.json({
      decision: result.decision,
      score: result.score,
      reasons: result.reasons,
      triggeredRules: result.triggeredRules,
      requirementsMissing: result.requirementsMissing,
      decisionId: decisionRow.id,
    });
  }
);

app.post(
  "/api/compliance/manual-override",
  authMiddleware,
  requireRole(["admin"]),
  async (req: Request, res: Response) => {
    const { contractorId, assignmentId, decision, reason } = req.body || {};
    if (!contractorId || !decision) {
      return res.status(400).json({ error: "contractorId and decision required" });
    }

    const dec = await prisma.complianceDecision.create({
      data: {
        contractorId,
        assignmentId,
        decision,
        score: 0,
        reasons: [reason || "manual override"],
        triggeredRules: ["MANUAL_OVERRIDE"],
        decidedBy: req.authUserId || "admin",
      },
    });

    await prisma.complianceAuditLog.create({
      data: {
        contractorId,
        assignmentId: assignmentId || null,
        decisionId: dec.id,
        action: "manual_override",
        actorType: "admin",
        actorId: req.authUserId,
        notes: reason || "",
      },
    });

    await prisma.contractor.update({
      where: { id: contractorId },
      data: { status: decision },
    });

    return res.json({ ok: true, decisionId: dec.id });
  }
);

app.get(
  "/api/compliance/contractors/:id/summary",
  authMiddleware,
  requireRole(["admin", "dispatcher"]),
  async (req: Request, res: Response) => {
    const contractorId = req.params.id;
    const contractor = await prisma.contractor.findUnique({
      where: { id: contractorId },
      include: {
        identityVerifications: { include: { files: true }, orderBy: { createdAt: "desc" } },
        livenessChecks: { orderBy: { createdAt: "desc" }, take: 5 },
        criminalBackgroundChecks: { orderBy: { createdAt: "desc" }, take: 3 },
        driverSafetyProfile: true,
        ratings: { orderBy: { createdAt: "desc" }, take: 20 },
        incidents: { orderBy: { occurredAt: "desc" }, take: 20 },
        complianceDecisions: { orderBy: { createdAt: "desc" }, take: 10 },
      },
    });

    if (!contractor) return res.status(404).json({ error: "contractor not found" });

    const avgRating =
      contractor.ratings.length > 0
        ? contractor.ratings.reduce((sum, r) => sum + r.score, 0) / contractor.ratings.length
        : null;

    const lastDecision = contractor.complianceDecisions[0] || null;
    const biometricPassport = contractor.identityVerifications.find((v) => isBiometricApprovedPassport(v));

    return res.json({
      contractor: {
        id: contractor.id,
        status: contractor.status,
        country: contractor.country,
      },
      identityVerifications: contractor.identityVerifications,
      hasStrongBiometricPassport: !!biometricPassport,
      livenessChecks: contractor.livenessChecks,
      criminalBackgroundChecks: contractor.criminalBackgroundChecks,
      driverSafetyProfile: contractor.driverSafetyProfile,
      incidents: contractor.incidents,
      ratings: contractor.ratings,
      averageRating: avgRating,
      complianceDecisions: contractor.complianceDecisions,
      lastDecision,
    });
  }
);

app.get(
