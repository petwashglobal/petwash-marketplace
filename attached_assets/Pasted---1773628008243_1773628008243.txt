function b64url(input: Buffer | string): string {
  return Buffer.from(input)
    .toString("base64")
    .replace(/\+/g, "-")
    .replace(/\//g, "_")
    .replace(/=+$/g, "");
}

interface SignedPayload {
  passId: string;
  userId: string;
  tokenVersion: number;
  purpose: "wallet-link" | "qr-redeem";
  nonce: string;
  issuedAt: number;
  expiresAt: number;
  machineId?: string | null;
  bayId?: "LEFT" | "RIGHT" | null;
}

function signPayload(payload: SignedPayload, secret: string): string {
  const body = b64url(JSON.stringify(payload));
  const sig = crypto.createHmac("sha256", secret).update(body).digest("base64url");
  return `${body}.${sig}`;
}

function verifyPayload(token: string, secret: string): SignedPayload {
  const [body, sig] = token.split(".");
  if (!body || !sig) throw new Error("INVALID_TOKEN_FORMAT");

  const expected = crypto.createHmac("sha256", secret).update(body).digest("base64url");
  if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) {
    throw new Error("INVALID_SIGNATURE");
  }

  const payload = JSON.parse(Buffer.from(body, "base64url").toString("utf8")) as SignedPayload;
  const now = Math.floor(Date.now() / 1000);
  if (payload.expiresAt < now) throw new Error("TOKEN_EXPIRED");
  return payload;
}

export function buildPassLinkToken(record: PetWashPassRecord): string {
  const now = Math.floor(Date.now() / 1000);
  return signPayload(
    {
      passId: record.passId,
      userId: record.userId,
      tokenVersion: record.qrTokenVersion,
      purpose: "wallet-link",
      nonce: crypto.randomUUID(),
      issuedAt: now,
      expiresAt: now + 60 * 60 * 24 * 3,
    },
    CONFIG.PASS_LINK_SECRET
  );
}

export function buildQrRedeemToken(record: PetWashPassRecord, machineId?: string, bayId?: "LEFT" | "RIGHT"): string {
  const now = Math.floor(Date.now() / 1000);
  return signPayload(
    {
      passId: record.passId,
      userId: record.userId,
      tokenVersion: record.qrTokenVersion,
      purpose: "qr-redeem",
      nonce: crypto.randomUUID(),
      issuedAt: now,
      expiresAt: now + 45,
      machineId: machineId || null,
      bayId: bayId || null,
    },
    CONFIG.PRESTIGE_QR_SECRET
  );
}

export class AppleWalletService {
  async generatePass(record: PetWashPassRecord): Promise<Buffer> {
    const qrToken = buildQrRedeemToken(record);

    const pass = await PKPass.from(
      {
        model: "./wallet/apple-model",
        certificates: {
          wwdr: fs.readFileSync(CONFIG.APPLE_WWDR_PATH),
          signerCert: fs.readFileSync(CONFIG.APPLE_SIGNER_CERT_PATH),
          signerKey: fs.readFileSync(CONFIG.APPLE_SIGNER_KEY_PATH),
          signerKeyPassphrase: CONFIG.APPLE_SIGNER_KEY_PASSPHRASE,
        },
      },
      {
        serialNumber: record.appleSerialNumber,
        passTypeIdentifier: CONFIG.APPLE_PASS_TYPE_IDENTIFIER,
        teamIdentifier: CONFIG.APPLE_TEAM_IDENTIFIER,
        organizationName: "Pet Wash Ltd",
        description: "PetWash Premium Pass",
        logoText: "PetWash",
        backgroundColor: "rgb(255,255,255)",
        foregroundColor: "rgb(26,26,26)",
        labelColor: "rgb(198,163,91)",
        authenticationToken: crypto.randomUUID(),
        webServiceURL: `${CONFIG.BASE_URL}/wallet/apple/webservice`,
      }
    );

    pass.primaryFields.add({
      key: "member",
      label: "Member",
      value: record.ownerName,
    });

    pass.secondaryFields.add({
      key: "tier",
      label: "Tier",
      value: `${record.tier} Tier`,
    });

    pass.auxiliaryFields.add({
      key: "credit",
      label: "Available Credit",
      value: `₪${record.availableCreditIls.toFixed(0)}`,
    });

    if (record.primaryPetName) {
      pass.backFields.add({
        key: "pet",
        label: "Primary Pet",
        value: record.primaryPetName,
      });
    }

    if (record.validUntil) {
      pass.backFields.add({
        key: "validUntil",
        label: "Valid Until",
        value: record.validUntil,
      });
    }

    pass.backFields.add({
      key: "memberId",
      label: "Member ID",
      value: record.passId,
    });

    pass.backFields.add({
      key: "support",
      label: "Support",
      value: "support@petwash.co.il",
    });

    pass.setBarcodes({
      message: qrToken,
      format: "PKBarcodeFormatQR",
      messageEncoding: "iso-8859-1",
      altText: record.passId,
    });

    return pass.getAsBuffer();
  }
}

export class GoogleWalletService {
  private auth() {
    const creds = JSON.parse(CONFIG.GOOGLE_SERVICE_ACCOUNT_JSON);
    return new google.auth.JWT(
      creds.client_email,
      undefined,
      creds.private_key,
      ["https://www.googleapis.com/auth/wallet_object.issuer"]
    );
  }

  private classId() {
    return `${CONFIG.GOOGLE_WALLET_ISSUER_ID}.petwash_premium_v1`;
  }

  async ensureClass() {
    const auth = this.auth();
    const wallet = google.walletobjects({ version: "v1", auth });

    try {
      await wallet.genericclass.get({ resourceId: this.classId() });
    } catch {
      await wallet.genericclass.insert({
        requestBody: {
          id: this.classId(),
          issuerName: "PetWash",
          reviewStatus: "UNDER_REVIEW",
        },
      });
    }
  }

  async createOrUpdateObject(record: PetWashPassRecord) {
    await this.ensureClass();

    const auth = this.auth();
    const wallet = google.walletobjects({ version: "v1", auth });
    const qrToken = buildQrRedeemToken(record);

    const objectId = record.googleObjectId;

    const requestBody = {
      id: objectId,
      classId: this.classId(),
      state: "ACTIVE",
      cardTitle: {
        defaultValue: { language: "en-US", value: "PetWash" },
      },
      subheader: {
        defaultValue: { language: "en-US", value: `${record.tier} Tier` },
      },
      header: {
        defaultValue: {
          language: "en-US",
          value: record.primaryPetName
            ? `${record.ownerName} • ${record.primaryPetName}`
            : record.ownerName,
        },
      },
      barcode: {
        type: "QR_CODE",
        value: qrToken,
        alternateText: record.passId,
      },
      textModulesData: [
        {
          id: "credit",
          header: "Available Credit",
          body: `₪${record.availableCreditIls.toFixed(0)}`,
        },
        {
          id: "memberId",
          header: "Member ID",
          body: record.passId,
        },
        ...(record.validUntil
          ? [{ id: "validUntil", header: "Valid Until", body: record.validUntil }]
          : []),
      ],
      validTimeInterval: record.validUntil
        ? { end: { date: record.validUntil } }
        : undefined,
      expiryNotification: { enableNotification: true },
    };

    try {
      await wallet.genericobject.get({ resourceId: objectId });
      await wallet.genericobject.update({ resourceId: objectId, requestBody });
    } catch {
      await wallet.genericobject.insert({ requestBody });
    }
  }

  buildSaveUrl(record: PetWashPassRecord): string {
    const creds = JSON.parse(CONFIG.GOOGLE_SERVICE_ACCOUNT_JSON);
    const qrToken = buildQrRedeemToken(record);

    const payload = {
      iss: creds.client_email,
      aud: "google",
      typ: "savetowallet",
      payload: {
        genericClasses: [
          {
            id: this.classId(),
            issuerName: "PetWash",
            reviewStatus: "UNDER_REVIEW",
          },
        ],
        genericObjects: [
          {
            id: record.googleObjectId,
            classId: this.classId(),
            state: "ACTIVE",
            cardTitle: {
              defaultValue: { language: "en-US", value: "PetWash" },
            },
            header: {
              defaultValue: { language: "en-US", value: record.ownerName },
            },
            subheader: {
              defaultValue: { language: "en-US", value: `${record.tier} Tier` },
            },
            barcode: {
              type: "QR_CODE",
              value: qrToken,
              alternateText: record.passId,
            },
            textModulesData: [
              {
                id: "credit",
                header: "Available Credit",
                body: `₪${record.availableCreditIls.toFixed(0)}`,
              },
            ],
          },
        ],
      },
    };

    const token = jwt.sign(payload, creds.private_key, { algorithm: "RS256" });
    return `https://pay.google.com/gp/v/save/${token}`;
  }
}

app.post("/wallet/redeem", async (req, res) => {
  try {
    const input = walletRedeemSchema.parse(req.body);
    const payload = verifyPayload(input.scannedToken, CONFIG.PRESTIGE_QR_SECRET);

    if (payload.purpose !== "qr-redeem") {
      return res.status(400).json({ error: "INVALID_PURPOSE" });
    }

    if (payload.machineId && payload.machineId !== input.kioskId) {
      return res.status(403).json({ error: "TOKEN_MACHINE_MISMATCH" });
    }

    if (payload.bayId && input.bayId && payload.bayId !== input.bayId) {
      return res.status(403).json({ error: "TOKEN_BAY_MISMATCH" });
    }

    await assertNonceUnusedAndConsume(payload.nonce);

    const record = await loadPassById(payload.passId);
    if (!record) return res.status(404).json({ error: "PASS_NOT_FOUND" });

    if (record.status !== "ACTIVE") {
      return res.status(403).json({ error: "PASS_NOT_ACTIVE" });
    }

    if (record.validUntil && new Date(record.validUntil).getTime() < Date.now()) {
      return res.status(403).json({ error: "PASS_EXPIRED" });
    }

    if (record.availableCreditIls < input.amountIls) {
      return res.status(402).json({ error: "INSUFFICIENT_BALANCE" });
    }

    await debitPassAtomically(record.passId, input.amountIls, {
      source: "K9000",
      kioskId: input.kioskId,
      bayId: input.bayId || null,
      nonce: payload.nonce,
    });

    await startMachine(input.kioskId, input.bayId);

    return res.json({
      ok: true,
      passId: record.passId,
      debitedIls: input.amountIls,
    });
  } catch (err: any) {
    return res.status(400).json({ error: err.message || "REDEEM_FAILED" });
  }
});
