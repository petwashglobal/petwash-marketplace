STEP 2 - ADD A GENERIC TIMEOUT WRAPPER
Add this helper once (anywhere common, for example utils/timeout.ts or utils/timeout.js):

function withTimeout(promise, ms, label) {
  let t;
  const timeout = new Promise((_, reject) => {
    t = setTimeout(() => reject(new Error(label + " timed out after " + ms + "ms")), ms);
  });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(t));
}

STEP 4 - NON-BLOCKING DB CONNECT WITH LOGS AND TIMEOUT
Wherever you do DB connect, do NOT let it block forever.
Wrap it like this (adjust dbConnect() for your library):

async function connectDbNonBlocking(dbConnectFn) {
  const t0 = Date.now();
  console.log("[DB] connecting...");
  try {
    await withTimeout(dbConnectFn(), 5000, "DB connect");
    console.log("[DB] connected in " + (Date.now() - t0) + "ms");
    healthState.db.ok = true;
    healthState.db.lastError = null;
    healthState.db.lastOkAt = new Date().toISOString();
  } catch (e) {
    console.error("[DB] connect failed in " + (Date.now() - t0) + "ms:", e && e.message ? e.message : e);
    healthState.db.ok = false;
    healthState.db.lastError = e && e.message ? e.message : String(e);
  }
}

Call it at startup but do not crash the process unless you want to:
- connectDbNonBlocking(() => prisma.$connect())
- connectDbNonBlocking(() => pool.connect())
- connectDbNonBlocking(() => sequelize.authenticate())
- connectDbNonBlocking(() => mongoose.connect(DATABASE_URL))

STEP 5 - FAST DB PING FUNCTION (NEVER HANGS)
Make a function that checks DB quickly with a short timeout.
Replace dbPingFn() with your actual "SELECT 1" or equivalent.

async function checkDbOnce(dbPingFn) {
  const t0 = Date.now();
  healthState.db.lastCheckAt = new Date().toISOString();
  try {
    await withTimeout(dbPingFn(), 2000, "DB ping");
    healthState.db.ok = true;
    healthState.db.lastError = null;
    healthState.db.lastOkAt = new Date().toISOString();
    healthState.db.lastMs = Date.now() - t0;
    return { ok: true, ms: healthState.db.lastMs };
  } catch (e) {
    healthState.db.ok = false;
    healthState.db.lastError = e && e.message ? e.message : String(e);
    healthState.db.lastMs = Date.now() - t0;
    return { ok: false, ms: healthState.db.lastMs, error: healthState.db.lastError };
  }
}

STEP 8 - MOST COMMON CLOUD RUN DB STALL CAUSES (CHECKLIST)
A) DATABASE_URL points to localhost or private host without routing
- must be publicly reachable or via VPC connector or Cloud SQL connector
B) SSL required but not configured
- often needs sslmode=require (Postgres) or equivalent TLS settings
C) Cloud SQL private IP without VPC connector
- Cloud Run needs VPC connector + correct egress
D) Too many DB connections due to scaling
- reduce pool sizes (Cloud Run scales horizontally)
- set small pool max (example 5) or Prisma connection_limit=5
