import express from 'express';
import axios from 'axios';

const router = express.Router();

// --- CONFIGURATION ---
const TIKTOK_CLIENT_KEY = process.env.TIKTOK_CLIENT_KEY;
const TIKTOK_CLIENT_SECRET = process.env.TIKTOK_CLIENT_SECRET;
const FB_APP_ID = process.env.FB_APP_ID;
const FB_APP_SECRET = process.env.FB_APP_SECRET;

// 1. TIKTOK CALLBACK (The 2025 Flow)
router.get('/api/auth/callback/tiktok', async (req, res) => {
  const { code } = req.query;

  try {
    // A. Exchange Code for Access Token
    const tokenResponse = await axios.post('https://open.tiktokapis.com/v2/oauth/token/', new URLSearchParams({
      client_key: TIKTOK_CLIENT_KEY,
      client_secret: TIKTOK_CLIENT_SECRET,
      code: code as string,
      grant_type: 'authorization_code',
      redirect_uri: 'https://your-pet-wash-app.com/api/auth/callback/tiktok'
    }).toString(), {
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' }
    });

    const { access_token, open_id } = tokenResponse.data;

    // B. Get User Info (Name, Avatar)
    const userResponse = await axios.get('https://open.tiktokapis.com/v2/user/info/', {
      headers: { Authorization: `Bearer ${access_token}` },
      params: { fields: 'open_id,union_id,avatar_url,display_name' }
    });

    // C. Log them into Pet Wash System
    const petOwner = {
      name: userResponse.data.data.user.display_name,
      avatar: userResponse.data.data.user.avatar_url,
      provider: 'tiktok',
      tiktokId: userResponse.data.data.user.open_id
    };

    // TODO: Save 'petOwner' to your database here
    
    res.redirect('/dashboard?login=success');

  } catch (error) {
    console.error('TikTok Login Error:', error);
    res.redirect('/login?error=tiktok_failed');
  }
});
