// Polished 2025 production version – uses MobileDatePicker, VATCalculatorService, real APIs, Nayax-only

import React, { useEffect, useMemo, useState } from "react";
import { useNavigate } from "react-router-dom";
import { MobileDatePicker } from "../../components/ui/mobile-date-picker";
import { useAuth } from "../../hooks/useAuth";
import { apiClient } from "../../lib/apiClient";
import { VATCalculatorService } from "../../../server/services/VATCalculatorService"; // path as in your project
import { Button } from "../../components/ui/button";
import { Spinner } from "../../components/ui/spinner";
import { ErrorBanner } from "../../components/ui/error-banner";

type SitterProfile = {
  id: string;
  name: string;
  photoUrl?: string;
  hourlyRate: number;
  rating: number;
  languages: string[];
  maxPets: number;
};

type Pet = {
  id: string;
  name: string;
  size: "small" | "medium" | "large";
  species: "dog" | "cat" | "other";
};

type BookingStep = "details" | "summary" | "confirmation";

const vatService = new VATCalculatorService();

export default function SitterSuiteBookingFlow() {
  const navigate = useNavigate();
  const { user, requireAuth } = useAuth();

  const [step, setStep] = useState<BookingStep>("details");
  const [sitters, setSitters] = useState<SitterProfile[]>([]);
  const [pets, setPets] = useState<Pet[]>([]);

  const [selectedSitterId, setSelectedSitterId] = useState<string | null>(null);
  const [selectedPetIds, setSelectedPetIds] = useState<string[]>([]);
  const [date, setDate] = useState<Date | null>(null);
  const [hours, setHours] = useState<number>(2);
  const [notes, setNotes] = useState("");
  const [address, setAddress] = useState("");

  const [isLoading, setIsLoading] = useState(true);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [bookingId, setBookingId] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;
    async function loadData() {
      try {
        setIsLoading(true);
        setError(null);

        const [sittersRes, petsRes] = await Promise.all([
          apiClient.get<SitterProfile[]>("/api/platforms/sitter_suite/sitters"),
          apiClient.get<Pet[]>("/api/pets/my")
        ]);

        if (cancelled) return;

        setSitters(sittersRes.data);
        setPets(petsRes.data);

        if (sittersRes.data.length > 0) {
          setSelectedSitterId(sittersRes.data[0].id);
        }
        if (petsRes.data.length > 0) {
          setSelectedPetIds([petsRes.data[0].id]);
        }
      } catch (e: any) {
        if (!cancelled) setError("לא הצלחנו לטעון את נתוני השמרטפים והחיות. נסה/י שוב מאוחר יותר.");
      } finally {
        if (!cancelled) setIsLoading(false);
      }
    }
    loadData();
    return () => {
      cancelled = true;
    };
  }, []);

  const selectedSitter = useMemo(
    () => sitters.find(s => s.id === selectedSitterId) || null,
    [sitters, selectedSitterId]
  );

  const netPrice = useMemo(() => {
    if (!selectedSitter || !hours || hours <= 0) return 0;
    const base = selectedSitter.hourlyRate * hours;
    return vatService.roundToCurrency(base);
  }, [selectedSitter, hours]);

  const vatAmount = useMemo(
    () => vatService.calculateVATAmount(netPrice),
    [netPrice]
  );

  const grossPrice = useMemo(
    () => vatService.calculateTotalWithVAT(netPrice),
    [netPrice]
  );

  const canContinueDetails = useMemo(() => {
    return (
      !!selectedSitter &&
      selectedPetIds.length > 0 &&
      !!date &&
      hours > 0 &&
      address.trim().length > 5
    );
  }, [selectedSitter, selectedPetIds, date, hours, address]);

  async function handleNextFromDetails() {
    const authed = await requireAuth();
    if (!authed) return;
    if (!canContinueDetails) {
      setError("יש למלא את כל הפרטים לפני המשך ההזמנה.");
      return;
    }
    setError(null);
    setStep("summary");
  }

  async function handleConfirmBooking() {
    if (!user) {
      const authed = await requireAuth();
      if (!authed) return;
    }
    if (!selectedSitter || !date) return;

    try {
      setIsSubmitting(true);
      setError(null);

      const payload = {
        platform: "sitter_suite",
        sitterId: selectedSitter.id,
        petIds: selectedPetIds,
        startAtIso: date.toISOString(),
        hours,
        address,
        notes,
        pricing: {
          currency: "ILS",
          netAmount: netPrice,
          vatAmount,
          vatRate: vatService.getCurrentRate(), // for audit trail
          totalAmount: grossPrice
        },
        payment: {
          provider: "NAYAX_ISRAEL",
          mode: "ESCROW_HOLD_72H",
          status: "PENDING_CUSTOMER_TAP"
        }
      };

      const res = await apiClient.post<{ bookingId: string }>(
        "/api/bookings/create",
        payload
      );

      setBookingId(res.data.bookingId);
      setStep("confirmation");
    } catch (e: any) {
      setError("אירעה שגיאה ביצירת ההזמנה. אין עסקה בחיוב. נסה/י שוב.");
    } finally {
      setIsSubmitting(false);
    }
  }

  function togglePet(id: string) {
    setSelectedPetIds(prev =>
      prev.includes(id) ? prev.filter(p => p !== id) : [...prev, id]
    );
  }

  function handleBack() {
    if (step === "summary") setStep("details");
    if (step === "confirmation") navigate("/sitter-suite");
  }

  if (isLoading) {
    return (
      <div className="flex h-full items-center justify-center">
        <Spinner label="טוען את נתוני השמרטפים והחיות שלך..." />
      </div>
    );
  }

  if (error && step === "details" && sitters.length === 0) {
    return (
      <div className="p-4">
        <ErrorBanner message={error} />
        <Button className="mt-4 w-full" onClick={() => window.location.reload()}>
          נסה/י שוב
        </Button>
      </div>
    );
  }

  return (
    <div className="mx-auto flex h-full max-w-xl flex-col bg-white px-4 pb-6 pt-4">
      {error && (
        <div className="mb-3">
          <ErrorBanner message={error} />
        </div>
      )}

      {step === "details" && (
        <>
          <h1 className="mb-3 text-xl font-semibold">
            הזמנת שמרטף/ית פרימיום לחיות המחמד שלך
          </h1>

          <section className="mb-4 rounded-2xl border border-slate-100 bg-slate-50/70 p-3">
            <div className="mb-2 text-xs font-semibold text-slate-500">
              בחר/י שמרטף/ית
            </div>
            <div className="flex gap-2 overflow-x-auto pb-1">
              {sitters.map(s => (
                <button
                  key={s.id}
                  type="button"
                  onClick={() => setSelectedSitterId(s.id)}
                  className={`flex min-w-[130px] flex-col rounded-2xl border p-2 text-left ${
                    s.id === selectedSitterId
                      ? "border-emerald-500 bg-emerald-50"
                      : "border-slate-200 bg-white"
                  }`}
                >
                  <div className="flex items-center gap-2">
                    {s.photoUrl ? (
                      <img
                        src={s.photoUrl}
                        alt={s.name}
                        className="h-8 w-8 rounded-full object-cover"
                      />
                    ) : (
                      <div className="flex h-8 w-8 items-center justify-center rounded-full bg-emerald-100 text-xs font-semibold text-emerald-700">
                        {s.name.charAt(0)}
                      </div>
                    )}
                    <div className="flex flex-col">
                      <span className="text-xs font-semibold">{s.name}</span>
                      <span className="text-[10px] text-slate-500">
                        ₪{s.hourlyRate.toFixed(0)} לשעה · ⭐ {s.rating.toFixed(1)}
                      </span>
                    </div>
                  </div>
                  <div className="mt-1 text-[10px] text-slate-500">
                    {s.languages.join(" · ")}
                  </div>
                </button>
              ))}
            </div>
          </section>

          <section className="mb-4 rounded-2xl border border-slate-100 bg-slate-50/70 p-3">
            <div className="mb-2 text-xs font-semibold text-slate-500">
              חיות מחמד להזמנה
            </div>
            <div className="flex flex-wrap gap-2">
              {pets.map(pet => {
                const active = selectedPetIds.includes(pet.id);
                return (
                  <button
                    key={pet.id}
                    type="button"
                    onClick={() => togglePet(pet.id)}
                    className={`rounded-full px-3 py-1 text-xs ${
                      active
                        ? "bg-emerald-500 text-white"
                        : "bg-white text-slate-700"
                    } border border-slate-200`}
                  >
                    {pet.name}
                  </button>
                );
              })}
            </div>
          </section>

          <section className="mb-4 rounded-2xl border border-slate-100 bg-slate-50/70 p-3">
            <div className="mb-2 text-xs font-semibold text-slate-500">
              תאריך ושעה
            </div>
            <MobileDatePicker
              value={date}
              onChange={setDate}
              minDate={new Date()}
              mode="datetime"
              locale="he-IL"
            />
            <div className="mt-3 flex items-center justify-between">
              <span className="text-xs text-slate-500">משך השמרטפות (שעות)</span>
              <input
                type="number"
                min={1}
                max={12}
                value={hours}
                onChange={e => setHours(Math.max(1, Number(e.target.value) || 1))}
                className="w-16 rounded-xl border border-slate-200 bg-white px-2 py-1 text-right text-sm"
              />
            </div>
          </section>

          <section className="mb-4 rounded-2xl border border-slate-100 bg-slate-50/70 p-3">
            <div className="mb-2 text-xs font-semibold text-slate-500">
              כתובת מלאה
            </div>
            <input
              value={address}
              onChange={e => setAddress(e.target.value)}
              placeholder="רחוב, מספר, עיר, קומה/כניסה"
              className="w-full rounded-xl border border-slate-200 bg-white px-3 py-2 text-sm"
            />
          </section>

          <section className="mb-4 rounded-2xl border border-slate-100 bg-slate-50/70 p-3">
            <div className="mb-2 text-xs font-semibold text-slate-500">
              הערות לשמרטף/ית (אופציונלי)
            </div>
            <textarea
              value={notes}
              onChange={e => setNotes(e.target.value)}
              rows={3}
              className="w-full resize-none rounded-xl border border-slate-200 bg-white px-3 py-2 text-sm"
              placeholder="הנחיות מיוחדות, רגישויות, מפתח, מערכת אזעקה ועוד"
            />
          </section>

          <div className="mt-auto pt-2">
            <div className="mb-2 flex items-center justify-between text-xs text-slate-600">
              <span>סכום לפני מע״מ</span>
              <span>₪{netPrice.toFixed(0)}</span>
            </div>
            <div className="mb-2 flex items-center justify-between text-xs text-slate-600">
              <span>מע״מ {vatService.getCurrentRate().toFixed(1)}%</span>
              <span>₪{vatAmount.toFixed(0)}</span>
            </div>
            <div className="mb-3 flex items-center justify-between text-sm font-semibold text-slate-900">
              <span>סה״כ לחיוב בנאייקס</span>
              <span>₪{grossPrice.toFixed(0)}</span>
            </div>
            <div className="mb-3 text-[10px] text-slate-500">
              החיוב מתבצע אך ורק דרך מערכת התשלומים Nayax Israel בתחנת Pet Wash™ או בהתאם להנחיות בהודעת האישור. סכום העסקה מוחזק ב־escrow ל־72 שעות להגנת שני הצדדים.
            </div>
            <Button
              className="h-11 w-full rounded-2xl text-sm font-semibold"
              disabled={!canContinueDetails}
              onClick={handleNextFromDetails}
            >
              המשך לסיכום
            </Button>
          </div>
        </>
      )}

      {step === "summary" && (
        <>
          <h1 className="mb-3 text-xl font-semibold">סיכום הזמנה</h1>
          <div className="mb-3 rounded-2xl border border-slate-100 bg-slate-50/70 p-3 text-sm">
            <div className="mb-1 font-semibold">{selectedSitter?.name}</div>
            <div className="text-xs text-slate-600">
              תאריך: {date ? date.toLocaleString("he-IL") : "-"}
            </div>
            <div className="text-xs text-slate-600">
              משך: {hours} שעות · חיות:{" "}
              {pets
                .filter(p => selectedPetIds.includes(p.id))
                .map(p => p.name)
                .join(", ")}
            </div>
            <div className="text-xs text-slate-600">כתובת: {address}</div>
          </div>
          <div className="mb-3 rounded-2xl border border-slate-100 bg-slate-50/70 p-3 text-sm">
            <div className="mb-1 text-xs font-semibold text-slate-500">
              פירוט מחיר (כולל מע״מ)
            </div>
            <div className="mb-1 flex items-center justify-between text-xs">
              <span>סכום לפני מע״מ</span>
              <span>₪{netPrice.toFixed(0)}</span>
            </div>
            <div className="mb-1 flex items-center justify-between text-xs">
              <span>מע״מ {vatService.getCurrentRate().toFixed(1)}%</span>
              <span>₪{vatAmount.toFixed(0)}</span>
            </div>
            <div className="mt-1 flex items-center justify-between text-sm font-semibold">
              <span>סה״כ לחיוב בנאייקס</span>
              <span>₪{grossPrice.toFixed(0)}</span>
            </div>
            <div className="mt-2 text-[10px] text-slate-500">
              לאחר אישור ההזמנה תקבל/י הודעה עם פרטי העסקה. החיוב מתבצע רק בנאייקס בתחנות Pet Wash™ או בקישור מאובטח מטעם החברה. אין גובים תשלום באמצעים אחרים.
            </div>
          </div>
          <div className="mt-auto flex gap-2 pt-2">
            <Button
              variant="outline"
              className="h-11 flex-1 rounded-2xl text-sm"
              onClick={handleBack}
            >
              חזרה
            </Button>
            <Button
              className="h-11 flex-1 rounded-2xl text-sm font-semibold"
              onClick={handleConfirmBooking}
              disabled={isSubmitting}
            >
              {isSubmitting ? "שולח הזמנה..." : "אישור והמשך"}
            </Button>
          </div>
        </>
      )}

      {step === "confirmation" && (
        <div className="flex h-full flex-col items-center justify-center text-center">
          <h1 className="mb-2 text-xl font-semibold">ההזמנה נקלטה בהצלחה</h1>
          <p className="mb-3 max-w-xs text-sm text-slate-600">
            השמרטף/ית יקבל/תקבל את פרטי ההזמנה שלך. מספר הזמנה: {bookingId || "בבדיקה"}. פרטי חיוב בנאייקס ישלחו בהודעה נפרדת.
          </p>
          <Button className="mt-2 rounded-2xl px-6" onClick={() => navigate("/dashboard")}>
            חזרה ללוח הבקרה
          </Button>
        </div>
      )}
    </div>
  );
}
