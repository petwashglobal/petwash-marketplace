function clamp(n: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, n));
}

export function normalizeProviderSearchFilters(query: any): ProviderSearchFilters {
  return {
    q: typeof query.q === "string" ? query.q.trim() : "",
    serviceType: (query.serviceType || "") as any,
    startDate: query.startDate || undefined,
    endDate: query.endDate || undefined,
    postcode: query.postcode || undefined,
    lat: query.lat ? Number(query.lat) : undefined,
    lng: query.lng ? Number(query.lng) : undefined,
    radiusKm: query.radiusKm ? clamp(Number(query.radiusKm), 1, 100) : 15,

    verifiedOnly: String(query.verifiedOnly || "") === "true",
    instantBookOnly: String(query.instantBookOnly || "") === "true",
    insuredOnly: String(query.insuredOnly || "") === "true",

    minRating: query.minRating ? clamp(Number(query.minRating), 0, 5) : 0,
    minReviews: query.minReviews ? Math.max(0, Number(query.minReviews)) : 0,
    priceMin: query.priceMin ? Math.max(0, Number(query.priceMin)) : 0,
    priceMax: query.priceMax ? Math.max(0, Number(query.priceMax)) : 999999,

    petType: (query.petType || "") as any,
    petSize: (query.petSize || "") as any,

    sort: (query.sort || "recommended") as ProviderSortMode,
    page: query.page ? Math.max(1, Number(query.page)) : 1,
    pageSize: query.pageSize ? clamp(Number(query.pageSize), 1, 50) : 20,
  };
}

export function haversineKm(aLat: number, aLng: number, bLat: number, bLng: number): number {
  const toRad = (v: number) => (v * Math.PI) / 180;
  const R = 6371;
  const dLat = toRad(bLat - aLat);
  const dLng = toRad(bLng - aLng);
  const lat1 = toRad(aLat);
  const lat2 = toRad(bLat);

  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLng / 2) ** 2;

  return 2 * R * Math.asin(Math.sqrt(h));
}

export function applyProviderFilters(
  providers: ProviderSearchItem[],
  filters: ProviderSearchFilters
): ProviderSearchItem[] {
  return providers.filter((p) => {
    if (filters.serviceType && !p.supportedServices.includes(filters.serviceType)) return false;
    if (filters.verifiedOnly && !p.verified) return false;
    if (filters.instantBookOnly && !p.instantBook) return false;
    if (filters.insuredOnly && !p.insured) return false;
    if ((filters.minRating || 0) > p.rating) return false;
    if ((filters.minReviews || 0) > p.reviewsCount) return false;
    if (p.startingPrice < (filters.priceMin || 0)) return false;
    if (p.startingPrice > (filters.priceMax || 999999)) return false;

    if (filters.q) {
      const haystack = [
        p.displayName,
        p.shortBio,
        p.city,
        p.suburb,
        p.postcode,
      ]
        .filter(Boolean)
        .join(" ")
        .toLowerCase();

      if (!haystack.includes(filters.q.toLowerCase())) return false;
    }

    // Open browse mode:
    // If no dates are provided, allow providers to show.
    // If dates are provided, only show providers available for requested dates.
    if (filters.startDate && filters.endDate && !p.availableForRequestedDates) return false;

    return true;
  });
}

export function scoreProvider(
  provider: ProviderSearchItem,
  filters: ProviderSearchFilters,
  userLocation?: { lat?: number; lng?: number }
): ProviderSearchItem {
  let distanceKm: number | undefined;

  if (
    typeof userLocation?.lat === "number" &&
    typeof userLocation?.lng === "number" &&
    typeof provider.lat === "number" &&
    typeof provider.lng === "number"
  ) {
    distanceKm = haversineKm(userLocation.lat, userLocation.lng, provider.lat, provider.lng);
  }

  provider.distanceKm = distanceKm;

  // Hard radius cutoff when location exists
  if (distanceKm != null && distanceKm > (filters.radiusKm || 15)) {
    provider.rankingScore = -999999;
    return provider;
  }

  const distanceScore =
    distanceKm == null ? 0.30 : Math.max(0, 1 - distanceKm / (filters.radiusKm || 15));

  const ratingScore = provider.rating / 5;
  const trustScore =
    (provider.verified ? 0.45 : 0) +
    (provider.insured ? 0.20 : 0) +
    Math.min(0.35, provider.reviewsCount / 400);

  const bookingVolumeScore = Math.min(1, provider.completedBookings / 1000);
  const priceScore = 1 / Math.max(1, provider.startingPrice);
  const responseScore =
    provider.responseTimeMinutes != null
      ? 1 / Math.max(1, provider.responseTimeMinutes)
      : 0.1;

  let score = 0;

  switch (filters.sort || "recommended") {
    case "closest":
      score =
        distanceScore * 0.70 +
        ratingScore * 0.15 +
        trustScore * 0.10 +
        bookingVolumeScore * 0.05;
      break;

    case "top_rated":
      score =
        ratingScore * 0.55 +
        trustScore * 0.20 +
        distanceScore * 0.15 +
        bookingVolumeScore * 0.10;
      break;

    case "lowest_price":
      score =
        priceScore * 0.60 +
        distanceScore * 0.15 +
        ratingScore * 0.15 +
        trustScore * 0.10;
      break;

    case "fastest_response":
      score =
        responseScore * 0.55 +
        ratingScore * 0.20 +
        trustScore * 0.15 +
        distanceScore * 0.10;
      break;

    case "recommended":
    default:
      score =
        distanceScore * 0.40 +
        ratingScore * 0.25 +
        trustScore * 0.15 +
        bookingVolumeScore * 0.10 +
        responseScore * 0.10;
      break;
  }

  if (provider.instantBook) score += 0.02;
  if (provider.verified) score += 0.015;
  if (provider.insured) score += 0.01;

  provider.rankingScore = score;
  return provider;
}

export async function runProviderSearch(filters: ProviderSearchFilters) {
  const location = await resolveSearchLocation(filters);
  const providers = await fetchMarketplaceProviders(filters);

  const scored = providers
    .map((p) => scoreProvider({ ...p }, filters, location))
    .filter((p) => p.rankingScore > -999999);

  const filtered = applyProviderFilters(scored, filters)
    .sort((a, b) => b.rankingScore - a.rankingScore);

  const page = filters.page || 1;
  const pageSize = filters.pageSize || 20;
  const offset = (page - 1) * pageSize;

  return {
    filters,
    total: filtered.length,
    page,
    pageSize,
    results: filtered.slice(offset, offset + pageSize),
    debug: {
      usedLocation: location.source !== "none",
      locationSource: location.source,
    },
  };
}

  try {
    const filters = normalizeProviderSearchFilters(req.query);
    const result = await runProviderSearch(filters);

      ok: false,
      error: "Provider search failed",
      reason: error?.message || "Unknown error",
    });

import type {
  ProviderSearchFilters,
  ProviderSearchResponse,
} from "../../../shared/provider-search-types";

export async function fetchProviderSearch(
  filters: ProviderSearchFilters
): Promise<ProviderSearchResponse> {
  const params = new URLSearchParams();

  Object.entries(filters).forEach(([key, value]) => {
    if (value === undefined || value === null || value === "" || value === false) {
      return;
    }
    params.set(key, String(value));
  });

  const response = await fetch(`/api/providers/search?${params.toString()}`, {
    credentials: "include",
  });

  if (!response.ok) {
    throw new Error(`Search request failed: ${response.status}`);
  }

  return response.json();
}

import { useEffect, useMemo, useState } from "react";
import type {

function parseFiltersFromUrl(): ProviderSearchFilters {
  const url = new URL(window.location.href);
  const q = url.searchParams;

  return {
    q: q.get("q") || "",
    serviceType: (q.get("serviceType") || "") as any,
    startDate: q.get("startDate") || undefined,
    endDate: q.get("endDate") || undefined,
    postcode: q.get("postcode") || undefined,
    radiusKm: q.get("radiusKm") ? Number(q.get("radiusKm")) : 15,
    verifiedOnly: q.get("verifiedOnly") === "true",
    instantBookOnly: q.get("instantBookOnly") === "true",
    insuredOnly: q.get("insuredOnly") === "true",
    minRating: q.get("minRating") ? Number(q.get("minRating")) : 0,
    minReviews: q.get("minReviews") ? Number(q.get("minReviews")) : 0,
    priceMin: q.get("priceMin") ? Number(q.get("priceMin")) : 0,
    priceMax: q.get("priceMax") ? Number(q.get("priceMax")) : 500,
    petType: (q.get("petType") || "") as any,
    petSize: (q.get("petSize") || "") as any,
    sort: (q.get("sort") || "recommended") as ProviderSortMode,
    page: q.get("page") ? Number(q.get("page")) : 1,
    pageSize: 20,
  };
}

function syncFiltersToUrl(filters: ProviderSearchFilters) {
  const url = new URL(window.location.href);
  url.search = "";

  Object.entries(filters).forEach(([key, value]) => {
    if (value === undefined || value === null || value === "" || value === false) {
      return;
    }
    url.searchParams.set(key, String(value));
  });

  window.history.replaceState({}, "", url.toString());
}

export function useProviderSearch() {
  const [filters, setFilters] = useState<ProviderSearchFilters>(parseFiltersFromUrl());
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<ProviderSearchResponse | null>(null);
  const [error, setError] = useState("");

  async function runSearch(nextFilters?: ProviderSearchFilters) {
    const merged = nextFilters || filters;
    setLoading(true);
    setError("");

    try {
      syncFiltersToUrl(merged);
      const result = await fetchProviderSearch(merged);
      setData(result);
    } catch (err: any) {
      setError(err?.message || "Search failed");
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    runSearch(filters);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const activeFilterCount = useMemo(() => {
    let count = 0;
    if (filters.verifiedOnly) count++;
    if (filters.instantBookOnly) count++;
    if (filters.insuredOnly) count++;
    if ((filters.minRating || 0) > 0) count++;
    if ((filters.minReviews || 0) > 0) count++;
    if ((filters.priceMin || 0) > 0 || (filters.priceMax || 500) < 500) count++;
    if ((filters.radiusKm || 15) !== 15) count++;
    return count;
  }, [filters]);

  return {
    filters,
    setFilters,
    loading,
    data,
    error,
    runSearch,
    activeFilterCount,
  };
}

import React, { useState } from "react";
import { useProviderSearch } from "../hooks/useProviderSearch";
import { ProviderSearchHero } from "../components/providers/ProviderSearchHero";
import { ProviderFilterPanel } from "../components/providers/ProviderFilterPanel";
import { ProviderCard } from "../components/providers/ProviderCard";

export function ProviderSearchPage() {
  const {
    filters,
    setFilters,
    loading,
    data,
    error,
    runSearch,
    activeFilterCount,
  } = useProviderSearch();

  const [mobileFiltersOpen, setMobileFiltersOpen] = useState(false);
  const results = data?.results || [];

  function resetFilters() {
    const next = {
      q: filters.q || "",
      serviceType: filters.serviceType || "",
      startDate: filters.startDate,
      endDate: filters.endDate,
      postcode: filters.postcode,
      radiusKm: 15,
      verifiedOnly: false,
      instantBookOnly: false,
      insuredOnly: false,
      minRating: 0,
      minReviews: 0,
      priceMin: 0,
      priceMax: 500,
      petType: "",
      petSize: "",
      sort: "recommended" as const,
      page: 1,
      pageSize: 20,
    };
    setFilters(next);
    runSearch(next);
  }

  return (
    <div className="pw-search-page">
      <ProviderSearchHero
        filters={filters}
        onChange={setFilters}
        onSearch={() => runSearch({ ...filters, page: 1 })}
      />

      <div className="pw-search-toolbar mobile-only">
        <button className="pw-btn" onClick={() => setMobileFiltersOpen(true)}>
          Filters {activeFilterCount > 0 ? `(${activeFilterCount})` : ""}
        </button>

        <select
          className="pw-input"
          value={filters.sort || "recommended"}
          onChange={(e) => {
            const next = { ...filters, sort: e.target.value as any, page: 1 };
            setFilters(next);
            runSearch(next);
          }}
        >
          <option value="recommended">Recommended</option>
          <option value="closest">Closest</option>
          <option value="top_rated">Top rated</option>
          <option value="lowest_price">Lowest price</option>
          <option value="fastest_response">Fastest response</option>
        </select>
      </div>

      <div className="pw-search-layout">
        <aside className="pw-search-sidebar desktop-only">
          <ProviderFilterPanel
            filters={filters}
            onChange={setFilters}
            onApply={() => runSearch({ ...filters, page: 1 })}
            onReset={resetFilters}
          />
        </aside>

        <main className="pw-search-results">
          {data && (
            <div className="pw-search-meta">
              <strong>{data.total}</strong> providers found
              {data.debug?.usedLocation ? " near your search" : ""}
            </div>
          )}

          {loading && <div className="pw-loading">Loading providers...</div>}
          {!loading && error && <div className="pw-error">{error}</div>}

          {!loading && !error && results.length === 0 && (
            <div className="pw-empty-state">
              <h3>No providers found</h3>
              <p>Try widening your radius, removing one filter, or changing dates.</p>
            </div>
          )}

          {!loading && !error && results.length > 0 && (
            <div className="pw-provider-grid">
              {results.map((item) => (
                <ProviderCard key={item.providerId} item={item} />
              ))}
            </div>
          )}
        </main>
      </div>

      {mobileFiltersOpen && (
        <div
          className="pw-mobile-sheet-backdrop"
          onClick={() => setMobileFiltersOpen(false)}
        >
          <div
            className="pw-mobile-sheet"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="pw-mobile-sheet-header">
              <h3>Filters</h3>
              <button className="pw-btn" onClick={() => setMobileFiltersOpen(false)}>
                Close
              </button>
            </div>

            <ProviderFilterPanel
              filters={filters}
              onChange={setFilters}
              onApply={() => {
                runSearch({ ...filters, page: 1 });
                setMobileFiltersOpen(false);
              }}
              onReset={resetFilters}
            />
          </div>
        </div>
      )}
    </div>
  );
}
