// Legal documents supported by the backend.
// These keys must match what the backend returns from:
// GET /api/legal/:docKey
export type LegalDocumentKey =
  | "terms_of_use"
  | "privacy_policy"
  | "cookie_policy"
  | "cancellation_policy"
  | "payments_security"
  | "marketing_consent"
  | "accessibility_statement";

export type ConsentCategory =
  | "necessary"
  | "functional"
  | "analytics"
  | "marketing";

export type PrivacyRequestType =
  | "access" // עיון
  | "delete" // מחיקה
  | "rectify"; // תיקון נתונים

export interface LegalDocumentContent {
  key: LegalDocumentKey;
  title: string;
  locale: "he" | "en";
  // Plain HTML string or markdown transformed on backend
  htmlContent: string;
  lastUpdatedIso?: string;
}

export interface ConsentState {
  necessary: boolean;
  functional: boolean;
  analytics: boolean;
  marketing: boolean;
  decisionAtIso?: string;
}

// Small helper to detect Hebrew direction.
function isHebrewLocale(locale: "he" | "en") {
  return locale === "he";
}

// Local storage key for cookie banner decisions
const COOKIE_CONSENT_STORAGE_KEY = "petwash_cookie_consent_2025";

// Basic API base path. Adjust if you use a different prefix.
const API_BASE = "/api";

// ---------------------------------
// 2. API helper functions (frontend)
// ---------------------------------

// Fetch a legal document from backend.
// Expected backend endpoint: GET /api/legal/:docKey
export async function fetchLegalDocument(
  docKey: LegalDocumentKey,
  locale: "he" | "en" = "he"
): Promise<LegalDocumentContent> {
  const res = await fetch(`${API_BASE}/legal/${docKey}?locale=${locale}`, {
    method: "GET",
    credentials: "include",
  });

  if (!res.ok) {
    throw new Error(`Failed to fetch legal document: ${docKey}`);
  }

  return (await res.json()) as LegalDocumentContent;
}

// Post cookie consent decision.
// Expected backend endpoint: POST /api/consents
export async function postConsent(input: {
  consentType: "cookies" | "marketing" | "other";
  categories?: ConsentCategory[];
  granted: boolean;
  locale?: "he" | "en";
}): Promise<{ ok: boolean }> {
  const res = await fetch(`${API_BASE}/consents`, {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(input),
  });

  if (!res.ok) {
    throw new Error("Failed to submit consent");
  }

  return (await res.json()) as { ok: boolean };
}

// Submit a privacy request.
// Expected backend endpoint: POST /api/privacy/requests
export async function postPrivacyRequest(input: {
  requestType: PrivacyRequestType;
  fullName: string;
  email: string;
  phone?: string;
  description?: string;
  locale?: "he" | "en";
}): Promise<{ ok: boolean }> {
  const res = await fetch(`${API_BASE}/privacy/requests`, {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(input),
  });

  if (!res.ok) {
    throw new Error("Failed to submit privacy request");
  }

  return (await res.json()) as { ok: boolean };
}

// 5. LegalDocumentView - generic viewer for documents
// --------------------------------------------------

interface LegalDocumentViewProps {
  docKey: LegalDocumentKey;
  locale?: "he" | "en";
  pageTitle?: string;
}

// LegalDocumentView - קומפוננטה גנרית להצגת מסמך משפטי אחד
export const LegalDocumentView: React.FC<LegalDocumentViewProps> = ({
  docKey,
  locale = "he",
  pageTitle,
}) => {
  const [loading, setLoading] = useState(true);
  const [doc, setDoc] = useState<LegalDocumentContent | null>(null);
  const [error, setError] = useState<string | null>(null);

  const rtl = isHebrewLocale(locale);

  useEffect(() => {
    let mounted = true;
    setLoading(true);
    setError(null);

    fetchLegalDocument(docKey, locale)
      .then((data) => {
        if (!mounted) return;
        setDoc(data);
      })
      .catch((err) => {
        console.error("Failed to load legal document", err);
        if (!mounted) return;
        setError(
          rtl
            ? "לא ניתן לטעון את המסמך כעת. נסה שוב מאוחר יותר."
            : "Unable to load this document at the moment. Please try again later."
        );
      })
      .finally(() => {
        if (!mounted) return;
        setLoading(false);
      });

    return () => {
      mounted = false;
    };
  }, [docKey, locale, rtl]);

  const effectiveTitle =
    pageTitle || doc?.title || (rtl ? "מסמך משפטי" : "Legal Document");

  return (
    <LegalShellLayout locale={locale} pageTitle={effectiveTitle}>
      {loading && (
        <p className="text-sm text-neutral-600">
          {rtl ? "טוען מסמך..." : "Loading document..."}
        </p>
      )}
      {!loading && error && (
        <p className="text-sm text-red-600" data-testid="legal-error">
          {error}
        </p>
      )}
      {!loading && !error && doc && (
        <article
          className="legal-document-content"
          // This uses htmlContent returned from backend.
          // Backend must sanitize HTML and remove unsafe elements.
          dangerouslySetInnerHTML={{ __html: doc.htmlContent }}
        />
      )}
    </LegalShellLayout>
  );
};

interface CookieBanner2025Props {
  locale?: "he" | "en";
}

// Helper to read consent from localStorage
function loadStoredConsent(): ConsentState | null {
  try {
    const raw = window.localStorage.getItem(COOKIE_CONSENT_STORAGE_KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw) as ConsentState;
    return parsed || null;
  } catch {
    return null;
  }
}

// Helper to save consent to localStorage
function storeConsent(state: ConsentState) {
  try {
    window.localStorage.setItem(
      COOKIE_CONSENT_STORAGE_KEY,
      JSON.stringify(state)
    );
  } catch {
    // ignore
  }
}

// CookieBanner2025 - באנר עוגיות מלא בהתאם לתיקון 13

  const rtl = isHebrewLocale(locale);
  const alignClass = rtl ? "text-right" : "text-left";

  const [isOpen, setIsOpen] = useState(false);
  const [consent, setConsent] = useState<ConsentState>({
    necessary: true, // חובה לפי החוק, לא ניתן לבטל
    functional: false,
    analytics: false,
    marketing: false,
    decisionAtIso: undefined,

  useEffect(() => {
    if (typeof window === "undefined") return;
    const stored = loadStoredConsent();
    if (!stored) {
      setIsOpen(true);
    } else {
      setConsent(stored);
      setIsOpen(false);
    }

      necessary: true,
      functional: grantAll ? true : consent.functional,
      analytics: grantAll ? true : consent.analytics,
      marketing: grantAll ? true : consent.marketing,

    storeConsent(newState);

        consentType: "cookies",
        granted: true,

          "necessary",
          ...(newState.functional ? ["functional"] : []),

          ...(newState.marketing ? ["marketing"] : []),

        locale,
      });

    setSubmitting(true);
    setMessage(null);
    setError(null);

    try {
      await postPrivacyRequest({
        requestType,
        fullName,
        email,
        phone: phone || undefined,
        description: description || undefined,
        locale,
      });

      setMessage(
        rtl
          ? "הבקשה התקבלה במערכת. נעדכן אותך בהתאם ללוחות הזמנים הקבועים בחוק."
          : "Your request has been received. We will get back to you within the time frames required by law."
      );
      setFullName("");
      setEmail("");
      setPhone("");
      setDescription("");
    } catch (err) {
      console.error("Failed to submit privacy request", err);
      setError(
        rtl
          ? "אירעה תקלה בשליחת הבקשה. נסו שוב מאוחר יותר."
          : "An error occurred while submitting your request. Please try again later."
      );
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <LegalShellLayout
      locale={locale}
      pageTitle={
        rtl ? "מרכז זכויות הפרטיות שלך" : "Your Data Protection Rights Center"
      }
    >
      <div className={`space-y-4 ${alignClass}`}>
        <p className="text-sm text-neutral-700">
          {rtl
            ? "במרכז זה תוכל לממש זכויות כמו עיון במידע, בקשת מחיקה או תיקון פרטים, בהתאם לחוק הגנת הפרטיות ותיקון 13. המידע בטופס זה יועבר לצוות המתאים, ואנו נחזור אליך במסגרת לוחות הזמנים הנדרשים."
            : "Here you can exercise your rights such as access to data, deletion requests or correction of details, in line with the Israeli Privacy Protection Law and Amendment 13. The information in this form is sent to the appropriate team and we will respond within the required time frames."}
        </p>

        <form
          onSubmit={handleSubmit}
          className="mt-4 space-y-4 rounded-2xl border border-neutral-200 bg-neutral-50 p-4"
        >
          {/* Request type */}
          <div className="space-y-2 text-sm">
            <label className="font-medium text-neutral-900">
              {rtl ? "סוג הבקשה" : "Request type"}
            </label>
            <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
              <label className="flex cursor-pointer items-center gap-2 rounded-xl bg-white px-3 py-2 text-xs hover:bg-neutral-50">
                <input
                  type="radio"
                  name="requestType"
                  value="access"
                  checked={requestType === "access"}
                  onChange={() => setRequestType("access")}
                />
                <span>
                  {rtl ? "עיון במידע" : "Access my data"}
                </span>
              </label>
              <label className="flex cursor-pointer items-center gap-2 rounded-xl bg-white px-3 py-2 text-xs hover:bg-neutral-50">
                <input
                  type="radio"
                  name="requestType"
                  value="delete"
                  checked={requestType === "delete"}
                  onChange={() => setRequestType("delete")}
                />
                <span>
                  {rtl ? "מחיקת נתונים" : "Delete my data"}
                </span>
              </label>
              <label className="flex cursor-pointer items-center gap-2 rounded-xl bg-white px-3 py-2 text-xs hover:bg-neutral-50">
                <input
                  type="radio"
                  name="requestType"
                  value="rectify"
                  checked={requestType === "rectify"}
                  onChange={() => setRequestType("rectify")}
                />
                <span>
                  {rtl ? "תיקון נתונים" : "Rectify my data"}
                </span>
              </label>
            </div>
          </div>

          {/* Full name */}
          <div className="text-sm">
            <label className="mb-1 block font-medium text-neutral-900">
              {rtl ? "שם מלא" : "Full name"}
            </label>
            <input
              type="text"
              required
              className="w-full rounded-xl border border-neutral-300 bg-white px-3 py-2 text-sm text-neutral-900 outline-none focus:border-emerald-500"
              value={fullName}
              onChange={(e) => setFullName(e.target.value)}
            />
          </div>

          {/* Email */}
          <div className="text-sm">
            <label className="mb-1 block font-medium text-neutral-900">
              {rtl ? "דואר אלקטרוני ליצירת קשר" : "Contact email"}
            </label>
            <input
              type="email"
              required
              className="w-full rounded-xl border border-neutral-300 bg-white px-3 py-2 text-sm text-neutral-900 outline-none focus:border-emerald-500"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
            />
          </div>

          {/* Phone (optional) */}
          <div className="text-sm">
            <label className="mb-1 block font-medium text-neutral-900">
              {rtl ? "טלפון (לא חובה)" : "Phone (optional)"}
            </label>
            <input
              type="tel"
              className="w-full rounded-xl border border-neutral-300 bg-white px-3 py-2 text-sm text-neutral-900 outline-none focus:border-emerald-500"
              value={phone}
              onChange={(e) => setPhone(e.target.value)}
            />
          </div>

          {/* Description */}
          <div className="text-sm">
            <label className="mb-1 block font-medium text-neutral-900">
              {rtl ? "פירוט הבקשה" : "Request details"}
            </label>
            <textarea
              className="min-h-[96px] w-full rounded-xl border border-neutral-300 bg-white px-3 py-2 text-sm text-neutral-900 outline-none focus:border-emerald-500"
              value={description}
              onChange={(e) => setDescription(e.target.value)}
            />
            <p className="mt-1 text-[11px] text-neutral-500">
              {rtl
                ? "ניתן לציין לדוגמה באילו שירותים השתמשת, האם יש מספר חשבון, תאריך פעילות משוער וכדומה."
                : "You may include for example which services you used, any account number, approximate dates of activity and similar details."}
            </p>
          </div>

          {/* Feedback messages */}
          {message && (
            <p className="text-xs text-emerald-700" data-testid="privacy-success">
              {message}
            </p>
          )}
          {error && (
            <p className="text-xs text-red-600" data-testid="privacy-error">
              {error}
            </p>
          )}

          {/* Submit */}
          <div className={rtl ? "text-left" : "text-right"}>
            <button
              type="submit"
              disabled={submitting}
              className="inline-flex items-center rounded-full bg-emerald-600 px-5 py-2 text-xs font-medium text-white hover:bg-emerald-700 disabled:opacity-60"
            >
              {submitting
                ? rtl
                  ? "שולח בקשה..."
                  : "Sending request..."
                : rtl
                ? "שלח בקשה"
                : "Submit request"}
            </button>
          </div>
        </form>

        <div className="rounded-2xl bg-neutral-100 p-3 text-[11px] text-neutral-700">
          <p>
            {rtl
              ? "הערה: מרכז זה הוא כלי טכני למימוש זכויות בהתאם לחוק. המענה בפועל ותוכן התשובה כפופים לפרשנות משפטית ולבדיקת הממונה על הגנת הפרטיות בחברה, אם מונה כזה."
              : "Note: this center is a technical tool to help you exercise your rights under the law. The actual response is subject to legal interpretation and review by the company's Data Protection Officer or legal team, if applicable."}
          </p>
        </div>
      </div>
    </LegalShellLayout>
  );
};

// 8. MarketingConsentToggle - reusable marketing consent UI
// ----------------------------------------------------------

interface MarketingConsentToggleProps {
  locale?: "he" | "en";
  initialValue?: boolean;
  onChange?: (value: boolean) => void;
}

// Reusable marketing consent control for profile or checkout forms
export const MarketingConsentToggle: React.FC<MarketingConsentToggleProps> = ({
  locale = "he",
  initialValue = false,
  onChange,
}) => {
  const rtl = isHebrewLocale(locale);
  const [checked, setChecked] = useState<boolean>(initialValue);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleToggle() {
    const newValue = !checked;
    setChecked(newValue);
    setSaving(true);
    setError(null);

    try {
      await postConsent({
        consentType: "marketing",
        granted: newValue,
        categories: newValue ? ["marketing"] : [],
        locale,
      });
      if (onChange) onChange(newValue);
    } catch (err) {
      console.error("Failed to update marketing consent", err);
      setError(
        rtl
          ? "לא ניתן לעדכן כרגע את ההעדפה. נסה שוב מאוחר יותר."
          : "Unable to update this preference at the moment. Please try again later."
      );
      // revert UI if failed
      setChecked(!newValue);
    } finally {
      setSaving(false);
    }
  }

  return (
    <div
      className={`rounded-2xl border border-neutral-200 bg-neutral-50 p-3 text-xs ${
        rtl ? "text-right" : "text-left"
      }`}
      dir={rtl ? "rtl" : "ltr"}
      data-testid="marketing-consent-toggle"
    >
      <div className="flex items-start gap-2">
        <button
          type="button"
          onClick={handleToggle}
          disabled={saving}
          className={`
            mt-1 inline-flex h-5 w-9 items-center rounded-full border
            ${checked ? "bg-emerald-500 border-emerald-500" : "bg-white border-neutral-300"}
          `}
          aria-pressed={checked}
        >
          <span
            className={`
              block h-4 w-4 rounded-full bg-white shadow transition-transform
              ${checked ? "translate-x-4" : "translate-x-0"}
            `}
          />
        </button>
        <div>
          <div className="font-medium text-neutral-900">
            {rtl
              ? "קבלת דיוור ועדכונים שיווקיים"
              : "Receive marketing updates and offers"}
          </div>
          <p className="mt-1 text-[11px] text-neutral-600">
            {rtl
              ? "באפשרותך לבחור אם לקבל מאיתנו עדכונים, הצעות והטבות בדואר אלקטרוני או בערוצים דיגיטליים אחרים. ניתן לשנות את ההעדפה בכל עת."
              : "You can choose whether to receive news, offers and benefits from us by email or other digital channels. You can change this preference at any time."}
          </p>
          {error && <p className="mt-1 text-[11px] text-red-600">{error}</p>}
        </div>
      </div>
    </div>
  );
};
