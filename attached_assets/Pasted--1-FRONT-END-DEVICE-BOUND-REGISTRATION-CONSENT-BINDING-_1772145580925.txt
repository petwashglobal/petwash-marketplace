    let registrationRequest = provider.createCredentialRegistrationRequest(challenge: challenge, name: "Emp_123", userID: userId)
    
    // CRITICAL: Set to .platform to ensure the key NEVER leaves this specific phone
    registrationRequest.authenticatorSelection.authenticatorAttachment = .platform
    registrationRequest.attestationPreference = .direct // Request hardware proof
    
    let controller = ASAuthorizationController(authorizationRequests: [registrationRequest])
    controller.performRequests()
}

import { verifyAuthenticationResponse } from '@simplewebauthn/server';

async function verifyClockIn(user, response, expectedChallenge) {
    const verification = await verifyAuthenticationResponse({
        response,
        expectedChallenge,
        expectedOrigin: ['ios:bundle-id', 'android:apk-key-hash'],
        expectedRPID: 'corp.app',
        authenticator: user.devices[response.id], // The public key saved during Step 1
        requireUserVerification: true, // Forces FaceID check
    });

    if (verification.verified) {
        // SYSTEM IS PERFECT: You now have mathematical proof that:
        // 1. The specific physical device was present.
        // 2. The user successfully passed FaceID/Biometrics.
        // 3. The time/location payload hasn't been tampered with.
        return saveClockInToDatabase({
            userId: user.id,
            timestamp: new Date(),
            verifiedHardwareId: response.id,
            signatureStamp: response.signature
        });
    }
}
