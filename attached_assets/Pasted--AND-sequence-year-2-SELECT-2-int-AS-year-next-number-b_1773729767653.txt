          AND sequence_year = $2
      )
      SELECT $2::int AS year, next_number::bigint AS number
      FROM next_val
    `, [documentType, year]);

    return {
      year,
      number: Number(row.rows[0].number),
    };
  }
}

export interface EnqueueJobInput {
  jobType: AsyncJobType;
  entityType: string;
  entityId: string;
  payload: Record<string, unknown>;
}

export class AsyncJobService {
  constructor(private db: any) {}

  async enqueue(input: EnqueueJobInput, trx?: any): Promise<void> {
    const runner = trx ?? this.db;
    await runner.insert("pw_async_jobs").values({
      jobType: input.jobType,
      entityType: input.entityType,
      entityId: input.entityId,
      payload: input.payload,
      status: "PENDING",
      attempts: 0,
      maxAttempts: 10,
    });
  }

  async claimBatch(workerId: string, limit = 10): Promise<AsyncJobRow[]> {
    const rows = await this.db.execute(`
      UPDATE pw_async_jobs
      SET status = 'PROCESSING',
          locked_at = NOW(),
          locked_by = $1,
          updated_at = NOW()
      WHERE id IN (
        SELECT id
        FROM pw_async_jobs
        WHERE status IN ('PENDING', 'FAILED')
          AND next_run_at <= NOW()
          AND attempts < max_attempts
        ORDER BY created_at ASC
        LIMIT $2
        FOR UPDATE SKIP LOCKED
      )
      RETURNING *
    `, [workerId, limit]);

    return rows.rows;
  }

  async markDone(jobId: string): Promise<void> {
    await this.db.update("pw_async_jobs")
      .set({
        status: "DONE",
        updatedAt: new Date(),
      })
      .where({ id: jobId });
  }

  async markFailed(jobId: string, attempts: number, errorMessage: string): Promise<void> {
    const nextDelaySeconds = Math.min(3600, Math.pow(2, attempts) * 30);

    await this.db.update("pw_async_jobs")
      .set({
        status: "FAILED",
        attempts,
        lastError: errorMessage.slice(0, 1000),
        nextRunAt: new Date(Date.now() + nextDelaySeconds * 1000),
        updatedAt: new Date(),
      })
      .where({ id: jobId });
  }
}

- store Drive file ID + SHA-256 back into DB

import { google } from "googleapis";

export interface GoogleDriveConfig {
  serviceAccountJson: string;
  taxDocumentsFolderId: string;
}

export class DriveArchiveService {
  private auth: any;
  private drive: any;
  private folderId: string;

  constructor(config: GoogleDriveConfig) {
    const creds = JSON.parse(config.serviceAccountJson);
    this.auth = new google.auth.JWT(
      creds.client_email,
      undefined,
      creds.private_key,
      [
        "https://www.googleapis.com/auth/drive.file",
        "https://www.googleapis.com/auth/drive.metadata",
      ]
    );
    this.drive = google.drive({ version: "v3", auth: this.auth });
    this.folderId = config.taxDocumentsFolderId;
  }

  async uploadPdf(params: {
    filename: string;
    pdfBuffer: Buffer;
    description?: string;
  }): Promise<{ fileId: string; sha256: string }> {
    const sha256 = crypto.createHash("sha256").update(params.pdfBuffer).digest("hex");

    const created = await this.drive.files.create({
      requestBody: {
        name: params.filename,
        parents: [this.folderId],
        description: params.description ?? "",
        mimeType: "application/pdf",
      },
      media: {
        mimeType: "application/pdf",
        body: ReadableFromBuffer(params.pdfBuffer),
      },
      fields: "id",
    });

    return {
      fileId: created.data.id,
      sha256,
    };
  }
}

function ReadableFromBuffer(buffer: Buffer) {
  const { Readable } = require("stream");
  return Readable.from(buffer);
}

/************************************************************************************************

export class ArchiveTaxDocumentJobHandler {
  constructor(
    private db: any,
    private renderer: TaxPdfRenderer,
    private drive: DriveArchiveService
  ) {}

  async run(job: AsyncJobRow): Promise<void> {
    const docRows = await this.db.select().from("pw_tax_documents").where({ id: job.payload.taxDocumentId });
    const doc = docRows[0] as TaxDocumentRow | undefined;
    if (!doc) throw new Error("TAX_DOCUMENT_NOT_FOUND");

    const pdf = await this.renderer.render(doc);
    const filename = `${doc.documentType}-${doc.sequenceYear}-${doc.sequenceNumber}.pdf`;

    const uploaded = await this.drive.uploadPdf({
      filename,
      pdfBuffer: pdf,
      description: `Pet Wash tax document ${doc.id}`,
    });

    await this.db.update("pw_tax_documents")
      .set({
        driveFileId: uploaded.fileId,
        driveFolderId: process.env.GOOGLE_DRIVE_TAX_FOLDER_ID ?? null,
        pdfSha256: uploaded.sha256,
        archiveStatus: "ARCHIVED",
        archiveAttempts: (doc.archiveAttempts ?? 0) + 1,
        archivedAt: new Date(),
      })
      .where({ id: doc.id });

    await this.db.insert("pw_external_exports").values({
      channel: "DRIVE",
      entityType: "TAX_DOCUMENT",
      entityId: doc.id,
      piiLevel: "FULL_INTERNAL",
      status: "DONE",
      externalRef: uploaded.fileId,
      payloadHash: uploaded.sha256,
    });
  }
}

export class AsyncJobWorker {
  constructor(
    private db: any,
    private asyncJobService: AsyncJobService,
    private archiveTaxDocumentHandler: ArchiveTaxDocumentJobHandler,
  ) {}

  async tick(workerId = `worker-${process.pid}`): Promise<void> {
    const jobs = await this.asyncJobService.claimBatch(workerId, 10);

    for (const job of jobs) {
      try {
        switch (job.jobType) {
          case "ARCHIVE_TAX_DOCUMENT_TO_DRIVE":
            await this.archiveTaxDocumentHandler.run(job);
            break;

          case "EXPORT_RECONCILIATION_TO_SHEETS":
            await this.handleReconciliationExport(job);
            break;

          case "EXPORT_BOOKING_TO_SHEETS":
            await this.handleBookingExport(job);
            break;

          case "CREATE_CALENDAR_EVENT":
            await this.handleCalendarEvent(job);
            break;

          case "SEND_GMAIL_FALLBACK":
            await this.handleGmailFallback(job);
            break;

            throw new Error(`UNKNOWN_JOB_TYPE:${job.jobType}`);
        }

        await this.asyncJobService.markDone(job.id);
      } catch (err: any) {
        await this.asyncJobService.markFailed(job.id, job.attempts + 1, err?.message ?? "UNKNOWN_JOB_FAILURE");

    }

/************************************************************************************************
SECTION 12 - GOOGLE MAPS POLICY + GUARDED WRAPPER
************************************************************************************************/

/*
Architect flagged Maps as missing from the policy.
Add strict wrapper and policy.

Maps is allowed for:
- address autocomplete
- geocoding
- place details
- reverse geocoding

Maps is NOT canonical for:
- booking truth
- pricing truth
- payment truth
*/

export interface MapsPolicyConfig {
  apiKey: string;
  timeoutMs?: number;
}

export class GuardedGoogleMapsService {
  constructor(private config: MapsPolicyConfig) {}

  private assertAllowedFeature(feature: "AUTOCOMPLETE" | "PLACE_DETAILS" | "GEOCODE" | "REVERSE_GEOCODE") {
    const allowed = ["AUTOCOMPLETE", "PLACE_DETAILS", "GEOCODE", "REVERSE_GEOCODE"];
    if (!allowed.includes(feature)) {
      throw new Error("MAPS_FEATURE_NOT_ALLOWED");
    }
  }

  async geocode(address: string): Promise<any> {
    this.assertAllowedFeature("GEOCODE");
    return this.fetchJson("https://maps.googleapis.com/maps/api/geocode/json", {
      address,
      key: this.config.apiKey,
    });
  }

  async placeDetails(placeId: string): Promise<any> {
    this.assertAllowedFeature("PLACE_DETAILS");
    return this.fetchJson("https://maps.googleapis.com/maps/api/place/details/json", {
      place_id: placeId,
      key: this.config.apiKey,
      fields: "place_id,formatted_address,geometry,name,address_component",
    });
  }

  async autocomplete(input: string): Promise<any> {
    this.assertAllowedFeature("AUTOCOMPLETE");
    return this.fetchJson("https://maps.googleapis.com/maps/api/place/autocomplete/json", {
      input,
      key: this.config.apiKey,
      components: "country:il",
      language: "en",
    });
  }

  async reverseGeocode(lat: number, lng: number): Promise<any> {
    this.assertAllowedFeature("REVERSE_GEOCODE");
    return this.fetchJson("https://maps.googleapis.com/maps/api/geocode/json", {
      latlng: `${lat},${lng}`,
      key: this.config.apiKey,
    });
  }

  private async fetchJson(baseUrl: string, query: Record<string, string>): Promise<any> {
    const url = new URL(baseUrl);
    Object.entries(query).forEach(([k, v]) => url.searchParams.set(k, v));

    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), this.config.timeoutMs ?? 8000);

    try {
      const res = await fetch(url.toString(), { signal: controller.signal });
      if (!res.ok) throw new Error(`MAPS_HTTP_${res.status}`);
      return await res.json();
    } finally {
      clearTimeout(timeout);
    }
  }
}

/************************************************************************************************
SECTION 14 - ADMIN SECRET PATH POLICY
************************************************************************************************/

/*
Architect flagged undocumented x-admin-secret path.
Document it and lock it down.

Rules:
- only enabled if explicitly configured
- must be logged on use
- must be rate-limited
- must never bypass immutable audit
- should be removable in production if full auth exists
*/

export function requireAdminAuth(req: any, res: any, next: any) {
  const authUser = req.user;
  const adminSecret = req.headers["x-admin-secret"];

  if (authUser?.role === "super_admin" || authUser?.role === "finance") {
    return next();
  }

  if (
    process.env.ENABLE_X_ADMIN_SECRET === "true" &&
    adminSecret &&
    adminSecret === process.env.X_ADMIN_SECRET
  ) {
    req.adminBypass = true;
    return next();
  }

  return res.status(403).json({ error: "FORBIDDEN" });
}

export async function logAdminPathUsage(db: any, req: any, route: string): Promise<void> {
  if (!req.adminBypass) return;

  await db.insert("audit_events").values({
    eventType: "ADMIN_SECRET_PATH_USED",
    payload: {
      route,
      ip: req.ip,
      userAgent: req.headers["user-agent"] ?? null,
      at: new Date().toISOString(),
    },
    createdAt: new Date(),
  });
}
