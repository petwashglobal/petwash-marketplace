import { initTRPC, TRPCError } from "@trpc/server";
import { z } from "zod";
import { auth } from "@clerk/nextjs/server"; 

// ==========================================
// 1. THE OCTOPUS SECURITY LAYER
// ==========================================
export const createContext = async (opts: { headers: Headers }) => {
  const session = auth(); 
  return {
    userId: session.userId,
    // Grants "Head Office" status if user has admin metadata in Clerk
    role: session.sessionClaims?.metadata?.role || "arm_operator", 
    orgId: session.orgId, 
    ...opts,
  };
};

const t = initTRPC.context<typeof createContext>().create();

// MIDDLEWARE: The "Bouncers"
const isAuthed = t.middleware(({ ctx, next }) => {
  if (!ctx.userId) throw new TRPCError({ code: "UNAUTHORIZED" });
  return next({ ctx: { userId: ctx.userId, role: ctx.role } });
});

const isHeadOffice = t.middleware(({ ctx, next }) => {
  if (ctx.role !== "admin") throw new TRPCError({ code: "FORBIDDEN", message: "Head Office Access Only" });
  console.log(`🚨 HEAD OFFICE OVERRIDE by ${ctx.userId}`);
  return next();
});

// ==========================================
// 2. THE 2025 ZOD SCHEMAS (Data Integrity)
// ==========================================
const GlobalTransactionSchema = z.object({
  locationId: z.string().min(3).describe("Unique Store ID"),
  amount: z.number().positive(),
  currency: z.enum(["USD", "GBP", "AUD", "EUR", "JPY"]),
  serviceType: z.enum(["wash", "groom", "product", "vet"]).default("wash"),
  // AI Context: Tags aid the AI in generating monthly reports
  aiTags: z.array(z.string()).optional(), 
});

const FranchiseOnboardSchema = z.object({
  ownerEmail: z.string().email(),
  region: z.enum(["NA", "EMEA", "APAC"]),
  biometricsEnabled: z.boolean().describe("True if FaceID scanner is installed"),
});

// ==========================================
// 3. THE ROUTER (Brain Functions)
// ==========================================
export const appRouter = t.router({

  // --- ARM 1: FRANCHISE OPERATIONS ---
  franchise: t.router({
    // Live Revenue Stream
    submitTransaction: t.procedure.use(isAuthed)
      .input(GlobalTransactionSchema)
      .mutation(async ({ ctx, input }) => {
        console.log(`💰 [${input.locationId}] Incoming: ${input.amount} ${input.currency}`);
        
        // Logic: If amount > 500, flag for Anti-Fraud AI check
        const riskLevel = input.amount > 500 ? "HIGH" : "LOW";
        
        return { status: "SECURED", txId: `tx_${Date.now()}`, riskLevel };
      }),
      
    // Supply Chain Request
    orderSupplies: t.procedure.use(isAuthed)
      .input(z.object({ item: z.string(), quantity: z.number() }))
      .mutation(({ input }) => {
        return { estimatedDelivery: "2 Days", tracking: "TRK-999" };
      }),
  }),

  // --- ARM 2: CUSTOMER MOBILE APP ---
  customer: t.router({
    getWaitTime: t.procedure
      .input(z.object({ storeId: z.string() }))
      .query(() => {
        return { minutes: 15, queueLength: 2 };
      }),
  }),

  // --- THE HEAD: GLOBAL COMMAND ---
  admin: t.router({
    // Biometric Approved Action
    globalBroadcast: t.procedure.use(isHeadOffice)
      .input(z.object({ message: z.string(), priority: z.enum(["info", "critical"]) }))
      .mutation(({ ctx, input }) => {
        console.log(`📢 GLOBAL BROADCAST: ${input.message}`);
        return { sentTo: "142 Locations", confirmed: true };
      }),

    createFranchise: t.procedure.use(isHeadOffice)
      .input(FranchiseOnboardSchema)
      .mutation(({ input }) => {
        return { welcomePackSent: true, dashboardUrl: `https://portal.petwash.com/${input.region}` };
      }),
  }),
});

export type AppRouter = typeof appRouter;
