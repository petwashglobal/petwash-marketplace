// --------- 4) Confirm top-up (webhook or admin-only callable) ---------
export const wallet_confirmTopUp = functions.https.onCall(async (data, context) => {
  const uid = mustAuth(context);
  const intentId = String(data?.intentId || "");
  if (!intentId) throw new functions.https.HttpsError("invalid-argument", "intentId required");

  // In production: verify payment provider signature + status and ensure it belongs to uid.
  const intentRef = db.collection("topup_intents").doc(intentId);
  const intentSnap = await intentRef.get();
  if (!intentSnap.exists) throw new functions.https.HttpsError("not-found", "Intent not found");

  const intent = intentSnap.data() as any;
  if (intent.uid !== uid) throw new functions.https.HttpsError("permission-denied", "Not yours");
  if (intent.status === "CONFIRMED") return { ok: true };

  const amount = Number(intent.amount || 0);

  // Atomic update using transaction
  await db.runTransaction(async (tx) => {
    tx.update(intentRef, { status: "CONFIRMED", confirmedAt: admin.firestore.FieldValue.serverTimestamp() });

    const wRef = db.collection("wallets").doc(uid);
    const wSnap = await tx.get(wRef);
    const bal = Number((wSnap.data() || {}).balance || 0);
    const nextBal = bal + amount;

    tx.set(wRef, { balance: nextBal, updatedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true });

    tx.create(db.collection("wallet_ledger").doc(), {
      uid,
      type: "TOP_UP",
      amount,
      meta: { intentId },
      createdAt: admin.firestore.FieldValue.serverTimestamp(),
    });
  });

  return { ok: true };
});

// Redeem tokens must be short-lived, signed, single-use.
export const wallet_redeem = functions.https.onCall(async (data, context) => {
  const uid = mustAuth(context);

  const channel = String(data?.channel || "kiosk"); // kiosk | station
  const redeemToken = String(data?.redeemToken || "");
  if (!redeemToken) throw new functions.https.HttpsError("invalid-argument", "redeemToken required");

  // Validate redeemToken (placeholder). Real: verify signature, expiry, stationId, amount, nonce.
  // Example token structure could contain amount and stationId, but always validate server-side.
  const amountToCharge = 55; // placeholder

  await db.runTransaction(async (tx) => {
    const wRef = db.collection("wallets").doc(uid);
    const wSnap = await tx.get(wRef);
    const bal = Number((wSnap.data() || {}).balance || 0);
    if (bal < amountToCharge) throw new functions.https.HttpsError("failed-precondition", "Insufficient balance");

    // Single-use token enforcement
    const usedRef = db.collection("redeem_tokens_used").doc(redeemToken);
    const usedSnap = await tx.get(usedRef);
    if (usedSnap.exists) throw new functions.https.HttpsError("already-exists", "Token already used");
    tx.set(usedRef, { uid, usedAt: admin.firestore.FieldValue.serverTimestamp(), channel });

    tx.set(
      wRef,
      { balance: bal - amountToCharge, updatedAt: admin.firestore.FieldValue.serverTimestamp() },
      { merge: true }
    );

    // Stamps: award stamps based on redeem logic (example: 1 stamp per redeem)
    const uRef = db.collection("users").doc(uid);
    const uSnap = await tx.get(uRef);
    const stamps = Number((uSnap.data() || {}).stamps || 0) + 1;
    const tier = tierForStamps(stamps);

    tx.set(
      uRef,
      { stamps, tier, updatedAt: admin.firestore.FieldValue.serverTimestamp() },
      { merge: true }
    );

    tx.create(db.collection("wallet_ledger").doc(), {
      uid,
      type: "REDEEM",
      amount: -amountToCharge,
      meta: { channel },
      createdAt: admin.firestore.FieldValue.serverTimestamp(),
    });
  });

  return { ok: true };
});

export const twoFactor_verifyEmailOtp = functions.https.onCall(async (data, context) => {
  const uid = mustAuth(context);
  const code = String(data?.code || "");
  if (!/^\d{6}$/.test(code)) throw new functions.https.HttpsError("invalid-argument", "Code invalid");

  const ref = db.collection("twofactor_email").doc(uid);
  const snap = await ref.get();
  if (!snap.exists) throw new functions.https.HttpsError("not-found", "No OTP request");

  const d = snap.data() as any;
  if (Date.now() > Number(d.expiresAt || 0)) throw new functions.https.HttpsError("deadline-exceeded", "Code expired");
  if (Number(d.attempts || 0) >= 5) throw new functions.https.HttpsError("resource-exhausted", "Too many attempts");

  const hash = crypto.createHash("sha256").update(code).digest("hex");
  if (hash !== d.hash) {
    await ref.set({ attempts: admin.firestore.FieldValue.increment(1) }, { merge: true });
    throw new functions.https.HttpsError("permission-denied", "Wrong code");
  }

  // Mark verified
  await db.collection("users").doc(uid).set(
    { email2faVerifiedAt: admin.firestore.FieldValue.serverTimestamp() },
    { merge: true }
  );

  // Invalidate OTP
  await ref.delete();

  return { ok: true };
});
