   - Admin accounts created without approval
   - Confusion and broken UX

   You control routing by:
   - user.role (customer, provider, staff, admin...)
   - provider_status (none, draft, pending_review, approved, rejected)
   - staff_status (none, pending, approved, rejected)
   - required steps (email verify, KYC, profile completeness)
   ========================================================= */

/* =========================================================
   3) POST LOGIN DECIDER (SERVER)
   ========================================================= */

import type { Request, Response } from "express";

type PostLoginDecision =
  | { redirectTo: string; reason: string }
  | { redirectTo: string; reason: string; toast?: string };

const ADMIN_APPROVER_EMAIL = "nir.h@petwash.co.il"; // you

export async function postLogin(req: Request, res: Response) {
  const userId = req.user?.id; // from verified auth token/session
  if (!userId) return res.status(401).json({ error: "AUTH_REQUIRED" });

  const user = await db.users.findById(userId);
  if (!user) return res.status(404).json({ error: "USER_NOT_FOUND" });

  // Hard block
  if (user.blocked) return res.json({ redirectTo: "/blocked", reason: "USER_BLOCKED" });

  // Global required gates
  const emailVerified = !!user.email_verified;
  const hasEmail = !!user.email;
  const hasPhone = !!user.phone_e164;

  // If your policy: email users must verify email before anything important
  if (hasEmail && !emailVerified) {
    return res.json({ redirectTo: "/verify-email", reason: "EMAIL_NOT_VERIFIED" });
  }

  // If user has not chosen an entry intent yet, send to role chooser.
  // This is the KEY: user decides "I am booking" vs "I want to be a provider"
  if (!user.role || user.role === "customer") {
    const hasProviderApp = await db.providerApps.existsForUser(userId);
    const hasStaffReq = await db.staffRequests.existsForUser(userId);

    // If they already started provider onboarding, continue it
    if (hasProviderApp) {
      const app = await db.providerApps.getByUser(userId);
      return res.json(routeProvider(app));
    }

    // If they requested staff/admin access, show pending page
    if (hasStaffReq) {
      const reqRow = await db.staffRequests.getLatestByUser(userId);
      return res.json(routeStaffRequest(reqRow));
    }

    // Default external user landing
    // If profile incomplete, go onboarding, otherwise go marketplace/home
    const profileComplete = await isCustomerProfileComplete(userId);
    if (!profileComplete) return res.json({ redirectTo: "/onboarding", reason: "CUSTOMER_PROFILE_INCOMPLETE" });

    return res.json({ redirectTo: "/home", reason: "CUSTOMER_READY" });
  }

  // Provider role
  if (user.role === "provider") {
    const app = await db.providerApps.getByUser(userId);
    return res.json(routeProvider(app));
  }

  // Staff role: must be approved
  if (user.role === "staff") {
    const ok = await isStaffApproved(userId);
    if (!ok) return res.json({ redirectTo: "/staff/pending", reason: "STAFF_NOT_APPROVED" });
    return res.json({ redirectTo: "/staff", reason: "STAFF_READY" });
  }

  // Admin/management: must be approved by you
  if (user.role === "admin" || user.role === "management" || user.role === "super_admin") {
    const ok = await isAdminApproved(userId);
    if (!ok) return res.json({ redirectTo: "/admin/pending", reason: "ADMIN_NOT_APPROVED" });
    return res.json({ redirectTo: "/admin", reason: "ADMIN_READY" });
  }

  // Fallback
  return res.json({ redirectTo: "/home", reason: "FALLBACK" });
}

function routeProvider(app: any): PostLoginDecision {
  if (!app) return { redirectTo: "/choose-role", reason: "PROVIDER_NO_APP" };

  if (app.status === "draft") return { redirectTo: "/provider/onboarding", reason: "PROVIDER_DRAFT" };
  if (app.status === "pending_review") return { redirectTo: "/provider/pending", reason: "PROVIDER_PENDING_REVIEW" };
  if (app.status === "rejected") return { redirectTo: "/provider/rejected", reason: "PROVIDER_REJECTED" };

  // approved
  // If provider still missing mandatory profile or payout setup, send to setup wizard
  if (!app.onboarding_complete) return { redirectTo: "/provider/setup", reason: "PROVIDER_SETUP_REQUIRED" };
  return { redirectTo: "/provider/dashboard", reason: "PROVIDER_APPROVED" };
}

function routeStaffRequest(reqRow: any): PostLoginDecision {
  if (!reqRow) return { redirectTo: "/staff/apply", reason: "NO_STAFF_REQUEST" };
  if (reqRow.status === "pending") return { redirectTo: "/staff/pending", reason: "STAFF_REQUEST_PENDING" };
  if (reqRow.status === "rejected") return { redirectTo: "/staff/rejected", reason: "STAFF_REQUEST_REJECTED" };
  return { redirectTo: "/staff", reason: "STAFF_REQUEST_APPROVED" };
}

async function isCustomerProfileComplete(userId: string) {
  const u = await db.users.findById(userId);
  // define your minimum: name + city + phone etc
  return !!(u.first_name && u.last_name && (u.phone_e164 || u.email));
}

async function isStaffApproved(userId: string) {
  const u = await db.users.findById(userId);
  return !!u.approved_at && !!u.approved_by;
}

async function isAdminApproved(userId: string) {
  const u = await db.users.findById(userId);
  if (!u.approved_at || !u.approved_by) return false;
  // enforce that only Nir can approve admin/management
  const approver = (u.approved_by || "").toLowerCase();
  return approver === ADMIN_APPROVER_EMAIL.toLowerCase();
}


/* =========================================================
   4) HOW USERS CHOOSE THEIR INTENT
   =========================================================
   Add a "Choose your path" screen:
   - "Book services" -> sets role customer, go onboarding/home
   - "Become a provider" -> creates provider_applications row (draft) and routes provider onboarding
   - "Staff/Admin access" -> creates staff_access_requests row (pending) and shows pending screen
   ========================================================= */

export async function chooseRole(req: Request, res: Response) {
  const userId = req.user?.id;
  const { intent } = req.body || {}; // "customer" | "provider" | "staff_request"
  if (!userId) return res.status(401).json({ error: "AUTH_REQUIRED" });

  if (intent === "customer") {
    await db.users.update(userId, { role: "customer", access_level: 1 });
    return res.json({ ok: true });
  }

  if (intent === "provider") {
    await db.users.update(userId, { role: "provider", access_level: 2 });
    await db.providerApps.ensureDraft(userId);
    return res.json({ ok: true });
  }

  if (intent === "staff_request") {
    // Do NOT set admin/staff role here.
    // Only create a request. User stays customer until approved.
    await db.staffRequests.create({
      user_id: userId,
      requested_role: "admin",
      status: "pending"
    });
    return res.json({ ok: true });
  }

  return res.status(400).json({ error: "INVALID_INTENT" });
}


/* =========================================================
   5) ADMIN APPROVAL ENDPOINT (ONLY YOU)
   ========================================================= */

export async function approveAdmin(req: Request, res: Response) {
  const approverEmail = req.user?.email?.toLowerCase();
  if (approverEmail !== ADMIN_APPROVER_EMAIL.toLowerCase()) {
    return res.status(403).json({ error: "FORBIDDEN" });
  }

  const { targetUserId, role } = req.body || {}; // admin | management | staff
  if (!targetUserId) return res.status(400).json({ error: "TARGET_REQUIRED" });

  // Approve and stamp
  await db.users.update(targetUserId, {
    role,
    access_level: role === "management" ? 8 : role === "admin" ? 6 : 4,
    approved_by: ADMIN_APPROVER_EMAIL,
    approved_at: new Date()
  });

  await db.audit.insert({
    actor_id: req.user.id,
    action: "ADMIN_APPROVAL",
    target_id: targetUserId,
    meta: { role }
  });

  return res.json({ ok: true });
}
