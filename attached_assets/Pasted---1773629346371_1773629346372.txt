/* ================================================================================================
SECTION 1 - SECURITY MODEL SUMMARY
===================================================================================================

OLD INSECURE MODEL
------------------
Barcode contained something static like:
  PETWASH:userId:serial

Problems:
- replayable forever
- screenshot reusable
- easy cloning risk
- no expiry
- no one-time burn
- no machine binding

NEW SECURE MODEL
----------------
Use two token types:

1. wallet-link token
   - purpose: distributing the pass by email / SMS / dashboard / link
   - signed
   - expires in 72 hours
   - if expired -> backend returns 410 Gone or similar

2. qr-redeem token
   - purpose: actual kiosk redemption
   - signed
   - expires in 45 seconds
   - contains nonce
   - nonce is burned on first successful use
   - optional machineId / bayId binding
   - optional tokenVersion to invalidate older passes

This is the correct modern model for wallet-backed redemption.
================================================================================================ */

import crypto from "crypto";

const PASS_LINK_SECRET = process.env.PASS_LINK_SECRET || "";
const PRESTIGE_QR_SECRET = process.env.PRESTIGE_QR_SECRET || "";

function b64url(input: Buffer | string): string {
  return Buffer.from(input)
    .toString("base64")
    .replace(/\+/g, "-")
    .replace(/\//g, "_")
    .replace(/=+$/g, "");
}

function signCompactPayload(payload: object, secret: string): string {
  const body = b64url(JSON.stringify(payload));
  const sig = crypto.createHmac("sha256", secret).update(body).digest("base64url");
  return `${body}.${sig}`;
}

function verifyCompactPayload<T>(token: string, secret: string): T {
  const [body, sig] = token.split(".");
  if (!body || !sig) throw new Error("INVALID_TOKEN_FORMAT");

  const expected = crypto.createHmac("sha256", secret).update(body).digest("base64url");

  // timing-safe compare is important
  if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) {
    throw new Error("INVALID_SIGNATURE");
  }

  const payload = JSON.parse(Buffer.from(body, "base64url").toString("utf8")) as T;

  const now = Math.floor(Date.now() / 1000);
  const expiresAt = (payload as any).expiresAt;
  if (typeof expiresAt !== "number" || expiresAt < now) {
    throw new Error("TOKEN_EXPIRED");
  }

  return payload;
}

function buildWalletLinkToken(record: PetWashPassRecord): string {
  const now = Math.floor(Date.now() / 1000);

  const payload: WalletLinkTokenPayload = {
    passId: record.passId,
    userId: record.userId,
    tokenVersion: record.qrTokenVersion,
    purpose: "wallet-link",
    nonce: crypto.randomUUID(),
    issuedAt: now,
    expiresAt: now + 60 * 60 * 24 * 3, // 72 hours
  };

  return signCompactPayload(payload, PASS_LINK_SECRET);
}

function buildQrRedeemToken(
  record: PetWashPassRecord,
  machineId?: string,
  bayId?: "LEFT" | "RIGHT"
): string {
  const now = Math.floor(Date.now() / 1000);

  const payload: QrRedeemTokenPayload = {
    passId: record.passId,
    userId: record.userId,
    tokenVersion: record.qrTokenVersion,
    purpose: "qr-redeem",
    nonce: crypto.randomUUID(),
    issuedAt: now,
    expiresAt: now + 45,
    machineId: machineId || null,
    bayId: bayId || null,
  };

  return signCompactPayload(payload, PRESTIGE_QR_SECRET);
}

function verifyWalletLinkToken(token: string): WalletLinkTokenPayload {
  const payload = verifyCompactPayload<WalletLinkTokenPayload>(token, PASS_LINK_SECRET);
  if (payload.purpose !== "wallet-link") throw new Error("INVALID_PURPOSE");
  return payload;
}

function verifyQrRedeemToken(token: string): QrRedeemTokenPayload {
  const payload = verifyCompactPayload<QrRedeemTokenPayload>(token, PRESTIGE_QR_SECRET);
  if (payload.purpose !== "qr-redeem") throw new Error("INVALID_PURPOSE");
  return payload;
}

app.post("/wallet/redeem", async (req, res) => {
  try {
    const body = req.body as WalletRedeemRequest;
    const payload = verifyQrRedeemToken(body.scannedToken);

    if (payload.machineId && payload.machineId !== body.kioskId) {
      return res.status(403).json({ error: "TOKEN_MACHINE_MISMATCH" });
    }

    if (payload.bayId && body.bayId && payload.bayId !== body.bayId) {
      return res.status(403).json({ error: "TOKEN_BAY_MISMATCH" });
    }

    await assertNonceUnusedAndConsume(payload.nonce, payload.passId, payload.expiresAt);

    const record = await loadPassById(payload.passId);
    if (!record) return res.status(404).json({ error: "PASS_NOT_FOUND" });

    if (record.status !== "ACTIVE") {
      return res.status(403).json({ error: "PASS_NOT_ACTIVE" });
    }

    if (record.validUntil && new Date(record.validUntil).getTime() < Date.now()) {
      return res.status(403).json({ error: "PASS_EXPIRED" });
    }

    if (record.availableCreditIls < body.amountIls) {
      return res.status(402).json({ error: "INSUFFICIENT_BALANCE" });
    }

    await debitPassAtomically(record.passId, body.amountIls, {
      source: "K9000",
      kioskId: body.kioskId,
      bayId: body.bayId || null,
      nonce: payload.nonce,
    });

    await startK9000Machine(body.kioskId, body.bayId);

    return res.json({
      ok: true,
      passId: record.passId,
      debitedIls: body.amountIls,
    });
  } catch (err: any) {
    return res.status(400).json({ error: err.message || "REDEEM_FAILED" });
  }
});
