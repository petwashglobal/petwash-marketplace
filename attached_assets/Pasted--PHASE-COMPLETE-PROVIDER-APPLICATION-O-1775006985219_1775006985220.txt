/* =========================================================
SECTION 1 - STATUS MACHINE
=========================================================

APPLICATION STATUS LIFECYCLE

submitted
-> saved instantly after user submits form

processing
-> async KYC and document analysis running

pending_review
-> human review needed

pending_resubmission
-> user must upload clearer / corrected files

approved
-> provider account activated

rejected
-> permanently declined

expired
-> applicant did not respond in time

withdrawn
-> applicant cancelled application

IMPORTANT RULES
- submit must NEVER block on external API calls
- processing must always end in:
  approved / pending_review / pending_resubmission / rejected
- pending_resubmission keeps same application_id
- every transition writes audit

-- MAIN APPLICATION TABLE
CREATE TABLE IF NOT EXISTS provider_applications (
  id SERIAL PRIMARY KEY,
  user_id TEXT,
  provider_type TEXT NOT NULL,
  applicant_name TEXT NOT NULL,
  applicant_email TEXT NOT NULL,
  applicant_mobile TEXT NOT NULL,

  status TEXT NOT NULL DEFAULT 'submitted',
  sub_status TEXT,

  document_file_path TEXT,
  selfie_file_path TEXT,

  reviewed_by TEXT,
  reviewed_at TIMESTAMP,
  manual_decision_reason TEXT,
  approved_as_provider_id TEXT,

  biometric_match_score DECIMAL(5,2),
  kyc_document_type VARCHAR(50),
  kyc_id_last_four VARCHAR(4),
  kyc_ocr_confidence DECIMAL(5,2),
  kyc_liveness_score DECIMAL(5,2),
  kyc_decision_flags JSONB,
  background_check_notes JSONB,
  biometric_failure_reason TEXT,
  biometric_verified_at TIMESTAMP,
  biometric_status VARCHAR(50),

  extracted_name TEXT,
  extracted_dob TEXT,
  extracted_document_number_last4 TEXT,
  extracted_expiry TEXT,
  extracted_country TEXT,

  fraud_flags JSONB,
  fraud_risk_level TEXT,

  communication_thread_id BIGINT,
  resubmission_count INT NOT NULL DEFAULT 0,
  last_requested_resubmission_at TIMESTAMP,
  last_resubmitted_at TIMESTAMP,

  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_provider_applications_status
  ON provider_applications(status);

CREATE INDEX IF NOT EXISTS idx_provider_applications_email
  ON provider_applications(applicant_email);

CREATE INDEX IF NOT EXISTS idx_provider_applications_mobile
  ON provider_applications(applicant_mobile);

CREATE INDEX IF NOT EXISTS idx_provider_applications_provider_type
  ON provider_applications(provider_type);


-- FILE VERSIONS
CREATE TABLE IF NOT EXISTS provider_application_files (
  id SERIAL PRIMARY KEY,
  application_id INT NOT NULL REFERENCES provider_applications(id) ON DELETE CASCADE,
  version_no INT NOT NULL,
  file_role TEXT NOT NULL, -- selfie | document
  file_path TEXT NOT NULL,
  mime_type TEXT,
  size_bytes BIGINT,
  uploaded_by TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE UNIQUE INDEX IF NOT EXISTS idx_provider_application_files_unique
  ON provider_application_files(application_id, version_no, file_role);


-- REVIEW QUEUE
CREATE TABLE IF NOT EXISTS provider_review_queue (
  id SERIAL PRIMARY KEY,
  application_id INT NOT NULL UNIQUE REFERENCES provider_applications(id) ON DELETE CASCADE,
  status TEXT NOT NULL DEFAULT 'open', -- open | assigned | completed | cancelled
  priority TEXT NOT NULL DEFAULT 'normal', -- low | normal | high | urgent
  assigned_to TEXT,
  assigned_at TIMESTAMP,
  due_at TIMESTAMP,
  review_reasons JSONB,
  unread_count INT NOT NULL DEFAULT 0,
  created_at TIMESTAMP DEFAULT NOW(),
  completed_at TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_provider_review_queue_status
  ON provider_review_queue(status);

CREATE INDEX IF NOT EXISTS idx_provider_review_queue_priority
  ON provider_review_queue(priority);

CREATE INDEX IF NOT EXISTS idx_provider_review_queue_due
  ON provider_review_queue(due_at);


-- INBOX / COMMUNICATION THREADS
CREATE TABLE IF NOT EXISTS provider_application_threads (
  id BIGSERIAL PRIMARY KEY,
  application_id INT NOT NULL UNIQUE REFERENCES provider_applications(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS provider_application_messages (
  id BIGSERIAL PRIMARY KEY,
  thread_id BIGINT NOT NULL REFERENCES provider_application_threads(id) ON DELETE CASCADE,
  application_id INT NOT NULL REFERENCES provider_applications(id) ON DELETE CASCADE,
  direction TEXT NOT NULL, -- outbound | inbound | internal_note
  channel TEXT NOT NULL, -- email | sms | system | internal_note
  subject TEXT,
  body TEXT NOT NULL,
  from_address TEXT,
  to_address TEXT,
  sent_by TEXT,
  delivery_status TEXT, -- queued | sent | failed | delivered | opened
  provider_visible BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_provider_application_messages_thread
  ON provider_application_messages(thread_id, created_at DESC);


-- RESUBMISSION REQUESTS
CREATE TABLE IF NOT EXISTS provider_application_resubmissions (
  id SERIAL PRIMARY KEY,
  application_id INT NOT NULL REFERENCES provider_applications(id) ON DELETE CASCADE,
  requested_by TEXT,
  request_reasons JSONB NOT NULL,
  secure_token TEXT NOT NULL UNIQUE,
  expires_at TIMESTAMP NOT NULL,
  fulfilled_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_provider_application_resubmissions_app
  ON provider_application_resubmissions(application_id);


-- AUDIT
CREATE TABLE IF NOT EXISTS provider_review_audit (
  id BIGSERIAL PRIMARY KEY,
  application_id INT NOT NULL REFERENCES provider_applications(id) ON DELETE CASCADE,
  event_type TEXT NOT NULL,
  actor_user_id TEXT,
  actor_role TEXT,
  payload JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_provider_review_audit_app
  ON provider_review_audit(application_id, created_at DESC);


-- MONITORING EVENTS
CREATE TABLE IF NOT EXISTS provider_workflow_events (
  id BIGSERIAL PRIMARY KEY,
  application_id INT,
  event_name TEXT NOT NULL,
  severity TEXT NOT NULL DEFAULT 'info', -- info | warning | critical
  payload JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_provider_workflow_events_name
  ON provider_workflow_events(event_name, created_at DESC);

      input.fromAddress ?? null,
      input.toAddress ?? null,
      input.sentBy ?? null,
      input.deliveryStatus ?? null,
      input.providerVisible ?? true,
    ]
  );

// server/services/providerDecisionEngine.ts

function normalizeDocumentType(raw?: string | null): string | null {
  if (!raw) return null;
  const v = raw.toLowerCase().trim();
  if (v.includes("passport")) return "passport";
  if (v.includes("id")) return "national_id";
  if (v.includes("license") || v.includes("licence")) return "drivers_license";
  return null;
}

function highRiskFraud(fraudFlags: string[]) {
  return fraudFlags.some((f) =>
    [
      "duplicate_document_number",
      "reused_selfie_hash",
      "multi_ip_velocity_high",
      "device_fingerprint_cluster",
    ].includes(f)
  );
}

function mediumRiskFraud(fraudFlags: string[]) {
  return fraudFlags.length > 0;
}

export function decideProviderKyc(input: {
  faceMatchScore: number;
  livenessPass: boolean;
  livenessScore: number;
  ocrConfidence: number;
  documentType?: string | null;
  extractedName?: string | null;
  extractedDob?: string | null;
  extractedExpiry?: string | null;
  extractedDocumentNumber?: string | null;
  selfieQuality?: "good" | "poor" | string | null;
  idQuality?: "good" | "poor" | string | null;
  submittedName?: string | null;
  submittedDob?: string | null;
  fraudFlags?: string[];
  rawBackground?: Record<string, any>;
}): KycDecisionResult {
  const flags: string[] = [];
  const fraudFlags = input.fraudFlags || [];
  const documentType = normalizeDocumentType(input.documentType);

  if (!input.extractedName?.trim()) flags.push("ocr_name_missing");
  if (!input.extractedDob?.trim()) flags.push("ocr_dob_missing");
  if (!input.extractedExpiry?.trim()) flags.push("ocr_expiry_missing");
  if (!input.extractedDocumentNumber?.trim()) flags.push("ocr_document_number_missing");
  if (!documentType) flags.push("document_type_unknown");
  if ((input.ocrConfidence ?? 0) < 50) flags.push("ocr_confidence_low");
  if ((input.idQuality ?? "good") !== "good") flags.push("id_document_poor_quality");
  if ((input.selfieQuality ?? "good") !== "good") flags.push("selfie_poor_quality");

  if (
    input.submittedName?.trim() &&
    input.extractedName?.trim() &&
    input.submittedName.trim().toLowerCase() !== input.extractedName.trim().toLowerCase()
  ) {
    flags.push("name_mismatch");
  }

  if (
    input.submittedDob?.trim() &&
    input.extractedDob?.trim() &&
    input.submittedDob.trim() !== input.extractedDob.trim()
  ) {
    flags.push("dob_mismatch");
  }

  let fraudRiskLevel: "low" | "medium" | "high" = "low";
  if (highRiskFraud(fraudFlags)) fraudRiskLevel = "high";
  else if (mediumRiskFraud(fraudFlags)) fraudRiskLevel = "medium";

  let status: KycDecisionResult["status"] = "pending_review";
  let reason = "KYC review required";

  // hard reject
  if (input.faceMatchScore < 55 || !input.livenessPass) {
    status = "rejected";
    reason = `Face score ${input.faceMatchScore.toFixed(1)} or liveness failed`;
  }
  // resubmission instead of rejection for quality problems
  else if (
    flags.includes("id_document_poor_quality") ||
    flags.includes("selfie_poor_quality") ||
    flags.includes("ocr_expiry_missing") ||
    flags.includes("ocr_document_number_missing")
  ) {
    status = "pending_resubmission";
    reason = `Better files required: ${flags.join(", ")}`;
  }
  // fraud always review or reject
  else if (fraudRiskLevel === "high") {
    status = "rejected";
    reason = `High fraud risk: ${fraudFlags.join(", ")}`;
  } else if (fraudRiskLevel === "medium") {
    status = "pending_review";
    reason = `Medium fraud risk: ${fraudFlags.join(", ")}`;
  }
  // strong pass
  else if (input.faceMatchScore >= 85 && input.livenessPass && flags.length === 0) {
    status = "approved";
    reason = `Strong pass: face ${input.faceMatchScore.toFixed(1)}, liveness passed, OCR complete`;
  }
  // review band
  else if (input.faceMatchScore >= 65) {
    status = "pending_review";
    reason = flags.length
      ? `Borderline or flagged: ${flags.join(", ")}`
      : `Borderline score: ${input.faceMatchScore.toFixed(1)}`;
  }
  // default reject
  else {
    status = "rejected";
    reason = `Insufficient confidence`;
  }

  return {
    status,
    reason,
    faceScore: input.faceMatchScore,
    livenessScore: input.livenessScore,
    ocrConfidence: input.ocrConfidence,
    documentType,
    idLastFour: input.extractedDocumentNumber
      ? input.extractedDocumentNumber.slice(-4)
      : null,
    flags,
    fraudFlags,
    fraudRiskLevel,
    backgroundCheckNotes: {
      ...(input.rawBackground || {}),
      decision: {
        status,
        reason,
        fraudRiskLevel,
      },
    },
  };
}
