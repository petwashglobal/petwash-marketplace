/* =========================================================
   2) PASSKEY (WEBAUTHN) SERVER ROUTES (EXPRESS)
   Using @simplewebauthn/server
   ========================================================= */

import crypto from "crypto";
import type { Request, Response } from "express";
import {
  generateRegistrationOptions,
  generateAuthenticationOptions,
  verifyRegistrationResponse,
  verifyAuthenticationResponse
} from "@simplewebauthn/server";

type RelyingParty = {
  rpID: string;        // eg "yourdomain.co.il"
  rpName: string;      // your app name
  origin: string;      // eg "https://yourdomain.co.il"
};

const RP: RelyingParty = {
  rpID: "yourdomain.co.il",
  rpName: "Your Platform",
  origin: "https://yourdomain.co.il"
};

function sha256Base64url(input: string) {
  const h = crypto.createHash("sha256").update(input, "utf8").digest("base64");
  return h.replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/g, "");
}

/**
 * 2.1 Start passkey registration
 * Requires user already identified (email verified, or phone verified)
 */
export async function startPasskeyRegistration(req: Request, res: Response) {
  const userId = req.user?.id; // from your auth middleware
  if (!userId) return res.status(401).json({ error: "AUTH_REQUIRED" });

  const user = await db.users.findById(userId);
  if (!user) return res.status(404).json({ error: "USER_NOT_FOUND" });

  const existing = await db.passkeys.listByUser(userId);

  const options = await generateRegistrationOptions({
    rpName: RP.rpName,
    rpID: RP.rpID,
    userID: userId,
    userName: user.email || user.phone_e164 || userId,
    attestationType: "none",
    // Prevent registering the same authenticator twice:
    excludeCredentials: existing.map((k: any) => ({
      id: k.credential_id,
      type: "public-key"
    })),
    authenticatorSelection: {
      userVerification: "required",     // forces Face ID/Touch ID when available
      residentKey: "preferred"
    }
  });

  await db.webauthnChallenges.insert({
    user_id: userId,
    purpose: "register",
    challenge: options.challenge,
    expires_at: new Date(Date.now() + 5 * 60 * 1000)
  });

  return res.json(options);
}

/**
 * 2.2 Finish passkey registration
 */
export async function finishPasskeyRegistration(req: Request, res: Response) {
  const userId = req.user?.id;
  if (!userId) return res.status(401).json({ error: "AUTH_REQUIRED" });

  const expected = await db.webauthnChallenges.getLatest(userId, "register");
  if (!expected) return res.status(400).json({ error: "NO_CHALLENGE" });

  const verification = await verifyRegistrationResponse({
    response: req.body,
    expectedChallenge: expected.challenge,
    expectedOrigin: RP.origin,
    expectedRPID: RP.rpID
  });

  if (!verification.verified || !verification.registrationInfo) {
    await db.authEvents.insert({ user_id: userId, event_type: "PASSKEY_REGISTER", success: false, reason: "VERIFY_FAILED" });
    return res.status(400).json({ error: "PASSKEY_VERIFY_FAILED" });
  }

  const { credential, credentialDeviceType, credentialBackedUp } = verification.registrationInfo;

  await db.passkeys.insert({
    user_id: userId,
    credential_id: credential.id,
    public_key: credential.publicKey,
    counter: credential.counter,
    transports: credential.transports || null,
    device_type: credentialDeviceType,
    backed_up: credentialBackedUp
  });

  await db.authEvents.insert({ user_id: userId, event_type: "PASSKEY_REGISTER", success: true });

  return res.json({ ok: true });
}

/**
 * 2.3 Start passkey login (by email)
 * This supports "email then Face ID" flow:
 * - user types email
 * - you find userId
 * - you issue auth options
 */
export async function startPasskeyLogin(req: Request, res: Response) {
  const { email, traceId } = req.body || {};
  if (!email) return res.status(400).json({ error: "EMAIL_REQUIRED" });

  const user = await db.users.findByEmail(email.toLowerCase().trim());
  if (!user) return res.status(404).json({ error: "USER_NOT_FOUND" });

  const keys = await db.passkeys.listByUser(user.id);
  if (!keys.length) return res.status(409).json({ error: "NO_PASSKEYS" });

  const options = await generateAuthenticationOptions({
    rpID: RP.rpID,
    userVerification: "required",
    allowCredentials: keys.map((k: any) => ({ id: k.credential_id, type: "public-key" }))
  });

  await db.webauthnChallenges.insert({
    user_id: user.id,
    purpose: "auth",
    challenge: options.challenge,
    expires_at: new Date(Date.now() + 5 * 60 * 1000)
  });

  return res.json({ userId: user.id, options, traceId: traceId || null });
}

/**
 * 2.4 Finish passkey login
 * - verifies challenge
 * - updates counter
 * - creates session cookie or JWT
 */
export async function finishPasskeyLogin(req: Request, res: Response) {
  const { userId, traceId } = req.body || {};
  if (!userId) return res.status(400).json({ error: "USER_ID_REQUIRED" });

  const expected = await db.webauthnChallenges.getLatest(userId, "auth");
  if (!expected) return res.status(400).json({ error: "NO_CHALLENGE" });

  const keys = await db.passkeys.listByUser(userId);
  const keyById = new Map(keys.map((k: any) => [k.credential_id, k]));

  const credentialId = req.body?.id; // from browser response
  const key = keyById.get(credentialId);
  if (!key) return res.status(400).json({ error: "UNKNOWN_CREDENTIAL" });

  const verification = await verifyAuthenticationResponse({
    response: req.body,
    expectedChallenge: expected.challenge,
    expectedOrigin: RP.origin,
    expectedRPID: RP.rpID,
    authenticator: {
      credentialID: key.credential_id,
      credentialPublicKey: key.public_key,
      counter: key.counter
    }
  });

  if (!verification.verified) {
    await db.authEvents.insert({ user_id: userId, event_type: "PASSKEY_LOGIN", success: false, reason: "VERIFY_FAILED", trace_id: traceId || null });
    return res.status(401).json({ error: "PASSKEY_AUTH_FAILED" });
  }

  // Update counter and last_used_at
  await db.passkeys.updateCounter(key.credential_id, verification.authenticationInfo.newCounter);

  await db.authEvents.insert({ user_id: userId, event_type: "PASSKEY_LOGIN", success: true, trace_id: traceId || null });

  // Create session (cookie) here:
  // res.cookie("session", createSessionToken(userId), { httpOnly: true, secure: true, sameSite: "lax" });

  return res.json({ ok: true, userId });
}


/* =========================================================
   3) CLIENT (BROWSER) PASSKEY CALLS
   Using @simplewebauthn/browser
   ========================================================= */

import { startRegistration, startAuthentication } from "@simplewebauthn/browser";

export async function clientEnrollPasskey() {
  const opts = await fetch("/api/auth/passkey/register/start", { method: "POST" }).then(r => r.json());
  const attResp = await startRegistration(opts);
  const res = await fetch("/api/auth/passkey/register/finish", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(attResp)
  });
  if (!res.ok) throw new Error("PASSKEY_ENROLL_FAILED");
  return true;
}

export async function clientPasskeyLoginByEmail(email: string) {
  const { userId, options } = await fetch("/api/auth/passkey/login/start", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ email })
  }).then(r => r.json());

  const asseResp = await startAuthentication(options);

  const res = await fetch("/api/auth/passkey/login/finish", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ userId, ...asseResp })
  });
  if (!res.ok) throw new Error("PASSKEY_LOGIN_FAILED");
  return true;
}

export async function startEmailMagicLink(req: Request, res: Response) {
  const { email, consentVersion, consentText } = req.body || {};
  if (!email) return res.status(400).json({ error: "EMAIL_REQUIRED" });

  // 4.1 Record "approved click" consent BEFORE sending link
  // (User clicked "I agree" to join)
  const consentHash = sha256Base64url(consentText || "");
  const user = await db.users.upsertByEmail(email.toLowerCase().trim());

  await db.userConsents.insert({
    user_id: user.id,
    consent_type: "terms",
    consent_version: consentVersion || "unknown",
    consent_text_hash: consentHash,
    accepted: true,
    ip: req.ip,
    user_agent: req.get("user-agent") || null,
    locale: req.get("accept-language") || null,
    source: "web",
    trace_id: req.get("x-trace-id") || null
  });

  // 4.2 Send magic link with one-time token
  const token = crypto.randomBytes(32).toString("hex");
  await db.magicLinks.insert({
    user_id: user.id,
    token_hash: sha256Base64url(token),
    expires_at: new Date(Date.now() + 15 * 60 * 1000)
  });

  const link = `https://yourdomain.co.il/auth/magic?token=${token}`;
  await emailProvider.send({

  });

  return res.json({ ok: true });
}

export async function finishEmailMagicLink(req: Request, res: Response) {

  if (!token) return res.status(400).json({ error: "TOKEN_REQUIRED" });

  const tokenHash = sha256Base64url(token);
  const record = await db.magicLinks.consume(tokenHash);
  if (!record) return res.status(401).json({ error: "INVALID_OR_EXPIRED" });
