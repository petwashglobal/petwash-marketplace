export function decide(input: {
faceScore: number;
livenessPass: boolean;
ocrConfidence: number;
documentType: string | null;
nameDetected: boolean;
dobDetected: boolean;
expiryDetected: boolean;
idDetected: boolean;
selfieQuality: “good” | “poor”;
idQuality: “good” | “poor”;
}): {
status: “approved” | “pending_review” | “rejected”;
flags: string[];
} {
const flags: string[] = [];

if (!input.nameDetected) flags.push(“ocr_name_missing”);
if (!input.dobDetected) flags.push(“ocr_dob_missing”);
if (!input.expiryDetected) flags.push(“ocr_expiry_missing”);
if (!input.idDetected) flags.push(“ocr_id_missing”);
if (!input.documentType) flags.push(“document_type_unknown”);
if (input.ocrConfidence < 50) flags.push(“ocr_low_confidence”);
if (input.selfieQuality !== “good”) flags.push(“selfie_poor”);
if (input.idQuality !== “good”) flags.push(“id_poor”);

if (!input.livenessPass || input.faceScore < 55) {
return { status: “rejected”, flags };
}

if (flags.length > 0) {
return { status: “pending_review”, flags };
}

if (input.faceScore >= 78 && input.livenessPass) {
return { status: “approved”, flags: [] };
}

return { status: “pending_review”, flags };
}
