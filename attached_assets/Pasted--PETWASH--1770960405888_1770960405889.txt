import crypto from "crypto";

export function nowIso() {
  return new Date().toISOString();
}

export function assert(condition: any, message: string) {
  if (!condition) throw new Error(message);
}

export function hmacSha256Hex(secret: string, payload: string) {
  return crypto.createHmac("sha256", secret).update(payload).digest("hex");
}

export function stablePayload(obj: Record<string, any>) {
  // stable stringify for signatures
  const keys = Object.keys(obj).sort();
  const stable: Record<string, any> = {};
  for (const k of keys) stable[k] = obj[k];
  return JSON.stringify(stable);
}

export function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

export function toInt(n: any, fallback = 0) {
  const v = Number(n);
  return Number.isFinite(v) ? Math.trunc(v) : fallback;
}

async function requireAuth(req: any) {
  const header = String(req.headers.authorization || "");
  const token = header.startsWith("Bearer ") ? header.slice(7) : "";
  assert(token, "Missing Authorization Bearer token");
  const decoded = await admin.auth().verifyIdToken(token);
  assert(decoded?.uid, "Invalid token");
  return decoded;
}

function tierForPoints(points: number): TierId {
  const tiers = CONFIG.TIERS.slice().sort((a, b) => a.minPoints - b.minPoints);
  let current: TierId = "BRONZE";
  for (const t of tiers) {
    if (points >= t.minPoints) current = t.id as TierId;
  }
  return current;
}

function computeSummaryFromEvents(input: {
  uid: string;
  currency: "ILS";
  events: WalletEvent[];
  prev?: WalletSummary | null;
}): WalletSummary {
  const { uid, currency, events, prev } = input;

  // Start from previous derived summary if provided
  let balance = prev?.balanceCents ?? 0;
  let available = prev?.availableCents ?? 0;
  let pending = prev?.pendingCents ?? 0;
  let lifetimeSpent = prev?.lifetimeSpentCents ?? 0;
  let stamps = prev?.stamps ?? 0;
  let points = prev?.tierPoints ?? 0;
  let tier: TierId = (prev?.tier as TierId) ?? "BRONZE";

  for (const e of events) {
    // Amount conventions:
    // TOPUP => +amount
    // PURCHASE => -amount (spend wallet), also add points/stamps based on spend
    // REDEEM => -amount (wallet spend) OR 0 if redeem uses package, decide by meta
    // ADJUSTMENT => +/-amount
    balance += e.amountCents;
    available += e.amountCents;

    if (e.type === "PURCHASE" || e.type === "REDEEM") {
      const spent = Math.max(0, -e.amountCents);
      lifetimeSpent += spent;

      // points and stamps
      const ilsSpent = spent / 100;
      points += Math.floor(ilsSpent * CONFIG.POINTS_PER_ILS_1);
      const earnedStamps = Math.floor(spent / CONFIG.STAMP_EVERY_CENTS_SPENT);
      if (earnedStamps > 0) stamps += earnedStamps;
    }

    if (e.type === "STAMP_EARNED") {
      const add = Math.max(0, e.meta?.count || 0);
      stamps += add;
    }
  }

  tier = tierForPoints(points);

  const now = nowIso();
  return {
    uid,
    currency,
    balanceCents: clamp(balance, -999999999, 999999999),
    availableCents: clamp(available, -999999999, 999999999),
    pendingCents: clamp(pending, 0, 999999999),
    lifetimeSpentCents: clamp(lifetimeSpent, 0, 999999999),
    stamps: clamp(stamps, 0, 999999999),
    tier,
    tierPoints: clamp(points, 0, 999999999),
    createdAt: prev?.createdAt || now,
    updatedAt: now,
  };
}

async function appendWalletEvent(params: {
  uid: string;
  type: WalletEvent["type"];
  platform: Platform;
  currency: "ILS";
  amountCents: number;
  createdBy: WalletEvent["createdBy"];
  refId?: string;
  meta?: Record<string, any>;
}) {
  const { uid, currency } = params;

  const walletRef = db.collection("wallets").doc(uid);
  const eventsCol = db.collection("walletEvents").doc(uid).collection("events");

  return await db.runTransaction(async (tx) => {
    const walletSnap = await tx.get(walletRef);
    const prev = walletSnap.exists ? (walletSnap.data() as WalletSummary) : null;

    // create wallet if missing
    if (!walletSnap.exists) {
      const base: WalletSummary = {
        uid,
        currency,
        balanceCents: 0,
        availableCents: 0,
        pendingCents: 0,
        lifetimeSpentCents: 0,
        stamps: 0,
        tier: "BRONZE",
        tierPoints: 0,
        createdAt: nowIso(),
        updatedAt: nowIso(),
      };
      tx.set(walletRef, base, { merge: false });
    }

    const eventId = `${Date.now()}_${Math.random().toString(16).slice(2)}`;
    const event: WalletEvent = {
      id: eventId,
      uid,
      type: params.type,
      platform: params.platform,
      currency: params.currency,
      amountCents: params.amountCents,
      refId: params.refId,
      meta: params.meta || {},
      createdAt: nowIso(),
      createdBy: params.createdBy,
    };

    tx.set(eventsCol.doc(eventId), event, { merge: false });

    // recompute summary using only this appended event on top of prev
    const next = computeSummaryFromEvents({
      uid,
      currency,
      events: [event],
      prev: prev || undefined,
    });

    // tier change event (optional)
    if (prev && prev.tier !== next.tier) {
      const tierEventId = `${Date.now()}_${Math.random().toString(16).slice(2)}_tier`;
      const tierEvent: WalletEvent = {
        id: tierEventId,
        uid,
        type: "TIER_CHANGED",
        platform: "admin",
        currency,
        amountCents: 0,
        createdAt: nowIso(),
        createdBy: "system",
        meta: { from: prev.tier, to: next.tier, points: next.tierPoints },
      };
      tx.set(eventsCol.doc(tierEventId), tierEvent, { merge: false });
    }

    tx.set(walletRef, next, { merge: true });

    return { event, wallet: next };
  });
}

async function sendPushToUser(uid: string, title: string, body: string, data?: Record<string, string>) {
  const tokensSnap = await db.collection("devices").doc(uid).collection("tokens").get();
  const tokens = tokensSnap.docs.map((d) => d.data().token).filter(Boolean);
  if (tokens.length === 0) return;

  await admin.messaging().sendEachForMulticast({
    tokens,
    notification: { title, body },
    data: data || {},
  });
}

/* -------------------------
   API: Payment webhook (server only)
   This endpoint should be locked down by verifying provider signature.
   When paid, append TOPUP event and update payment status.
   ------------------------- */
app.post("/api/payments/webhook", async (req, res) => {
  try {
    const { paymentId, status } = req.body || {};
    assert(typeof paymentId === "string", "paymentId required");

    const payRef = db.collection("payments").doc(paymentId);
    const snap = await payRef.get();
    assert(snap.exists, "payment not found");
    const pay = snap.data() as any;

    if (pay.status === "paid") return res.json({ ok: true });

    if (status !== "paid") {
      await payRef.set({ status: "failed", updatedAt: nowIso() }, { merge: true });
      return res.json({ ok: true });
    }

    await payRef.set({ status: "paid", updatedAt: nowIso() }, { merge: true });

    const r = await appendWalletEvent({
      uid: pay.uid,
      type: "TOPUP",
      platform: "admin",
      currency: "ILS",
      amountCents: pay.amountCents,
      createdBy: "system",
      refId: paymentId,
      meta: { provider: pay.provider },
    });

    await sendPushToUser(pay.uid, "Wallet topped up", `+₪${(pay.amountCents / 100).toFixed(0)}`, {
      kind: "wallet",
      paymentId,
    });

    return res.json({ ok: true, wallet: r.wallet });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "webhook error" });
  }
});

/* -------------------------
   API: Redeem at Kiosk/Station (signed request)
   This is what kiosk or station calls.
   - Uses HMAC signature so kiosk does not need Firebase Auth.
   - Prevent replay using nonce + ts and store used nonces.
   ------------------------- */
app.post("/api/kiosk/redeem", async (req, res) => {
  try {
    const body = req.body as KioskRedeemRequest;

    const required = ["uid", "sku", "nonce", "ts", "signature"];
    for (const k of required) assert((body as any)[k], `${k} required`);

    const { uid, sku, amountCents, stationId, kioskId, nonce, ts, signature } = body;

    // time window
    const now = Date.now();
    assert(Math.abs(now - ts) < 2 * 60 * 1000, "Request expired");

    // replay protection
    const nonceId = `${uid}_${nonce}`;
    const nonceRef = db.collection("redeemNonces").doc(nonceId);
    const nonceSnap = await nonceRef.get();
    assert(!nonceSnap.exists, "Replay detected");

    // verify signature
    const payload = stablePayload({ uid, sku, amountCents: amountCents || 0, stationId: stationId || "", kioskId: kioskId || "", nonce, ts });
    const expected = hmacSha256Hex(CONFIG.KIOSK_HMAC_SECRET, payload);
    assert(expected === signature, "Bad signature");

    // map SKU to debit amount
    let debitCents = 0;
    if (sku === "WASH_SINGLE") debitCents = 5500;
    else if (sku === "WASH_5") debitCents = 27500;
    else if (sku === "WASH_10") debitCents = 55000;
    else if (sku === "WALLET_CREDIT_CUSTOM") debitCents = clamp(toInt(amountCents, 0), 500, 5000000);
    else assert(false, "Unknown sku");

    // store nonce first in transaction style approach
    await nonceRef.set({ uid, nonce, ts, createdAt: nowIso() }, { merge: false });

    // redeem means spend wallet
    const result = await appendWalletEvent({
      uid,
      type: "REDEEM",
      platform: kioskId ? "kiosk" : "station",
      currency: "ILS",
      amountCents: -debitCents,
      createdBy: kioskId ? "kiosk" : "station",
      refId: `${kioskId || stationId || "unknown"}_${Date.now()}`,
      meta: { sku, kioskId, stationId },
    });

    await sendPushToUser(uid, "Redeemed", `-₪${(debitCents / 100).toFixed(0)} at ${kioskId ? "kiosk" : "station"}`, {
      kind: "redeem",
      sku,
    });

    return res.json({ ok: true, redeemId: result.event.id, newBalanceCents: result.wallet.balanceCents });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "redeem error" });
  }
});

/* =====================================================================================
   FILE: firestore.rules
   ===================================================================================== */

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }
    function isOwner(uid) { return isSignedIn() && request.auth.uid == uid; }
    function isAdmin() { return isSignedIn() && request.auth.token.admin == true; }

    // Users can read their profile, update only safe fields
    match /users/{uid} {
      allow read: if isOwner(uid) || isAdmin();
      allow create: if isOwner(uid);
      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).changedKeys().hasOnly([
          "displayName","locale","twoFactorMode","notificationPrefs","phone"
        ]);
      allow delete: if false;
    }

    // Wallet summary is read only for user, write only by server
    match /wallets/{uid} {
      allow read: if isOwner(uid) || isAdmin();
      allow write: if false;
    }

    // Wallet events are read only for user, write only by server
    match /walletEvents/{uid}/events/{eventId} {
      allow read: if isOwner(uid) || isAdmin();
      allow write: if false;
    }

    // Rewards read for user, write only by server
    match /rewards/{uid}/items/{rewardId} {
      allow read: if isOwner(uid) || isAdmin();
      allow write: if false;
    }

    // Device tokens: user can write their own tokens
    match /devices/{uid}/tokens/{tokenId} {
      allow read: if isOwner(uid) || isAdmin();
      allow create, update: if isOwner(uid);
      allow delete: if isOwner(uid) || isAdmin();
    }

    // Stations and kiosks: read public, write admin
    match /stations/{stationId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /kiosks/{kioskId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Payments: user can read their own, write only server/admin
    match /payments/{paymentId} {
      allow read: if isSignedIn() && request.auth.uid == resource.data.uid || isAdmin();
      allow write: if false;
    }

    match /auditLogs/{logId} {
      allow read: if isAdmin();
      allow write: if false;
    }

    match /redeemNonces/{nonceId} {
      allow read: if false;
      allow write: if false;
    }
  }
}

export async function apiGet(path: string, token?: string) {
  const r = await fetch(path, {
    headers: token ? { Authorization: `Bearer ${token}` } : {},
  });
  return await r.json();
}

export async function apiPost(path: string, body: any, token?: string) {
  const r = await fetch(path, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: JSON.stringify(body),
  });
  return await r.json();
}

   D) Booking state machine backend validated

   - Reject invalid transitions server side

   E) Rewards
   - Tier based perks plus stamp cycle rewards

   - FCM tokens per device
   - Notify on topup, redeem, tier upgrade, booking updates

   - Web: Tailwind, container queries optional
