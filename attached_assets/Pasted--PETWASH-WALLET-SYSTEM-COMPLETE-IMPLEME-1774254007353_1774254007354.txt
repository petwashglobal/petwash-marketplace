function generateQRPayload(userId: string) {
  return `petwash:${userId}:${uuid()}`;
}

import PKPass from "passkit-generator";

async function generateApplePass(user: any) {
  const pass = await PKPass.from({
    model: "./apple-pass-model", // folder with images + pass.json template
    certificates: {
      wwdr: fs.readFileSync("./certs/wwdr.pem"),
      signerCert: fs.readFileSync(process.env.APPLE_CERT_PATH!),
      signerKey: fs.readFileSync(process.env.APPLE_CERT_PATH!),
      signerKeyPassphrase: process.env.APPLE_CERT_PASSWORD
    }
  });

  const qrValue = generateQRPayload(user.id);

  pass.primaryFields.add({
    key: "name",
    label: "MEMBER",
    value: user.name
  });

  pass.secondaryFields.add({
    key: "tier",
    label: "TIER",
    value: user.tier
  });

  pass.auxiliaryFields.add({
    key: "points",
    label: "POINTS",
    value: `${user.points}`
  });

  pass.backFields.add([
    {
      key: "reward",
      label: "NEXT REWARD",
      value: `${user.nextReward} pts`
    },
    {
      key: "since",
      label: "MEMBER SINCE",
      value: user.memberSince
    },
    {
      key: "site",
      label: "WEBSITE",
      value: "https://petwash.co.il"
    }
  ]);

  pass.setBarcodes({
    message: qrValue,
    format: "PKBarcodeFormatQR",
    messageEncoding: "iso-8859-1"
  });

  return pass.getAsBuffer();
}

/*
========================================================
GOOGLE WALLET JWT
========================================================
*/

function generateGoogleWalletJWT(user: any) {
  const objectId = `${process.env.GOOGLE_ISSUER_ID}.${user.id}`;

  const payload = {
    iss: process.env.GOOGLE_SERVICE_ACCOUNT_EMAIL,
    aud: "google",
    typ: "savetowallet",
    payload: {
      genericObjects: [
        {
          id: objectId,
          classId: `${process.env.GOOGLE_ISSUER_ID}.petwash`,
          state: "active",
          cardTitle: { defaultValue: { value: "PetWash Prestige" } },
          subheader: { defaultValue: { value: user.name } },
          header: { defaultValue: { value: user.tier } },
          barcode: {
            type: "QR_CODE",
            value: generateQRPayload(user.id)
          },
          textModulesData: [
            { header: "Points", body: `${user.points}` },
            { header: "Next Reward", body: `${user.nextReward}` }
          ]
        }
      ]
    }
  };

  const token = jwt.sign(payload, process.env.GOOGLE_PRIVATE_KEY!, {
    algorithm: "RS256"
  });

  return `https://pay.google.com/gp/v/save/${token}`;
}

// Apple Wallet
app.get("/wallet/apple/:userId", async (req, res) => {
  const user = users[req.params.userId];
  if (!user) return res.sendStatus(404);

  const pass = await generateApplePass(user);

  res.set({
    "Content-Type": "application/vnd.apple.pkpass",
    "Content-Disposition": "attachment; filename=petwash.pkpass"
  });

  res.send(pass);
});

// Google Wallet
app.get("/wallet/google/:userId", (req, res) => {
  const user = users[req.params.userId];
  if (!user) return res.sendStatus(404);

  const url = generateGoogleWalletJWT(user);
  res.json({ url });
});
