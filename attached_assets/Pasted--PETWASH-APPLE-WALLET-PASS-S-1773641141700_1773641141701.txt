STEP 5 — APPLE WALLET PASS GENERATION
=====================================================================

Recommended Node library:

passkit-generator

Install:

npm install passkit-generator

Example usage:

const { PKPass } = require("passkit-generator");

async function buildAppleWalletPass(record, qrToken) {

  const pass = await PKPass.from({
    model: "./passModel",
    certificates: {
      wwdr: process.env.APPLE_WALLET_WWDR_CERT,
      signerCert: process.env.APPLE_WALLET_CERT_PATH,
      signerKey: process.env.APPLE_WALLET_CERT_PATH,
      signerKeyPassphrase: process.env.APPLE_WALLET_CERT_PASSWORD
    }
  });

  pass.primaryFields.add({
    key: "owner",
    label: "Member",
    value: record.ownerName
  });

  pass.secondaryFields.add({
    key: "credit",
    label: "Available Credit",
    value: `₪${record.availableCredit}`
  });

  pass.auxiliaryFields.add({
    key: "memberId",
    label: "Member ID",
    value: record.passId
  });

  pass.setBarcodes({
    message: qrToken,
    format: "PKBarcodeFormatQR"
  });

  return pass.getAsBuffer();
}
