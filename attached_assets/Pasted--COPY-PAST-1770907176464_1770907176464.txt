/*****************************************************************************************
COPY-PASTE FULL PACKAGE (Backend + OTP + SMS + Email + Rate Limit + Device Trust + Logging
+ Minimal Mobile/Web UI flow)
------------------------------------------------------------------------------------------
Goal:
- One OTP per session (single source of truth)
- Deliver via SMS, Email, or Both (same code)
- Verify once, then stamp session as verified for ALL platforms via "Octopus Brain"

Stack:
- Backend: Node 20 + Express + TypeScript
- OTP Store: Redis (TTL)
- SMS: Twilio Messaging (NOT Twilio Verify)
- Email: SendGrid (or swap to SES)
- Auth: JWT access + refresh (basic example)
- Rate limits: IP + per session attempts
- Session record: Firestore (optional) OR Postgres OR any DB (example included)

How to use:
1) Create a folder "octopus-2fa"
2) Create files exactly as marked below
3) Fill .env
4) Run backend
5) Plug frontend calls into your app

------------------------------------------------------------------------------------------
FILES
------------------------------------------------------------------------------------------
1) backend/package.json

------------------------------------------------------------------------------------------
IMPORTANT NOTES
------------------------------------------------------------------------------------------
- Twilio error 21408 (geo permissions) is a Twilio Console setting. Enable Israel (+972).
- Use Twilio Messaging to send SMS. Do NOT use Twilio Verify if you want one unified OTP.
- OTP is stored in Redis with TTL. Firestore is used only to stamp verified sessions.
- This is the correct scalable pattern for multi-product platforms.

import crypto from "crypto";

export function uuid(): string {
  return crypto.randomUUID();
}

export function nowIso(): string {
  return new Date().toISOString();
}

export function normalizeEmail(email: string): string {
  return String(email || "").trim().toLowerCase();
}

export function normalizePhone(phone: string): string {
  // Keep + and digits only
  const p = String(phone || "").trim();
  const cleaned = p.replace(/[^\d+]/g, "");
  return cleaned;
}

export function safeName(name: string): string {
  return String(name || "").trim().replace(/\s+/g, " ");
}

export function isHebrewLocale(locale?: string): boolean {
  return String(locale || "").toLowerCase().startsWith("he");
}

export function formatText(locale: "he" | "en", en: string, he: string): string {
  return locale === "he" ? he : en;
}

export function generateOtp6(): string {
  return String(Math.floor(100000 + Math.random() * 900000));
}

export function constantTimeEq(a: string, b: string): boolean {
  const aa = Buffer.from(a);
  const bb = Buffer.from(b);
  if (aa.length !== bb.length) return false;
  return crypto.timingSafeEqual(aa, bb);
}

import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import { CONFIG } from "../config.js";

export type AuthedRequest = Request & {
  user?: { userId: string; role?: string };
};

export function requireAuth(req: AuthedRequest, res: Response, next: NextFunction) {
  try {
    const auth = String(req.headers.authorization || "");
    const token = auth.startsWith("Bearer ") ? auth.slice(7) : "";
    if (!token) return res.status(401).json({ ok: false, error: "missing_token" });

    const decoded = jwt.verify(token, CONFIG.JWT_ACCESS_SECRET) as any;
    if (!decoded?.userId) return res.status(401).json({ ok: false, error: "invalid_token" });

    req.user = { userId: String(decoded.userId), role: decoded.role ? String(decoded.role) : undefined };
    next();
  } catch {
    return res.status(401).json({ ok: false, error: "unauthorized" });
  }
}

/* =======================================================================================
FILE: backend/src/middleware/rateLimit.ts
---------------------------------------------------------------------------------------
Simple Redis-based rate limiting:
- IP bucket for /2fa/request
- Session attempt limit for /2fa/verify
- Cooldown between requests per user
======================================================================================= */
import { Request, Response, NextFunction } from "express";
import { redis } from "../redis.js";
import { CONFIG } from "../config.js";

function ipKey(ip: string) {
  return `rl:ip:${ip}`;
}

export async function limitIpRequests(req: Request, res: Response, next: NextFunction) {
  try {
    const ip = (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() || req.socket.remoteAddress || "unknown";
    const key = ipKey(ip);
    const count = await redis.incr(key);
    if (count === 1) {
      await redis.expire(key, 600); // 10 minutes
    }
    if (count > CONFIG.IP_MAX_REQUESTS_PER_10MIN) {
      return res.status(429).json({ ok: false, error: "rate_limited_ip" });
    }
    next();
  } catch {
    return res.status(500).json({ ok: false, error: "rate_limit_error" });
  }
}

export async function enforceUserCooldown(userId: string) {
  const key = `rl:userCooldown:${userId}`;
  const exists = await redis.get(key);
  if (exists) return false;
  await redis.set(key, "1", "EX", CONFIG.OTP_MIN_SECONDS_BETWEEN_REQUESTS);
  return true;
}

export async function bumpVerifyAttempt(userId: string, sessionId: string) {
  const key = `2fa:attempts:${userId}:${sessionId}`;
  const n = await redis.incr(key);
  if (n === 1) await redis.expire(key, CONFIG.OTP_TTL_SEC);
  return n;
}

/* =======================================================================================
FILE: backend/src/services/otpService.ts
======================================================================================= */
import { redis } from "../redis.js";
import { CONFIG } from "../config.js";
import { generateOtp6, constantTimeEq } from "../utils.js";

function otpKey(userId: string, sessionId: string) {
  return `2fa:otp:${userId}:${sessionId}`;
}

export async function createOtp(userId: string, sessionId: string) {
  const code = generateOtp6();
  await redis.set(otpKey(userId, sessionId), code, "EX", CONFIG.OTP_TTL_SEC);
  return code;
}

export async function verifyOtp(userId: string, sessionId: string, input: string) {
  const key = otpKey(userId, sessionId);
  const stored = await redis.get(key);
  if (!stored) return { ok: false as const, reason: "expired_or_missing" as const };

  const good = constantTimeEq(stored, String(input || ""));
  if (!good) return { ok: false as const, reason: "invalid_code" as const };

  await redis.del(key);
  return { ok: true as const };
}

/* =======================================================================================
FILE: backend/src/services/sessionStore.ts
---------------------------------------------------------------------------------------
This is the "Octopus Brain stamp" that makes 2FA apply to ALL platforms.
You can swap Firestore with Postgres easily.
======================================================================================= */
import { CONFIG } from "../config.js";
import { nowIso } from "../utils.js";

// Optional Firestore
import admin from "firebase-admin";

let firestoreReady = false;
function ensureFirestore() {
  if (firestoreReady) return;
  if (!CONFIG.FIREBASE_PROJECT_ID) {
    // If you do not want Firestore, you can disable session stamping here.
    firestoreReady = false;
    return;
  }
  if (admin.apps.length === 0) {
    admin.initializeApp({
      projectId: CONFIG.FIREBASE_PROJECT_ID
    });
  }
  firestoreReady = true;
}

export type SessionRecord = {
  sessionId: string;
  userId: string;
  method: "sms" | "email" | "both";
  locale: "he" | "en";
  createdAt: string;
  verified2fa: boolean;
  verifiedAt?: string;
  ip?: string;
  deviceId?: string;
  deviceTrusted?: boolean;
  meta?: Record<string, any>;
};

export async function createSession(rec: SessionRecord) {
  ensureFirestore();
  if (!firestoreReady) return;

  const db = admin.firestore();
  await db.collection("sessions").doc(rec.sessionId).set(rec, { merge: false });
}

export async function markSessionVerified(userId: string, sessionId: string) {
  ensureFirestore();
  if (!firestoreReady) return;

  const db = admin.firestore();
  await db.collection("sessions").doc(sessionId).set(
    {
      userId,
      verified2fa: true,
      verifiedAt: nowIso()
    },
    { merge: true }
  );
}

export async function setTrustedDevice(userId: string, deviceId: string) {
  ensureFirestore();
  if (!firestoreReady) return;

  const db = admin.firestore();
  await db.collection("trustedDevices").doc(`${userId}:${deviceId}`).set(
    {
      userId,
      deviceId,
      trusted: true,
      trustedAt: nowIso()
    },
    { merge: true }
  );
}

import { Router } from "express";
import { uuid, normalizeEmail, normalizePhone, safeName, isHebrewLocale } from "../utils.js";
import { createOtp, verifyOtp } from "../services/otpService.js";
import { sendSms } from "../services/smsService.js";
import { sendEmail } from "../services/emailService.js";
import { createSession, markSessionVerified, setTrustedDevice } from "../services/sessionStore.js";
import { enforceUserCooldown, bumpVerifyAttempt } from "../middleware/rateLimit.js";
import { CONFIG } from "../config.js";

export const twofaRouter = Router();

/**
 * POST /2fa/request
 * Body:
 *  userId: string (from your main auth system, or temp id before login)
 *  method: "sms" | "email" | "both"
 *  phone?: string
 *  email?: string
 *  name?: string
 *  locale?: "he" | "en" | "he-IL" | etc
 *  deviceId?: string (stable device identifier from app)
 *  meta?: any
 */
twofaRouter.post("/request", async (req, res) => {
  try {
    const userId = String(req.body?.userId || "").trim();
    const method = String(req.body?.method || "").trim().toLowerCase();
    const locale: "he" | "en" = isHebrewLocale(req.body?.locale) ? "he" : "en";

    if (!userId) return res.status(400).json({ ok: false, error: "missing_userId" });
    if (!["sms", "email", "both"].includes(method)) return res.status(400).json({ ok: false, error: "invalid_method" });

    const okCooldown = await enforceUserCooldown(userId);
    if (!okCooldown) return res.status(429).json({ ok: false, error: "cooldown_active" });

    const phone = normalizePhone(req.body?.phone || "");
    const email = normalizeEmail(req.body?.email || "");
    const name = safeName(req.body?.name || "");
    const deviceId = String(req.body?.deviceId || "").trim();

    if (method === "sms" || method === "both") {
      if (!phone.startsWith("+")) return res.status(400).json({ ok: false, error: "invalid_phone" });
    }
    if (method === "email" || method === "both") {
      if (!email.includes("@")) return res.status(400).json({ ok: false, error: "invalid_email" });
    }

    const sessionId = uuid();
    const code = await createOtp(userId, sessionId);

    // Send
    if (method === "sms" || method === "both") {
      await sendSms(phone, code, locale);
    }
    if (method === "email" || method === "both") {
      await sendEmail(email, code, locale);
    }

    // Stamp session in Octopus Brain store (optional)
    const ip =
      (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() || req.socket.remoteAddress || "unknown";

    await createSession({
      sessionId,
      userId,
      method: method as any,
      locale,
      createdAt: new Date().toISOString(),
      verified2fa: false,
      ip,
      deviceId: deviceId || undefined,
      deviceTrusted: false,
      meta: {
        name: name || undefined,
        phone: phone || undefined,
        email: email || undefined,
        platform: req.body?.meta?.platform || undefined
      }
    });

    return res.json({
      ok: true,
      sessionId,
      ttlSec: CONFIG.OTP_TTL_SEC
    });
  } catch (e: any) {
    return res.status(500).json({ ok: false, error: e?.message || "server_error" });
  }
});

/**
 * POST /2fa/verify
 * Body:
 *  userId: string
 *  sessionId: string
 *  code: string
 *  trustThisDevice?: boolean
 *  deviceId?: string
 */
twofaRouter.post("/verify", async (req, res) => {
  try {
    const userId = String(req.body?.userId || "").trim();
    const sessionId = String(req.body?.sessionId || "").trim();
    const code = String(req.body?.code || "").trim();
    const trustThisDevice = Boolean(req.body?.trustThisDevice);
    const deviceId = String(req.body?.deviceId || "").trim();

    if (!userId || !sessionId || !code) return res.status(400).json({ ok: false, error: "missing_fields" });

    // Attempt limit
    const attempts = await bumpVerifyAttempt(userId, sessionId);
    if (attempts > CONFIG.OTP_MAX_ATTEMPTS) {
      return res.status(429).json({ ok: false, error: "too_many_attempts" });
    }

    const result = await verifyOtp(userId, sessionId, code);
    if (!result.ok) {
      return res.status(401).json({ ok: false, error: result.reason });
    }

    // Stamp verified in Octopus Brain store (Firestore)
    await markSessionVerified(userId, sessionId);

    // Device trust (optional)
    if (trustThisDevice && deviceId) {
      await setTrustedDevice(userId, deviceId);
    }

    return res.json({ ok: true });
  } catch (e: any) {
    return res.status(500).json({ ok: false, error: e?.message || "server_error" });
  }
});

import { Router } from "express";
import jwt from "jsonwebtoken";
import { CONFIG } from "../config.js";

export const authRouter = Router();

function signAccess(userId: string, role?: string) {
  return jwt.sign({ userId, role }, CONFIG.JWT_ACCESS_SECRET, { expiresIn: CONFIG.JWT_ACCESS_EXPIRES_SEC });
}
function signRefresh(userId: string) {
  return jwt.sign({ userId }, CONFIG.JWT_REFRESH_SECRET, { expiresIn: CONFIG.JWT_REFRESH_EXPIRES_SEC });
}

authRouter.post("/login-demo", async (req, res) => {
  // DEMO ONLY:
  // In production: validate password, then require 2FA, then issue tokens.
  const userId = String(req.body?.userId || "").trim();
  if (!userId) return res.status(400).json({ ok: false, error: "missing_userId" });

  const accessToken = signAccess(userId, "user");
  const refreshToken = signRefresh(userId);

  return res.json({ ok: true, accessToken, refreshToken });
});

export type TwoFAMethod = "sms" | "email" | "both";

export async function request2FA(args: {
  apiBase: string;
  userId: string;
  method: TwoFAMethod;
  phone?: string;
  email?: string;
  name?: string;
  locale?: string;
  deviceId?: string;
  meta?: any;
}) {
  const res = await fetch(`${args.apiBase}/2fa/request`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(args)
  });
  const data = await res.json();
  if (!res.ok) throw new Error(data?.error || "request_failed");
  return data as { ok: true; sessionId: string; ttlSec: number };
}

export async function verify2FA(args: {
  apiBase: string;
  userId: string;
  sessionId: string;
  code: string;
  trustThisDevice?: boolean;
  deviceId?: string;
}) {
  const res = await fetch(`${args.apiBase}/2fa/verify`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(args)
  });
  const data = await res.json();
  if (!res.ok) throw new Error(data?.error || "verify_failed");
  return data as { ok: true };
}

type Props = {
  apiBase: string;        // ex: https://api.petwash.co.il
  userId: string;         // your user id from main auth
  deviceId?: string;      // stable id if you have it
  locale?: string;        // "he-IL" or "en"
};

export function TwoFactorUI(props: Props) {
  const [method, setMethod] = useState<TwoFAMethod>("sms");
  const [phone, setPhone] = useState("");
  const [email, setEmail] = useState("");
  const [sessionId, setSessionId] = useState("");
  const [code, setCode] = useState("");
  const [trust, setTrust] = useState(true);
  const [status, setStatus] = useState("");

  async function onRequest() {
    setStatus("Requesting code...");
    try {
      const r = await request2FA({
        apiBase: props.apiBase,
        userId: props.userId,
        method,
        phone,
        email,
        name: "User",
        locale: props.locale || "he-IL",
        deviceId: props.deviceId,
        meta: { platform: "app" }
      });
      setSessionId(r.sessionId);
      setStatus(`Code sent. Session: ${r.sessionId}`);
    } catch (e: any) {
      setStatus(`Request failed: ${e?.message || "error"}`);
    }
  }

  async function onVerify() {
    setStatus("Verifying...");
    try {
      await verify2FA({
        apiBase: props.apiBase,
        userId: props.userId,
        sessionId,
        code,
        trustThisDevice: trust,
        deviceId: props.deviceId
      });
      setStatus("Verified. Continue to app.");
    } catch (e: any) {
      setStatus(`Verify failed: ${e?.message || "error"}`);
    }
  }

  return (
    <div style={{ maxWidth: 420, fontFamily: "system-ui", padding: 16 }}>
      <h3>Two Factor Verification</h3>

      <label>Method</label>
      <select value={method} onChange={(e) => setMethod(e.target.value as TwoFAMethod)}>
        <option value="sms">SMS</option>
        <option value="email">Email</option>
        <option value="both">Both</option>
      </select>

      {(method === "sms" || method === "both") && (
        <>
          <label>Phone (+972...)</label>
          <input value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+9725..." />
        </>
      )}

      {(method === "email" || method === "both") && (
        <>
          <label>Email</label>
          <input value={email} onChange={(e) => setEmail(e.target.value)} placeholder="name@email.com" />
        </>
      )}

      <button onClick={onRequest} style={{ marginTop: 10 }}>Send code</button>

      <hr />

      <label>Session ID</label>
      <input value={sessionId} onChange={(e) => setSessionId(e.target.value)} placeholder="auto-filled" />

      <label>Code</label>
      <input value={code} onChange={(e) => setCode(e.target.value)} placeholder="123456" />

      <label style={{ display: "flex", gap: 8, alignItems: "center" }}>
        <input type="checkbox" checked={trust} onChange={(e) => setTrust(e.target.checked)} />
        Trust this device
      </label>

      <button onClick={onVerify} style={{ marginTop: 10 }}>Verify</button>

      <p style={{ marginTop: 12 }}>{status}</p>
    </div>
  );
}
