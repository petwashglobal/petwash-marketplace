import React, { useMemo } from "react";
import PhoneInput, {
  getCountries,
  isValidPhoneNumber,
  parsePhoneNumber
} from "react-phone-number-input";
import "react-phone-number-input/style.css";

type IntlPhoneProps = {
  value: string; // E.164
  onChange: (v: string) => void;
  defaultCountry?: ApprovedCountry;
  allowedCountries?: ApprovedCountry[];
  label?: string;
};

export function IntlPhone({
  value,
  onChange,
  defaultCountry = "IL",
  allowedCountries = [...APPROVED_COUNTRIES],
  label = "Mobile"
}: IntlPhoneProps) {
  const allowed = useMemo(() => {
    const supported = new Set(getCountries());
    return allowedCountries.filter((c) => supported.has(c as any));
  }, [allowedCountries]);

  const safeDefault = allowed.includes(defaultCountry) ? defaultCountry : (allowed[0] || "IL");

  const helper = useMemo(() => {
    if (!value) return "Enter mobile number";
    try {
      const ok = isValidPhoneNumber(value);
      if (!ok) return "Invalid number";
      const pn = parsePhoneNumber(value);
      return `Saved: ${pn?.number || value}`;
    } catch {
      return "Invalid number";
    }
  }, [value]);

  return (
    <div style={{ display: "grid", gap: 8 }}>
      <label style={{ fontSize: 13, fontWeight: 600 }}>{label}</label>
      <div style={{ border: "1px solid rgba(0,0,0,0.12)", borderRadius: 12, padding: "10px 12px" }}>
        <PhoneInput
          international
          withCountryCallingCode
          defaultCountry={safeDefault as any}
          countries={allowed as any}
          value={value}
          onChange={(v) => onChange(v || "")}
          placeholder="e.g. +972 50 123 4567"
        />
      </div>
      <div style={{ fontSize: 12, opacity: 0.7 }}>{helper}</div>
    </div>
  );
}

/* =========================================================
   3) THE ONLY VALID FORMAT IN YOUR SYSTEM: E.164
   ========================================================= */
import { isValidPhoneNumber as _isValid, parsePhoneNumber as _parse } from "react-phone-number-input";

export function normalizeAndValidateE164(phoneE164: string) {
  const phone = (phoneE164 || "").trim();
  if (!phone.startsWith("+")) throw new Error("PHONE_MUST_START_WITH_PLUS");
  if (!_isValid(phone)) throw new Error("INVALID_PHONE");
  const pn = _parse(phone);
  return {
    phoneE164: pn?.number || phone,
    country: (pn?.country || null) as string | null,
    callingCode: pn?.countryCallingCode || null,
    nationalNumber: pn?.nationalNumber || null
  };
}

/* =========================================================
   4) WEB PHONE AUTH - CORRECT OTP FLOW (Firebase)
   IMPORTANT:
   - Web requires reCAPTCHA verifier (invisible)
   - iOS/Android native does NOT use web recaptcha
   ========================================================= */
import { getAuth, RecaptchaVerifier, signInWithPhoneNumber, ConfirmationResult } from "firebase/auth";

let confirmationResult: ConfirmationResult | null = null;

export async function webSendOtp(phoneE164: string, traceId: string) {
  const auth = getAuth();

  // Create recaptcha verifier once
  const elId = "recaptcha-container";
  if (!(window as any).recaptchaVerifier) {
    (window as any).recaptchaVerifier = new RecaptchaVerifier(auth, elId, {
      size: "invisible"
    });
  }

  const appVerifier = (window as any).recaptchaVerifier;
  confirmationResult = await signInWithPhoneNumber(auth, phoneE164, appVerifier);

  return { ok: true, traceId };
}

export async function webVerifyOtp(code: string, traceId: string) {
  if (!confirmationResult) throw new Error("NO_CONFIRMATION_SESSION");
  const cred = await confirmationResult.confirm(code);

  // Get Firebase ID token and exchange for server session cookie
  const idToken = await cred.user.getIdToken(true);

  const resp = await fetch("/api/auth/session", {
    method: "POST",
    headers: { "Content-Type": "application/json", "X-Trace-Id": traceId },
    body: JSON.stringify({ idToken })
  });
  if (!resp.ok) throw new Error("SESSION_EXCHANGE_FAILED");

  return { ok: true, uid: cred.user.uid, traceId };
}

/* ================================
   SERVER COUNTRY APPROVAL CHECK
   ================================ */
export function assertApprovedCountry(country: string | null) {
  if (!country) throw new Error("COUNTRY_UNKNOWN");
  if (!APPROVED_COUNTRIES.includes(country as any)) throw new Error("COUNTRY_NOT_APPROVED");
}
