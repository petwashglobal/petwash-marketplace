DATABASE_URL=postgres://user:pass@localhost:5432/appdb

# Security
JWT_SECRET=replace_me_long_random
HMAC_STAMP_SECRET=replace_me_long_random
ALLOWED_ORIGINS=https://your-domain.com,http://localhost:5173

# Google Service Account JSON (base64-encoded)
# echo -n "$(cat service-account.json | base64)" then paste here
GOOGLE_SERVICE_ACCOUNT_B64=

# Google Sheets
GOOGLE_SHEETS_SPREADSHEET_ID=your_sheet_id
GOOGLE_SHEETS_VIP_TAB=VIP_KYC
GOOGLE_SHEETS_PROVIDER_TAB=PROVIDERS

# Google Drive (optional)
GOOGLE_DRIVE_FOLDER_ID=your_drive_folder_id

# Optional: turn on/off Google writes without breaking core flow
FEATURE_GOOGLE_SHEETS=true
FEATURE_GOOGLE_DRIVE=true

  if (!v) throw new Error(`Missing env var: ${name}`);
  return v;
}
