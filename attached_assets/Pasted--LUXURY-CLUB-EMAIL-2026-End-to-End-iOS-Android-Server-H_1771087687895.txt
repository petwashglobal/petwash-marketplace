function escapeHtml(s: string) {
  return String(s || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}
function escapeAttr(s: string) {
  return escapeHtml(s).replace(/`/g, "&#096;");
}

import * as functions from "firebase-functions";
import * as admin from "firebase-admin";

admin.initializeApp();

/**
 * Firestore recommended structure:
 * users/{uid} -> { email, displayName, locale, marketingOptIn }
 * events/{eventId} -> { kind, title, subtitle, imageUrl, ctaUrl, tierLabel, tierAccent, benefits[], moments[] }
 *
 * No fake claims:
 * - This function only sends content that exists in Firestore event doc or safe defaults.
 */

export const sendLuxuryEventEmail = functions.https.onCall(async (data, context) => {
  if (!context.auth) throw new functions.https.HttpsError("unauthenticated", "Sign in required");

  const uid = context.auth.uid;
  const eventId = String(data?.eventId || "").trim();
  if (!eventId) throw new functions.https.HttpsError("invalid-argument", "eventId is required");

  const userSnap = await admin.firestore().collection("users").doc(uid).get();
  const user = userSnap.data() || {};
  const toEmail = String(user.email || "").trim();
  if (!toEmail) throw new functions.https.HttpsError("failed-precondition", "No email on user profile");

  // Optional: respect marketingOptIn for promo events
  const marketingOptIn = user.marketingOptIn !== false;

  const evSnap = await admin.firestore().collection("events").doc(eventId).get();
  if (!evSnap.exists) throw new functions.https.HttpsError("not-found", "Event not found");
  const ev = evSnap.data() || {};

  const kind = String(ev.kind || "club_event"); // "welcome" | "tier_upgrade" | "gift" | "club_event"
  const isMarketing = kind === "club_event" || kind === "promo";
  if (isMarketing && !marketingOptIn) {
    return { ok: true, skipped: true, reason: "marketing_opt_out" };
  }

  const brandName = String(ev.brandName || "YourBrand");
  const programName = String(ev.programName || "Prestige Club");

  const viewInBrowserUrl = String(ev.viewInBrowserUrl || "https://yourdomain.com/email/" + eventId);
  const unsubscribeUrl = String(ev.unsubscribeUrl || "https://yourdomain.com/unsubscribe?uid=" + uid);

  const subjectLine = String(ev.subjectLine || "Welcome to " + programName);
  const preheader = String(ev.preheader || "Your membership is ready. Wallet, benefits, and premium moments.");
  const heroTitle = String(ev.title || "Welcome to Prestige");
  const heroSubtitle = String(ev.subtitle || "A premium membership built for calm, quality, and smart rewards.");
  const heroImageUrl = String(ev.imageUrl || "https://yourcdn.com/images/hero.jpg");
  const logoUrl = String(ev.logoUrl || "https://yourcdn.com/images/logo.png");

  const primaryCtaUrl = String(ev.ctaUrl || "https://yourdomain.com/app");
  const primaryCtaLabel = String(ev.ctaLabel || "Open membership");

  const secondaryCtaUrl = ev.secondaryCtaUrl ? String(ev.secondaryCtaUrl) : "";
  const secondaryCtaLabel = ev.secondaryCtaLabel ? String(ev.secondaryCtaLabel) : "View benefits";

  const tierLabel = String(ev.tierLabel || "Pearl");
  const tierAccent = (String(ev.tierAccent || "gold") as any);

  const walletBalanceText = ev.walletBalanceText ? String(ev.walletBalanceText) : "";
  const memberSinceText = ev.memberSinceText ? String(ev.memberSinceText) : "Member since 2026";

  const benefits = Array.isArray(ev.benefits) ? ev.benefits.slice(0, 6) : [
    { title: "Digital wallet", desc: "Balance, top-ups and clear tracking." },
    { title: "Premium gifting", desc: "Send elegant digital gifts for any occasion." },
    { title: "Tier privileges", desc: "Unlock higher status and exclusive advantages." },
    { title: "Activity timeline", desc: "Every action logged with timestamps." },
  ];

  const moments = Array.isArray(ev.moments) ? ev.moments.slice(0, 3) : [
    { label: "Fast redemption", desc: "QR redemption in a smooth, controlled flow." },
    { label: "Calm notifications", desc: "Status updates and important reminders, no noise." },
  ];

  const html = LUXURY_EMAIL_HTML({
    brandName,
    programName,
    recipientName: String(user.displayName || ""),
    subjectLine,
    preheader,
    heroTitle,
    heroSubtitle,
    primaryCtaLabel,
    primaryCtaUrl,
    secondaryCtaLabel: secondaryCtaUrl ? secondaryCtaLabel : undefined,
    secondaryCtaUrl: secondaryCtaUrl || undefined,
    cardTierLabel: tierLabel,
    cardTierAccent: tierAccent,
    walletBalanceText,
    memberSinceText,
    benefits,
    moments,
    footerLine1: "This message confirms a club event and access to your membership experience.",
    footerLine2: "If you did not request this email, you can ignore it.",
    logoUrl,
    heroImageUrl,
    jewelImageUrl: ev.jewelImageUrl ? String(ev.jewelImageUrl) : undefined,
    supportEmail: String(ev.supportEmail || "support@yourdomain.com"),
    unsubscribeUrl,
    viewInBrowserUrl,
  });

  // Provider selection
  const provider = String(functions.config()?.email?.provider || "sendgrid").toLowerCase();
  const from = String(functions.config()?.email?.from || `${brandName} <noreply@yourdomain.com>`);

  if (provider === "sendgrid") {
    // SendGrid
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const sg = require("@sendgrid/mail");
    const key = String(functions.config()?.sendgrid?.key || "");
    if (!key) throw new functions.https.HttpsError("failed-precondition", "Missing SendGrid key");
    sg.setApiKey(key);

    await sg.send({
      to: toEmail,
      from,
      subject: subjectLine,
      html,
    });

  } else if (provider === "resend") {
    // Resend
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const { Resend } = require("resend");
    const key = String(functions.config()?.resend?.key || "");
    if (!key) throw new functions.https.HttpsError("failed-precondition", "Missing Resend key");

    const resend = new Resend(key);
    await resend.emails.send({
      from,
      to: toEmail,
      subject: subjectLine,
      html,
    });

  } else {
    throw new functions.https.HttpsError("failed-precondition", "Unknown email provider");
  }

  // Audit log (server truth)
  await admin.firestore().collection("email_audit").add({
    uid,
    eventId,
    toEmail,
    subjectLine,
    sentAt: admin.firestore.FieldValue.serverTimestamp(),
    provider,
    kind,
  });

  return { ok: true, sent: true };
});
