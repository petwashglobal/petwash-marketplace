      function applyLang(lang){
        const isHe = (lang === "he");
        document.documentElement.lang = lang;
        document.documentElement.dir = isHe ? "rtl" : "ltr";
        $("#brandSub").textContent = dict[lang].brandSub;

        $$("[data-i18n]").forEach(el=>{
          const k = el.getAttribute("data-i18n");
          if(k && dict[lang][k]) el.textContent = dict[lang][k];
        });

        // Update demo membership copy direction-sensitive
        $("#memberSub").textContent = isHe
          ? "חבר מאז 2026 · ארנק והטבות מסתנכרנים בין שירותים"
          : "Member since 2026 · Wallet and benefits synced across services";
        $("#walletLabel").textContent = isHe ? "יתרה בארנק" : "Wallet Balance";

        // Buttons active feel
        if(isHe){
          $("#btnHE").classList.add("btnPrimary");
          $("#btnEN").classList.remove("btnPrimary");
        } else {
          $("#btnEN").classList.add("btnPrimary");
          $("#btnHE").classList.remove("btnPrimary");
        }
      }

      // Language buttons
      $("#btnHE").addEventListener("click", ()=> applyLang("he"));
      $("#btnEN").addEventListener("click", ()=> applyLang("en"));

      // Init language
      applyLang("he");
    })();
