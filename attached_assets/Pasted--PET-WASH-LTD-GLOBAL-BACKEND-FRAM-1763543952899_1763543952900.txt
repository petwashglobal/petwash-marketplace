/* ===========================================================
   1. DRIZZLE SCHEMAS
   =========================================================== */
import { pgTable, text, timestamp, varchar, integer, boolean, uuid, float } from "drizzle-orm/pg-core";

export const contractors = pgTable("contractors", {
  id: uuid("id").primaryKey().defaultRandom(),
  fullName: text("full_name").notNull(),
  email: varchar("email", { length: 255 }).notNull(),
  phone: varchar("phone", { length: 50 }),
  country: varchar("country", { length: 50 }),
  
  roleType: varchar("role_type", { length: 50 }).notNull(), // sitter, driver, groomer, courier, etc
  status: varchar("status", { length: 50 }).default("pending"), // pending, active, blocked
  riskScore: float("risk_score"),

  createdAt: timestamp("created_at").defaultNow().notNull(),
  updatedAt: timestamp("updated_at").defaultNow().notNull()
});

export const identityDocuments = pgTable("identity_documents", {
  id: uuid("id").primaryKey().defaultRandom(),
  contractorId: uuid("contractor_id").references(() => contractors.id),
  documentType: varchar("document_type", { length: 50 }).notNull(),
  documentNumber: varchar("document_number", { length: 100 }).notNull(),
  issuedCountry: varchar("issued_country", { length: 50 }),
  expiryDate: varchar("expiry_date", { length: 50 }),
  createdAt: timestamp("created_at").defaultNow()
});

export const drivers = pgTable("drivers", {
  id: uuid("id").primaryKey().defaultRandom(),
  contractorId: uuid("contractor_id").references(() => contractors.id),
  vehicleType: varchar("vehicle_type", { length: 100 }),
  licenseNumber: varchar("license_number", { length: 100 }),
  licenseExpiry: varchar("license_expiry", { length: 50 }),
  areasOfService: text("areas_of_service"),
  isActive: boolean("is_active").default(true),
  createdAt: timestamp("created_at").defaultNow()
});

export const ratings = pgTable("ratings", {
  id: uuid("id").primaryKey().defaultRandom(),
  contractorId: uuid("contractor_id").references(() => contractors.id),
  givenByUserId: uuid("given_by_user_id"),
  score: integer("score"),
  category: varchar("category", { length: 50 }), // communication, reliability, etc
  comment: text("comment"),
  createdAt: timestamp("created_at").defaultNow()
});

import express from "express";

/* ---------------------- CONTRACTORS ---------------------- */
router.post("/api/contractors", async (req, res) => {
  const db = req.db;
  const { fullName, email, phone, country, roleType } = req.body;
  const result = await db.insert(contractors).values({
    fullName, email, phone, country, roleType
  }).returning();
  res.json(result[0]);
});

router.get("/api/contractors", async (req, res) => {
  const db = req.db;
  const result = await db.select().from(contractors);
  res.json(result);
});

router.get("/api/contractors/:id", async (req, res) => {
  const db = req.db;
  const { id } = req.params;
  const result = await db.select().from(contractors).where(contractors.id.eq(id));
  res.json(result[0] || {});
});

   =========================================================== */
router.post("/api/compliance/evaluate", async (req, res) => {
  const db = req.db;
  const { contractorId } = req.body;

  const contractor = await db.query.contractors.findFirst({ where: (c) => c.id.eq(contractorId) });

  const docs = await db.query.identityDocuments.findMany({ where: (d) => d.contractorId.eq(contractorId) });
  const driver = await db.query.drivers.findFirst({ where: (d) => d.contractorId.eq(contractorId) });
  const contractorRatings = await db.query.ratings.findMany({ where: (r) => r.contractorId.eq(contractorId) });

  // SIMPLE RISK ENGINE
  let riskScore = 0.1;
  if (contractorRatings.length > 0) {
    const avg = contractorRatings.reduce((a, r) => a + r.score, 0) / contractorRatings.length;
    riskScore = Math.max(0.05, 1.2 - avg * 0.2);
  }

  const eligible = riskScore < 0.7;

  res.json({
    contractorId,
    eligible,
    riskScore,
    reasons: eligible ? ["clean profile", "acceptable rating pattern"] : ["high risk score"],
    recommendedLimits: {
      maxBookingsPerDay: eligible ? 10 : 2,
      verificationTier: eligible ? "tier_2" : "tier_1"
    }
  });
});
