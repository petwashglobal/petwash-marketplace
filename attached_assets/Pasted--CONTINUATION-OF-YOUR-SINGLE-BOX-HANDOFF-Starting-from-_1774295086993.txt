async function placeBookingAndCapturePayment(input: BookingPlacedInput) {
  return withDbTransaction(async () => {
    // 1. Validate input
    if (input.totalAmount < 0) throw new Error("Invalid total");
    if (!input.customerUserId) throw new Error("Missing customer");

    // 2. Load users
    const customer = await loadUser(input.customerUserId);
    if (!customer) throw new Error("Customer not found");

    const provider = input.providerUserId
      ? await loadUser(input.providerUserId)
      : null;

    // 3. Generate references
    const bookingId = randomUUID();
    const bookingReference = generateBookingReference();
    const transactionId = randomUUID();
    const transactionReference = generateTransactionReference("PWT");

    // 4. Create booking record
    await dbInsert("bookings", {
      id: bookingId,
      booking_reference: bookingReference,
      customer_user_id: input.customerUserId,
      provider_user_id: input.providerUserId ?? null,
      pet_id: input.petId ?? null,
      service_type: input.serviceType,
      service_name: input.serviceName,
      status: provider ? "placed" : "provider_pending",
      scheduled_start_at: input.scheduledStartAt,
      scheduled_end_at: input.scheduledEndAt ?? null,
      location_text: input.locationText ?? null,
      subtotal_amount: input.subtotalAmount,
      discount_amount: input.discountAmount,
      tax_amount: input.taxAmount,
      total_amount: input.totalAmount,
      provider_gross_amount: calculateProviderGross(input.totalAmount),
      platform_fee_amount: calculatePlatformFee(input.totalAmount),
      provider_net_amount: calculateProviderNet(input.totalAmount),
      currency: input.currency,
      promo_code: input.promoCode ?? null,
      placed_at: new Date().toISOString(),
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    });

    // 5. Create payment transaction
    const auditHash = createAuditHash({
      bookingReference,
      transactionReference,
      totalAmount: input.totalAmount,
      customerUserId: input.customerUserId,
      externalPaymentReference: input.externalPaymentReference ?? null,
      stampedAt: new Date().toISOString(),
    });

    await dbInsert("payment_transactions", {
      id: transactionId,
      transaction_reference: transactionReference,
      user_id: input.customerUserId,
      booking_id: bookingId,
      transaction_type: "booking_payment",
      payment_provider: input.paymentProvider,
      payment_method: input.paymentMethod,
      payment_status: "captured",
      subtotal_amount: input.subtotalAmount,
      discount_amount: input.discountAmount,
      tax_amount: input.taxAmount,
      total_amount: input.totalAmount,
      currency: input.currency,
      external_payment_reference: input.externalPaymentReference ?? null,
      audit_hash: auditHash,
      transaction_stamped_at: new Date().toISOString(),
      raw_gateway_payload: {
        provider: input.paymentProvider,
        method: input.paymentMethod,
      },
      metadata_json: {
        promoCode: input.promoCode ?? null,
      },
      created_at: new Date().toISOString(),
    });

    // 6. Create customer receipt / invoice
    const customerDocumentId = randomUUID();
    const customerDocReference = generateTransactionReference("PWR");

    const receiptPayload = buildCustomerBookingReceiptPayload({
      bookingReference,
      transactionReference,
      customer,
      provider,
      petName: input.petName ?? null,
      serviceName: input.serviceName,
      scheduledStartAt: input.scheduledStartAt,
      scheduledEndAt: input.scheduledEndAt ?? null,
      locationText: input.locationText ?? null,
      paymentMethod: input.paymentMethod,
      subtotalAmount: input.subtotalAmount,
      discountAmount: input.discountAmount,
      taxAmount: input.taxAmount,
      totalAmount: input.totalAmount,
      currency: input.currency,
      nextSteps: provider
        ? "Your booking has been placed. The provider will review and confirm shortly."
        : "Your booking has been placed. Provider assignment is pending.",
    });

    await dbInsert("financial_documents", {
      id: customerDocumentId,
      document_reference: customerDocReference,
      user_id: input.customerUserId,
      booking_id: bookingId,
      transaction_id: transactionId,
      document_type: "customer_receipt",
      issued_by_entity: "PetWash",
      document_payload_json: receiptPayload,
      rendered_html: renderCustomerBookingEmailHtml(receiptPayload),
      rendered_pdf_url: null,
      issued_at: new Date().toISOString(),
    });

    // 7. Record provider earnings if provider exists
    if (provider) {
      await dbInsert("provider_earnings", {
        id: randomUUID(),
        provider_user_id: provider.id,
        booking_id: bookingId,
        transaction_id: transactionId,
        gross_amount: calculateProviderGross(input.totalAmount),
        platform_fee_amount: calculatePlatformFee(input.totalAmount),
        net_amount: calculateProviderNet(input.totalAmount),
        currency: input.currency,
        settlement_status: "pending",
        recorded_at: new Date().toISOString(),
      });

      // Provider-side earnings notice document
      const providerDocumentId = randomUUID();
      const providerDocReference = generateTransactionReference("PWPAY");

      const providerPayload = buildProviderBookingNoticePayload({
        bookingReference,
        customerFirstName: customer.first_name ?? customer.display_name ?? "Customer",
        petName: input.petName ?? null,
        serviceName: input.serviceName,
        scheduledStartAt: input.scheduledStartAt,
        scheduledEndAt: input.scheduledEndAt ?? null,
        locationText: input.locationText ?? null,
        grossAmount: calculateProviderGross(input.totalAmount),
        platformFeeAmount: calculatePlatformFee(input.totalAmount),
        netAmount: calculateProviderNet(input.totalAmount),
        currency: input.currency,
      });

      await dbInsert("financial_documents", {
        id: providerDocumentId,
        document_reference: providerDocReference,
        user_id: provider.id,
        booking_id: bookingId,
        transaction_id: transactionId,
        document_type: "provider_earnings_notice",
        issued_by_entity: "PetWash",
        document_payload_json: providerPayload,
        rendered_html: renderProviderBookingEmailHtml(providerPayload),
        rendered_pdf_url: null,
        issued_at: new Date().toISOString(),
      });
    }

    // 8. Write audit event
    await dbInsert("audit_events", {
      id: randomUUID(),
      actor_user_id: input.customerUserId,
      entity_type: "booking",
      entity_id: bookingId,
      action_key: "booking_placed_and_paid",
      before_json: null,
      after_json: {
        bookingReference,
        transactionReference,
        paymentStatus: "captured",
      },
      audit_hash: auditHash,
      ip_address: null,
      user_agent: null,
      created_at: new Date().toISOString(),
    });

    // 9. Queue notifications AFTER commit-ready data exists
    await dispatchNotifications({
      userId: input.customerUserId,
      bookingId,
      transactionId,
      eventType: "booking_placed",
      templateKey: "customer_booking_placed",
      channels: ["email", "sms", "push"],
      payload: receiptPayload,
    });

    if (provider) {
      await dispatchNotifications({
        userId: provider.id,
        bookingId,
        transactionId,
        eventType: "provider_booking_assigned",
        templateKey: "provider_new_booking_assigned",
        channels: ["email", "sms", "push"],
        payload: buildProviderBookingNoticePayload({
          bookingReference,
          customerFirstName: customer.first_name ?? customer.display_name ?? "Customer",
          petName: input.petName ?? null,
          serviceName: input.serviceName,
          scheduledStartAt: input.scheduledStartAt,
          scheduledEndAt: input.scheduledEndAt ?? null,
          locationText: input.locationText ?? null,
          grossAmount: calculateProviderGross(input.totalAmount),
          platformFeeAmount: calculatePlatformFee(input.totalAmount),
          netAmount: calculateProviderNet(input.totalAmount),
          currency: input.currency,
        }),
      });
    }

    return {
      bookingId,
      bookingReference,
      transactionId,
      transactionReference,
      auditHash,
    };
  });
}

async function dispatchNotifications(job: NotificationJob): Promise<void> {
  const user = await loadUser(job.userId);
  if (!user) return;

  const prefs = await loadNotificationPreferences(job.userId);

  const effectiveChannels: NotificationChannel[] = [];

  for (const channel of job.channels) {
    if (channel === "email" && user.email && prefs.email_enabled && user.email_notifications_enabled) {
      effectiveChannels.push("email");
    }
    if (channel === "sms" && user.phone_e164 && prefs.sms_enabled && user.sms_notifications_enabled) {
      effectiveChannels.push("sms");
    }
    if (channel === "push" && prefs.push_enabled && user.push_notifications_enabled) {
      effectiveChannels.push("push");
    }
  }

  const template = resolveNotificationTemplate(job.templateKey, user.preferred_language, job.payload);

  for (const channel of effectiveChannels) {
    const eventId = randomUUID();

    await dbInsert("notification_events", {
      id: eventId,
      user_id: job.userId,
      booking_id: job.bookingId ?? null,
      transaction_id: job.transactionId ?? null,
      event_type: job.eventType,
      channel,
      template_key: job.templateKey,
      destination:
        channel === "email"
          ? user.email ?? null
          : channel === "sms"
            ? user.phone_e164 ?? null
            : user.id,
      subject: template.subject,
      payload_json: job.payload,
      send_status: "queued",
      provider_message_id: null,
      failure_reason: null,
      sent_at: null,
      created_at: new Date().toISOString(),
    });

    try {
      if (channel === "email") {
        const providerMessageId = await sendEmail({
          to: user.email!,
          subject: template.subject,
          html: template.html,
          text: template.text,
        });

        await markNotificationSent(eventId, providerMessageId);
      }

      if (channel === "sms") {
        const providerMessageId = await sendSms({
          to: user.phone_e164!,
          message: template.smsText,
        });

        await markNotificationSent(eventId, providerMessageId);
      }

      if (channel === "push") {
        const providerMessageId = await sendPush({
          userId: user.id,
          title: template.pushTitle,
          body: template.pushBody,
          data: template.pushData,
        });

        await markNotificationSent(eventId, providerMessageId);
      }
    } catch (error: any) {
      await markNotificationFailed(eventId, error?.message ?? "Unknown failure");
    }
  }
}

async function markNotificationSent(notificationEventId: UUID, providerMessageId: string | null) {
  await dbExecute(
    `
      UPDATE notification_events
      SET send_status = 'sent',
          provider_message_id = $2,
          sent_at = NOW()
      WHERE id = $1
    `,
    [notificationEventId, providerMessageId]
  );
}

async function markNotificationFailed(notificationEventId: UUID, failureReason: string) {
  await dbExecute(
    `
      UPDATE notification_events
      SET send_status = 'failed',
          failure_reason = $2
      WHERE id = $1
    `,
    [notificationEventId, failureReason]
  );
}

interface NotificationTemplate {
  subject: string;
  html: string;
  text: string;
  smsText: string;
  pushTitle: string;
  pushBody: string;
  pushData: Record<string, unknown>;
}

function resolveNotificationTemplate(
  templateKey: string,
  language: string,
  payload: Record<string, unknown>
): NotificationTemplate {
  switch (templateKey) {
    case "customer_booking_placed":
      return buildCustomerBookingPlacedTemplate(payload);

    case "provider_new_booking_assigned":
      return buildProviderNewBookingAssignedTemplate(payload);

    case "customer_prestige_joined":
      return buildCustomerPrestigeJoinedTemplate(payload);

    case "customer_egift_purchased":
      return buildCustomerEGiftPurchasedTemplate(payload);

    case "provider_registration_received":
      return buildProviderRegistrationReceivedTemplate(payload);

    case "provider_registration_approved":
      return buildProviderRegistrationApprovedTemplate(payload);

    default:
      return {
        subject: "PetWash update",
        html: "<p>PetWash update</p>",
        text: "PetWash update",
        smsText: "PetWash update",
        pushTitle: "PetWash",
        pushBody: "You have a new update",
        pushData: {},
      };
  }
}

    <ul>
      <li>Pet: ${payload.petName ?? "Not specified"}</li>

      <li>Location: ${payload.locationText ?? "Not specified"}</li>

function randomUUID(): UUID {
  return cryptoRandomUuid();
}

function calculateProviderGross(totalAmount: number): number {
  return roundMoney(totalAmount);
}

function calculatePlatformFee(totalAmount: number): number {
  return roundMoney(totalAmount * 0.15);
}

function calculateProviderNet(totalAmount: number): number {
  return roundMoney(calculateProviderGross(totalAmount) - calculatePlatformFee(totalAmount));
}

function roundMoney(value: number): number {
  return Math.round(value * 100) / 100;
}

function createAuditHash(payload: Record<string, unknown>): string {
  // Replace with real SHA-256 hashing in production
  return `sha256_${Buffer.from(JSON.stringify(payload)).toString("base64").slice(0, 32)}`;
}

function formatMoney(amount: number, currency: string): string {
  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency,
  }).format(amount);
}

function formatDateTime(value: string): string {
  return new Date(value).toLocaleString("en-AU", {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

function formatShortDate(value: string): string {
  return new Date(value).toLocaleDateString("en-AU");
}

function addMonths(date: Date, months: number): Date {
  const copy = new Date(date);
  copy.setMonth(copy.getMonth() + months);
  return copy;
}

function generateGiftCode(): string {
  const rand = Math.random().toString(36).slice(2, 8).toUpperCase();
  return `PWGIFT-${rand}`;
}

function generateGiftQrToken(code: string): string {
  return `petwash:egift:${code}:${Date.now()}`;
}
