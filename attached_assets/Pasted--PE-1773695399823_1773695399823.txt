------------------------------------------------------------------------------------------------

- אין "לצרוב" את זה כהכנסה מוכרת לשירות לפני הצריכה, אלא אם רו"ח מורה אחרת למסלול מוצר ספציפי

4.4 ב-Provider commission model
------------------------------------------------------------------------------------------------
אם Pet Wash היא רק marketplace:
- gross customer charge != platform revenue
- platform revenue = commission
- vat calculation for Pet Wash revenue applies on the platform’s taxable revenue component
- provider payable נרשם בנפרד

7. ארכיטקטורת תשלום - זרימות
************************************************************************************************

7.1 K9000 direct sale via Nayax
------------------------------------------------------------------------------------------------
Client
-> Nayax Terminal
-> Card Network
-> Nayax settlement
-> Nayax webhook / poll
-> Pet Wash backend
-> Create MACHINE_DIRECT_SALE transaction
-> Calculate VAT + net revenue
-> Issue receipt / accounting event
-> Unlock machine / confirm completed wash
-> Audit ledger

7.2 Wallet top-up
------------------------------------------------------------------------------------------------
Client
-> App/Web
-> Payment processor
-> webhook/confirmation
-> Create WALLET_TOPUP transaction
-> Increase wallet liability
-> Issue receipt / top-up record
-> No service consumption yet
-> Audit ledger

7.3 Wallet redeem at K9000
------------------------------------------------------------------------------------------------
Client
-> Apple Wallet / Google Wallet / App QR
-> Kiosk scanner
-> Backend validates signed token
-> Check balance / status / expiry / anti-replay
-> Start machine
-> Create WALLET_REDEEM_K9000 transaction
-> Reduce wallet liability
-> Recognize service revenue + VAT
-> Audit ledger
-> Push wallet update

7.4 Provider booking
------------------------------------------------------------------------------------------------
Client
-> Booking checkout
-> Payment auth/capture
-> Create PROVIDER_BOOKING_CHARGE
-> Branch logic:
   if commercialModel = MARKETPLACE_COMMISSION
      create provider payable
      create platform commission
      create payout schedule
   else if commercialModel = PRINCIPAL
      recognize platform gross revenue
      create subcontractor payable
-> Audit ledger

9. קוד ייחוס - חישובי מע"מ, נטו, עמלה, Wallet

const VAT_RATE_ILS_2026 = 0.18;

function round2(n) {
  return Math.round((Number(n) + Number.EPSILON) * 100) / 100;
}

function splitVatIncluded(gross) {
  const net = gross / (1 + VAT_RATE_ILS_2026);
  const vat = gross - net;
  return {
    gross: round2(gross),
    net: round2(net),
    vat: round2(vat),
  };
}

function calcNayaxDirectSale({
  grossPaidByCustomer,
  nayaxFeeGross,
}) {
  const sale = splitVatIncluded(grossPaidByCustomer);

  return {
    customerPaidGross: round2(grossPaidByCustomer),
    serviceNetRevenue: sale.net,
    outputVat: sale.vat,
    processorFeeGross: round2(nayaxFeeGross),
    companyCashAfterProcessor: round2(grossPaidByCustomer - nayaxFeeGross),
  };
}

function calcMarketplaceCommission({
  bookingGross,
  commissionRate,          // 0.15
  commissionIsVatIncluded = true,
}) {
  const commissionGross = round2(bookingGross * commissionRate);

  if (commissionIsVatIncluded) {
    const commissionSplit = splitVatIncluded(commissionGross);
    return {
      bookingGross: round2(bookingGross),
      commissionGross: commissionSplit.gross,
      commissionNet: commissionSplit.net,
      commissionVat: commissionSplit.vat,
      providerPayable: round2(bookingGross - commissionGross),
    };
  }

  const commissionVat = round2(commissionGross * VAT_RATE_ILS_2026);
  return {
    bookingGross: round2(bookingGross),
    commissionGross: round2(commissionGross + commissionVat),
    commissionNet: round2(commissionGross),
    commissionVat,
    providerPayable: round2(bookingGross - commissionGross - commissionVat),
  };
}

function createWalletTopupLedger({
  userId,
  amountGross,
  processor,
  processorTxnId,
}) {
  return {
    payment: {
      transactionType: "WALLET_TOPUP",
      transactionState: "SETTLED",
      userId,
      processor,
      processorTxnId,
      grossAmount: round2(amountGross),
      netAmount: 0,
      vatAmount: 0,
      processorFeeAmount: 0,
      metadata: {
        note: "Stored value top-up. Liability created, service not yet consumed.",
      },
    },
    walletLedger: {
      direction: "CREDIT",
      amount: round2(amountGross),
      reason: "TOPUP",
    },
  };
}

function createWalletRedeemK9000({
  userId,
  serviceGross,
  machineId,
  sourceRef,
}) {
  const split = splitVatIncluded(serviceGross);

  return {
    payment: {
      transactionType: "WALLET_REDEEM_K9000",
      transactionState: "CAPTURED",
      userId,
      machineId,
      sourceRef,
      grossAmount: split.gross,
      netAmount: split.net,
      vatAmount: split.vat,
      metadata: {
        note: "Wallet consumed for K9000 wash service",
      },
    },
    walletLedger: {
      direction: "DEBIT",
      amount: split.gross,
      reason: "K9000_SERVICE_CONSUMPTION",
    },
  };
}

function createMachineDirectSale({
  userId = null,
  machineId,
  processorTxnId,
  grossPaidByCustomer,
  nayaxFeeGross = 0,
}) {
  const result = calcNayaxDirectSale({
    grossPaidByCustomer,
    nayaxFeeGross,
  });

  return {
    payment: {
      transactionType: "MACHINE_DIRECT_SALE",
      transactionState: "SETTLED",
      userId,
      machineId,
      processor: "NAYAX",
      processorTxnId,
      grossAmount: result.customerPaidGross,
      netAmount: result.serviceNetRevenue,
      vatAmount: result.outputVat,
      processorFeeAmount: result.processorFeeGross,
      metadata: {
        note: "Direct anonymous/public kiosk sale",
      },
    },
  };
}

************************************************************************************************
10. Webhook processing - Nayax
************************************************************************************************

/*
מטרת ה-webhook:
- לאשר שעסקה הצליחה
- לזהות machine / terminal / merchant transaction
- ליצור transaction במערכת
- לייצר מסמך / רישום accounting / קבלה
- להניע machine logic אם התשלום הוא authorization להפעלה
*/

async function handleNayaxWebhook(event) {
  // Validate signature if provided by Nayax
  // Log raw payload immutably

  const {
    transactionId,
    status,
    amount,
    machineId,
    customerMode,
    walletTopup,
  } = normalizeNayaxPayload(event);

  if (status !== "APPROVED") {
    await writeAuditStamp({
      relatedRef: transactionId,
      stampType: "NAYAX_FAILED_OR_DECLINED",
      payload: event,
    });
    return { ok: true };
  }

  // Branch 1 - direct machine sale
  if (customerMode === "PUBLIC_WALKIN" && !walletTopup) {
    const tx = createMachineDirectSale({
      machineId,
      processorTxnId: transactionId,
      grossPaidByCustomer: amount,
      nayaxFeeGross: estimateProcessorFee(amount),
    });

    await persistPayment(tx.payment);
    await issueReceiptOrAccountingRecord(tx.payment);
    await writeAuditStamp({
      relatedRef: transactionId,
      stampType: "MACHINE_DIRECT_SALE_RECORDED",
      payload: tx,
    });

    return { ok: true };
  }

  // Branch 2 - wallet topup
  if (walletTopup === true) {
    const tx = createWalletTopupLedger({
      userId: event.userId,
      amountGross: amount,
      processor: "NAYAX",
      processorTxnId: transactionId,
    });

    await persistPayment(tx.payment);
    await applyWalletCredit(event.userId, tx.walletLedger.amount, tx.walletLedger);
    await issueTopupReceipt(tx.payment);
    await writeAuditStamp({
      relatedRef: transactionId,
      stampType: "WALLET_TOPUP_RECORDED",
      payload: tx,
    });

    return { ok: true };
  }

  return { ok: true };
}

************************************************************************************************
11. QR / Wallet / Kiosk flow
************************************************************************************************

/*
לקוח רשום לא משלם ב-Nayax אם הוא בוחר לממש wallet/points/credit.
במקרה כזה:
- QR/Pass נועד לזהות ולאשר
- backend הוא מקור האמת
- machine לא מחליטה לבד
*/

function generateSignedRedeemToken({ userId, passId, machineId = null, ttlSeconds = 45 }) {
  const now = Math.floor(Date.now() / 1000);

  const payload = {
    userId,
    passId,
    machineId,
    iat: now,
    exp: now + ttlSeconds,
    nonce: crypto.randomUUID(),
    purpose: "K9000_REDEEM",
  };

  const body = Buffer.from(JSON.stringify(payload)).toString("base64url");
  const sig = crypto.createHmac("sha256", process.env.PRESTIGE_QR_SECRET).update(body).digest("base64url");
  return `${body}.${sig}`;
}

function verifySignedRedeemToken(token) {
  const [body, sig] = token.split(".");
  if (!body || !sig) throw new Error("INVALID_TOKEN_FORMAT");

  const expected = crypto.createHmac("sha256", process.env.PRESTIGE_QR_SECRET).update(body).digest("base64url");
  if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) {
    throw new Error("INVALID_SIGNATURE");
  }

  const payload = JSON.parse(Buffer.from(body, "base64url").toString("utf8"));
  const now = Math.floor(Date.now() / 1000);
  if (payload.exp < now) throw new Error("TOKEN_EXPIRED");

  return payload;
}

async function redeemAtK9000({ token, machineId, amountGross }) {
  const payload = verifySignedRedeemToken(token);

  if (payload.machineId && payload.machineId !== machineId) {
    throw new Error("TOKEN_MACHINE_MISMATCH");
  }

  await assertNonceUnusedAndConsume(payload.nonce);

  const wallet = await loadWalletByUserId(payload.userId);
  if (!wallet) throw new Error("WALLET_NOT_FOUND");
  if (wallet.availableCredit < amountGross) throw new Error("INSUFFICIENT_WALLET");

  const tx = createWalletRedeemK9000({
    userId: payload.userId,
    serviceGross: amountGross,
    machineId,
    sourceRef: payload.passId,
  });

  await db.transaction(async (trx) => {
    await persistPayment(tx.payment, trx);
    await applyWalletDebit(payload.userId, tx.walletLedger.amount, tx.walletLedger, trx);
    await issueServiceDocument(tx.payment, trx);
    await startMachine(machineId, trx);
    await writeAuditStamp({
      relatedRef: tx.payment.sourceRef || tx.payment.processorTxnId || payload.passId,
      stampType: "K9000_WALLET_REDEEM",
      payload: {
        machineId,
        amountGross,
        payment: tx.payment,
        walletLedger: tx.walletLedger,
      },
    }, trx);
  });

  return { ok: true };
}

const COMMERCIAL_MODEL = {
  MARKETPLACE_COMMISSION: "MARKETPLACE_COMMISSION",
  PRINCIPAL: "PRINCIPAL",
};

async function processProviderBookingCharge({
  bookingId,
  userId,
  providerId,
  bookingGross,
  commissionRate = 0.15,
  commercialModel,
}) {
  if (commercialModel === COMMERCIAL_MODEL.MARKETPLACE_COMMISSION) {
    const result = calcMarketplaceCommission({
      bookingGross,
      commissionRate,
      commissionIsVatIncluded: true,
    });

    await db.transaction(async (trx) => {
      await persistPayment({
        transactionType: "PROVIDER_BOOKING_CHARGE",
        transactionState: "SETTLED",
        userId,
        providerId,
        sourceRef: bookingId,
        grossAmount: result.bookingGross,
        netAmount: 0,
        vatAmount: 0,
        providerPayoutAmount: result.providerPayable,
        metadata: {
          commercialModel: "MARKETPLACE_COMMISSION",
        },
      }, trx);

      await persistPayment({
        transactionType: "PROVIDER_COMMISSION",
        transactionState: "CAPTURED",
        userId,
        providerId,
        sourceRef: bookingId,
        grossAmount: result.commissionGross,
        netAmount: result.commissionNet,
        vatAmount: result.commissionVat,
        providerPayoutAmount: result.providerPayable,
        metadata: {
          commercialModel: "MARKETPLACE_COMMISSION",
        },
      }, trx);

      await createProviderPayout({
        providerId,
        bookingId,
        grossBookingAmount: result.bookingGross,
        platformCommissionAmount: result.commissionNet,
        commissionVatAmount: result.commissionVat,
        providerPayableAmount: result.providerPayable,
      }, trx);

      await issueCommissionDocument({
        bookingId,
        providerId,
        commissionGross: result.commissionGross,
        commissionVat: result.commissionVat,
      }, trx);

      await writeAuditStamp({
        relatedRef: bookingId,
        stampType: "PROVIDER_MARKETPLACE_FLOW",
        payload: result,
      }, trx);
    });

    return { ok: true, commercialModel };
  }

  if (commercialModel === COMMERCIAL_MODEL.PRINCIPAL) {
    const split = splitVatIncluded(bookingGross);

    await db.transaction(async (trx) => {
      await persistPayment({
        transactionType: "PROVIDER_BOOKING_CHARGE",
        transactionState: "SETTLED",
        userId,
        providerId,
        sourceRef: bookingId,
        grossAmount: split.gross,
        netAmount: split.net,
        vatAmount: split.vat,
        metadata: {
          commercialModel: "PRINCIPAL",
        },
      }, trx);

      await createProviderPayableAsSubcontractor({
        providerId,
        bookingId,
        amount: decideSubcontractorAmount(bookingGross),
      }, trx);

      await issueServiceDocument({
        sourceRef: bookingId,
        grossAmount: split.gross,
        netAmount: split.net,
        vatAmount: split.vat,
      }, trx);

      await writeAuditStamp({
        relatedRef: bookingId,
        stampType: "PROVIDER_PRINCIPAL_FLOW",
        payload: split,
      }, trx);
    });

    return { ok: true, commercialModel };
  }

  throw new Error("UNKNOWN_COMMERCIAL_MODEL");
}

async function refundMachineSale({ paymentId, reason }) {
  await db.transaction(async (trx) => {
    const payment = await loadPayment(paymentId, trx);
    if (!payment) throw new Error("PAYMENT_NOT_FOUND");

    await persistPayment({
      transactionType: "REFUND",
      transactionState: "SETTLED",
      userId: payment.userId,
      machineId: payment.machineId,
      sourceRef: paymentId,
      grossAmount: payment.grossAmount,
      netAmount: -Math.abs(payment.netAmount),
      vatAmount: -Math.abs(payment.vatAmount),
      metadata: { reason },
    }, trx);

    await issueCreditNote(payment, reason, trx);

    await writeAuditStamp({
      relatedRef: paymentId,
      stampType: "REFUND_ISSUED",
      payload: { reason },
    }, trx);
  });
}

async function cancelBookingWithImmutableStamp({ bookingId, cancelledBy, role, reason }) {
  await db.transaction(async (trx) => {
    await markBookingCancelled(bookingId, cancelledBy, reason, trx);

    await writeAuditStamp({
      relatedRef: bookingId,
      stampType: "CANCELLATION",
      payload: {
        cancelledBy,
        role,
        reason,
        timestamp: new Date().toISOString(),
      },
    }, trx);
  });
}
