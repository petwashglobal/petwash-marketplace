generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  CUSTOMER
  PROVIDER
  ADMIN
  FINANCE
}

enum Platform {
  PETSITTER
  PETTREK
  ACADEMY
  PETWASH_HUB
}

enum BookingStatus {
  DRAFT
  CONFIRMED
  IN_PROGRESS
  COMPLETED
  CANCELLED
}

enum LedgerType {
  BOOKING_CREATED
  PAYMENT_CAPTURED
  WALLET_DEBIT
  WALLET_CREDIT
  PROVIDER_EARNING
  PLATFORM_FEE
  INVOICE_ISSUED
}

model User {
  id        String   @id @default(cuid())
  email     String?  @unique
  phone     String?  @unique
  fullName  String?
  role      Role     @default(CUSTOMER)
  kycStatus String   @default("pending")
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  wallet    Wallet?
  provider  Provider?
  bookings  Booking[]
}

model Provider {
  id           String   @id @default(cuid())
  userId       String   @unique
  city         String
  cityNormalized String
  services     Platform[]
  rating       Float    @default(5)
  approved     Boolean  @default(false)
  visible      Boolean  @default(false)
  createdAt    DateTime @default(now())

  user         User     @relation(fields: [userId], references: [id])
  bookings     Booking[]
}

model Wallet {
  id        String   @id @default(cuid())
  userId    String   @unique
  balance   Int      @default(0)
  petwashCredits  Int @default(0)
  petsitterCredits Int @default(0)
  pettrekCredits   Int @default(0)
  academyCredits   Int @default(0)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  user      User     @relation(fields: [userId], references: [id])
  ledger    Ledger[]
}

model Booking {
  id        String   @id @default(cuid())
  platform  Platform
  status    BookingStatus @default(DRAFT)
  userId    String
  providerId String?
  price     Int
  platformFee Int
  providerShare Int
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  user      User     @relation(fields: [userId], references: [id])
  provider  Provider? @relation(fields: [providerId], references: [id])
  ledger    Ledger[]
}

model Ledger {
  id        String   @id @default(cuid())
  type      LedgerType
  bookingId String?
  walletId  String?
  amount    Int
  platform  Platform?
  createdAt DateTime @default(now())

  booking   Booking? @relation(fields: [bookingId], references: [id])
  wallet    Wallet?  @relation(fields: [walletId], references: [id])
}

model Invoice {
  id        String   @id @default(cuid())
  bookingId String   @unique
  docNumber String?
  allocationNumber String?
  createdAt DateTime @default(now())

  booking   Booking @relation(fields: [bookingId], references: [id])
}
