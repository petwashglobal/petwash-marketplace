// =========================
// CONFIG
// =========================

// Base URL of your API (behind Google / Cloud Run / Load Balancer)
const BASE_URL = "https://api.petwash.co.il"; // change if needed

// Admin login for the platform
const ADMIN_EMAIL = "support@petwash.co.il"; // adjust
const ADMIN_PASSWORD = "ChangeMe123!";       // adjust

// If your Node version is older than 18, you need to install node-fetch and import it.
// In Node 18+ fetch exists globally.

// Small helper for sleep
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// =========================
// HTTP helper
// =========================

async function api(path, method = "GET", body, token) {
  const url = `${BASE_URL}${path}`;
  const opts = {
    method,
    headers: {
      "Content-Type": "application/json",
    },
  };
  if (token) {
    opts.headers["Authorization"] = `Bearer ${token}`;
  }
  if (body !== undefined) {
    opts.body = JSON.stringify(body);
  }

  const res = await fetch(url, opts);
  const text = await res.text();
  let json;
  try {
    json = text ? JSON.parse(text) : {};
  } catch (e) {
    json = { raw: text };
  }

  if (!res.ok) {
    console.error(`API ERROR ${method} ${path}:`, res.status, json);
    throw new Error(`API error ${res.status} on ${path}`);
  }
  return json;
}

// =========================
// LOGIN AS ADMIN
// =========================

async function loginAdmin() {
  console.log("🔐 Logging in as admin...");
  const data = await api("/auth/login", "POST", {
    email: ADMIN_EMAIL,
    password: ADMIN_PASSWORD,
  });
  if (!data.accessToken) {
    throw new Error("No accessToken in login response");
  }
  console.log("✅ Admin login OK");
  return data.accessToken;
}

// =========================
// CREATE CONTRACTORS
// =========================

async function createContractor(token, profile) {
  console.log(`👤 Creating contractor: ${profile.displayName}...`);
  const data = await api("/contractors", "POST", profile, token);
  console.log(`✅ Contractor created: ${data.id || data.contractorId}`);
  return data;
}

function baseIsraeliContractorPayload({
  displayName,
  email,
  phone,
  roleTag,
}) {
  const now = new Date().toISOString();
  return {
    displayName,
    legalName: `${displayName} Legal Name`,
    countryOfOperation: "IL",
    primaryCity: "Tel Aviv",
    contact: {
      email,
      phoneE164: phone,
      whatsappOptIn: true,
    },
    languageCodes: ["he", "en"],
    meta: {
      roleTag, // your backend can ignore this or store as metadata
    },
    // you can add extra fields if your schema requires
    createdAt: now,
    updatedAt: now,
  };
}

// =========================
// IDENTITY / DOCUMENTS / LIVENESS / CRIMINAL
// =========================

async function uploadDocument(token, contractorId, doc) {
  console.log(`📄 Uploading document ${doc.type} for ${contractorId}...`);
  const data = await api(
    `/identity/document`,
    "POST",
    {
      contractorId,
      type: doc.type,
      country: "IL",
      // Here we send metadata. Real file upload may be via presigned URL.
      fileName: doc.fileName,
      storageKey: doc.storageKey || null,
      verified: doc.verified || false,
      expiresAt: doc.expiresAt || null,
    },
    token
  );
  console.log(`✅ Document ${doc.type} stored id=${data.id || "?"}`);
  return data;
}

async function createLivenessCheck(token, contractorId, ok = true) {
  console.log(`🧑‍💻 Liveness check for ${contractorId}...`);
  const data = await api(
    `/identity/liveness`,
    "POST",
    {
      contractorId,
      provider: "OTHER",
      status: ok ? "PASSED" : "FAILED",
      isLive: ok,
      faceMatchScore: ok ? 95 : 50,
    },
    token
  );
  console.log(`✅ Liveness: ${data.status}`);
  return data;
}

async function createCriminalCheck(token, contractorId, blocked = false) {
  console.log(`⚖️ Criminal check for ${contractorId}...`);
  const hits = blocked
    ? [
        {
          category: "ANIMAL_CRUELTY",
          description: "Example record – should block.",
          isSpentOrCleared: false,
        },
      ]
    : [];

  const data = await api(
    `/identity/criminal`,
    "POST",
    {
      contractorId,
      provider: "OTHER",
      status: blocked ? "FAILED" : "PASSED",
      hits,
    },
    token
  );
  console.log(
    `✅ Criminal check: ${data.status}, hits=${data.hits ? data.hits.length : 0}`
  );
  return data;
}

// =========================
// DRIVER PROFILE
// =========================

async function createDriverProfile(token, contractorId, ok = true) {
  console.log(`🚗 Creating driver profile for ${contractorId}...`);
  const licenseExpiry = new Date();
  licenseExpiry.setFullYear(licenseExpiry.getFullYear() + 3);

  const data = await api(
    `/drivers`,
    "POST",
    {
      contractorId,
      selfDeclaredCleanRecord: ok,
      licenseVerified: ok,
      licenseExpiry: licenseExpiry.toISOString(),
      activeDrivingBans: [],
    },
    token
  );

  console.log("✅ Driver profile created");
  return data;
}

// =========================
// AVAILABILITY
// =========================

async function createAvailabilityProfile(token, contractorId) {
  console.log(`🕒 Creating availability for ${contractorId}...`);

  const body = {
    contractorId,
    timezone: "Asia/Jerusalem",
    weeklyRules: [
      // Sunday to Thursday, 09:00-18:00
      { dayOfWeek: 0, fromMinutes: 9 * 60, toMinutes: 18 * 60 },
      { dayOfWeek: 1, fromMinutes: 9 * 60, toMinutes: 18 * 60 },
      { dayOfWeek: 2, fromMinutes: 9 * 60, toMinutes: 18 * 60 },
      { dayOfWeek: 3, fromMinutes: 9 * 60, toMinutes: 18 * 60 },
      { dayOfWeek: 4, fromMinutes: 9 * 60, toMinutes: 18 * 60 },
    ],
    blackoutRanges: [],
    maxJobsPerDay: 5,
  };

  const data = await api(`/availability/profile`, "POST", body, token);
  console.log("✅ Availability profile created");
  return data;
}

// =========================
// RATINGS SEED
// =========================

async function seedRatings(token, contractorId, starsArray) {
  console.log(`⭐ Seeding ratings for ${contractorId}...`);
  for (const stars of starsArray) {
    await api(
      `/ratings`,
      "POST",
      {
        contractorId,
        jobId: null,
        stars,
        tags: stars >= 4 ? ["friendly", "professional"] : ["late"],
        comment:
          stars >= 4 ? "Very good test service" : "Not great in this test",
      },
      token
    );
    await sleep(50);
  }
  console.log("✅ Ratings seeded");
}

// =========================
// INCIDENTS SEED
// =========================

async function seedIncidents(token, contractorId, critical = false) {
  console.log(`🚨 Seeding incidents for ${contractorId}...`);
  if (!critical) {
    console.log("✅ No critical incidents needed for this profile");
    return;
  }
  await api(
    `/incidents`,
    "POST",
    {
      contractorId,
      status: "OPEN",
      severity: "CRITICAL",
      title: "Test critical safety incident",
      description: "Test incident to verify BLOCK logic in compliance engine",
    },
    token
  );
  console.log("✅ Critical incident created");
}

// =========================
// COMPLIANCE EVALUATE
// =========================

async function evaluateCompliance(
  token,
  contractorId,
  role,
  serviceType,
  startIso,
  endIso
) {
  console.log(
    `🧠 Evaluating compliance for ${contractorId} role=${role} service=${serviceType}...`
  );

  const data = await api(
    `/compliance/evaluate`,
    "POST",
    {
      contractorId,
      role,
      serviceType,
      startIso,
      endIso,
    },
    token
  );

  console.log("✅ Compliance result:", JSON.stringify(data, null, 2));
  return data;
}
